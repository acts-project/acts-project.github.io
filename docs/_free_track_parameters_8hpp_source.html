89 FreeTrackParameters(const Vector4& pos4, double phi, double theta,
90 double qOverP, std::optional<FreeMatrix> cov,
91 ParticleHypothesis particleHypothesis)
92 : m_params(FreeVector::Zero()),
93 m_cov(std::move(cov)),

96 m_params[eFreePos0] = pos4[ePos0];

101 m_params[eFreeDir1] = dir[eMom1];

158 double absoluteMomentum() const {
159 return m_particleHypothesis.extractMomentum(m_params[eFreeQOverP]);

161
163 double transverseMomentum() const {
164 // direction vector w/ arbitrary normalization can be parametrized as
165 // [f*sin(theta)*cos(phi), f*sin(theta)*sin(phi), f*cos(theta)]
166 // w/ f,sin(theta) positive, the transverse magnitude is then
167 // sqrt(f^2*sin^2(theta)) = f*sin(theta)
168 double transverseMagnitude2 =
169 square(m_params[eFreeDir0]) + square(m_params[eFreeDir1]);
170 // absolute magnitude is f by construction
171 double magnitude2 = transverseMagnitude2 + square(m_params[eFreeDir2]);
172 // such that we can extract sin(theta) = f*sin(theta) / f
173 return std::sqrt(transverseMagnitude2 / magnitude2) * absoluteMomentum();
174 }
175
177 Vector3 momentum() const { return absoluteMomentum() * direction(); }
178
181 double charge() const {
182 return m_particleHypothesis.extractCharge(get<eFreeQOverP>());
183 }

199 return reflected;
