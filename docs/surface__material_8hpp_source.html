39template <concepts::detector detector_t, typename view_t>
40auto material_grid(const detector_t& detector, const dindex index,
41 const view_t& view,
42 const styling::grid_style& style =
43 styling::tableau_colorblind::grid_style) {
44 using scalar_t = dscalar<typename detector_t::algebra_type>;
45
46 const auto& sf_desc = detector.surface(index);
47 const auto& link = sf_desc.material();
48
49 // Proactively calculate the reference radius for a cylinder grid
50 // (will only be used if the volume actually holds a barrel grid)
51 auto r =
52 detector.mask_store().template visit<detray::detail::outer_radius_getter>(
53 sf_desc.mask());
54
55 scalar_t cyl_ref_radius{0.f};
56 if (r.has_value()) {
57 cyl_ref_radius = *r;
58 }
59
60 return svgtools::conversion::grid(detector.material_store(), link, view,
61 cyl_ref_radius, style);
62}

65struct material_converter {
73 template <typename mat_coll_t, typename index_t,
74 concepts::transform3D transform3_t>
75 DETRAY_HOST inline auto operator()(const mat_coll_t& mat_coll,
76 const index_t& index,
77 const transform3_t& /*unused*/) const {
78 using material_t = typename mat_coll_t::value_type;
79
80 std::vector<std::vector<actsvg::proto::material_slab>> m_matrix;
81
82 if constexpr (concepts::surface_material<material_t> &&
83 concepts::material_map<material_t>) {
84 using loc_bin_idx_t = typename material_t::loc_bin_index;
85 using algebra_t = typename material_t::local_frame_type::algebra_type;
86
87 const auto material_map = mat_coll.at(index);
88
89 // Create the bin associations
90 dindex edges0 = material_map.template get_axis<0>().nbins();
91 dindex edges1 = material_map.template get_axis<1>().nbins();
92
93 // In the svg convention the phi axis has to be the second axis to
94 // loop over
95 constexpr bool is_cyl{
96 std::is_same_v<typename material_t::local_frame_type,
97 detray::cylindrical2D<algebra_t>> ||
98 std::is_same_v<typename material_t::local_frame_type,
99 detray::concentric_cylindrical2D<algebra_t>>};
100 if constexpr (is_cyl) {
101 dindex tmp = edges0;
102 edges0 = edges1;
103 edges1 = tmp;
104 }
105
106 m_matrix.reserve(edges1);
107
108 // Material map is always 2-dimensional
109 for (dindex j = 0u; j < edges1; ++j) {
110 std::vector<actsvg::proto::material_slab> m_matrix_row;
111 m_matrix_row.reserve(edges1);
112
113 for (dindex i = 0u; i < edges0; ++i) {
114 loc_bin_idx_t bin_idx{i, j};
115 if constexpr (is_cyl) {
116 bin_idx = {j, i};
117 }
118
119 const auto& mat_slab = material_map.bin(bin_idx).ref();
120 const auto& mat = mat_slab.get_material();
121
122 actsvg::proto::material_slab p_mat_slab;
123 p_mat_slab._properties = {
124 static_cast<actsvg::scalar>(mat.X0()),
125 static_cast<actsvg::scalar>(mat.L0()),
126 static_cast<actsvg::scalar>(mat.Ar()),
127 static_cast<actsvg::scalar>(mat.Z()),
128 static_cast<actsvg::scalar>(mat.mass_density()),
129 static_cast<actsvg::scalar>(mat_slab.thickness())};
130
131 m_matrix_row.push_back(p_mat_slab);
132 }
133
134 m_matrix.push_back(std::move(m_matrix_row));
135 }
136 }
137
138 return m_matrix;
139 }
140};

150template <concepts::detector detector_t, typename view_t>
151auto surface_material(const detector_t& detector,
152 const detray::geometry::surface<detector_t>& d_surface,
153 const view_t& view,
154 const styling::surface_material_style& style =
155 styling::tableau_colorblind::material_style) {
156 // Convert grid, if present
157 auto [p_grid, grid_type] = svgtools::conversion::material_grid(
158 detector, d_surface.index(), view, style._grid_style);
159
160 actsvg::proto::surface_material p_material;
161
162 // Create the surface material
163 if (p_grid.has_value()) {
164 // Fill material data
165 auto m_matrix = d_surface.template visit_material<material_converter>(
166 d_surface.transform({}));
167
168 p_material = {m_matrix, *p_grid};
169 p_material._material_ranges =
170 actsvg::proto::material_ranges(p_material._material_matrix);
171
172 svgtools::styling::apply_style(p_material, style);
173 }
174
175 return p_material;
176}
