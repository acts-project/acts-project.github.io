138 double d = Acts::detail::wrap_periodic(deg, -180.0, 360.0);
139
140 // If the angle was mapped to 0 from a non-0 multiple of 360, set it to 360
141 if (constexpr double eps = 1e-6; std::abs(d) < eps && std::abs(deg) > eps) {
