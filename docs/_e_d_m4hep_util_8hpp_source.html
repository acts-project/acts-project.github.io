232template <Acts::TrackProxyConcept track_proxy_t>
233void writeTrack(const Acts::GeometryContext& gctx,
234 const Acts::MagneticFieldContext& mctx, track_proxy_t track,
235 edm4hep::MutableTrack to,
236 const Acts::MagneticFieldProvider& magneticField,
237 const TrackerHitLookup& hitLookup = {},
238 const Acts::Logger& logger = Acts::getDummyLogger()) {
239 ACTS_VERBOSE("Converting track to EDM4hep");
240 to.setChi2(track.chi2());
241 to.setNdf(track.nDoF());
242 to.setNholes(static_cast<std::int32_t>(track.nHoles()));
243
244 auto fieldCache = magneticField.makeCache(mctx);
245
246 std::vector<edm4hep::TrackState> outTrackStates;
247 outTrackStates.reserve(track.nTrackStates());
248 // Source track states kept in parallel with outTrackStates, so that the hit
249 // lookup below can be run in the same order the states are emitted.
250 std::vector<Acts::AnyConstTrackStateProxy> outStates;
251 outStates.reserve(track.nTrackStates());
252
253 ACTS_VERBOSE("Converting " << track.nTrackStates() << " track states");
254
255 for (auto state : track.trackStatesReversed()) {
256 if (!state.typeFlags().isMeasurement()) {
257 continue;
258 }
259
260 outStates.push_back(Acts::AnyConstTrackStateProxy{state});
261
262 Acts::BoundTrackParameters params{state.referenceSurface().getSharedPtr(),
263 state.parameters(), state.covariance(),
264 track.particleHypothesis()};
265
266 outTrackStates.push_back(writeTrackState(gctx, edm4hep::TrackState::AtOther,
267 params, magneticField, fieldCache,
268 logger));
269 }
270 // At this point the measurement states are ordered outside-in (last hit
271 // first, first hit last), following Acts' reverse track state iteration.
272 if (!outTrackStates.empty()) {
273 outTrackStates.front().location = edm4hep::TrackState::AtLastHit;
274 outTrackStates.back().location = edm4hep::TrackState::AtFirstHit;
275 }
276
277 // Track state that represents the IP parameters. The reference point ends up
278 // at the track's own reference surface, or at an ad-hoc perigee created at
279 // its position if that surface is not already a perigee.
280 Acts::BoundTrackParameters trackParams{
281 track.referenceSurface().getSharedPtr(), track.parameters(),
282 track.covariance(), track.particleHypothesis()};
283
284 ACTS_VERBOSE("Writing track level quantities as IP track state");
285 edm4hep::TrackState ipState =
286 writeTrackState(gctx, edm4hep::TrackState::AtIP, trackParams,
287 magneticField, fieldCache, logger);
288
289 // Emit track states following the EDM4hep/LCIO convention used by
290 // k4ActsTracking: the IP state first, then the measurement states ordered
291 // inside-out (first hit ... last hit). Reverse the outside-in buffers so the
292 // measurement states, and their tracker hits, come out in that order.
293 std::ranges::reverse(outTrackStates);
294 std::ranges::reverse(outStates);
295
296 to.addToTrackStates(ipState);
297 for (std::size_t i = 0; i < outTrackStates.size(); ++i) {
298 to.addToTrackStates(outTrackStates[i]);
299 // Resolve the tracker hit here rather than in the loop above, so that the
300 // callback is invoked in the same order the states are emitted.
301 if (hitLookup) {
302 if (std::optional<edm4hep::TrackerHit> hit = hitLookup(outStates[i]);
303 hit.has_value()) {
304 to.addToTrackerHits(hit.value());
305 }
306 }
307 }
308}

356template <Acts::TrackProxyConcept track_proxy_t>
357void readTrack(const Acts::GeometryContext& gctx,
358 const Acts::MagneticFieldContext& mctx,
359 const edm4hep::Track& from, track_proxy_t& track,
360 const Acts::MagneticFieldProvider& magneticField,
361 const Acts::Logger& logger = Acts::getDummyLogger()) {
362 using namespace Acts;
363 ACTS_VERBOSE("Reading track from EDM4hep");
364 TrackStatePropMask mask = TrackStatePropMask::Smoothed;
365
366 auto fieldCache = magneticField.makeCache(mctx);
367 auto bzAtPosition = [&](const Acts::Vector3& position) {
368 return detail::localBz(magneticField, fieldCache, position, logger);
369 };
370
371 std::optional<edm4hep::TrackState> ipState;
372
373 ACTS_VERBOSE("Reading " << from.trackStates_size()
374 << " track states (including IP state)");
375 // Track states are written IP-first, followed by the measurement states
376 // ordered inside-out (first hit ... last hit). Iterate forward and skip the
377 // IP state so the reconstructed Acts track state order matches the input that
378 // was passed to writeTrack.
379 for (std::size_t i = 0; i < from.trackStates_size(); ++i) {
380 auto trackState = from.getTrackStates(i);
381 if (trackState.location == edm4hep::TrackState::AtIP) {
382 ipState = trackState;
383 continue;
384 }
385
386 auto params = detail::unpackTrackState(trackState);
387
388 auto ts = track.appendTrackState(mask);
389 ts.typeFlags().setIsMeasurement();
390
391 // Evaluate the local field where the particle actually is, matching what
392 // writeTrackState does
393 auto converted = detail::convertTrackParametersFromEdm4hep(
394 bzAtPosition(detail::pointOfClosestApproach(gctx, params)), params);
395
396 ts.smoothed() = converted.parameters();
397 ts.smoothedCovariance() =
398 converted.covariance().value_or(BoundMatrix::Zero());
399 ts.setReferenceSurface(params.surface);
400 }
401
402 if (!ipState.has_value()) {
403 ACTS_ERROR("Did not find IP state in edm4hep input");
404 throw std::runtime_error{"Did not find IP state in edm4hep input"};
405 }
406
407 detail::Parameters params = detail::unpackTrackState(ipState.value());
408
409 // Evaluate the local field where the particle actually is, matching what
410 // writeTrackState does
411 auto converted = detail::convertTrackParametersFromEdm4hep(
412 bzAtPosition(detail::pointOfClosestApproach(gctx, params)), params);
413
414 ACTS_VERBOSE("IP state parameters: " << converted.parameters().transpose());
415 ACTS_VERBOSE("-> covariance:\n"
416 << converted.covariance().value_or(BoundMatrix::Zero()));
417
418 track.parameters() = converted.parameters();
419 track.covariance() = converted.covariance().value_or(BoundMatrix::Zero());
420 track.setReferenceSurface(params.surface);
421
422 track.chi2() = from.getChi2();
423 track.nDoF() = from.getNdf();
424 track.nHoles() = static_cast<unsigned int>(from.getNholes());
425 track.nMeasurements() = track.nTrackStates();
426}
