107template <Acts::TrackProxyConcept track_proxy_t>
108void writeTrack(const Acts::GeometryContext& gctx, track_proxy_t track,
109 edm4hep::MutableTrack to, double Bz,
110 const Acts::Logger& logger = Acts::getDummyLogger()) {
111 ACTS_VERBOSE("Converting track to EDM4hep");
112 to.setChi2(track.chi2());
113 to.setNdf(track.nDoF());
114
115 std::vector<edm4hep::TrackState> outTrackStates;
116 outTrackStates.reserve(track.nTrackStates());
117
118 auto setParameters = [](edm4hep::TrackState& trackState,
119 const detail::Parameters& params) {
120 trackState.D0 = params.values[0];
121 trackState.Z0 = params.values[1];
122 trackState.phi = params.values[2];
123 trackState.tanLambda = params.values[3];
124 trackState.omega = params.values[4];
125 trackState.time = params.values[5];
126
127 if (params.covariance) {
128 detail::packCovariance(params.covariance.value(),
129 trackState.covMatrix.data());
130 }
131 };
132
133 ACTS_VERBOSE("Converting " << track.nTrackStates() << " track states");
134
135 for (const auto& state : track.trackStatesReversed()) {
136 auto typeFlags = state.typeFlags();
137 if (!typeFlags.isMeasurement()) {
138 continue;
139 }
140
141 edm4hep::TrackState& trackState = outTrackStates.emplace_back();
142 trackState.location = edm4hep::TrackState::AtOther;
143
144 Acts::BoundTrackParameters params{state.referenceSurface().getSharedPtr(),
145 state.parameters(), state.covariance(),
146 track.particleHypothesis()};
147
148 // Convert to LCIO track parametrization expected by EDM4hep
149 detail::Parameters converted =
150 detail::convertTrackParametersToEdm4hep(gctx, Bz, params);
151
152 // Write the converted parameters to the EDM4hep track state
153 setParameters(trackState, converted);
154 ACTS_VERBOSE("- parameters: " << state.parameters().transpose() << " -> "
155 << converted.values.transpose());
156 ACTS_VERBOSE("- covariance: \n"
157 << state.covariance() << "\n->\n"
158 << converted.covariance.value());
159
160 // Converted parameters are relative to an ad-hoc perigee surface created at
161 // the hit location
162 auto center = converted.surface->center(gctx);
163 trackState.referencePoint.x = center.x();
164 trackState.referencePoint.y = center.y();
165 trackState.referencePoint.z = center.z();
166 ACTS_VERBOSE("- ref surface ctr: " << center.transpose());
167 }
168 outTrackStates.front().location = edm4hep::TrackState::AtLastHit;
169 outTrackStates.back().location = edm4hep::TrackState::AtFirstHit;
170
171 // Add a track state that represents the IP parameters
172 auto& ipState = outTrackStates.emplace_back();
173
174 // Convert the track parameters at the IP
175 Acts::BoundTrackParameters trackParams{
176 track.referenceSurface().getSharedPtr(), track.parameters(),
177 track.covariance(), track.particleHypothesis()};
178
179 // Convert to LCIO track parametrization expected by EDM4hep
180 auto converted =
181 detail::convertTrackParametersToEdm4hep(gctx, Bz, trackParams);
182 setParameters(ipState, converted);
183 ipState.location = edm4hep::TrackState::AtIP;
184 ACTS_VERBOSE("Writing track level quantities as IP track state");
185 ACTS_VERBOSE("- parameters: " << track.parameters().transpose());
186 ACTS_VERBOSE(" -> " << converted.values.transpose());
187 ACTS_VERBOSE("- covariance: \n"
188 << track.covariance() << "\n->\n"
189 << converted.covariance.value());
190
191 // Write the converted parameters to the EDM4hep track state
192 // The reference point is at the location of the reference surface of the
193 // track itself, but if that's not a perigee surface, another ad-hoc perigee
194 // at the position will be created.
195 auto center = converted.surface->center(gctx);
196 ipState.referencePoint.x = center.x();
197 ipState.referencePoint.y = center.y();
198 ipState.referencePoint.z = center.z();
199
200 ACTS_VERBOSE("- ref surface ctr: " << center.transpose());
201
202 for (auto& trackState : outTrackStates) {
203 to.addToTrackStates(trackState);
204 }
205}

212template <Acts::TrackProxyConcept track_proxy_t>
213void readTrack(const edm4hep::Track& from, track_proxy_t& track, double Bz,
214 const Acts::Logger& logger = Acts::getDummyLogger()) {
215 using namespace Acts;
216 ACTS_VERBOSE("Reading track from EDM4hep");
217 TrackStatePropMask mask = TrackStatePropMask::Smoothed;
218
219 std::optional<edm4hep::TrackState> ipState;
220
221 auto unpack =
222 [](const edm4hep::TrackState& trackState) -> detail::Parameters {
223 detail::Parameters params;
224 params.covariance = BoundMatrix::Zero();
225 params.values = BoundVector::Zero();
226 detail::unpackCovariance(trackState.covMatrix.data(),
227 params.covariance.value());
228 params.values[0] = trackState.D0;
229 params.values[1] = trackState.Z0;
230 params.values[2] = trackState.phi;
231 params.values[3] = trackState.tanLambda;
232 params.values[4] = trackState.omega;
233 params.values[5] = trackState.time;
234
235 Vector3 center = {
236 trackState.referencePoint.x,
237 trackState.referencePoint.y,
238 trackState.referencePoint.z,
239 };
240 params.surface = Acts::Surface::makeShared<PerigeeSurface>(center);
241
242 return params;
243 };
244
245 ACTS_VERBOSE("Reading " << from.trackStates_size()
246 << " track states (including IP state)");
247 // We write the trackstates out outside in, need to reverse iterate to get the
248 // same order
249 for (std::size_t i = from.trackStates_size() - 1;
250 i <= from.trackStates_size(); i--) {
251 auto trackState = from.getTrackStates(i);
252 if (trackState.location == edm4hep::TrackState::AtIP) {
253 ipState = trackState;
254 continue;
255 }
256
257 auto params = unpack(trackState);
258
259 auto ts = track.appendTrackState(mask);
260 ts.typeFlags().setIsMeasurement();
261
262 auto converted = detail::convertTrackParametersFromEdm4hep(Bz, params);
263
264 ts.smoothed() = converted.parameters();
265 ts.smoothedCovariance() =
266 converted.covariance().value_or(BoundMatrix::Zero());
267 ts.setReferenceSurface(params.surface);
268 }
269
270 if (!ipState.has_value()) {
271 ACTS_ERROR("Did not find IP state in edm4hep input");
272 throw std::runtime_error{"Did not find IP state in edm4hep input"};
273 }
274
275 detail::Parameters params = unpack(ipState.value());
276
277 auto converted = detail::convertTrackParametersFromEdm4hep(Bz, params);
278
279 ACTS_VERBOSE("IP state parameters: " << converted.parameters().transpose());
280 ACTS_VERBOSE("-> covariance:\n"
281 << converted.covariance().value_or(BoundMatrix::Zero()));
282
283 track.parameters() = converted.parameters();
284 track.covariance() = converted.covariance().value_or(BoundMatrix::Zero());
285 track.setReferenceSurface(params.surface);
286
287 track.chi2() = from.getChi2();
288 track.nDoF() = from.getNdf();
289 track.nMeasurements() = track.nTrackStates();
290}
