50template <concepts::detector detector_t>
51inline dindex_range overlaps_removal(
52 dvector<intersection_record<detector_t>> &intersection_records,
53 const float tol = 1e-4f * unit<float>::mm) {
54 // Document any overlaps on the range
55 constexpr auto inv_idx{detray::detail::invalid_value<dindex>()};
56 dindex_range overlap_idx{inv_idx, inv_idx};
57
58 const std::size_t n_rec{intersection_records.size()};
59 std::size_t n_eq_intrs{1u}; //< Number of consecutive overlapping inters.
60 for (std::size_t i = 0u; i < n_rec - 1u; ++i) {
61 const auto &rec = intersection_records.at(i);
62 const auto &next_rec = intersection_records.at(i + 1u);
63
64 // This record and the following are overlapping: Count until we reach
65 // the end of the overlapping surfaces or end of trace
66 if (math::fabs(next_rec.intersection().path() - rec.intersection().path()) <
67 tol &&
68 i != (n_rec - 2u)) {
69 ++n_eq_intrs;
70 continue;
71 }
72
73 // No overlap: continue
74 if (n_eq_intrs == 1u) {
75 continue;
76 }
77 // Found two overlapping surfaces
78 if (n_eq_intrs == 2u) {
79 // Two overlapping portals form a valid, connected volume boundary
80 if (const auto &prev_rec = intersection_records.at(i - 1u);
81 !(rec.intersection().surface().is_portal() &&
82 prev_rec.intersection().surface().is_portal())) {
83 auto prev_sf_desc = prev_rec.intersection().surface();
84 auto sf_desc = rec.intersection().surface();
85
86 // Other types of surfaces must not overlap!
87 std::stringstream err_stream;
88 err_stream << "The following surfaces overlap at\n"
89 << "POS:\n"
90 << "glob: " << prev_rec.template position<double>()
91 << ", loc: " << prev_rec.intersection().local()
92 << "\nvs.\nglob: " << rec.template position<double>()
93 << ", loc: " << rec.intersection().local() << std::endl;
94 err_stream << "SURFACES:\n -> " << prev_sf_desc << std::endl;
95 err_stream << " -> " << sf_desc << std::endl;
96
97 // TODO: Fix wire_chamber geometry
98 // throw std::invalid_argument(err_stream.str());
99 DETRAY_ERROR_HOST(err_stream.str());
100 overlap_idx = {static_cast<dindex>(i - 1u), static_cast<dindex>(i)};
101 }
102 // Reset to find the next range
103 n_eq_intrs = 1u;
104 continue;
105 }
106
107 // Found more than two overlapping surfaces: Remove oversized portal
108 // intersections
109 std::size_t first{(i - (n_eq_intrs - 1u))};
110 std::size_t last{i};
111
112 // Map of the volume index the portals link to and the portal
113 // intersection indices in the intersection trace
114 std::multimap<std::size_t, std::size_t> pt_buckets{};
115
116 bool is_all_portals{true};
117 for (std::size_t j = first; j <= last; ++j) {
118 const auto &intr = intersection_records.at(j).intersection();
119 if (!intr.surface().is_portal()) {
120 is_all_portals = false;
121 }
122 pt_buckets.insert({static_cast<std::size_t>(intr.volume_link()), j});
123 }
124
125 // Remove buckets that contain only one portal: This is the valid exit
126 // portal that the overlapping portals need to be matched against
127 auto exit_idx{std::numeric_limits<std::size_t>::max()};
128 const auto n_erased =
129 std::erase_if(pt_buckets, [&pt_buckets, &exit_idx](const auto &item) {
130 const auto &[vol_link, rec_idx] = item;
131 if (pt_buckets.count(vol_link) == 1u) {
132 exit_idx = rec_idx;
133 return true;
134 }
135 return false;
136 });
137
138 // This is not a case of oversized portals, treat as actual overlaps
139 if (n_erased != 1u || pt_buckets.empty() || !is_all_portals) {
140 DETRAY_ERROR_HOST(
141 "Portals extrude volumes: Could not resolve which portals to remove "
142 "from trace (correct exit portal has to be kept!).\n Overlapping "
143 "intersection indices: "
144 << first << " to " << last);
145
146 overlap_idx = {static_cast<dindex>(first), static_cast<dindex>(last)};
147 } else {
148 // Keep only the portal that forms the correct volume crossing
149 // with the portal intersection at 'exit_idx'
150 const auto &exit_rec = intersection_records.at(exit_idx);
151 auto itr = intersection_records.cbegin();
152
153 // Remove elements indiceated by ordered map with higher index first
154 for (const auto &[v_link, idx] : pt_buckets | std::views::reverse) {
155 // All these portals link against the exit portal
156 assert(v_link == exit_rec.volume());
157
158 auto test_elem = std::next(itr, static_cast<int>(idx));
159 if (exit_rec.intersection().volume_link() != test_elem->volume()) {
160 intersection_records.erase(test_elem);
161 }
162 }
163 }
164
165 // Reset
166 n_eq_intrs = 1u;
167 }
168
169 return overlap_idx;
170}

197template <dindex invalid_value = dindex_invalid, bool check_sorted_trace = true,
198 typename entry_type = std::pair<dindex, dindex>>
199inline bool check_connectivity(
200 std::vector<std::pair<entry_type, entry_type>> trace,
201 dindex start_volume = 0u) {
203 std::stringstream err_stream;
204
206 auto print_err = [](const std::stringstream &stream) {
207 std::cerr << "\n<<<<<<<<<<<<<<< ERROR in connectivity check\n" << std::endl;
208 std::cerr << stream.str() << std::endl;
209 std::cerr << "\n>>>>>>>>>>>>>>>\n" << std::endl;
210 };
211
212 // There must always be portals!
213 if (trace.empty()) {
214 err_stream << "Trace empty!";
215 print_err(err_stream);
216
217 return false;
218 }
219 // Keep record of leftovers
220 std::stringstream record_stream;
221
222 // Where are we on the trace?
223 dindex current_volume = start_volume;
224
225 // If the intersection trace comes from the ray gun/trace intersections
226 // function it should be sorted, which is the stronger constraint
227 using vector_t = decltype(trace);
228 using records_iterator_t = typename vector_t::iterator;
229 using index_t = std::iter_difference_t<records_iterator_t>;
230 std::function<records_iterator_t(index_t)> get_connected_record;
231 if constexpr (check_sorted_trace) {
232 // Get the next record
233 get_connected_record =
234 [&trace, &current_volume](index_t next) -> records_iterator_t {
235 // Make sure that the record contains the volume that is currently
236 // being checked for connectivity
237 if (auto rec = trace.begin() + next;
238 rec != trace.end() &&
239 ((std::get<1>(rec->first) == current_volume) ||
240 (std::get<1>(rec->second) == current_volume))) {
241 return rec;
242 }
243 return trace.end();
244 };
245 } else {
246 // Search for the existence of a fitting record over the entire trace
247 get_connected_record =
248 [&trace, &current_volume](index_t /*next*/) -> records_iterator_t {
249 return std::ranges::find_if(
250 trace, [&](const std::pair<entry_type, entry_type> &rec) -> bool {
251 return (std::get<1>(rec.first) == current_volume) ||
252 (std::get<1>(rec.second) == current_volume);
253 });
254 };
255 }
256
257 // Init chain search
258 index_t i{0};
259 auto record = get_connected_record(i);
260
261 // Check first volume index, which has no partner otherwise
262 if (std::get<1>(record->first) != start_volume) {
263 err_stream << "First record does not start at given initial volume: "
264 << std::get<1>(record->first) << " vs. " << start_volume;
265
266 print_err(err_stream);
267
268 return false;
269 }
270
271 // Walk along the trace as long as a connection is found
272 while (record != trace.end()) {
273 auto first_vol = std::get<1>(record->first);
274 auto second_vol = std::get<1>(record->second);
275
276 record_stream << "On volume: " << current_volume << " and record ("
277 << first_vol << ", " << second_vol << ")";
278
279 // update to next volume
280 current_volume = (current_volume == first_vol ? second_vol : first_vol);
281
282 record_stream << " -> next volume: " << current_volume << std::endl;
283
284 // Don't search this key again -> only one potential key with current
285 // index left
286 if constexpr (!check_sorted_trace) {
287 trace.erase(record);
288 }
289
290 // find connected record for the current volume
291 record = get_connected_record(++i);
292 }
293
294 // There are unconnected elements left (we didn't leave world before
295 // termination)
296 if (current_volume != invalid_value) {
297 err_stream << "Didn't leave world or unconnected elements left in trace:"
298 << "\n\nValid connections that were found:" << std::endl;
299 err_stream << record_stream.str();
300
301 err_stream << "\nPairs left to match:" << std::endl;
302 for (auto j = static_cast<std::size_t>(i); j < trace.size(); ++j) {
303 auto first_vol = std::get<1>(trace[j].first);
304 auto second_vol = std::get<1>(trace[j].second);
305
306 err_stream << "(" << first_vol << ", " << second_vol << ")" << std::endl;
307 }
308
309 print_err(err_stream);
310
311 return false;
312 }
313
314 return true;
315}

337template <concepts::detector detector_t>
338inline auto trace_intersections(
339 const dvector<intersection_record<detector_t>> &intersection_records,
340 dindex start_volume = 0u) {
342 using trace_entry = std::pair<dindex, dindex>;
344 std::vector<std::pair<trace_entry, trace_entry>> portal_trace = {};
346 std::vector<trace_entry> module_trace = {};
348 std::stringstream record_stream;
350 std::stringstream err_stream;
351 bool error_code{true};
352
354 struct record {
355 const intersection_record<detector_t> &entry;
356
359 inline bool is_invalid() const {
360 return entry.intersection().surface().identifier().is_invalid();
361 }
362 inline auto surface_idx() const {
363 return entry.intersection().surface().index();
364 }
365 inline auto surface_volume_idx() const {
366 return entry.intersection().surface().volume();
367 }
368 inline auto &inters() const { return entry.intersection(); }
369 inline auto volume_idx() const { return entry.volume(); }
370 inline auto volume_link() const {
371 return entry.intersection().volume_link();
372 }
373 inline auto dist() const { return entry.intersection().path(); }
374 inline bool is_portal() const {
375 return entry.intersection().surface().is_portal();
376 }
377 inline bool is_sensitive() const {
378 return entry.intersection().surface().is_sensitive();
379 }
380 inline bool is_passive() const {
381 return entry.intersection().surface().is_passive();
382 }
384 };
385
387 auto print_err = [&error_code](const std::stringstream &stream) {
388 std::cerr << "\n<<<<<<<<<<<<<<< ERROR intersection trace\n" << std::endl;
389 std::cerr << stream.str() << std::endl;
390 std::cerr << "\n>>>>>>>>>>>>>>>\n" << std::endl;
391 error_code = false;
392 };
393
394 // No intersections found by ray
395 if (intersection_records.empty()) {
396 err_stream << "No surfaces found in detector!";
397 print_err(err_stream);
398
399 return std::make_tuple(portal_trace, module_trace, error_code);
400 }
401
402 // If there is only one surface in the trace, it must be a portal
403 if (intersection_records.size() == 1u) {
404 const record rec{intersection_records.at(0u)};
405
406 // No exit potal
407 if (!rec.is_portal()) {
408 if (rec.is_invalid()) {
409 err_stream << "No surfaces found in detector!";
410 print_err(err_stream);
411 } else {
412 const std::string sf_type{rec.is_sensitive() ? "sensitive" : "passive"};
413
414 err_stream << "We don't leave the detector by portal!" << std::endl;
415 err_stream << "Only found single " << sf_type
416 << " surface: portal(s) missing!";
417
418 print_err(err_stream);
419 }
420
421 return std::make_tuple(portal_trace, module_trace, error_code);
422 }
423 }
424
425 // Go through recorded intersection (two at a time)
426 dindex current_vol = start_volume;
427 // The first entry is the dummy record that preserves initial track
428 // parameters, skip it
429 const std::size_t start_idx{
430 record{intersection_records.at(0)}.is_invalid() ? 1u : 0u};
431 for (std::size_t rec = start_idx; rec < (intersection_records.size() - 1u);) {
432 const record current_rec = record{intersection_records.at(rec)};
433 const record next_rec = record{intersection_records.at(rec + 1u)};
434
435 // Keep a debug stream
436 record_stream << current_rec.volume_idx() << "\t" << current_rec.inters()
437 << std::endl;
438
439 // If the current record is not a portal, add an entry to the module
440 // trace and continue in more fine-grained steps (sensitive/passive
441 // surfaces do not come in pairs)
442 if (!current_rec.is_portal()) {
443 // Check that the surface was found in the volume it claims to
444 // belong to
445 const bool is_in_volume =
446 (current_rec.volume_idx() == current_rec.surface_volume_idx()) &&
447 (current_rec.surface_volume_idx() == current_vol);
448 if (is_in_volume) {
449 module_trace.emplace_back(current_rec.surface_idx(),
450 current_rec.volume_idx());
451 } else {
452 err_stream << "\n(!!) Surface " << current_rec.surface_idx()
453 << " outside of its volume (Found in: " << current_vol
454 << ", belongs in: " << current_rec.surface_volume_idx()
455 << ")\n";
456
457 err_stream << record_stream.str();
458
459 print_err(err_stream);
460
461 return std::make_tuple(portal_trace, module_trace, error_code);
462 }
463 ++rec;
464 continue;
465 }
466 // If the record is a portal, the current volume switches
467 // the portals in the pair may be unordered, so check both
468 else if (current_vol == current_rec.volume_idx()) {
469 current_vol = current_rec.volume_link();
470 } else if (current_vol == next_rec.volume_idx()) {
471 current_vol = next_rec.volume_link();
472 }
473
474 record_stream << next_rec.volume_idx() << "\t" << next_rec.inters()
475 << std::endl;
476
477 // Check that also the second surface was found in the volume it claims
478 // to belong to
479 const bool is_in_volume =
480 next_rec.volume_idx() == next_rec.surface_volume_idx();
481 // Is this doublet connected via a valid portal intersection?
482 const bool is_valid =
483 (current_rec.inters() == next_rec.inters()) &&
484 (current_rec.volume_idx() == next_rec.volume_link()) &&
485 (next_rec.volume_idx() == current_rec.volume_link());
486 // Is this indeed a portal crossing, i.e. changing volumes?
487 const bool is_self_link = current_rec.volume_idx() == next_rec.volume_idx();
488 // Is the record doublet we picked made up of a portal and a module?
489 const bool is_mixed = (current_rec.is_portal() && !next_rec.is_portal()) ||
490 (next_rec.is_portal() && !current_rec.is_portal());
491
492 if (!is_in_volume) {
493 record_stream << "\n(!!) Surface outside of its volume (Found: "
494 << next_rec.volume_idx()
495 << ", belongs in: " << next_rec.surface_volume_idx() << ")"
496 << std::endl;
497 }
498 if (!is_valid) {
499 record_stream << "\n(!!) Not a valid portal crossing ("
500 << current_rec.volume_idx() << " <-> "
501 << next_rec.volume_idx() << "):\nPortals are not "
502 << "connected, either geometrically or by linking!"
503 << std::endl;
504 }
505 if (is_self_link) {
506 record_stream << "\n(!!) Found portal crossing inside volume ("
507 << current_rec.volume_idx() << ")!" << std::endl;
508 }
509 if (is_mixed) {
510 record_stream << "\n(!!) Portal crossing involves module surface ("
511 << current_rec.volume_idx() << " <-> "
512 << next_rec.volume_idx() << ")! The second surface in"
513 << " this portal crossing is not a portal!" << std::endl;
514 }
515 if (is_in_volume && is_valid && !is_self_link && !is_mixed) {
516 // Insert into portal trace
517 trace_entry lower{current_rec.surface_idx(), current_rec.volume_idx()};
518 trace_entry upper{next_rec.surface_idx(), next_rec.volume_idx()};
519 portal_trace.emplace_back(lower, upper);
520 }
521 // Something went wrong
522 else {
523 // Print search log
524 err_stream << "\nError in portal matching:\n" << std::endl;
525 err_stream << "volume id\t(intersection info)" << std::endl;
526
527 err_stream << record_stream.str() << std::endl;
528
529 err_stream << "-----\nINFO: Ray terminated at portal x-ing "
530 << (rec + 1) / 2 << ":\n"
531 << current_rec.inters() << " <-> " << next_rec.inters()
532 << std::endl;
533
534 const record rec_front{record{intersection_records.front()}.is_invalid()
535 ? intersection_records[1u]
536 : intersection_records.front()};
537 const record rec_back{intersection_records.back()};
538 err_stream << "Start volume : " << start_volume << std::endl;
539 err_stream << "- first recorded intersection: (sf id:"
540 << rec_front.surface_idx() << ", dist:" << rec_front.dist()
541 << ")," << std::endl;
542 err_stream << "- last recorded intersection: (sf id:"
543 << rec_back.surface_idx() << ", dist:" << rec_back.dist()
544 << "),";
545
546 print_err(err_stream);
547
548 return std::make_tuple(portal_trace, module_trace, error_code);
549 }
550
551 // Advance to inspect next pair
552 rec += 2u;
553 }
554
555 // Look at the last entry, which is a single portal
556 if (const record rec_back{intersection_records.back()};
557 !rec_back.is_portal()) {
558 err_stream << "We don't leave the detector by portal!";
559 print_err(err_stream);
560 } else {
561 trace_entry lower(rec_back.surface_idx(), rec_back.volume_idx());
562 trace_entry upper(rec_back.surface_idx(), rec_back.volume_link());
563 portal_trace.emplace_back(lower, upper);
564 }
565
566 return std::make_tuple(portal_trace, module_trace, error_code);
567}

585template <dindex invalid_value = dindex_invalid,
586 typename entry_type = std::pair<dindex, dindex>>
587inline auto build_adjacency(
588 const std::vector<std::pair<entry_type, entry_type>> &portal_trace,
589 const std::vector<entry_type> &module_trace,
590 std::map<dindex, std::map<dindex, dindex>> &adj_list,
591 std::unordered_set<dindex> &obj_hashes) {
592 // Every module that was recorded adds a link to the mother volume
593 for (const auto &record : module_trace) {
594 const auto sf_index = std::get<0>(record);
595 const auto vol_index = std::get<1>(record);
596 // Check whether we have seen this module in this volume before
597 if (obj_hashes.find(sf_index) == obj_hashes.end()) {
598 adj_list[vol_index][vol_index]++;
599 obj_hashes.insert(sf_index);
600 }
601 }
602
603 // Portal in first volume links to second volume in the record
604 for (const auto &record : portal_trace) {
605 const auto pt_index_1 = std::get<0>(record.first);
606 const auto vol_index_1 = std::get<1>(record.first);
607 const auto pt_index_2 = std::get<0>(record.second);
608 const auto vol_index_2 = std::get<1>(record.second);
609
610 if (obj_hashes.find(pt_index_1) == obj_hashes.end()) {
611 adj_list[vol_index_1][vol_index_2]++;
612 obj_hashes.insert(pt_index_1);
613 }
614 // Assume the return link for now (filter out portal that leaves world)
615 if (vol_index_2 != invalid_value &&
616 obj_hashes.find(pt_index_2) == obj_hashes.end()) {
617 adj_list[vol_index_2][vol_index_1]++;
618 obj_hashes.insert(pt_index_2);
619 }
620 }
621
622 return adj_list;
623}

641template <dindex invalid_value = dindex_invalid,
642 typename entry_type = std::pair<dindex, dindex>>
643inline auto build_adjacency(
644 const std::vector<std::pair<entry_type, entry_type>> &portal_trace,
645 const std::vector<entry_type> &module_trace, dvector<dindex> &adj_matrix,
646 std::unordered_set<dindex> &obj_hashes) {
647 const auto dim = static_cast<dindex>(math::sqrt(adj_matrix.size()));
648
649 // Every module that was recorded adds a link to the mother volume
650 for (const auto &record : module_trace) {
651 const auto sf_index = std::get<0>(record);
652 const auto vol_index = std::get<1>(record);
653 // Check whether we have seen this module in this volume before
654 if (obj_hashes.find(sf_index) == obj_hashes.end()) {
655 adj_matrix[dim * vol_index + vol_index]++;
656 obj_hashes.insert(sf_index);
657 }
658 }
659
660 // Portal in first volume links to second volume in the record
661 for (const auto &record : portal_trace) {
662 const auto pt_index_1 = std::get<0>(record.first);
663 const auto vol_index_1 = std::get<1>(record.first);
664 const auto pt_index_2 = std::get<0>(record.second);
665 const auto vol_index_2 = std::get<1>(record.second);
666
667 if (obj_hashes.find(pt_index_1) == obj_hashes.end()) {
668 dindex mat_elem_vol1{dindex_invalid};
669 // Assume the return link for now (filtering out portals that leave world)
670 if (vol_index_2 != invalid_value) {
671 mat_elem_vol1 = dim * vol_index_1 + vol_index_2;
672
673 if (obj_hashes.find(pt_index_2) == obj_hashes.end()) {
674 adj_matrix[dim * vol_index_2 + vol_index_1]++;
675 obj_hashes.insert(pt_index_2);
676 }
677 } else {
678 mat_elem_vol1 = dim * vol_index_1 + dim - 1;
679 }
680 adj_matrix[mat_elem_vol1]++;
681 obj_hashes.insert(pt_index_1);
682 }
683 }
684
685 return adj_matrix;
686}

696template <concepts::detector detector_t>
697inline bool check_trace(
698 const dvector<intersection_record<detector_t>> &intersection_trace,
699 const dindex start_index, dvector<dindex> &adj_mat_scan,
700 std::unordered_set<dindex> &obj_hashes) {
701 using nav_link_t = typename detector_t::surface_type::navigation_link;
702 static constexpr auto leaving_world{
703 detray::detail::invalid_value<nav_link_t>()};
704
705 // Create a trace of the volume indices that were encountered
706 // and check that portal intersections are connected
707 auto [portal_trace, surface_trace, err_code] =
708 detector_scanner::trace_intersections(intersection_trace, start_index);
709
710 DETRAY_DEBUG_HOST("portal_trace " << portal_trace.size() << ", surface_trace "
711 << surface_trace.size());
712
713 // Is the succession of volumes consistent ?
714 err_code = err_code &&
715 detector_scanner::check_connectivity<leaving_world>(portal_trace);
716
717 if (!adj_mat_scan.empty()) {
718 // Build an adjacency matrix from this trace that can be checked
719 // against the geometry hash (see 'track_geometry_changes')
720 detector_scanner::build_adjacency<leaving_world>(
721 portal_trace, surface_trace, adj_mat_scan, obj_hashes);
722 }
723
724 return err_code;
725}

741template <concepts::detector detector_t, typename trajectory_t>
742inline void display_error(
743 const typename detector_t::geometry_context gctx, const detector_t &det,
744 const typename detector_t::name_map &vol_names,
745 const std::string &test_name, const trajectory_t &test_track,
746 const dvector<intersection_record<detector_t>> &truth_trace,
747 const detray::svgtools::styling::style &svg_style,

813inline std::string print_adj(const dvector<dindex> &adjacency_matrix) {
814 std::size_t dim = static_cast<dindex>(math::sqrt(adjacency_matrix.size()));
815 std::stringstream out_stream{};
816
817 for (std::size_t i = 0u; i < dim - 1; ++i) {
818 out_stream << "[>>] Node with index " << i << std::endl;
819 out_stream << " -> edges: " << std::endl;
820 for (std::size_t j = 0u; j < dim; ++j) {
821 const auto degr = adjacency_matrix[dim * i + j];
822 if (degr == 0) {
823 continue;
824 }
825 std::string n_occur =
826 degr > 1 ? "\t\t\t\t(" + std::to_string(degr) + "x)" : "";
827
828 // Edge that leads out of the detector world
829 if (j == dim - 1 && degr != 0) {
830 out_stream << " -> leaving world " + n_occur << std::endl;
831 } else {
832 out_stream << " -> " << std::to_string(j) + "\t" + n_occur
833 << std::endl;
834 }
835 }
836 }
837
838 return out_stream.str();
839}
