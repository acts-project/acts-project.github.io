426 template <class Point>
427 T interpolate(const Point& point) const
428 requires(Concepts::interpolatable<T, Point, std::array<double, DIM>,
429 std::array<double, DIM>>)
430 {
431 // there are 2^DIM corner points used during the interpolation
432 constexpr std::size_t nCorners = 1 << DIM;
433
434 // construct vector of pairs of adjacent bin centers and values
435 std::array<value_type, nCorners> neighbors{};
436
437 // get local indices for current bin
438 // value of bin is interpreted as being the field value at its lower left
439 // corner
440 const auto& llIndices = localBinsFromPosition(point);
441
442 // get global indices for all surrounding corner points
443 const auto& closestIndices = rawClosestPointsIndices(llIndices);
444
445 // get values on grid points
446 std::size_t i = 0;
447 for (std::size_t index : closestIndices) {
448 neighbors.at(i++) = at(index);
449 }
450
451 return Acts::interpolate(point, lowerLeftBinEdge(llIndices),
452 upperRightBinEdge(llIndices), neighbors);
453 }

521 std::size_t size(bool fullCounter = true) const {
522 index_t nBinsArray = numLocalBins();
523 std::size_t current_size = 1;
524 // add under-and overflow bins for each axis and multiply all bins
525 if (fullCounter) {
526 for (const auto& value : nBinsArray) {
527 current_size *= value + 2;
528 }
529 }
530 // ignore under-and overflow bins for each axis and multiply all bins
531 else {
532 for (const auto& value : nBinsArray) {
533 current_size *= value;
534 }
535 }
536 return current_size;
537 }
