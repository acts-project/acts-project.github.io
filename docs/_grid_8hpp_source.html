434 template <class Point>
435 T interpolate(const Point& point) const {
436 // get local indices for current bin
437 // value of bin is interpreted as being the field value at its lower left
438 // corner
439 const auto llIndices = localBinsFromPosition(point);
440
441 // get global indices for all surrounding corner points
442 const auto closestIndices =
443 m_axes.getNeighborHoodIndices(llIndices, {0, 1});
444
445 // there are 2^DIM corner points used during the interpolation
446 constexpr std::size_t nCorners = 1 << DIM;
447
448 // construct vector of pairs of adjacent bin centers and values
449 std::array<value_type, nCorners> neighbors{};
450
451 // get values on grid points
452 std::size_t i = 0;
453 for (const auto index : closestIndices) {
454 neighbors.at(i) = at(index);
455 ++i;
456 }
457
458 return Acts::interpolate(point, m_axes.getLowerLeftBinEdge(llIndices),
459 m_axes.getUpperRightBinEdge(llIndices), neighbors);
460 }

678 return b.visit([&]<typename AxisTypeB>(
679 const AxisTypeB& axisB) -> std::unique_ptr<IGrid> {

681 return std::make_unique<GridType>(axisA, axisB);
