159 BinUtility& operator+=(const BinUtility& gbu) {
160 const std::vector<BinningData>& bData = gbu.binningData();
161
162 m_transform = m_transform * gbu.transform();
163 m_itransform = m_transform.inverse();
164 if (m_binningData.size() + bData.size() > 3) {
165 throw std::runtime_error{"BinUtility does not support dim > 3"};
166 }
167 m_binningData.insert(m_binningData.end(), bData.begin(), bData.end());
168 return (*this);
169 }
170
174 bool operator==(const BinUtility& other) const {
175 return (m_transform.isApprox(other.m_transform) &&
176 m_binningData == other.binningData());
177 }

194 std::array<std::size_t, 3> binTriple(const Vector3& position) const {
196 const Vector3 bPosition = m_itransform * position;
197 // get the dimension
198 std::size_t mdim = m_binningData.size();
200 std::size_t bin0 = m_binningData[0].searchGlobal(bPosition);
201 std::size_t bin1 = mdim > 1 ? m_binningData[1].searchGlobal(bPosition) : 0;
202 std::size_t bin2 = mdim > 2 ? m_binningData[2].searchGlobal(bPosition) : 0;
204 return {{bin0, bin1, bin2}};
205 }

336 std::size_t serialize(const std::array<std::size_t, 3>& bin) const {
337 std::size_t serializedBin = bin[0];
338 if (m_binningData.size() == 2) {
339 serializedBin += bin[1] * m_binningData[0].bins();
340 } else if (m_binningData.size() == 3) {
341 serializedBin +=
342 (bin[1] * m_binningData[0].bins() * bin[2] * m_binningData[1].bins());
343 }
344 return serializedBin;
345 }
