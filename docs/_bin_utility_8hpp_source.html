125 BinUtility& operator+=(const BinUtility& gbu) {
126 const std::vector<BinningData>& bData = gbu.binningData();
127
128 m_transform = m_transform * gbu.transform();
129 m_itransform = m_transform.inverse();
130 if (m_binningData.size() + bData.size() > 3) {
131 throw std::runtime_error{"BinUtility does not support dim > 3"};
132 }
133 m_binningData.insert(m_binningData.end(), bData.begin(), bData.end());
134 return (*this);
135 }
136
140 bool operator==(const BinUtility& other) const {
141 return (m_transform.isApprox(other.m_transform) &&
142 m_binningData == other.binningData());
143 }

147 const std::vector<BinningData>& binningData() const { return m_binningData; }
148
151 std::size_t bins() const { return bins(0) * bins(1) * bins(2); }
152
160 std::array<std::size_t, 3> binTriple(const Vector3& position) const {
162 const Vector3 bPosition = m_itransform * position;
163 // get the dimension
164 std::size_t mdim = m_binningData.size();
166 std::size_t bin0 = m_binningData[0].searchGlobal(bPosition);
167 std::size_t bin1 = mdim > 1 ? m_binningData[1].searchGlobal(bPosition) : 0;
168 std::size_t bin2 = mdim > 2 ? m_binningData[2].searchGlobal(bPosition) : 0;
170 return {{bin0, bin1, bin2}};
171 }
172
179 std::size_t bin(const Vector3& position, std::size_t ba = 0) const {
180 if (ba >= m_binningData.size()) {
181 return 0;
182 }
183 std::size_t bEval = m_binningData[ba].searchGlobal(m_itransform * position);
184 return bEval;
185 }
186
198 std::size_t bin(const Vector2& lposition, std::size_t ba = 0) const {
199 if (ba >= m_binningData.size()) {
200 return 0;
201 }
202 return m_binningData[ba].searchLocal(lposition);
203 }

216 std::size_t bin(float value, std::size_t ba = 0) const {

232 int nextDirection(const Vector3& position, const Vector3& direction,
233 std::size_t ba = 0) const {

235 return 0;

238 }
239
244 bool inside(const Vector3& position) const {
246 const Vector3& bPosition = m_itransform * position;
247 return std::ranges::all_of(m_binningData, [&](const auto& bData) {
248 return bData.inside(bPosition);
249 });
250 }
251
254 std::size_t dimensions() const { return m_binningData.size(); }

261 std::size_t max(std::size_t ba = 0) const {
262 if (ba >= m_binningData.size()) {
263 return 0;

266 }

273 std::size_t bins(std::size_t ba) const {

279
283 const Transform3& transform() const { return m_transform; }
284
290 AxisDirection binningValue(std::size_t ba = 0) const {

302 std::size_t serialize(const std::array<std::size_t, 3>& bin) const {
303 std::size_t serializedBin = bin[0];
304 if (m_binningData.size() == 2) {
305 serializedBin += bin[1] * m_binningData[0].bins();
306 } else if (m_binningData.size() == 3) {
307 serializedBin +=
308 (bin[1] * m_binningData[0].bins() * bin[2] * m_binningData[1].bins());
309 }
310 return serializedBin;
311 }

319 std::ostream& toStream(std::ostream& sl,
320 const std::string& indent = "") const {
321 sl << indent << "BinUtility for " << m_binningData.size()
322 << "- dimensional array:" << std::endl;
323 for (auto [ibd, bd] : enumerate(m_binningData)) {
324 sl << indent << "dimension : " << ibd << std::endl;
325 sl << bd.toString(indent) << std::endl;
326 }
327 return sl;
328 }

335 std::string toString(const std::string& indent = "") const {
