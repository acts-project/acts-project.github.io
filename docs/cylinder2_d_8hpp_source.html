86 template <concepts::algebra algebra_t>
87 DETRAY_HOST_DEVICE constexpr result_type<dbool<algebra_t>> check_boundaries(
88 const bounds_type<dscalar<algebra_t>> &bounds,
89 const dtransform3D<algebra_t> &trf, const dpoint3D<algebra_t> &glob_p,
90 const dscalar<algebra_t> tol =
91 std::numeric_limits<dscalar<algebra_t>>::epsilon(),
92 const dscalar<algebra_t> edge_tol = 0.f) const {
93 // Rotate to local cartesian
94 const auto loc_p = cartesian3D<algebra_t>::global_to_local(trf, glob_p, {});
95
96 // Only need the z-position for the check
97 return check_boundaries(
98 bounds,
99 dpoint2D<algebra_t>{static_cast<dscalar<algebra_t>>(0.f), loc_p[2]},
100 tol, edge_tol);
101 }

117 template <concepts::scalar scalar_t, concepts::point point_t>
118 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
119 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
120 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
121 const scalar_t edge_tol = 0.f) const {
122 auto inside_mask{((bounds[e_lower_z] - tol) <= loc_p[1] &&
123 loc_p[1] <= (bounds[e_upper_z] + tol))};
124
125 decltype(inside_mask) inside_edge{false};
126 if (detail::any_of(edge_tol > 0.f)) {
127 const scalar_t full_tol{tol + edge_tol};
128 inside_edge = ((bounds[e_lower_z] - full_tol) <= loc_p[1] &&
129 loc_p[1] <= (bounds[e_upper_z] + full_tol));
130 }
131
132 return result_type<decltype(inside_mask)>{inside_mask, inside_edge};
133 }

152 template <concepts::scalar scalar_t>
153 DETRAY_HOST_DEVICE constexpr scalar_t area(
154 const bounds_type<scalar_t> &bounds) const {
155 return 2.f * constant<scalar_t>::pi * bounds[e_r] *
156 (bounds[e_upper_z] - bounds[e_lower_z]);
157 }

165 template <concepts::scalar scalar_t>
166 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
167 const bounds_type<scalar_t> &bounds,
168 const bounds_type<scalar_t> &o_bounds) const {
169 bounds_type<scalar_t> new_bounds{};
170
171 new_bounds[e_r] = math::max(bounds[e_r], o_bounds[e_r]);
172 new_bounds[e_lower_z] = math::min(bounds[e_lower_z], o_bounds[e_lower_z]);
173 new_bounds[e_upper_z] = math::max(bounds[e_upper_z], o_bounds[e_upper_z]);
174
175 return new_bounds;
176 }

187 template <concepts::algebra algebra_t>
188 DETRAY_HOST_DEVICE inline darray<dscalar<algebra_t>, 6> local_min_bounds(
189 const bounds_type<dscalar<algebra_t>> &bounds,
190 const dscalar<algebra_t> env =
191 std::numeric_limits<dscalar<algebra_t>>::epsilon()) const {
192 assert(env > 0.f);
193
194 const dscalar<algebra_t> xy_bound{bounds[e_r] + env};
195 return {-xy_bound, -xy_bound, bounds[e_lower_z] - env,
196 xy_bound, xy_bound, bounds[e_upper_z] + env};
197 }

201 template <concepts::algebra algebra_t>
202 DETRAY_HOST_DEVICE dpoint3D<algebra_t> centroid(
203 const bounds_type<dscalar<algebra_t>> &bounds) const {
204 return {static_cast<dscalar<algebra_t>>(0.f),
205 static_cast<dscalar<algebra_t>>(0.f),
206 0.5f * (bounds[e_lower_z] + bounds[e_upper_z])};
207 }

230 template <concepts::scalar scalar_t>
231 DETRAY_HOST constexpr bool check_consistency(
232 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
233 constexpr auto tol{10.f * std::numeric_limits<scalar_t>::epsilon()};
234
235 if (bounds[e_r] < tol) {
236 os << "DETRAY ERROR (HOST): Radius must be in the range (0, "
237 "numeric_max)"
238 << std::endl;
239 return false;
240 }
241 if (bounds[e_lower_z] >= bounds[e_upper_z] ||
242 math::fabs(bounds[e_lower_z] - bounds[e_upper_z]) < tol) {
243 os << "DETRAY ERROR (HOST): Neg. half length must be smaller than "
244 "pos. half "
245 "length.";
246 return false;
247 }
248
249 return true;
250 }
