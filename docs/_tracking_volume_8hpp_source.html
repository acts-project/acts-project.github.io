190 template <SurfaceVisitor visitor_t>
191 void visitSurfaces(visitor_t&& visitor, bool restrictToSensitives) const {
192 if (restrictToSensitives) {
193 apply([&visitor](const Surface& surface) {
194 if (surface.geometryId().sensitive() == 0) {
195 return;
196 }
197 visitor(&surface);
198 });
199
200 } else {
201 apply(overloaded{
202 [&visitor](const Surface& surface) { visitor(&surface); },
203 [&visitor](const Portal& portal) { visitor(&portal.surface()); },
204 [&visitor](const BoundarySurface& bs) {
205 visitor(&bs.surfaceRepresentation());
206 }});
207 }
208 }
