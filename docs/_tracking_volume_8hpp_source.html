204 template <SurfaceVisitor visitor_t>
205 void visitSurfaces(visitor_t&& visitor, bool restrictToSensitives) const {
206 if (restrictToSensitives) {
207 apply([&visitor](const Surface& surface) {
208 if (surface.geometryId().sensitive() == 0) {
209 return;
210 }
211 visitor(&surface);
212 });
213
214 } else {
215 apply(overloaded{
216 [&visitor](const Surface& surface) { visitor(&surface); },
217 [&visitor](const Portal& portal) { visitor(&portal.surface()); },
218 [&visitor](const BoundarySurface& bs) {
219 visitor(&bs.surfaceRepresentation());
220 }});
221 }
222 }
223
233 template <SurfaceVisitor visitor_t>
234 void visitSurfaces(visitor_t&& visitor) const {
235 visitSurfaces(std::forward<visitor_t>(visitor), true);
236 }
