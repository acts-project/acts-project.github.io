150template <typename grid_type>
151nlohmann::json toJson(const grid_type& grid) {
152 nlohmann::json jGrid;
153
154 auto axes = grid.axes();
155 nlohmann::json jAxes;
156 for (unsigned int ia = 0u; ia < grid_type::DIM; ++ia) {
157 auto jAxis = AxisJsonConverter::toJson(*axes[ia]);
158 jAxes.push_back(jAxis);
159 }
160 jGrid["axes"] = jAxes;
161
162 nlohmann::json jData;
163 // 1D connections
164 if constexpr (grid_type::DIM == 1u) {
165 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
166 typename grid_type::index_t lbin;
167 lbin[0u] = ib0;
168 jData.push_back(std::tie(lbin, grid.atLocalBins(lbin)));
169 }
170 }
171 // 2D connections
172 if constexpr (grid_type::DIM == 2u) {
173 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
174 for (unsigned int ib1 = 1u; ib1 <= axes[1u]->getNBins(); ++ib1) {
175 typename grid_type::index_t lbin;
176 lbin[0u] = ib0;
177 lbin[1u] = ib1;
178 jData.push_back(std::tie(lbin, grid.atLocalBins(lbin)));
179 }
180 }
181 }
182 jGrid["data"] = jData;
183
184 return jGrid;
185}

197template <typename grid_type>
198nlohmann::json toJsonDetray(const grid_type& grid, bool swapAxis = false) {
199 nlohmann::json jGrid;
200 // Get the grid axes & potentially swap them
201 auto axes = grid.axes();
202 if (swapAxis && grid_type::DIM == 2u) {
203 std::swap(axes[0u], axes[1u]);
204 }
205
206 nlohmann::json jAxes;
207
208 // Fill the axes in the order they are
209 for (unsigned int ia = 0u; ia < grid_type::DIM; ++ia) {
210 auto jAxis = AxisJsonConverter::toJsonDetray(*axes[ia]);
211 jAxis["label"] = ia;
212 jAxes.push_back(jAxis);
213 }
214 jGrid["axes"] = jAxes;
215
216 nlohmann::json jData;
217 // 1D connections
218 if constexpr (grid_type::DIM == 1u) {
219 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
220 // Lookup bin
221 typename grid_type::index_t lbin;
222 lbin[0u] = ib0;
223 nlohmann::json jBin;
224 jBin["content"] = grid.atLocalBins(lbin);
225 // Corrected bin for detray
226 lbin[0u] = ib0 - 1u;
227 jBin["loc_index"] = lbin;
228 jData.push_back(jBin);
229 }
230 }
231
232 // 2D connections
233 if constexpr (grid_type::DIM == 2u) {
234 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
235 for (unsigned int ib1 = 1u; ib1 <= axes[1u]->getNBins(); ++ib1) {
236 typename grid_type::index_t lbin;
237 // Lookup bin - respect swap (if it happened) for the lookup
238 lbin[0u] = swapAxis ? ib1 : ib0;
239 lbin[1u] = swapAxis ? ib0 : ib1;
240 nlohmann::json jBin;
241 jBin["content"] = grid.atLocalBins(lbin);
242 // Corrected bin for detray
243 lbin[0u] = ib0 - 1u;
244 lbin[1u] = ib1 - 1u;
245 jBin["loc_index"] = lbin;
246 jData.push_back(jBin);
247 }
248 }
249 }
250 jGrid["bins"] = jData;
251
252 return jGrid;
253}

266template <typename axis_generator_type,
267 typename value_type = std::vector<std::size_t>>
268auto fromJson(const nlohmann::json& jGrid,
269 const axis_generator_type& aGenerator) {
270 // Generate the grid
271 using GridType = typename axis_generator_type::template grid_type<value_type>;
272 GridType grid(aGenerator());
273 nlohmann::json jData = jGrid["data"];
274 // Index filling
275 if constexpr (GridType::DIM == 1u) {
276 for (const auto& jd : jData) {
277 std::array<std::size_t, 1u> lbin = jd[0u];
278 if (!jd[1u].is_null()) {
279 grid.atLocalBins(lbin) = jd[1u].get<value_type>();
280 }
281 }
282 }
283 if constexpr (GridType::DIM == 2u) {
284 for (const auto& jd : jData) {
285 std::array<std::size_t, 2u> lbin = jd[0u];
286 if (!jd[1u].is_null()) {
287 grid.atLocalBins(lbin) = jd[1u].get<value_type>();
288 }
289 }
290 }
291 return grid;
292}
