143template <typename grid_type>
144nlohmann::json toJson(const grid_type& grid) {
145 nlohmann::json jGrid;
146
147 auto axes = grid.axes();
148 nlohmann::json jAxes;
149 for (unsigned int ia = 0u; ia < grid_type::DIM; ++ia) {
150 auto jAxis = AxisJsonConverter::toJson(*axes[ia]);
151 jAxes.push_back(jAxis);
152 }
153 jGrid["axes"] = jAxes;
154
155 nlohmann::json jData;
156 // 1D connections
157 if constexpr (grid_type::DIM == 1u) {
158 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
159 typename grid_type::index_t lbin;
160 lbin[0u] = ib0;
161 jData.push_back(std::tie(lbin, grid.atLocalBins(lbin)));
162 }
163 }
164 // 2D connections
165 if constexpr (grid_type::DIM == 2u) {
166 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
167 for (unsigned int ib1 = 1u; ib1 <= axes[1u]->getNBins(); ++ib1) {
168 typename grid_type::index_t lbin;
169 lbin[0u] = ib0;
170 lbin[1u] = ib1;
171 jData.push_back(std::tie(lbin, grid.atLocalBins(lbin)));
172 }
173 }
174 }
175 jGrid["data"] = jData;
176
177 return jGrid;
178}

191template <typename axis_generator_type,
192 typename value_type = std::vector<std::size_t>>
193auto fromJson(const nlohmann::json& jGrid,
194 const axis_generator_type& aGenerator) {
195 // Generate the grid
196 using GridType = typename axis_generator_type::template grid_type<value_type>;
197 GridType grid(aGenerator());
198 nlohmann::json jData = jGrid["data"];
199 // Index filling
200 if constexpr (GridType::DIM == 1u) {
201 for (const auto& jd : jData) {
202 std::array<std::size_t, 1u> lbin = jd[0u];
203 if (!jd[1u].is_null()) {
204 grid.atLocalBins(lbin) = jd[1u].get<value_type>();
205 }
206 }
207 }
208 if constexpr (GridType::DIM == 2u) {
209 for (const auto& jd : jData) {
210 std::array<std::size_t, 2u> lbin = jd[0u];
211 if (!jd[1u].is_null()) {
212 grid.atLocalBins(lbin) = jd[1u].get<value_type>();
213 }
214 }
215 }
216 return grid;
217}
