24template <concepts::algebra algebra_t>
25struct scattering_helper {
26 public:
28 using scalar_type = dscalar<algebra_t>;
30 using vector3_type = dvector3D<algebra_t>;
31
38 template <typename generator_t>
39 DETRAY_HOST inline vector3_type operator()(const vector3_type& dir,
40 const scalar_type angle,
41 generator_t& generator) const {
42 // Generate theta and phi for random scattering
43 const scalar_type r_theta{
44 angle == 0.f
45 ? 0.f
46 : std::normal_distribution<scalar_type>(0.f, angle)(generator)};
47
48 const scalar_type r_phi{std::uniform_real_distribution<scalar_type>(
49 -constant<scalar_type>::pi, constant<scalar_type>::pi)(generator)};
50
51 // xaxis of curvilinear plane
52 const vector3_type u =
53 unit_vectors<vector3_type>().make_curvilinear_unit_u(dir);
54
55 vector3_type new_dir = axis_rotation<algebra_t>(u, r_theta)(dir);
56 return axis_rotation<algebra_t>(dir, r_phi)(new_dir);
57 }
58};
