82static constexpr PdgParticle makeNucleusGroundState(PdgParticle pdg) {
83 if (!isNucleus(pdg)) {
84 throw std::invalid_argument("PDG must represent a nucleus");
85 }
86 // set isomer level to zero
87 const auto value = static_cast<std::int32_t>(pdg);
88 return static_cast<PdgParticle>((value / 10) * 10);
89}

98static constexpr std::pair<std::int32_t, std::int32_t> extractNucleusZandA(
99 PdgParticle pdg) {
100 if (!isNucleus(pdg)) {
101 throw std::invalid_argument("PDG must represent a nucleus");
102 }
103 const auto value = static_cast<std::int32_t>(pdg);
104 // proton number respects the charge
105 int Z = (value / 10000) % 1000;
106 // atomic number is always positive
107 int A = std::abs((value / 10) % 1000);
108 return std::make_pair(Z, A);
109}
