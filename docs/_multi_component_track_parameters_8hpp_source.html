69 static MultiComponentBoundTrackParameters createCurvilinear(
70 const GeometryContext& geoCtx,
71 const std::vector<ConstructionTuple>& curvi,
72 ParticleHypothesis particleHypothesis) {
73 // Construct and average surface
74 Vector3 avgPos = Vector3::Zero();
75 Vector3 avgDir = Vector3::Zero();
76 for (const auto& [w, pos4, dir, qop, cov] : curvi) {
77 avgPos += w * pos4.template segment<3>(0);
78 avgDir += w * dir;
79 }
80
81 std::shared_ptr<PlaneSurface> s =
82 CurvilinearSurface(avgPos, avgDir).planeSurface();
83
84 std::vector<Component> bound;
85 bound.reserve(curvi.size());
86
87 // Project the position onto the surface, keep everything else as is
88 for (const auto& [w, pos4, dir, qop, cov] : curvi) {
89 const Intersection3D closestIntersection =
90 s->intersect(geoCtx, pos4.template segment<3>(eFreePos0), dir,
91 BoundaryTolerance::Infinite())
92 .closest();
93 const Vector3& newPos = closestIntersection.position();
94
95 ParametersVector bv =
96 transformFreeToCurvilinearParameters(pos4[eTime], dir, qop);
97
98 // Because of the projection this should never fail
99 bv.template segment<2>(eBoundLoc0) =
100 *(s->globalToLocal(geoCtx, newPos, dir));
101
102 bound.emplace_back(w, bv, cov);
103 }
104
105 return MultiComponentBoundTrackParameters(
106 s, bound,
107 [](const Component& cmp) {
108 return std::tie(std::get<0>(cmp), std::get<1>(cmp),
109 *std::get<2>(cmp));
110 },
111 particleHypothesis);
112 }

149 template <std::ranges::range component_range_t, typename projector_t>
150 MultiComponentBoundTrackParameters(std::shared_ptr<const Surface> surface,
151 const component_range_t& cmps,
152 const projector_t& proj,
153 ParticleHypothesis particleHypothesis)
154 requires detail::ComponentRangeAndProjectorWithoutCovarianceConcept<
155 component_range_t, projector_t>
156 : m_surface(std::move(surface)),
157 m_particleHypothesis(particleHypothesis) {
158 if (cmps.begin() == cmps.end()) {
159 throw std::invalid_argument(
160 "Cannot construct MultiComponentBoundTrackParameters without "
161 "components");
162 }
163
164 for (const auto& cmp : cmps) {
165 const auto& [weight, parameters] = proj(cmp);
166 m_weights.push_back(weight);
167 m_parameters.push_back(parameters);
168 }
169 }

176 template <std::ranges::range component_range_t, typename projector_t>
177 MultiComponentBoundTrackParameters(std::shared_ptr<const Surface> surface,
178 const component_range_t& cmps,
179 const projector_t& proj,
180 ParticleHypothesis particleHypothesis)
181 requires detail::ComponentRangeAndProjectorWithCovarianceConcept<
182 component_range_t, projector_t>
183 : m_surface(std::move(surface)),
184 m_particleHypothesis(particleHypothesis),
185 m_hasCovariance(true) {
186 if (cmps.begin() == cmps.end()) {
187 throw std::invalid_argument(
188 "Cannot construct MultiComponentBoundTrackParameters without "
189 "components");
190 }
191
192 for (const auto& cmp : cmps) {
193 const auto& [weight, parameters, covariance] = proj(cmp);
194 m_weights.push_back(weight);
195 m_parameters.push_back(parameters);
196 m_covariances.push_back(covariance);
197 }
198 }

265 std::vector<Component> toComponents() const {
266 std::vector<Component> cmps;
267 cmps.reserve(size());
268 for (std::size_t i = 0; i < size(); ++i) {
