96template <concepts::detector detector_t, typename mask_shape_t = line_square>
97class wire_layer_generator final
98 : public surface_factory_interface<detector_t> {
99 using algebra_t = typename detector_t::algebra_type;
100 using scalar_t = dscalar<algebra_t>;
101 using transform3_t = dtransform3D<algebra_t>;
102 using point3_t = dpoint3D<algebra_t>;
103 using vector3_t = dvector3D<algebra_t>;
104
105 public:
108 DETRAY_HOST
109 explicit wire_layer_generator(
110 const wire_layer_generator_config<scalar_t> &cfg)
111 : m_cfg{cfg},
112 m_layer_central_rad{m_cfg.inner_layer_radius() + m_cfg.cell_size()} {}
113
115 DETRAY_HOST
116 auto size() const -> dindex override {
117 const scalar_t circumference{2.f * constant<scalar_t>::pi *
118 m_layer_central_rad};
119 return static_cast<dindex>(
120 std::floor(circumference / 2.f * m_cfg.cell_size()));
121 }
122
125 DETRAY_HOST
126 void clear() override { /*Do nothing*/ };
127 DETRAY_HOST
128 void push_back(
129 surface_data<detector_t> && /*unused*/) override { /*Do nothing*/ }
130 DETRAY_HOST
132 auto push_back(std::vector<surface_data<detector_t>> && /*unused*/)
133 -> void override { /*Do nothing*/ }
134
135
143 DETRAY_HOST
145 auto operator()(typename detector_t::volume_type &volume,
146 typename detector_t::surface_lookup_container &surfaces,
147 typename detector_t::transform_container &transforms,
148 typename detector_t::mask_container &masks,
149 typename detector_t::geometry_context ctx = {})
150 -> dindex_range override {
151 DETRAY_VERBOSE_HOST("Generate modules for wire chamber layer...");
152 DETRAY_VERBOSE_HOST("-> Generate " << size() << " surfaces");
153
154 using surface_t = typename detector_t::surface_type;
155 using nav_link_t = typename surface_t::navigation_link;
156 using mask_link_t = typename surface_t::mask_link;
157 using material_link_t = typename surface_t::material_link;
158
159 // Volume that this layer will be added to
160 auto volume_idx{volume.index()};
161
162 const dindex surfaces_offset{static_cast<dindex>(surfaces.size())};
163 constexpr auto invalid_src_link{detail::invalid_value<std::uint64_t>()};
164
165 // The type id of the surface mask shape (drift cell or straw tube)
166 constexpr auto mask_id{
167 types::id<typename detector_t::masks, mask<mask_shape_t, algebra_t>>};
168 // Modules link back to mother volume in navigation
169 const auto mask_volume_link{static_cast<nav_link_t>(volume_idx)};
170
171 // Cell center positions
172 detray::dvector<point3_t> cell_centers{};
173 // Distance between wires in phi
174 const scalar_t delta = 2.f * m_cfg.cell_size() / m_layer_central_rad;
175 scalar_t phi{0.f};
176 while (phi <= 2.f * constant<scalar_t>::pi) {
177 const scalar_t x = m_layer_central_rad * math::cos(phi);
178 const scalar_t y = m_layer_central_rad * math::sin(phi);
179 const scalar_t z = 0.f;
180
181 cell_centers.push_back({x, y, z});
182 phi += delta;
183 }
184
185 // Generate a wire surface at every position
186 for (const point3_t &cell_center : cell_centers) {
187 // Mask link for the cell
188 mask_link_t mask_link{mask_id, {masks.template size<mask_id>(), 1u}};
189 // The material will be added in a later step
190 material_link_t material_link{surface_t::material_id::e_none,
191 dindex_invalid};
192 // Link the placement transform to the surface
193 const auto trf_index = transforms.size(ctx);
194
195 surfaces.push_back({trf_index, mask_link, material_link, volume_idx,
196 surface_id::e_sensitive},
197 invalid_src_link);
198
199 // The wire bounds
200 masks.template emplace_back<mask_id>(empty_context{}, mask_volume_link,
201 m_cfg.cell_size(),
202 m_cfg.half_length());
203
204 // Build the transform
205 vector3_t z_axis{0.f, 0.f, 1.f};
206 const vector3_t r_axis = vector::normalize(cell_center);
207 z_axis = axis_rotation<algebra_t>(r_axis, m_cfg.stereo_angle())(z_axis);
208 const vector3_t x_axis =
209 unit_vectors<vector3_t>().make_curvilinear_unit_u(z_axis);
210
211 transforms.emplace_back(ctx, cell_center, z_axis, x_axis);
212 }
213
214 return {surfaces_offset, static_cast<dindex>(surfaces.size())};
215 }
216
217 private:
219 wire_layer_generator_config<scalar_t> m_cfg{};
222 scalar_t m_layer_central_rad{};
223};
