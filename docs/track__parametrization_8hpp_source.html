19enum bound_indices : unsigned int {
20 // Local position on the reference surface.
21 // This is intentionally named different from the position components in
22 // the other data vectors, to clarify that this is defined on a surface
23 // while the others are defined in free space.
24 e_bound_loc0 = 0u,
25 e_bound_loc1 = 1u,
26 // Direction angles
27 e_bound_phi = 2u,
28 e_bound_theta = 3u,
29 // Global inverse-momentum-like parameter, i.e. q/p or 1/p
30 // The naming is inconsistent for the case of neutral track parameters where
31 // the value is interpreted as 1/p not as q/p. This is intentional to avoid
32 // having multiple aliases for the same element and for lack of an
33 // acceptable
34 // common name.
35 e_bound_qoverp = 4u,
36 e_bound_time = 5u,
37 // Last uninitialized value contains the total number of components
38 e_bound_size,
39};

47enum free_indices : unsigned int {
48 // Spatial position
49 // The spatial position components must be stored as one continuous block.
50 e_free_pos0 = 0u,
51 e_free_pos1 = e_free_pos0 + 1u,
52 e_free_pos2 = e_free_pos0 + 2u,
53 // Time
54 e_free_time = 3u,
55 // (Unit) direction
56 // The direction components must be stored as one continuous block.
57 e_free_dir0 = 4u,
58 e_free_dir1 = e_free_dir0 + 1u,
59 e_free_dir2 = e_free_dir0 + 2u,
60 // Global inverse-momentum-like parameter, i.e. q/p or 1/p
61 // See BoundIndices for further information
62 e_free_qoverp = 7u,
63 // Last uninitialized value contains the total number of components
64 e_free_size,
65};
