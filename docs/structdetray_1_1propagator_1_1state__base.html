template<typename stepper_t, typename navigator_t, concepts::actor_chain actor_chain_t>
template<bool states_as_reference = false>
struct detray::propagator< stepper_t, navigator_t, actor_chain_t >::state_base< states_as_reference >

Propagation that state aggregates a stepping and a navigation state.

It also keeps references to the actor states.
