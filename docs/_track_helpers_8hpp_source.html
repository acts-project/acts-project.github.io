63template <TrackProxyConcept track_proxy_t>
64Result<typename track_proxy_t::ConstTrackStateProxy> findFirstMeasurementState(
65 const track_proxy_t &track) {
66 using TrackStateProxy = typename track_proxy_t::ConstTrackStateProxy;
67
68 // TODO specialize if track is forward linked
69
70 std::optional<TrackStateProxy> firstMeasurementOpt;
71
72 for (const auto &trackState : track.trackStatesReversed()) {
73 if (trackState.typeFlags().isMeasurement()) {
74 firstMeasurementOpt = trackState;
75 }
76 }
77
78 if (firstMeasurementOpt) {
79 return Result<TrackStateProxy>::success(*firstMeasurementOpt);
80 }
81
82 return Result<TrackStateProxy>::failure(
83 TrackExtrapolationError::CompatibleTrackStateNotFound);
84}

115template <TrackProxyConcept track_proxy_t,
116 typename smoother_t = GainMatrixSmoother>
117Result<void> smoothTrack(
118 const GeometryContext &geoContext, track_proxy_t &track,
119 const Logger &logger = *getDefaultLogger("TrackSmoother", Logging::INFO),
120 smoother_t smoother = GainMatrixSmoother()) {
121 auto &trackContainer = track.container();
122 auto &trackStateContainer = trackContainer.trackStateContainer();
123
124 auto last = findLastMeasurementState(track);
125 if (!last.ok()) {
126 ACTS_DEBUG("no last track state found");
127 return last.error();
128 }
129
130 auto smoothingResult =
131 smoother(geoContext, trackStateContainer, last->index(), logger);
132
133 if (!smoothingResult.ok()) {
134 ACTS_DEBUG("Smoothing track " << track.index() << " failed with error "
135 << smoothingResult.error());
136 return smoothingResult.error();
137 }
138
139 return Result<void>::success();
140}

151template <TrackContainerFrontend track_container_t>
152Result<void> smoothTracks(
153 const GeometryContext &geoContext, const track_container_t &trackContainer,
154 const Logger &logger = *getDefaultLogger("TrackSmoother", Logging::INFO)) {
155 Result<void> result = Result<void>::success();
156
157 for (const auto &track : trackContainer) {
158 auto smoothingResult = smoothTrack(geoContext, track, logger);
159
160 // Only keep the first error
161 if (!smoothingResult.ok() && result.ok()) {
162 result = smoothingResult.error();
163 }
164 }
165
166 return result;

181template <TrackProxyConcept track_proxy_t>
182Result<std::pair<typename track_proxy_t::ConstTrackStateProxy, double>>
183findTrackStateForExtrapolation(
184 const GeometryContext &geoContext, const track_proxy_t &track,
185 const Surface &referenceSurface, TrackExtrapolationStrategy strategy,
186 const Logger &logger = *getDefaultLogger("TrackExtrapolation",
187 Logging::INFO)) {
188 using TrackStateProxy = typename track_proxy_t::ConstTrackStateProxy;
189
190 // Intersect the reference surface with the trajectory at a track state.
191 // Returns `std::nullopt` if the state carries no parameters at all and can
192 // therefore not be started from; an invalid intersection means the state has
193 // parameters but does not reach the reference surface.
194 auto intersect =
195 [&](const TrackStateProxy &state) -> std::optional<Intersection3D> {
196 if (!state.hasSmoothed() && !state.hasFiltered() && !state.hasPredicted()) {
197 return std::nullopt;
198 }
199
200 // `parameters` picks smoothed over filtered over predicted, which is what
201 // `TrackProxy::createParametersFromState` starts the propagation from. The
202 // distance has to be measured on the same parameters.
203 const FreeVector freeVector = transformBoundToFreeParameters(
204 state.referenceSurface(), geoContext, state.parameters());
205
206 return referenceSurface
207 .intersect(geoContext, freeVector.template segment<3>(eFreePos0),
208 freeVector.template segment<3>(eFreeDir0),
209 BoundaryTolerance::None(), s_onSurfaceTolerance)
210 .closest();
211 };
212
213 switch (strategy) {
214 case TrackExtrapolationStrategy::first: {
215 ACTS_VERBOSE("looking for first track state");
216
217 auto first = findFirstMeasurementState(track);
218 if (!first.ok()) {
219 ACTS_DEBUG("no first track state found");
220 return first.error();
221 }
222
223 std::optional<Intersection3D> intersection = intersect(*first);
224 if (!intersection.has_value()) {
225 ACTS_DEBUG("first track state carries no parameters");
226 return Result<std::pair<TrackStateProxy, double>>::failure(
227 TrackExtrapolationError::CompatibleTrackStateNotFound);
228 }
229 if (!intersection->isValid()) {
230 ACTS_DEBUG("no intersection found");
231 return Result<std::pair<TrackStateProxy, double>>::failure(
232 TrackExtrapolationError::ReferenceSurfaceUnreachable);
233 }
234
235 ACTS_VERBOSE("found intersection at " << intersection->pathLength());
236 return std::pair(*first, intersection->pathLength());
237 }
238
239 case TrackExtrapolationStrategy::last: {
240 ACTS_VERBOSE("looking for last track state");
241
242 auto last = findLastMeasurementState(track);
243 if (!last.ok()) {
244 ACTS_DEBUG("no last track state found");
245 return last.error();
246 }
247
248 std::optional<Intersection3D> intersection = intersect(*last);
249 if (!intersection.has_value()) {
250 ACTS_DEBUG("last track state carries no parameters");
251 return Result<std::pair<TrackStateProxy, double>>::failure(
252 TrackExtrapolationError::CompatibleTrackStateNotFound);
253 }
254 if (!intersection->isValid()) {
255 ACTS_DEBUG("no intersection found");
256 return Result<std::pair<TrackStateProxy, double>>::failure(
257 TrackExtrapolationError::ReferenceSurfaceUnreachable);
258 }
259
260 ACTS_VERBOSE("found intersection at " << intersection->pathLength());
261 return std::pair(*last, intersection->pathLength());
262 }
263
264 case TrackExtrapolationStrategy::firstOrLast: {
265 ACTS_VERBOSE("looking for first or last track state");
266
267 auto first = findFirstMeasurementState(track);
268 if (!first.ok()) {
269 ACTS_DEBUG("no first track state found");
270 return first.error();
271 }
272
273 auto last = findLastMeasurementState(track);
274 if (!last.ok()) {
275 ACTS_DEBUG("no last track state found");
276 return last.error();
277 }
278
279 std::optional<Intersection3D> intersectionFirstOpt = intersect(*first);
280 std::optional<Intersection3D> intersectionLastOpt = intersect(*last);
281
282 if (!intersectionFirstOpt.has_value() &&
283 !intersectionLastOpt.has_value()) {
284 ACTS_DEBUG("neither first nor last track state carries parameters");
285 return Result<std::pair<TrackStateProxy, double>>::failure(
286 TrackExtrapolationError::CompatibleTrackStateNotFound);
287 }
288
289 // an end without parameters cannot be started from, so it loses the
290 // comparison below through the infinite path length of an invalid
291 // intersection
292 Intersection3D intersectionFirst =
293 intersectionFirstOpt.value_or(Intersection3D::Invalid());
294 Intersection3D intersectionLast =
295 intersectionLastOpt.value_or(Intersection3D::Invalid());
296
297 double absDistanceFirst = std::abs(intersectionFirst.pathLength());
298 double absDistanceLast = std::abs(intersectionLast.pathLength());
299
300 if (intersectionFirst.isValid() && absDistanceFirst <= absDistanceLast) {
301 ACTS_VERBOSE("using first track state with intersection at "
302 << intersectionFirst.pathLength());
303 return std::pair(*first, intersectionFirst.pathLength());
304 }
305
306 if (intersectionLast.isValid() && absDistanceLast <= absDistanceFirst) {
307 ACTS_VERBOSE("using last track state with intersection at "
308 << intersectionLast.pathLength());
309 return std::pair(*last, intersectionLast.pathLength());
310 }
311
312 ACTS_DEBUG("no intersection found");
313 return Result<std::pair<TrackStateProxy, double>>::failure(
314 TrackExtrapolationError::ReferenceSurfaceUnreachable);
315 }
316 }
317
318 // unreachable
319 return Result<std::pair<TrackStateProxy, double>>::failure(
320 TrackExtrapolationError::CompatibleTrackStateNotFound);
321}

337template <TrackProxyConcept track_proxy_t, typename propagator_t,
338 typename propagator_options_t>
339Result<void> extrapolateTrackToReferenceSurface(
340 track_proxy_t &track, const Surface &referenceSurface,
341 const propagator_t &propagator, propagator_options_t options,
342 TrackExtrapolationStrategy strategy,
343 const Logger &logger = *getDefaultLogger("TrackExtrapolation",
344 Logging::INFO)) {
345 auto findResult = findTrackStateForExtrapolation(
346 options.geoContext, track, referenceSurface, strategy, logger);
347
348 if (!findResult.ok()) {
349 ACTS_DEBUG("failed to find track state for extrapolation");
350 return findResult.error();
351 }
352
353 auto &[trackState, distance] = *findResult;
354
355 options.direction = Direction::fromScalarZeroAsPositive(distance);
356
357 BoundTrackParameters parameters = track.createParametersFromState(trackState);
358 ACTS_VERBOSE("extrapolating track to reference surface at distance "
359 << distance << " with direction " << options.direction
360 << " with starting parameters " << parameters);
361
362 auto propagateResult =
363 propagator.template propagate<propagator_options_t, ForcedSurfaceReached>(
364 parameters, referenceSurface, options);
365
366 if (!propagateResult.ok()) {
367 ACTS_DEBUG("failed to extrapolate track: " << propagateResult.error());
368 return propagateResult.error();
369 }
370
371 track.setReferenceSurface(referenceSurface.getSharedPtr());
372 track.parameters() = propagateResult->endParameters.value().parameters();
373 track.covariance() =
374 propagateResult->endParameters.value().covariance().value();
375
376 return Result<void>::success();
377}

393template <TrackContainerFrontend track_container_t, typename propagator_t,
394 typename propagator_options_t>
395Result<void> extrapolateTracksToReferenceSurface(
396 const track_container_t &trackContainer, const Surface &referenceSurface,
397 const propagator_t &propagator, propagator_options_t options,
398 TrackExtrapolationStrategy strategy,
399 const Logger &logger = *getDefaultLogger("TrackExtrapolation",
400 Logging::INFO)) {
401 Result<void> result = Result<void>::success();
402
403 for (const auto &track : trackContainer) {
404 auto extrapolateResult = extrapolateTrackToReferenceSurface(
405 track, referenceSurface, propagator, options, strategy, logger);
406
407 // Only keep the first error
408 if (!extrapolateResult.ok() && result.ok()) {
409 result = extrapolateResult.error();
410 }
411 }
412
413 return result;
414}

420template <TrackProxyConcept track_proxy_t>
421void calculateTrackQuantities(track_proxy_t track)
422 requires(!track_proxy_t::ReadOnly)
423{
424 track.chi2() = 0;
425 track.nDoF() = 0;
426
427 track.nHoles() = 0;
428 track.nMeasurements() = 0;
429 track.nSharedHits() = 0;
430 track.nOutliers() = 0;
431
432 for (const auto &trackState : track.trackStatesReversed()) {
433 ConstTrackStateTypeMap typeFlags = trackState.typeFlags();
434
435 if (typeFlags.isHole()) {
436 track.nHoles()++;
437 } else if (typeFlags.isOutlier()) {
438 track.nOutliers()++;
439 } else if (typeFlags.isMeasurement()) {
440 if (typeFlags.isSharedHit()) {
441 track.nSharedHits()++;
442 }
443 track.nMeasurements()++;
444 track.chi2() += trackState.chi2();
445 track.nDoF() += trackState.calibratedSize();
446 }
447 }
448}

457template <TrackProxyConcept track_proxy_t>
458void trimTrackFront(track_proxy_t track, bool trimHoles, bool trimOutliers,
459 bool trimMaterial, bool trimOtherNoneMeasurement)
460 requires(!track_proxy_t::ReadOnly)
461{
462 using TrackStateProxy = typename track_proxy_t::TrackStateProxy;
463
464 // TODO specialize if track is forward linked
465
466 std::optional<TrackStateProxy> front;
467
468 for (TrackStateProxy trackState : track.trackStatesReversed()) {
469 TrackStateTypeMap typeFlags = trackState.typeFlags();
470 bool isHole = typeFlags.isHole();
471 bool isOutlier = typeFlags.isOutlier();
472 bool isMaterial = typeFlags.isMaterial();
473 bool isOtherNoneMeasurement =
474 !typeFlags.hasMeasurement() && !isHole && !isOutlier && !isMaterial;
475 if (trimHoles && isHole) {
476 continue;
477 }
478 if (trimOutliers && isOutlier) {
479 continue;
480 }
481 if (trimMaterial && isMaterial) {
482 continue;
483 }
484 if (trimOtherNoneMeasurement && isOtherNoneMeasurement) {
485 continue;
486 }
487
488 front = trackState;
489 }
490
491 if (front.has_value()) {
492 front.value().previous() = TrackStateProxy::kInvalid;
493 }
494}

503template <TrackProxyConcept track_proxy_t>
504void trimTrackBack(track_proxy_t track, bool trimHoles, bool trimOutliers,
505 bool trimMaterial, bool trimOtherNoneMeasurement)
506 requires(!track_proxy_t::ReadOnly)
507{
508 using TrackStateProxy = typename track_proxy_t::TrackStateProxy;
509
510 std::optional<TrackStateProxy> back;
511
512 for (TrackStateProxy trackState : track.trackStatesReversed()) {
513 back = trackState;
514
515 TrackStateTypeMap typeFlags = trackState.typeFlags();
516 bool isHole = typeFlags.isHole();
517 bool isOutlier = typeFlags.isOutlier();
518 bool isMaterial = typeFlags.isMaterial();
519 bool isOtherNoneMeasurement =
520 !typeFlags.hasMeasurement() && !isHole && !isOutlier && !isMaterial;
521 if (trimHoles && isHole) {
522 continue;
523 }
524 if (trimOutliers && isOutlier) {
525 continue;
526 }
527 if (trimMaterial && isMaterial) {
528 continue;
529 }
530 if (trimOtherNoneMeasurement && isOtherNoneMeasurement) {
531 continue;
532 }
533
534 break;
535 }
536
537 if (back.has_value()) {
538 track.tipIndex() = back.value().index();

565template <std::size_t nMeasurementDim,
566 TrackStateProxyConcept track_state_proxy_t>
567std::pair<Vector<nMeasurementDim>, SquareMatrix<nMeasurementDim>>
568calculatePredictedResidual(track_state_proxy_t trackState) {
569 using MeasurementVector = Vector<nMeasurementDim>;
570 using MeasurementMatrix = SquareMatrix<nMeasurementDim>;
571
572 if (!trackState.hasPredicted()) {
573 throw std::invalid_argument("track state has no predicted parameters");
574 }
575 if (!trackState.hasCalibrated()) {
576 throw std::invalid_argument("track state has no calibrated parameters");
577 }
578
579 auto subspaceHelper =
580 trackState.template projectorSubspaceHelper<nMeasurementDim>();
581
582 auto measurement = trackState.template calibrated<nMeasurementDim>();
583 auto measurementCovariance =
584 trackState.template calibratedCovariance<nMeasurementDim>();
585 MeasurementVector predicted =
586 subspaceHelper.projectVector(trackState.predicted());
587 MeasurementMatrix predictedCovariance =
588 subspaceHelper.projectMatrix(trackState.predictedCovariance());
589
590 MeasurementVector residual = measurement - predicted;
591 MeasurementMatrix residualCovariance =
592 measurementCovariance + predictedCovariance;
593
594 return {residual, residualCovariance};
595}

675template <TrackStateProxyConcept track_state_proxy_t>
676double calculatePredictedChi2(track_state_proxy_t trackState) {
677 if (!trackState.hasPredicted()) {
678 throw std::invalid_argument("track state has no predicted parameters");
679 }
680 if (!trackState.hasCalibrated()) {
681 throw std::invalid_argument("track state has no calibrated parameters");
682 }
683
684 return visit_measurement(
685 trackState.calibratedSize(),
686 [&]<std::size_t measdim>(
687 std::integral_constant<std::size_t, measdim>) -> double {
688 auto [residual, residualCovariance] =
689 calculatePredictedResidual<measdim>(trackState);
690
691 return (residual.transpose() * residualCovariance.inverse() * residual)
692 .eval()(0, 0);
693 });
694}

760template <TrackStateProxyConcept track_state_proxy_t>
761[[deprecated(
762 "Use calculateUnbiasedParametersCovariance(const AnyConstTrackStateProxy&) "
763 "instead; the templated form instantiates expensive Eigen code in every "

766 track_state_proxy_t trackState) {
767 // Explicitly select the non-template overload taking a type-erased proxy.
768 // A plain call here would re-resolve to this very template (the wrapped

772 calculateUnbiasedParametersCovariance;
773 return impl(AnyConstTrackStateProxy{trackState});
