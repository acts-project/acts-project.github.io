64template <TrackProxyConcept track_proxy_t>
65Result<typename track_proxy_t::ConstTrackStateProxy> findFirstMeasurementState(
66 const track_proxy_t &track) {
67 using TrackStateProxy = typename track_proxy_t::ConstTrackStateProxy;
68
69 // TODO specialize if track is forward linked
70
71 std::optional<TrackStateProxy> firstMeasurementOpt;
72
73 for (const auto &trackState : track.trackStatesReversed()) {
74 if (trackState.typeFlags().isMeasurement()) {
75 firstMeasurementOpt = trackState;
76 }
77 }
78
79 if (firstMeasurementOpt) {
80 return Result<TrackStateProxy>::success(*firstMeasurementOpt);
81 }
82
83 return Result<TrackStateProxy>::failure(
84 TrackExtrapolationError::CompatibleTrackStateNotFound);
85}

116template <TrackProxyConcept track_proxy_t,
117 typename smoother_t = GainMatrixSmoother>
118Result<void> smoothTrack(
119 const GeometryContext &geoContext, track_proxy_t &track,
120 const Logger &logger = *getDefaultLogger("TrackSmoother", Logging::INFO),
121 smoother_t smoother = GainMatrixSmoother()) {
122 auto &trackContainer = track.container();
123 auto &trackStateContainer = trackContainer.trackStateContainer();
124
125 auto last = findLastMeasurementState(track);
126 if (!last.ok()) {
127 ACTS_DEBUG("no last track state found");
128 return last.error();
129 }
130
131 auto smoothingResult =
132 smoother(geoContext, trackStateContainer, last->index(), logger);
133
134 if (!smoothingResult.ok()) {
135 ACTS_DEBUG("Smoothing track " << track.index() << " failed with error "
136 << smoothingResult.error());
137 return smoothingResult.error();
138 }
139
140 return Result<void>::success();
141}

152template <TrackContainerFrontend track_container_t>
153Result<void> smoothTracks(
154 const GeometryContext &geoContext, const track_container_t &trackContainer,
155 const Logger &logger = *getDefaultLogger("TrackSmoother", Logging::INFO)) {
156 Result<void> result = Result<void>::success();
157
158 for (const auto &track : trackContainer) {
159 auto smoothingResult = smoothTrack(geoContext, track, logger);
160
161 // Only keep the first error
162 if (!smoothingResult.ok() && result.ok()) {
163 result = smoothingResult.error();
164 }
165 }
166
167 return result;

182template <TrackProxyConcept track_proxy_t>
183Result<std::pair<typename track_proxy_t::ConstTrackStateProxy, double>>
184findTrackStateForExtrapolation(
185 const GeometryContext &geoContext, const track_proxy_t &track,
186 const Surface &referenceSurface, TrackExtrapolationStrategy strategy,
187 const Logger &logger = *getDefaultLogger("TrackExtrapolation",
188 Logging::INFO)) {
189 using TrackStateProxy = typename track_proxy_t::ConstTrackStateProxy;
190
191 // Intersect the reference surface with the trajectory at a track state.
192 // Returns `std::nullopt` if the state carries no parameters at all and can
193 // therefore not be started from; an invalid intersection means the state has
194 // parameters but does not reach the reference surface.
195 auto intersect =
196 [&](const TrackStateProxy &state) -> std::optional<Intersection3D> {
197 if (!state.hasSmoothed() && !state.hasFiltered() && !state.hasPredicted()) {
198 return std::nullopt;
199 }
200
201 // `parameters` picks smoothed over filtered over predicted, which is what
202 // `TrackProxy::createParametersFromState` starts the propagation from. The
203 // distance has to be measured on the same parameters.
204 const FreeVector freeVector = transformBoundToFreeParameters(
205 state.referenceSurface(), geoContext, state.parameters());
206
207 return referenceSurface
208 .intersect(geoContext, freeVector.template segment<3>(eFreePos0),
209 freeVector.template segment<3>(eFreeDir0),
210 BoundaryTolerance::None(), s_onSurfaceTolerance)
211 .closest();
212 };
213
214 switch (strategy) {
215 case TrackExtrapolationStrategy::first: {
216 ACTS_VERBOSE("looking for first track state");
217
218 auto first = findFirstMeasurementState(track);
219 if (!first.ok()) {
220 ACTS_DEBUG("no first track state found");
221 return first.error();
222 }
223
224 std::optional<Intersection3D> intersection = intersect(*first);
225 if (!intersection.has_value()) {
226 ACTS_DEBUG("first track state carries no parameters");
227 return Result<std::pair<TrackStateProxy, double>>::failure(
228 TrackExtrapolationError::CompatibleTrackStateNotFound);
229 }
230 if (!intersection->isValid()) {
231 ACTS_DEBUG("no intersection found");
232 return Result<std::pair<TrackStateProxy, double>>::failure(
233 TrackExtrapolationError::ReferenceSurfaceUnreachable);
234 }
235
236 ACTS_VERBOSE("found intersection at " << intersection->pathLength());
237 return std::pair(*first, intersection->pathLength());
238 }
239
240 case TrackExtrapolationStrategy::last: {
241 ACTS_VERBOSE("looking for last track state");
242
243 auto last = findLastMeasurementState(track);
244 if (!last.ok()) {
245 ACTS_DEBUG("no last track state found");
246 return last.error();
247 }
248
249 std::optional<Intersection3D> intersection = intersect(*last);
250 if (!intersection.has_value()) {
251 ACTS_DEBUG("last track state carries no parameters");
252 return Result<std::pair<TrackStateProxy, double>>::failure(
253 TrackExtrapolationError::CompatibleTrackStateNotFound);
254 }
255 if (!intersection->isValid()) {
256 ACTS_DEBUG("no intersection found");
257 return Result<std::pair<TrackStateProxy, double>>::failure(
258 TrackExtrapolationError::ReferenceSurfaceUnreachable);
259 }
260
261 ACTS_VERBOSE("found intersection at " << intersection->pathLength());
262 return std::pair(*last, intersection->pathLength());
263 }
264
265 case TrackExtrapolationStrategy::firstOrLast: {
266 ACTS_VERBOSE("looking for first or last track state");
267
268 auto first = findFirstMeasurementState(track);
269 if (!first.ok()) {
270 ACTS_DEBUG("no first track state found");
271 return first.error();
272 }
273
274 auto last = findLastMeasurementState(track);
275 if (!last.ok()) {
276 ACTS_DEBUG("no last track state found");
277 return last.error();
278 }
279
280 std::optional<Intersection3D> intersectionFirstOpt = intersect(*first);
281 std::optional<Intersection3D> intersectionLastOpt = intersect(*last);
282
283 if (!intersectionFirstOpt.has_value() &&
284 !intersectionLastOpt.has_value()) {
285 ACTS_DEBUG("neither first nor last track state carries parameters");
286 return Result<std::pair<TrackStateProxy, double>>::failure(
287 TrackExtrapolationError::CompatibleTrackStateNotFound);
288 }
289
290 // an end without parameters cannot be started from, so it loses the
291 // comparison below through the infinite path length of an invalid
292 // intersection
293 Intersection3D intersectionFirst =
294 intersectionFirstOpt.value_or(Intersection3D::Invalid());
295 Intersection3D intersectionLast =
296 intersectionLastOpt.value_or(Intersection3D::Invalid());
297
298 double absDistanceFirst = std::abs(intersectionFirst.pathLength());
299 double absDistanceLast = std::abs(intersectionLast.pathLength());
300
301 if (intersectionFirst.isValid() && absDistanceFirst <= absDistanceLast) {
302 ACTS_VERBOSE("using first track state with intersection at "
303 << intersectionFirst.pathLength());
304 return std::pair(*first, intersectionFirst.pathLength());
305 }
306
307 if (intersectionLast.isValid() && absDistanceLast <= absDistanceFirst) {
308 ACTS_VERBOSE("using last track state with intersection at "
309 << intersectionLast.pathLength());
310 return std::pair(*last, intersectionLast.pathLength());
311 }
312
313 ACTS_DEBUG("no intersection found");
314 return Result<std::pair<TrackStateProxy, double>>::failure(
315 TrackExtrapolationError::ReferenceSurfaceUnreachable);
316 }
317 }
318
319 // unreachable
320 return Result<std::pair<TrackStateProxy, double>>::failure(
321 TrackExtrapolationError::CompatibleTrackStateNotFound);
322}

338template <TrackProxyConcept track_proxy_t, typename propagator_t,
339 typename propagator_options_t>
340Result<void> extrapolateTrackToReferenceSurface(
341 track_proxy_t &track, const Surface &referenceSurface,
342 const propagator_t &propagator, propagator_options_t options,
343 TrackExtrapolationStrategy strategy,
344 const Logger &logger = *getDefaultLogger("TrackExtrapolation",
345 Logging::INFO)) {
346 auto findResult = findTrackStateForExtrapolation(
347 options.geoContext, track, referenceSurface, strategy, logger);
348
349 if (!findResult.ok()) {
350 ACTS_DEBUG("failed to find track state for extrapolation");
351 return findResult.error();
352 }
353
354 auto &[trackState, distance] = *findResult;
355
356 options.direction = Direction::fromScalarZeroAsPositive(distance);
357
358 BoundTrackParameters parameters = track.createParametersFromState(trackState);
359 ACTS_VERBOSE("extrapolating track to reference surface at distance "
360 << distance << " with direction " << options.direction
361 << " with starting parameters " << parameters);
362
363 auto propagateResult =
364 propagator.template propagate<propagator_options_t, ForcedSurfaceReached>(
365 parameters, referenceSurface, options);
366
367 if (!propagateResult.ok()) {
368 ACTS_DEBUG("failed to extrapolate track: " << propagateResult.error());
369 return propagateResult.error();
370 }
371
372 track.setReferenceSurface(referenceSurface.getSharedPtr());
373 track.parameters() = propagateResult->endParameters.value().parameters();
374 track.covariance() =
375 propagateResult->endParameters.value().covariance().value();
376
377 return Result<void>::success();
378}

394template <TrackContainerFrontend track_container_t, typename propagator_t,
395 typename propagator_options_t>
396Result<void> extrapolateTracksToReferenceSurface(
397 const track_container_t &trackContainer, const Surface &referenceSurface,
398 const propagator_t &propagator, propagator_options_t options,
399 TrackExtrapolationStrategy strategy,
400 const Logger &logger = *getDefaultLogger("TrackExtrapolation",
401 Logging::INFO)) {
402 Result<void> result = Result<void>::success();
403
404 for (const auto &track : trackContainer) {
405 auto extrapolateResult = extrapolateTrackToReferenceSurface(
406 track, referenceSurface, propagator, options, strategy, logger);
407
408 // Only keep the first error
409 if (!extrapolateResult.ok() && result.ok()) {
410 result = extrapolateResult.error();
411 }
412 }
413
414 return result;
415}

421template <TrackProxyConcept track_proxy_t>
422void calculateTrackQuantities(track_proxy_t track)
423 requires(!track_proxy_t::ReadOnly)
424{
425 track.chi2() = 0;
426 track.nDoF() = 0;
427
428 track.nHoles() = 0;
429 track.nMeasurements() = 0;
430 track.nSharedHits() = 0;
431 track.nOutliers() = 0;
432
433 // The split hit column is optional, so only count if the track has it
434 unsigned int nSplitHits = 0;
435
436 for (const auto &trackState : track.trackStatesReversed()) {
437 ConstTrackStateTypeMap typeFlags = trackState.typeFlags();
438
439 if (typeFlags.isHole()) {
440 track.nHoles()++;
441 } else if (typeFlags.isOutlier()) {
442 track.nOutliers()++;
443 } else if (typeFlags.isMeasurement()) {
444 if (typeFlags.isSharedHit()) {
445 track.nSharedHits()++;
446 }
447 if (typeFlags.isSplitHit()) {
448 nSplitHits++;
449 }
450 track.nMeasurements()++;
451 track.chi2() += trackState.chi2();
452 track.nDoF() += trackState.calibratedSize();
453 }
454 }
455
456 if constexpr (requires { track.nSplitHits(); }) {
457 if (track.hasColumn(detail_tp::kSplitHitsKey)) {
458 track.nSplitHits() = nSplitHits;
459 }
460 }
461}

470template <TrackProxyConcept track_proxy_t>
471void trimTrackFront(track_proxy_t track, bool trimHoles, bool trimOutliers,
472 bool trimMaterial, bool trimOtherNoneMeasurement)
473 requires(!track_proxy_t::ReadOnly)
474{
475 using TrackStateProxy = typename track_proxy_t::TrackStateProxy;
476
477 // TODO specialize if track is forward linked
478
479 std::optional<TrackStateProxy> front;
480
481 for (TrackStateProxy trackState : track.trackStatesReversed()) {
482 TrackStateTypeMap typeFlags = trackState.typeFlags();
483 bool isHole = typeFlags.isHole();
484 bool isOutlier = typeFlags.isOutlier();
485 bool isMaterial = typeFlags.isMaterial();
486 bool isOtherNoneMeasurement =
487 !typeFlags.hasMeasurement() && !isHole && !isOutlier && !isMaterial;
488 if (trimHoles && isHole) {
489 continue;
490 }
491 if (trimOutliers && isOutlier) {
492 continue;
493 }
494 if (trimMaterial && isMaterial) {
495 continue;
496 }
497 if (trimOtherNoneMeasurement && isOtherNoneMeasurement) {
498 continue;
499 }
500
501 front = trackState;
502 }
503
504 if (front.has_value()) {
505 front.value().previous() = TrackStateProxy::kInvalid;
506 }
507}

516template <TrackProxyConcept track_proxy_t>
517void trimTrackBack(track_proxy_t track, bool trimHoles, bool trimOutliers,
518 bool trimMaterial, bool trimOtherNoneMeasurement)
519 requires(!track_proxy_t::ReadOnly)
520{
521 using TrackStateProxy = typename track_proxy_t::TrackStateProxy;
522
523 std::optional<TrackStateProxy> back;
524
525 for (TrackStateProxy trackState : track.trackStatesReversed()) {
526 back = trackState;
527
528 TrackStateTypeMap typeFlags = trackState.typeFlags();
529 bool isHole = typeFlags.isHole();
530 bool isOutlier = typeFlags.isOutlier();
531 bool isMaterial = typeFlags.isMaterial();
532 bool isOtherNoneMeasurement =
533 !typeFlags.hasMeasurement() && !isHole && !isOutlier && !isMaterial;
534 if (trimHoles && isHole) {
535 continue;
536 }
537 if (trimOutliers && isOutlier) {
538 continue;
539 }
540 if (trimMaterial && isMaterial) {
541 continue;
542 }
543 if (trimOtherNoneMeasurement && isOtherNoneMeasurement) {
544 continue;
545 }
546
547 break;
548 }
549
550 if (back.has_value()) {
551 track.tipIndex() = back.value().index();
552 }
553}

578template <std::size_t nMeasurementDim,
579 TrackStateProxyConcept track_state_proxy_t>
580std::pair<Vector<nMeasurementDim>, SquareMatrix<nMeasurementDim>>
581calculatePredictedResidual(track_state_proxy_t trackState) {
582 using MeasurementVector = Vector<nMeasurementDim>;
583 using MeasurementMatrix = SquareMatrix<nMeasurementDim>;
584
585 if (!trackState.hasPredicted()) {
586 throw std::invalid_argument("track state has no predicted parameters");
587 }
588 if (!trackState.hasCalibrated()) {
589 throw std::invalid_argument("track state has no calibrated parameters");
590 }
591
592 auto subspaceHelper =
593 trackState.template projectorSubspaceHelper<nMeasurementDim>();
594
595 auto measurement = trackState.template calibrated<nMeasurementDim>();
596 auto measurementCovariance =
597 trackState.template calibratedCovariance<nMeasurementDim>();
598 MeasurementVector predicted =
599 subspaceHelper.projectVector(trackState.predicted());
600 MeasurementMatrix predictedCovariance =
601 subspaceHelper.projectMatrix(trackState.predictedCovariance());
602
603 MeasurementVector residual = measurement - predicted;
604 MeasurementMatrix residualCovariance =
605 measurementCovariance + predictedCovariance;
606
607 return {residual, residualCovariance};
608}

619 using MeasurementVector = Vector<nMeasurementDim>;

688template <TrackStateProxyConcept track_state_proxy_t>
689double calculatePredictedChi2(track_state_proxy_t trackState) {
690 if (!trackState.hasPredicted()) {
691 throw std::invalid_argument("track state has no predicted parameters");
692 }
693 if (!trackState.hasCalibrated()) {
694 throw std::invalid_argument("track state has no calibrated parameters");
695 }
696
697 return visit_measurement(
698 trackState.calibratedSize(),
699 [&]<std::size_t measdim>(
700 std::integral_constant<std::size_t, measdim>) -> double {
701 auto [residual, residualCovariance] =
702 calculatePredictedResidual<measdim>(trackState);
703
704 return (residual.transpose() * residualCovariance.inverse() * residual)
705 .eval()(0, 0);
706 });
707}

773template <TrackStateProxyConcept track_state_proxy_t>
774[[deprecated(
