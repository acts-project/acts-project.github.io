77struct transform_payload {
79 std::array<io::scalar, 3u> tr{};
80 // Column major rotation matrix
83 std::array<io::scalar, 9u> rot{};
84
85 private:
86 DETRAY_HOST friend inline std::ostream& operator<<(
87 std::ostream& os, const transform_payload& transform) {
89 const auto& r = transform.rot;
91 os << "rot: ";
92 os << std::fixed << std::setw(4);
93 auto print_line = [&](std::size_t i) {
94 os << r[i] << " " << r[i + 3] << " " << r[i + 6] << "\n";
95 };
96 print_line(0);
97 os << " ";
98 print_line(1);
99 os << " ";
100 print_line(2);
101
103 const auto& t = transform.tr;
104
105 os << "tr: ";
106 os << t[0] << " " << t[1] << " " << t[2];
107 return os;
108 }
109};

247struct axis_payload {
249 axis::binning binning{axis::binning::e_regular};
251 axis::bounds bounds{axis::bounds::e_closed};
253 axis::label label{axis::label::e_r};
254
256 std::size_t bins{0u};
258 std::vector<io::scalar> edges{};
259
260 private:
261 DETRAY_HOST friend inline std::ostream& operator<<(std::ostream& os,
262 const axis_payload& axis) {
263 os << "axis_payload{binning: " << axis.binning
264 << ", bounds: " << axis.bounds << ", label: " << axis.label
265 << ", bins: " << axis.bins << ", edges: [";
266 for (std::size_t i = 0; i < axis.edges.size(); ++i) {
267 if (i > 0) {
268 os << ", ";
269 }
270 os << axis.edges[i];
271 }
272 os << "]}";
273 return os;
274 }
