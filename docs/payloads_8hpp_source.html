95struct transform_payload {
97 std::array<io::scalar, 3u> tr{};
98 // Column major rotation matrix
101 std::array<io::scalar, 9u> rot{};
102
103 private:
104 DETRAY_HOST friend inline std::ostream& operator<<(
105 std::ostream& os, const transform_payload& transform) {
107 const auto& r = transform.rot;
109 os << "rot: ";
110 os << std::fixed << std::setw(4);
111 auto print_line = [&](std::size_t i) {
112 os << r[i] << " " << r[i + 3] << " " << r[i + 6] << "\n";
113 };
114 print_line(0);
115 os << " ";
116 print_line(1);
117 os << " ";
118 print_line(2);
119
121 const auto& t = transform.tr;
122
123 os << "tr: ";
124 os << t[0] << " " << t[1] << " " << t[2];
125 return os;
126 }
127};

266struct axis_payload {

270 axis::bounds bounds{axis::bounds::e_closed};

273
275 std::size_t bins{0u};

279 private:
280 DETRAY_HOST friend inline std::ostream& operator<<(std::ostream& os,
281 const axis_payload& axis) {
282 os << "axis_payload{binning: " << axis.binning
283 << ", bounds: " << axis.bounds << ", label: " << axis.label
284 << ", bins: " << axis.bins << ", edges: [";
285 for (std::size_t i = 0; i < axis.edges.size(); ++i) {
286 if (i > 0) {
287 os << ", ";
288 }
289 os << axis.edges[i];
290 }
291 os << "]}";
