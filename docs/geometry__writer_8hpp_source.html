46 template <detray::concepts::detector detector_t>
47 static geo_header_payload header_to_payload(const detector_t& det,
48 const std::string_view det_name) {
49 geo_header_payload header_data;
50
51 header_data.common = detail::basic_converter::to_payload(det_name, tag);
52
53 header_data.sub_header.emplace();
54 auto& geo_sub_header = header_data.sub_header.value();
55 geo_sub_header.n_volumes = det.volumes().size();
56 geo_sub_header.n_surfaces = det.surfaces().size();
57
58 return header_data;
59 }

66 template <detray::concepts::detector detector_t>
67 static payload_type to_payload(const detector_t& det,
68 const typename detector_t::name_map& names) {
69 payload_type det_data;
70 det_data.volumes.reserve(det.volumes().size());
71
72 for (const auto& vol : det.volumes()) {
73 if (names.contains(vol.index())) {
74 det_data.volumes.push_back(to_payload(vol, det, names.at(vol.index())));
75 } else {
76 det_data.volumes.push_back(to_payload(vol, det, ""));
77 }
78 }
79
80 return det_data;
81 }

87 template <detray::concepts::detector detector_t>
88 static transform_payload to_payload(
89 const typename detector_t::transform3_type& trf) {
90 transform_payload trf_data;
91
92 const auto& t = trf.translation();
93 const auto& x = trf.x();
94 const auto& y = trf.y();
95 const auto& z = trf.z();
96
97 trf_data.tr = {t[0], t[1], t[2]};
98 trf_data.rot = {x[0], x[1], x[2], y[0], y[1], y[2], z[0], z[1], z[2]};
99
100 return trf_data;
101 }

107 template <typename mask_t>
108 requires(!std::is_same_v<typename mask_t::shape, void>)
109 static mask_payload to_payload(const mask_t& m) {
110 mask_payload mask_data;
111
112 mask_data.shape = io::detail::get_id<typename mask_t::shape>();
113
114 mask_data.volume_link =
115 detail::basic_converter::to_payload(m.volume_link());
116
117 mask_data.boundaries.resize(mask_t::boundaries::e_size);
118 std::ranges::copy(m.values(), std::begin(mask_data.boundaries));
119
120 return mask_data;
121 }

128 template <detray::concepts::detector detector_t>
129 static surface_payload to_payload(const geometry::surface<detector_t>& sf,
130 std::size_t sf_idx) {
131 using algebra_t = typename detector_t::algebra_type;
132 surface_payload sf_data;
133
134 sf_data.index_in_coll = sf_idx;
135 sf_data.type = sf.id();
136 sf_data.identifier = sf.identifier().value();
137 sf_data.transform = to_payload<detector_t>(sf.transform({}));
138 sf_data.masks = sf.template visit_mask<get_mask_link_payload>();
139 if (sf.has_material()) {
140 sf_data.material =
141 sf.template visit_material<get_material_link_payload<algebra_t>>();
142 }
143 sf_data.source = sf.source();
144
145 return sf_data;
146 }

154 template <detray::concepts::detector detector_t>
155 static volume_payload to_payload(
156 const typename detector_t::volume_type& vol_desc, const detector_t& det,
157 const std::string_view name) {
158 volume_payload vol_data;
159
160 vol_data.index = detail::basic_converter::to_payload(vol_desc.index());
161 vol_data.name = name;
162 vol_data.transform =
163 to_payload<detector_t>(det.transform_store().at(vol_desc.transform()));
164 vol_data.type = vol_desc.id();
165
166 // Count the surfaces belonging to this volume
167 std::size_t sf_idx{0};
168
169 for (const auto& sf_desc : det.surfaces()) {
170 if (sf_desc.volume() == vol_desc.index()) {
171 vol_data.surfaces.push_back(
172 to_payload(geometry::surface{det, sf_desc}, sf_idx++));
173 }
174 }
175
176 // Only run the query, if object type is contained in volume
177 const auto& link = vol_desc.accel_link();
178 // Initialize the std::optional
179 if (link.size() > 1u) {
180 vol_data.acc_links.emplace();
181 }
182 // Skip the first acceleration structure which exists in every volume
183 // and is handled automatically during detector building
184 for (unsigned int i = 1u; i < link.size(); ++i) {
185 const auto& l = link[i];
186 if (!l.is_invalid()) {
187 const auto aclp =
188 det.accelerator_store().template visit<get_acc_link_payload>(l);
189 vol_data.acc_links->push_back(aclp);
190 }
191 }
192
193 return vol_data;
194 }

198 struct get_mask_link_payload {
199 template <typename mask_group_t, typename idx_range_t>
200 inline auto operator()(const mask_group_t& mask_group,
201 const idx_range_t& idx_range) const {
202 std::vector<mask_payload> payloads{};
203
204 for (const auto& mask : detray::ranges::subrange(mask_group, idx_range)) {
205 payloads.push_back(geometry_writer::to_payload(mask));
206 }
207
208 return payloads;
209 }
210 };

213 template <detray::concepts::algebra algebra_t>
214 struct get_material_link_payload {
215 template <typename material_group_t, typename index_t>
216 constexpr auto operator()(const material_group_t& /*unused*/,
217 const index_t& index) const {
218 using material_t = typename material_group_t::value_type;
219
220 // Find the correct material type index
221 if constexpr (detray::concepts::grid<material_t>) {
222 return detail::basic_converter::to_payload(
223 io::detail::get_id<material_t>(), index);
224 } else {
225 return detail::basic_converter::to_payload(
226 io::detail::get_id<algebra_t, material_t>(), index);
227 }
228 }
229 };
230
232 struct get_acc_link_payload {
233 template <typename acc_group_t, typename index_t>
234 constexpr auto operator()(const acc_group_t& /*unused*/,
235 const index_t& index) const {
236 using accel_t = typename acc_group_t::value_type;
237
238 auto id{acc_links_payload::type_id::unknown};
239
240 // Only convert grids
241 if constexpr (detray::concepts::surface_grid<accel_t>) {
242 id = io::detail::get_id<accel_t>();
243 }
244
245 return detail::basic_converter::to_payload(id, index);
246 }
247 };
