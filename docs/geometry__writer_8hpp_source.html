60 template <detray::concepts::detector detector_t>
61 static payload_type to_payload(const detector_t& det,
62 const typename detector_t::name_map& names) {
63 payload_type det_data;
64 det_data.volumes.reserve(det.volumes().size());
65
66 for (const auto& vol : det.volumes()) {
67 if (names.contains(vol.index())) {
68 det_data.volumes.push_back(to_payload(vol, det, names.at(vol.index())));
69 } else {
70 det_data.volumes.push_back(to_payload(vol, det, ""));
71 }
72 }
73
74 return det_data;
75 }

81 template <detray::concepts::detector detector_t>
82 static transform_payload to_payload(
83 const typename detector_t::transform3_type& trf) {
84 transform_payload trf_data;
85
86 const auto& t = trf.translation();
87 const auto& x = trf.x();
88 const auto& y = trf.y();
89 const auto& z = trf.z();
90
91 trf_data.tr = {t[0], t[1], t[2]};
92 trf_data.rot = {x[0], x[1], x[2], y[0], y[1], y[2], z[0], z[1], z[2]};
93
94 return trf_data;
95 }

101 template <typename mask_t>
102 requires(!std::is_same_v<typename mask_t::shape, void>)
103 static mask_payload to_payload(const mask_t& m) {
104 mask_payload mask_data;
105
106 mask_data.shape = io::detail::get_id<typename mask_t::shape>();
107
108 mask_data.volume_link =
109 detail::basic_converter::to_payload(m.volume_link());
110
111 mask_data.boundaries.resize(mask_t::boundaries::e_size);
112 std::ranges::copy(m.values(), std::begin(mask_data.boundaries));
113
114 return mask_data;

122 template <detray::concepts::detector detector_t>
123 static surface_payload to_payload(const geometry::surface<detector_t>& sf,
124 std::size_t sf_idx) {
125 using algebra_t = typename detector_t::algebra_type;
126 surface_payload sf_data;
127
128 sf_data.index_in_coll = sf_idx;
129 sf_data.type = sf.id();
130 sf_data.identifier = sf.identifier().value();
131 sf_data.transform = to_payload<detector_t>(sf.transform({}));
132 sf_data.masks = sf.template visit_mask<get_mask_link_payload>();
133 if (sf.has_material()) {
134 sf_data.material =
135 sf.template visit_material<get_material_link_payload<algebra_t>>();
136 }
137 sf_data.source = sf.source();
138
139 return sf_data;
140 }

148 template <detray::concepts::detector detector_t>
149 static volume_payload to_payload(
150 const typename detector_t::volume_type& vol_desc, const detector_t& det,
151 const std::string_view name) {
152 volume_payload vol_data;
153
154 vol_data.index = detail::basic_converter::to_payload(vol_desc.index());
155 vol_data.name = name;
156 vol_data.transform =
157 to_payload<detector_t>(det.transform_store().at(vol_desc.transform()));
158 vol_data.type = vol_desc.id();
159
160 // Count the surfaces belonging to this volume
161 std::size_t sf_idx{0};
162
163 for (const auto& sf_desc : det.surfaces()) {
164 if (sf_desc.volume() == vol_desc.index()) {
165 vol_data.surfaces.push_back(
166 to_payload(geometry::surface{det, sf_desc}, sf_idx++));
167 }
168 }
169
170 // Only run the query, if object type is contained in volume
171 const auto& link = vol_desc.accel_link();
172 // Initialize the std::optional
173 if (link.size() > 1u) {
174 vol_data.acc_links.emplace();
175 }
176 // Skip the first acceleration structure which exists in every volume
177 // and is handled automatically during detector building
178 for (unsigned int i = 1u; i < link.size(); ++i) {
179 const auto& l = link[i];
180 if (!l.is_invalid()) {
181 const auto aclp =
182 det.accelerator_store().template visit<get_acc_link_payload>(l);
183 vol_data.acc_links->push_back(aclp);
184 }
185 }
186
187 return vol_data;
188 }

192 struct get_mask_link_payload {
193 template <typename mask_group_t, typename idx_range_t>
194 inline auto operator()(const mask_group_t& mask_group,
195 const idx_range_t& idx_range) const {
196 std::vector<mask_payload> payloads{};
197
198 for (const auto& mask : detray::ranges::subrange(mask_group, idx_range)) {
199 payloads.push_back(geometry_writer::to_payload(mask));
200 }
201
202 return payloads;
203 }
204 };

207 template <detray::concepts::algebra algebra_t>
208 struct get_material_link_payload {
209 template <typename material_group_t, typename index_t>
210 constexpr auto operator()(const material_group_t& /*unused*/,
211 const index_t& index) const {
212 using material_t = typename material_group_t::value_type;
213
214 // Find the correct material type index
215 if constexpr (detray::concepts::grid<material_t>) {
216 return detail::basic_converter::to_payload(
217 io::detail::get_id<material_t>(), index);
218 } else {
219 return detail::basic_converter::to_payload(
220 io::detail::get_id<algebra_t, material_t>(), index);
221 }
222 }
223 };
224
226 struct get_acc_link_payload {
227 template <typename acc_group_t, typename index_t>
228 constexpr auto operator()(const acc_group_t& /*unused*/,
229 const index_t& index) const {
230 using accel_t = typename acc_group_t::value_type;
231
232 auto id{acc_links_payload::type_id::unknown};
233
234 // Only convert grids
235 if constexpr (detray::concepts::surface_grid<accel_t>) {
236 id = io::detail::get_id<accel_t>();
237 }
238
239 return detail::basic_converter::to_payload(id, index);
240 }
241 };
