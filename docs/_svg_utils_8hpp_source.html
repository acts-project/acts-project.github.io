60 std::tuple<actsvg::style::fill, actsvg::style::stroke> fillAndStroke() const {
61 actsvg::style::fill fll;
62 fll._fc._rgb = fillColor;
63 fll._fc._opacity = static_cast<actsvg::scalar>(fillOpacity);
64 fll._fc._hl_rgb = highlightColor;
65 fll._fc._highlight = highlights;
66
67 actsvg::style::stroke str;
68 str._sc._rgb = strokeColor;
69 str._sc._hl_rgb = highlightStrokeColor;
70 str._width = static_cast<actsvg::scalar>(strokeWidth);
71 str._hl_width = static_cast<actsvg::scalar>(highlightStrokeWidth);
72 str._dasharray = strokeDasharray;
73
74 return {fll, str};
75 }
76
79 std::tuple<actsvg::style::fill, actsvg::style::stroke, actsvg::style::font>
80 fillStrokeFont() const {
81 auto [fll, str] = fillAndStroke();
82
83 actsvg::style::font fnt;
84 fnt._size = fontSize;
85 fnt._fc._rgb = fontColor;
86
87 return std::tie(fll, str, fnt);
88 }
89};

147inline static actsvg::svg::object measure(double xStart, double yStart,
148 double xEnd, double yEnd,
149 const std::string& variable = "",
150 double value = 0.,
151 const std::string& unit = "") {
152 std::string mlabel = "";
153 if (!variable.empty()) {
154 mlabel = variable + " = ";
155 }
156 if (value != 0.) {
157 mlabel += actsvg::utils::to_string(static_cast<actsvg::scalar>(value));
158 }
159 if (!unit.empty()) {
160 mlabel += " ";
161 mlabel += unit;
162 }
163 return actsvg::draw::measure(
164 "measure",
165 {static_cast<actsvg::scalar>(xStart),
166 static_cast<actsvg::scalar>(yStart)},
167 {static_cast<actsvg::scalar>(xEnd), static_cast<actsvg::scalar>(yEnd)},
168 actsvg::style::stroke(), actsvg::style::marker({"o"}),
169 actsvg::style::marker({"|<<"}), actsvg::style::font(), mlabel);
170}

200inline static actsvg::svg::object infoBox(
201 double xPos, double yPos, const std::string& title, const Style& titleStyle,
202 const std::vector<std::string>& info, const Style& infoStyle,
203 const actsvg::svg::object& object,
204 const std::vector<std::string>& highlights = {"mouseover", "mouseout"}) {
205 auto [titleFill, titleStroke, titleFont] = titleStyle.fillStrokeFont();
206 auto [infoFill, infoStroke, infoFont] = infoStyle.fillStrokeFont();
207
208 actsvg::style::stroke stroke;
209
210 return actsvg::draw::connected_info_box(
211 object._id + "_infoBox",
212 {static_cast<actsvg::scalar>(xPos), static_cast<actsvg::scalar>(yPos)},
213 title, titleFill, titleFont, info, infoFill, infoFont, stroke, object,
214 highlights);
215}

222inline static void toFile(const std::vector<actsvg::svg::object>& objects,
223 const std::string& fileName) {
