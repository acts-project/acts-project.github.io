112 void assignSpacePointIndices(
113 std::span<const SpacePointIndex2> spacePointIndices)
114 requires(!ReadOnly)
115 {
116 if (m_index >= m_container->size()) {
117 throw std::out_of_range("Index out of range in SeedContainer2");
118 }
119 if (m_container->m_spacePointCounts[m_index] != 0) {
120 throw std::logic_error("Space points already assigned to the seed");
121 }
122
123 m_container->m_spacePointOffsets[m_index] =
124 static_cast<std::uint32_t>(m_container->m_spacePoints.size());
125 m_container->m_spacePointCounts[m_index] =
126 static_cast<std::uint8_t>(spacePointIndices.size());
127 m_container->m_spacePoints.insert(m_container->m_spacePoints.end(),
128 spacePointIndices.begin(),
129 spacePointIndices.end());
130 }
131
135 [[nodiscard]] std::size_t size() const noexcept {
136 return spacePointIndices().size();
137 }

141 [[nodiscard]]
142 bool empty() const noexcept {
143 return size() == 0;
144 }
