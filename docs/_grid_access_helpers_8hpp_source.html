30template <typename cast_container, typename Array, std::size_t... idx>
31void fillCasts(const Vector3& position, const cast_container& globalCasts,
32 Array& ra, std::index_sequence<idx...> /*indices*/) {
33 ((ra[idx] = VectorHelpers::cast(position, globalCasts[idx])), ...);
34}
48template <typename grid_type, typename cast_container>
49typename grid_type::point_t castPosition(const Vector3& position,
50 const cast_container& globalCasts) {
51 // Fill the grid point from global
52 typename grid_type::point_t casted{};
53 fillCasts(position, globalCasts, casted,
54 std::make_integer_sequence<std::size_t, grid_type::DIM>{});
55 return casted;
56}

65template <typename Array, typename local_indices, std::size_t... idx>
66void fillLocal(const Vector2& lposition, const local_indices& laccess,
67 Array& ra, std::index_sequence<idx...> /*indices*/) {
68 ((ra[idx] = lposition[laccess[idx]]), ...);
69}
70
80template <typename grid_type, typename local_indices>
81typename grid_type::point_t accessLocal(const Vector2& lposition,
82 const local_indices& laccess) {
83 if constexpr (grid_type::DIM > 2u) {
84 throw std::invalid_argument(
85 "GridAccessHelper: only 1-D and 2-D grids are possible for local "
86 "access.");
87 }
88 // Fill the grid point from local according to the accessors
89 typename grid_type::point_t accessed{};
90 fillLocal(lposition, laccess, accessed,
91 std::make_integer_sequence<std::size_t, grid_type::DIM>{});
92 return accessed;
93}

145template <AxisDirection... Args>
146class GlobalSubspace : public IGlobalToGridLocal {
147 public:
149 using grid_local_t = std::array<double, sizeof...(Args)>;
150
152 static_assert(sizeof...(Args) > 0,
153 "GlobalSubspace: cannot have an empty binning value list.");
154
156 static_assert(sizeof...(Args) <= 3,
157 "GlobalSubspace: cannot have more than 3 binning values.");
158
159 // Constructor
160 GlobalSubspace() = default;
161
163 static constexpr std::array<AxisDirection, sizeof...(Args)> axisDirs = {
164 Args...};
165
171 grid_local_t toGridLocal(const Vector3& position) const {
172 // Fill the grid point from global
173 grid_local_t glocal{};
174 GridAccessHelpers::fillCasts(
175 position, axisDirs, glocal,
176 std::make_integer_sequence<std::size_t, sizeof...(Args)>{});
177 return glocal;
178 }

181// The bound to grid local transformation, if only access of a subspace
182// is requested
184template <std::size_t... Args>
185class LocalSubspace : public IBoundToGridLocal {
186 public:
188 using grid_local_t = std::array<double, sizeof...(Args)>;
189
191 static_assert(sizeof...(Args) == 1 || sizeof...(Args) == 2,
192 "LocalSubspace: only 1 or 2 accessors are allowed.");
193
195 static_assert(((Args < 2) && ...),
196 "LocalSubspace: local access needs to be 0u or 1u");
197
198 // Constructor
199 LocalSubspace() = default;
200
202 static constexpr std::array<std::size_t, sizeof...(Args)> accessors = {
203 Args...};
204
210 grid_local_t toGridLocal(const Vector2& lposition) const {
211 // Fill the grid point from local according to the accessors
212 grid_local_t accessed{};
213 GridAccessHelpers::fillLocal(
214 lposition, accessors, accessed,
215 std::make_integer_sequence<std::size_t, sizeof...(Args)>{});
216 return accessed;
217 }
