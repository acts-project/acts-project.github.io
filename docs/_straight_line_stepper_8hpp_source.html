324 bool prepareCurvilinearState(State& state) const {
325 // test whether the accumulated path has still its initial value.
326 if (state.pathAccumulated != 0) {
327 return true;
328 }
329
330 // dr/ds :
331 state.derivative.template head<3>() = direction(state);
332 // dt / ds
333 state.derivative(eFreeTime) = fastHypot(
334 1., state.particleHypothesis.mass() / absoluteMomentum(state));
335 // d (dr/ds) / ds : == 0
336 state.derivative.template segment<3>(4) = Acts::Vector3::Zero().transpose();
337 // d qop / ds == 0
338 state.derivative(eFreeQOverP) = 0.;
339
340 return true;
341 }

406 // This is simply ignored if `nullptr`.
411 Result<double> step(State& state, Direction propDir,
412 const IVolumeMaterial* material) const {
413 static_cast<void>(material);
414
415 // use the adjusted step size
416 const auto h = state.stepSize.value() * propDir;
417 const auto m = state.particleHypothesis.mass();
418 const auto p = absoluteMomentum(state);
419 // time propagates along distance as 1/b = sqrt(1 + m²/p²)
420 const auto dtds = fastHypot(1., m / p);
421 // Update the track parameters according to the equations of motion
422 Vector3 dir = direction(state);
423 state.pars.template segment<3>(eFreePos0) += h * dir;
424 state.pars[eFreeTime] += h * dtds;
425
426 // Propagate the jacobian
427 if (state.covTransport) {
428 // The step transport matrix in global coordinates
429 FreeMatrix D = FreeMatrix::Identity();
430 D.block<3, 3>(0, 4) = SquareMatrix<3>::Identity() * h;
431 // Extend the calculation by the time propagation
432 // Evaluate dt/dlambda
433 D(3, 7) = h * m * m * state.pars[eFreeQOverP] / dtds;
434 // Set the derivative factor the time
435 state.derivative(3) = dtds;
436 // Update jacobian and derivative
437 state.jacTransport = D * state.jacTransport;
438 state.derivative.template head<3>() = dir;
439 }
440
441 // state the path length
442 state.pathAccumulated += h;
443 ++state.nSteps;
444 ++state.nStepTrials;
445
446 ++state.statistics.nAttemptedSteps;
447 ++state.statistics.nSuccessfulSteps;
448 if (propDir != Direction::fromScalarZeroAsPositive(h)) {
449 ++state.statistics.nReverseSteps;
450 }
451 state.statistics.pathLength += h;
452 state.statistics.absolutePathLength += std::abs(h);
453
454 return h;
455 }
