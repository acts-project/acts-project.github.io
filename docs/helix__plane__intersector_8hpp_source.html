69 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
70 const trajectory_type<algebra_t> &h, const dtransform3D<algebra_t> &trf,
71 const scalar_t /*unused*/ = 0.f) const {
72 using point3_t = dpoint3D<algebra_t>;
73 using vector3_t = dvector3D<algebra_t>;
74
75 // Surface normal
76 const vector3_t sn = trf.z();
77 // Surface translation
78 const point3_t st = trf.translation();
79
80 // For a plane parallel to the magnetic field, the projected helix is
81 // a circle. Reject planes outside that circle before attempting to find
82 // a root: bracket expansion cannot succeed when no intersection exists.
83 const vector3_t field = h.b_field();
84 const scalar_t curvature = -h.qop() * h.B();
85 if (vector::dot(sn, field) == 0.f && curvature != 0.f &&
86 h.radius() * h.B() >= 1e-6f) {
87 const vector3_t field_axis = vector::normalize(field);
88 // The transverse circle centre is offset from the initial position.
89 const vector3_t center_offset =
90 (1.f / curvature) * vector::cross(field_axis, h.dir());
91 const scalar_t center_distance = math::fabs(
92 vector::dot(sn, h.pos() - st) + vector::dot(sn, center_offset));
93 const scalar_t radius = h.radius();
94 // Keep near-tangent cases within the solver's tolerance, with an
95 // additional allowance for floating-point roundoff.
96 const scalar_t margin = convergence_tolerance +
97 64.f * std::numeric_limits<scalar_t>::epsilon() *
98 (center_distance + radius);
99 if (std::isfinite(center_distance) && std::isfinite(radius) &&
100 center_distance - radius > margin) {
101 constexpr scalar_t inv = detail::invalid_value<scalar_t>();
102 return {inv, point3_t{inv, inv, inv}, inv};
103 }
104 }
105
106 // Starting point on the helix for the Newton iteration
107 const vector3_t dist{st - h.pos(0.f)};
108 scalar_t denom{vector::dot(sn, h.dir(0.5f * vector::norm(dist)))};
109 scalar_t s_ini;
110 if (denom == 0.f) {
111 s_ini = vector::norm(dist);
112 } else {
113 s_ini = vector::dot(sn, dist) / denom;
114 }
115
117 auto plane_inters_func = [&h, &st, &sn](const scalar_t x) {
118 // f(s) = sn * (h.pos(s) - st) == 0
119 const scalar_t f_s{vector::dot(sn, (h.pos(x) - st))};
120 // f'(s) = sn * h.dir(s)
121 const scalar_t df_s{vector::dot(sn, h.dir(x))};
122
123 return std::make_tuple(f_s, df_s);
124 };
125
126 // Run the root finding algorithm
127 scalar_t s;
128 scalar_t ds;
129 if (run_rtsafe) {
130 std::tie(s, ds) =
131 newton_raphson_safe(plane_inters_func, s_ini, convergence_tolerance,
132 max_n_tries, max_path);
133 } else {
134 std::tie(s, ds) =
135 newton_raphson(plane_inters_func, s_ini, convergence_tolerance,
136 max_n_tries, max_path);
137 }
138
139 return {s, h.pos(s), ds};
140 }
141
143 scalar_t convergence_tolerance{1.f * unit<scalar_t>::um};
145 std::size_t max_n_tries{1000u};
147 scalar_t max_path{5.f * unit<scalar_t>::m};
149 bool run_rtsafe{true};
150};
