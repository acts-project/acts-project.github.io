60template <typename... output_converter_ts>
61void convert_from_payload(detray::io::detector_payload& payload,
62 const detray::io::detector_writer_config& cfg) {
63 static_assert(sizeof...(output_converter_ts) > 0);
64
65 // Check that the required geometry data is there
66 if (payload.geometry.volumes.empty()) {
67 const bool has_grids{payload.surface_grids.has_value() &&
68 payload.surface_grids->is_valid()};
69 const bool has_hom_mat{payload.homogeneous_material.has_value() &&
70 payload.homogeneous_material->is_valid()};
71 const bool has_mat_maps{payload.material_maps.has_value() &&
72 payload.material_maps->is_valid()};
73 if (!has_grids && !has_hom_mat && !has_mat_maps) {
74 std::string err{"Empty payload!"};
75 DETRAY_FATAL_HOST(err);
76 throw std::invalid_argument(err);
77 }
78 DETRAY_WARN_HOST(
79 "No geometry data found in detector payload: Converting single "
80 "detector components...");
81 }
82
83 // In case the conversion is started from an external payload:
84 // Fill the sub-headers (no further detector data available at this level)
85 detail::fill_sub_header_payloads(payload, cfg);
86 payload.header.source = cfg.source();
87
88 // How to open the file
89 const std::ios_base::openmode out_mode{std::ios_base::out |
90 std::ios_base::binary};
91 const auto mode =
92 cfg.replace_files() ? (out_mode | std::ios_base::trunc) : out_mode;
93
94 const auto file_path = detray::io::create_path(cfg.path());
95
96 if (cfg.compactify_json()) {
97 DETRAY_ERROR_HOST("Compactifying json files is not yet implemented");
98 }
99
100 // Hold all required output converters
101 detray::io::detail::detector_components_output_converter payload_converter{};
102
103 // Register the converters that match the output format
104 (detray::io::detail::add_output_file_converters<output_converter_ts>(
105 payload_converter, cfg, file_path, mode),
106 ...);
107
108 // Make sure that the detector data will be written
109 if (payload_converter.converters().empty()) {
110 const std::string err_msg{
111 "No converter registered for the configured output format: Please "
112 "check that the output format matches the converter types!"};
113 DETRAY_FATAL_HOST(err_msg);
114 throw std::invalid_argument(err_msg);
115 }
116
117 payload_converter.convert(payload);
118}

133template <class detector_t, typename... output_converter_ts>
134void write_detector(const detector_t& det,
135 const typename detector_t::name_map& names,
136 detray::io::detector_writer_config& cfg,
137 detray::io::detector_payload& payload) noexcept(false) {
138 DETRAY_INFO_HOST("Writing detray detector...");
139
140 if (det.material_store().all_empty() &&
141 !payload.homogeneous_material.has_value() &&
142 !payload.material_maps.has_value()) {
143 cfg.write_material(false);
144 }
145
146 // Collect all data about the detector in the detector payload
147 detray::io::detail::detector_components_writer<detector_t>
148 detector_components;
149 detector_components.write(det, names, cfg, payload);
150
151 // Convert the detector payload to the output format
152 if constexpr (sizeof...(output_converter_ts) > 0) {
153 // Check that the required geometry data is there
154 if (payload.geometry.volumes.empty()) {
155 const std::string err_msg{"No geometry data found in detector payload"};
156 DETRAY_FATAL_HOST(err_msg);
157 throw std::runtime_error(err_msg);
158 }
159
160 detray::io::convert_from_payload<output_converter_ts...>(payload, cfg);
161 } else {
162 DETRAY_INFO_HOST(
163 "No output converters given to 'detector_writer'. Only the payload "
164 "will be filled...");
165 }
166
167 DETRAY_INFO_HOST("Done writing: " << det.name(names)
168 << " (Source: " << cfg.source() << ")");
169}

179template <class detector_t, typename... output_converter_ts>
180void write_detector(const detector_t& det,
181 const typename detector_t::name_map& names,
182 detray::io::detector_writer_config& cfg) noexcept(false) {
183 // No converters present: Output files cannot be written
184 if constexpr (sizeof...(output_converter_ts) == 0) {
185 const std::string err_msg{
186 "No converters compiled into the 'write_detector' function: Cannot "
187 "write output files!\nUse e.g. the 'write_detector_json' function for "
188 "json files or add custom output converter types to 'write_detector'."};
189 DETRAY_FATAL_HOST(err_msg);
190 throw std::invalid_argument(err_msg);
191 }
192
193 // Empty payload to be filled: All data comes from the detector
194 detray::io::detector_payload payload{};
195
196 detray::io::write_detector<detector_t, output_converter_ts...>(det, names,
197 cfg, payload);
198}
