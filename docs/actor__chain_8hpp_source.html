34template <concepts::actor... actors_t>
35class actor_chain {
36 public:
38 using actor_tuple = dtuple<actors_t...>;
39
41 using state_tuple =
42 detail::unique_t<detail::tuple_cat_t<detail::state_tuple_t<actors_t>...>>;

45 using state_ref_tuple = detail::unique_t<

52 template <typename actor_states_t, typename propagator_state_t>
53 DETRAY_HOST_DEVICE void operator()(actor_states_t &states,
54 propagator_state_t &p_state) const {
55 DETRAY_VERBOSE_HOST_DEVICE("Run actors...");
56 run(states, p_state, std::make_index_sequence<sizeof...(actors_t)>{});
57 }
58
60 DETRAY_HOST_DEVICE constexpr const actor_tuple &actors() const {
61 return m_actors;
62 }
63
65 DETRAY_HOST_DEVICE
66 static constexpr auto make_default_actor_states() {
67 // Only possible if each state is default initializable (including
68 // obsevers to the actors in actors_t)
69 if constexpr (detail::tuple_all_v<std::is_default_constructible,
70 state_tuple>) {
71 return state_tuple();
72 } else {
73 return std::nullopt;
74 }
75 }
76
80 DETRAY_HOST_DEVICE static constexpr state_ref_tuple setup_actor_states(
81 state_tuple &t) {
82 return setup_actor_states(
83 t, std::make_index_sequence<detail::tuple_size_v<state_tuple>>{});
84 }
85
86 private:
92 template <concepts::actor actor_t, typename actor_states_t,
93 typename propagator_state_t>
94 DETRAY_HOST_DEVICE inline void run(const actor_t &actr,
95 actor_states_t &states,
96 propagator_state_t &p_state) const {
97 if constexpr (!typename actor_t::is_comp_actor()) {
98 // No actor state defined (empty)
99 if constexpr (std::same_as<typename actor_t::state,
100 detray::base_actor::state>) {
101 actr(p_state);
102 } else {
103 actr(detail::get<typename actor_t::state &>(states), p_state);
104 }
105 } else {
106 actr(states, p_state);
107 }
108 }
109
114 template <typename actor_states_t, typename propagator_state_t,
115 std::size_t... indices>
116 DETRAY_HOST_DEVICE inline void run(
117 actor_states_t &states, propagator_state_t &p_state,
118 std::index_sequence<indices...> /*ids*/) const {
119 (run(detail::get<indices>(m_actors), states, p_state), ...);
120 }
121
123 template <std::size_t... indices>
124 DETRAY_HOST_DEVICE static constexpr state_ref_tuple setup_actor_states(
125 state_tuple &t, std::index_sequence<indices...> /*ids*/) {
126 return detray::tie(detail::get<indices>(t)...);
127 }
128
130 DETRAY_NO_UNIQUE_ADDRESS actor_tuple m_actors = {};
131};
