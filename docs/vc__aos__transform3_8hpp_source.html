149 DETRAY_HOST_DEVICE
150 transform3(const matrix44 &m, const matrix44 &m_inv)
151 : _data{m}, _data_inv{m_inv} {
152 // The assertion will not hold for (casts to) int
153 if constexpr (std::floating_point<scalar_type>) {
154 [[maybe_unused]] constexpr auto epsilon{
155 std::numeric_limits<float>::epsilon()};
156 assert(algebra::approx_equal(invert(m), m_inv, 16.f * epsilon, 1e-6f));
157 }
158 }

163 DETRAY_HOST_DEVICE
164 explicit transform3(const array_type<16> &ma) {
165 // The values that are not set here, are known to be zero or one
166 // and never used explicitly
167 _data[e_x] = column_t{ma[0], ma[4], ma[8]};
168 _data[e_y] = column_t{ma[1], ma[5], ma[9]};
169 _data[e_z] = column_t{ma[2], ma[6], ma[10]};
170 _data[e_t] = column_t{ma[3], ma[7], ma[11]};
171
172 _data_inv = invert(_data);
173 }

211 DETRAY_HOST_DEVICE
212 constexpr scalar_type determinant(const matrix44 &m) const {
213 return -m[e_z][0] * m[e_y][1] * m[e_x][2] +
214 m[e_y][0] * m[e_z][1] * m[e_x][2] +
215 m[e_z][0] * m[e_x][1] * m[e_y][2] -
216 m[e_x][0] * m[e_z][1] * m[e_y][2] -
217 m[e_y][0] * m[e_x][1] * m[e_z][2] +
218 m[e_x][0] * m[e_y][1] * m[e_z][2];
219 }
220
226 DETRAY_HOST_DEVICE
227 constexpr matrix44 invert(const matrix44 &m) const {
228 matrix44 i;
229 i[e_x][0] = -m[e_z][1] * m[e_y][2] + m[e_y][1] * m[e_z][2];
230 i[e_x][1] = m[e_z][1] * m[e_x][2] - m[e_x][1] * m[e_z][2];
231 i[e_x][2] = -m[e_y][1] * m[e_x][2] + m[e_x][1] * m[e_y][2];
232 // i[e_x][3] = 0;
233 i[e_y][0] = m[e_z][0] * m[e_y][2] - m[e_y][0] * m[e_z][2];
234 i[e_y][1] = -m[e_z][0] * m[e_x][2] + m[e_x][0] * m[e_z][2];
235 i[e_y][2] = m[e_y][0] * m[e_x][2] - m[e_x][0] * m[e_y][2];
236 // i[e_y][3] = 0;
237 i[e_z][0] = -m[e_z][0] * m[e_y][1] + m[e_y][0] * m[e_z][1];
238 i[e_z][1] = m[e_z][0] * m[e_x][1] - m[e_x][0] * m[e_z][1];
239 i[e_z][2] = -m[e_y][0] * m[e_x][1] + m[e_x][0] * m[e_y][1];
240 // i[e_z][3] = 0;
241 i[e_t][0] =
242 m[e_t][0] * m[e_z][1] * m[e_y][2] - m[e_z][0] * m[e_t][1] * m[e_y][2] -
243 m[e_t][0] * m[e_y][1] * m[e_z][2] + m[e_y][0] * m[e_t][1] * m[e_z][2] +
244 m[e_z][0] * m[e_y][1] * m[e_t][2] - m[e_y][0] * m[e_z][1] * m[e_t][2];
245 i[e_t][1] =
246 m[e_z][0] * m[e_t][1] * m[e_x][2] - m[e_t][0] * m[e_z][1] * m[e_x][2] +
247 m[e_t][0] * m[e_x][1] * m[e_z][2] - m[e_x][0] * m[e_t][1] * m[e_z][2] -
248 m[e_z][0] * m[e_x][1] * m[e_t][2] + m[e_x][0] * m[e_z][1] * m[e_t][2];
249 i[e_t][2] =
250 m[e_t][0] * m[e_y][1] * m[e_x][2] - m[e_y][0] * m[e_t][1] * m[e_x][2] -
251 m[e_t][0] * m[e_x][1] * m[e_y][2] + m[e_x][0] * m[e_t][1] * m[e_y][2] +
252 m[e_y][0] * m[e_x][1] * m[e_t][2] - m[e_x][0] * m[e_y][1] * m[e_t][2];
253 // i[e_t][3] = 1;
254 const scalar_type idet{scalar_type(1.f) / determinant(i)};
255
256 i[e_x] = i[e_x] * idet;
257 i[e_y] = i[e_y] * idet;
258 i[e_z] = i[e_z] * idet;
259 i[e_t] = i[e_t] * idet;
260
261 return i;
262 }

269 template <concepts::vector2D vector2_type>
270 DETRAY_HOST_DEVICE constexpr auto rotate(const matrix44 &m,
271 const vector2_type &v) const {
272 return m[e_x] * v[0] + m[e_y] * v[1];
273 }
274
280 template <concepts::vector3D vector3_type>
281 DETRAY_HOST_DEVICE constexpr auto rotate(const matrix44 &m,
282 const vector3_type &v) const {
283 return m[e_x] * v[0] + m[e_y] * v[1] + m[e_z] * v[2];
284 }

288 DETRAY_HOST_DEVICE
289 constexpr auto rotation() const {
290 using matrix_t = algebra::storage::matrix<array_t, scalar_type, 3u, 3u>;
291
292 matrix_t submatrix;
293 for (unsigned int icol = 0; icol < 3; ++icol) {
294 submatrix[icol] = _data[icol];
295 }
296 return submatrix;
297 }
