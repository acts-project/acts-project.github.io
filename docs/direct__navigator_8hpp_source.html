37template <concepts::detector detector_t, typename surface_sequence_t = void>
38class direct_navigator {
39 using algebra_t = typename detector_t::algebra_type;
40 using scalar_t = dscalar<algebra_t>;
41
42 public:
44 using detector_type = detector_t;
46 using context_type = detector_type::geometry_context;
47
48 // Use an external surface sequence type (rather than a single surface type),
49 // perigee/external surface flows instantiate direct_navigator with
50 // single_view-like range wrappers.
52 using surface_sequence_type = std::conditional_t<
53 std::same_as<surface_sequence_t, void>,
54 vecmem::device_vector<typename detector_t::surface_type>,
55 surface_sequence_t>;
56
57 static_assert(detray::ranges::range<surface_sequence_type>,
58 "Surface sequence must be iterable");
59
61 using surface_type = detray::ranges::range_value_t<surface_sequence_type>;
63 using intersection_type =
64 intersection2D<surface_type, algebra_t, !intersection::contains_pos>;
66 using inspector_type = navigation::void_inspector;
67
69 class state
70 : public navigation::base_state<state, detector_type, 2u, inspector_type,
71 intersection_type> {
72 friend class direct_navigator;
73
74 template <typename state_t>
75 friend constexpr void navigation::update_status(state_t &,
76 const navigation::config &);
77
78 using base_type = navigation::base_state<state, detector_type, 2u,
79 inspector_type, intersection_type>;
80
81 // Internal surface index
82 using dist_t = std::int_least16_t;
83
84 public:
86 using value_type = intersection_type;
88 using sequence_type = surface_sequence_type;
89
91 using view_type =
92 std::conditional_t<std::same_as<surface_sequence_t, void>,
93 detail::get_view_t<surface_sequence_type>,
94 surface_sequence_type>;
96 using const_view_type =
97 std::conditional_t<std::same_as<surface_sequence_t, void>,
98 detail::get_view_t<const surface_sequence_type>,
99 surface_sequence_type>;

105 DETRAY_HOST_DEVICE constexpr state(const detector_t &det,
106 const view_type &sequence)
107 : base_type(det), m_sequence(sequence) {
108 assert(!m_sequence.empty());
109 reset();
110 }
111
113 DETRAY_HOST_DEVICE void reset() {
114 // Remove old navigation information
115 clear_cache();
116
117 // Set the index into the external surface sequence to the beginning
118 // or end of the container
119 m_next_external = is_forward()
120 ? dist_t{0}
121 : static_cast<dist_t>(m_sequence.size() - 1u);
122
123 // Update the target with the next external surface
124 this->target().set_surface(next_external());
125
126 // The next candidate is always stored in the second cache entry
127 this->next_index(1u);
128 this->last_index(1u);
129 }
130
132 DETRAY_HOST_DEVICE
133 constexpr auto n_candidates() const -> dindex { return 1u; }
134
136 DETRAY_HOST_DEVICE
137 constexpr scalar_t external_tol() const {
138 return std::numeric_limits<scalar_t>::max();
139 }
140
143 DETRAY_HOST_DEVICE
144 constexpr void set_direction(const navigation::direction dir) {
145 base_type::set_direction(dir);
146 }
147
149 DETRAY_HOST_DEVICE
150 constexpr bool has_next_external() const {
151 return (is_forward() &&
152 m_next_external != static_cast<dist_t>(m_sequence.size())) ||
153 (!is_forward() && m_next_external != -1);
154 }
155
158 DETRAY_HOST_DEVICE
159 constexpr surface_type next_external() {
160 assert(has_next_external());
161 return m_sequence[static_cast<unsigned int>(m_next_external)];
162 }
163
165 DETRAY_HOST_DEVICE
166 constexpr void advance() {
167 // The target has become the current candidate
168 this->candidates()[0] = this->target();
169
170 assert(has_next_external());
171 if (is_forward()) {
172 m_next_external++;
173 } else {
174 m_next_external--;
175 }
176
177 // Update the target with the next external surface
178 // Could make the target invalid -> exit navigation
179 if (has_next_external()) {
180 // If the next external can be indexed, the cast is safe
181 this->target().set_surface(
182 m_sequence[static_cast<unsigned int>(m_next_external)]);
183 }
184
185 assert(
186 !has_next_external() ||
187 (has_next_external() && (this->target().surface().is_sensitive() ||
188 this->target().surface().has_material())));
189 }
190
192 DETRAY_HOST_DEVICE constexpr void clear_cache() {
193 base_type::clear_cache();
194 this->next_index(1);
195 this->last_index(1);
196 }
197
199 DETRAY_HOST_DEVICE
200 constexpr bool finished() const {
201 // Normal exit for this navigation?
202 bool is_finished = base_type::finished();
203
204 // All external surfaces were visited?
205 is_finished =
206 is_finished &&
207 ((is_forward() &&
208 m_next_external == static_cast<dist_t>(m_sequence.size())) ||
209 (!is_forward() && m_next_external == -1));
210
211 return is_finished;
212 }
213
214 private:
216 DETRAY_HOST_DEVICE
217 constexpr bool is_forward() const {
218 return this->direction() == navigation::direction::e_forward;
219 }
220
222 sequence_type m_sequence;
223
225 dist_t m_next_external{0};
226 };

233 template <typename track_t>
234 DETRAY_HOST_DEVICE inline void init(const track_t &track, state &navigation,
235 const navigation::config &cfg,
236 const context_type &ctx) const {
237 DETRAY_VERBOSE_HOST_DEVICE("Called 'init()':");
238 assert(navigation.has_next_external());
239
240 // Clean up state
241 navigation.clear_cache();
242
243 // Update the next candidate
244 update(track, navigation, cfg, ctx);
245
246 DETRAY_VERBOSE_HOST_DEVICE("Init complete!");
247 }

256 template <typename track_t>
257 DETRAY_HOST_DEVICE inline bool update(const track_t &track, state &navigation,
258 const navigation::config &cfg,
259 const context_type &ctx = {}) const {
260 DETRAY_VERBOSE_HOST_DEVICE("Called 'update()'");
261 DETRAY_DEBUG_HOST(" -> Trust level: " << navigation.trust_level());
262
263 // Do not resurrect a failed/finished navigation state
264 assert(navigation.is_alive() ||
265 navigation.status() == navigation::status::e_unknown);
266 assert(!track.is_invalid());
267
268 // If the last external was reached, the navigation is finished
269 constexpr bool is_init{true};
270 if (!navigation.has_next_external()) {
271 DETRAY_VERBOSE_HOST_DEVICE("No next target in surface sequence: Exit");
272 navigation.exit();
273 return !is_init;
274 }
275
276 const detector_type &det = navigation.detector();
277
278 // Update only the current candidate and the corresponding next target
279 if (navigation.trust_level() != navigation::trust_level::e_full) {
280 // Use infinite tolerance to hit the target surface
281 constexpr auto inf{std::numeric_limits<float>::max()};
282 const intersection::config intr_cfg{
283 inf, inf, inf, cfg.intersection.path_tolerance, -inf};
284
285 // Tangential to the track direction
286 const detray::detail::ray<algebra_t> tangential{
287 track.pos(),
288 static_cast<scalar_t>(navigation.direction()) * track.dir()};
289
290 // Update the current target. If it cannot be reached, direct
291 // navigation is broken
292 if (!update_candidate(tangential, det, navigation, intr_cfg, ctx)) {
293 // If update fails but the track is already on the target within
294 // tolerance, treat the target as reached instead of aborting.
295 if (target_reached_within_tolerance(track, navigation, cfg)) {
296 navigation.target().set_path(static_cast<scalar_t>(0.f));
297 navigation.target().set_status(intersection::status::e_inside);
298 } else {
299 navigation.abort("Could not reach current target");
300 return !is_init;
301 }
302 }
303
304 // Update navigation flow on the new candidate information and set
305 // the next target if the surface has been reached
306 navigation::update_status(navigation, cfg);
307 // Set full trust, even if a portal was reached (no volume switch)
308 navigation.trust_level(navigation::trust_level::e_full);
309
310 // The work is done if the track has not reached the surface yet
311 if (navigation.status() == navigation::status::e_towards_object) {
312 DETRAY_VERBOSE_HOST_DEVICE("Update complete (towards object):");
313 DETRAY_DEBUG_HOST("\n"
314 << detray::navigation::print_state(navigation)
315 << detray::navigation::print_candidates(
316 navigation, cfg, track.pos(), track.dir()));
317
318 return !is_init;
319 }
320
321 // Otherwise, track is on surface: Update the next target
322 if (navigation.has_next_external() &&
323 !update_candidate(tangential, det, navigation, intr_cfg, ctx)) {
324 navigation.abort("Could not find new target after surface was reached");
325 return !is_init;
326 }
327
328 // At this point, the track has to be on surface:
329 // Set volume index to the volume the current surface is in
330 navigation.set_volume(navigation.current().surface().volume());
331
332 // Set full trust again: no volume switch needed
333 navigation.trust_level(navigation::trust_level::e_full);
334
335 DETRAY_VERBOSE_HOST_DEVICE("Update complete (on surface):");
336 DETRAY_DEBUG_HOST("\n"
337 << detray::navigation::print_state(navigation)
338 << detray::navigation::print_candidates(
339 navigation, cfg, track.pos(), track.dir()));
340
341 // Return true to reset the step size of the RKN algorithm
342 return is_init;
343 }
344
345 DETRAY_VERBOSE_HOST_DEVICE(" -> Full trust: Nothing left to do");
346
347 return !is_init;
348 }

350 private:
353 DETRAY_HOST_DEVICE DETRAY_INLINE constexpr bool update_candidate(
354 const detray::detail::ray<algebra_t> &tangential, const detector_t &det,
355 state &navigation, const intersection::config &intr_cfg,
356 [[maybe_unused]] const context_type &ctx) const {
357 if constexpr (std::same_as<surface_sequence_t, void>) {
358 // Update a detector surface.
359 return navigation::update_candidate(navigation.target(), tangential, det,
360 intr_cfg, navigation.external_tol(),
361 ctx);
362 } else {
363 // Update a standalone surface.
364 return update_external_candidate(navigation.target(), tangential,
365 navigation.next_external(), intr_cfg,
366 navigation.external_tol());
367 }
368 }
369
371 DETRAY_HOST_DEVICE DETRAY_INLINE constexpr bool update_external_candidate(
372 typename state::value_type &candidate,
373 const detray::detail::ray<algebra_t> &tangential, const surface_type &sf,
374 const intersection::config &cfg,
375 const scalar_t external_mask_tolerance) const {
376 constexpr ray_intersector<typename surface_type::mask_type::shape,
377 algebra_t, !intersection::contains_pos>
378 intersector{};
379
380 // Intersect the standalone surface.
381 typename decltype(intersector)::result_type result{};
382 if constexpr (concepts::cylindrical<typename surface_type::mask_type>) {
383 result = intersector.point_of_intersection(
384 tangential, sf.transform(), sf.mask(), cfg.overstep_tolerance);
385 } else {
386 result = intersector.point_of_intersection(tangential, sf.transform(),
387 cfg.overstep_tolerance);
388 }
389
390 // Build resulting intersection candidate(s).
391 if constexpr (decltype(intersector)::n_solutions > 1) {
392 resolve_mask(candidate, tangential, result[0], candidate.surface(),
393 sf.mask(), sf.transform(), cfg, external_mask_tolerance);
394 } else {
395 resolve_mask(candidate, tangential, result, candidate.surface(),
396 sf.mask(), sf.transform(), cfg, external_mask_tolerance);
397 }
398
399 return candidate.is_probably_inside();
400 }

403 template <typename track_t>
404 DETRAY_HOST_DEVICE DETRAY_INLINE constexpr bool
405 target_reached_within_tolerance(const track_t &track, const state &navigation,
406 const navigation::config &cfg) const {
407 if constexpr (std::same_as<surface_sequence_t, void>) {
408 return false;
409 } else {
410 const surface_type target_sf = navigation.target().surface();
411 const auto inside = target_sf.mask().is_inside(
412 target_sf.transform(), track.pos(),
413 static_cast<scalar_t>(cfg.intersection.path_tolerance));
414
415 if constexpr (std::is_scalar_v<decltype(inside)>) {
416 return inside;
417 } else {
418 return detray::detail::any_of(inside);
419 }
420 }
421 }
