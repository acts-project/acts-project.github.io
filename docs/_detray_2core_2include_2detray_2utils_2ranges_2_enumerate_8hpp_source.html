34template <std::input_iterator range_itr_t, std::incrementable incr_t>
35class enumerate_view : public detray::ranges::view_interface<
36 enumerate_view<range_itr_t, incr_t>> {
37 private:
43 struct iterator {
44 using itr_value_t = std::iter_value_t<range_itr_t>;
45 using itr_ref_t = std::iter_reference_t<range_itr_t>;
46 using itr_ptr_t = typename std::iterator_traits<range_itr_t>::pointer;
47
48 using difference_type = std::iter_difference_t<range_itr_t>;
49 using value_type = std::pair<incr_t, itr_ref_t>;
50 using pointer = value_type *;
51 using reference = value_type;
52 using iterator_category =
53 typename std::iterator_traits<range_itr_t>::iterator_category;
54
55 constexpr iterator()
56 requires std::default_initializable<range_itr_t>
57 = default;
58
59 DETRAY_HOST_DEVICE
60 explicit iterator(range_itr_t iter, incr_t offset = 0)
61 : m_iter(iter), m_i{offset}, m_offset{offset} {};
62
65 DETRAY_HOST_DEVICE constexpr auto operator++() -> iterator & {
66 ++m_i;
67 ++m_iter;
68 return *this;
69 }
70
71 DETRAY_HOST_DEVICE constexpr auto operator++(int) -> iterator {
72 auto tmp(*this);
73 ++(*this);
74 return tmp;
75 }
77
80 DETRAY_HOST_DEVICE constexpr auto operator--() -> iterator &
81 requires std::bidirectional_iterator<range_itr_t>
82 {
83 --m_i;
84 --m_iter;
85 return *this;
86 }
87
88 DETRAY_HOST_DEVICE constexpr auto operator--(int) -> iterator
89 requires std::bidirectional_iterator<range_itr_t>
90 {
91 auto tmp(*this);
92 ++(*this);
93 return tmp;
94 }
96
98 DETRAY_HOST_DEVICE constexpr auto operator*() const {
99 return std::pair<incr_t, itr_ref_t>(m_i, *m_iter);
100 }
101
104 DETRAY_HOST_DEVICE constexpr auto operator+=(const difference_type j)
105 -> iterator &
106 requires std::random_access_iterator<range_itr_t>
107 {
108 m_iter += j;
109 m_i += static_cast<incr_t>(j);
110 return *this;
111 }
112
115 DETRAY_HOST_DEVICE constexpr auto operator-=(const difference_type j)
116 -> iterator &
117 requires std::random_access_iterator<range_itr_t>
118 {
119 m_iter -= j;
120 m_i -= static_cast<incr_t>(j);
121 return *this;
122 }
123
125 DETRAY_HOST_DEVICE constexpr auto operator[](const difference_type i) const
126 requires std::random_access_iterator<range_itr_t>
127 {
128 // check narrowing
129 const incr_t index{m_offset + static_cast<incr_t>(i)};
130 return std::pair<incr_t, itr_ref_t>(index, m_iter[i]);
131 }
132
133 private:
135 DETRAY_HOST_DEVICE
136 friend constexpr auto operator==(const iterator &lhs, const iterator &rhs)
137 -> bool {
138 return (lhs.m_iter == rhs.m_iter);
139 }
140
142 DETRAY_HOST_DEVICE friend constexpr auto operator!=(const iterator &lhs,
143 const iterator &rhs)
144 -> bool {
145 return (lhs.m_iter != rhs.m_iter);
146 }
147
149 DETRAY_HOST_DEVICE
150 friend constexpr auto operator<=>(const iterator &lhs, const iterator &rhs)
151 requires detray::ranges::random_access_iterator<range_itr_t>
152 {
153#if defined(__apple_build_version__)
154 const auto l{lhs.m_iter};
155 const auto r{rhs.m_iter};
156 if (l < r || (l == r && l < r)) {
157 return std::strong_ordering::less;
158 }
159 if (l > r || (l == r && l > r)) {
160 return std::strong_ordering::greater;
161 }
162 return std::strong_ordering::equivalent;
163#else
164 return (lhs.m_iter <=> rhs.m_iter);
165#endif
166 }
167
172 DETRAY_HOST_DEVICE friend constexpr auto operator+(const iterator &itr,
173 const difference_type j)
174 -> iterator
175 requires std::random_access_iterator<range_itr_t>
176 {
177 return {itr.m_iter + j, itr.m_i + static_cast<incr_t>(j)};
178 }
179
184 DETRAY_HOST_DEVICE friend constexpr auto operator+(const difference_type j,
185 const iterator &itr)
186 -> iterator
187 requires std::random_access_iterator<range_itr_t>
188 {
189 return itr + j;
190 }
191
196 DETRAY_HOST_DEVICE friend constexpr auto operator-(const iterator &itr,
197 const difference_type j)
198 -> iterator
199 requires std::random_access_iterator<range_itr_t>
200 {
201 return itr + (-j);
202 }
203
206 DETRAY_HOST_DEVICE friend constexpr auto operator-(const iterator &lhs,
207 const iterator &rhs)
208 -> difference_type
209 requires std::random_access_iterator<range_itr_t>
210 {
211 return lhs.m_iter - rhs.m_iter;
212 }
213
214 range_itr_t m_iter{};
215 incr_t m_i{0};
216 incr_t m_offset{0};
217 };

234 DETRAY_HOST_DEVICE constexpr explicit enumerate_view(range_t &&rng)
235 : m_begin{detray::ranges::begin(rng)},
236 m_end{detray::ranges::end(rng), static_cast<incr_t>(rng.size())} {}
237
241 template <detray::ranges::range range_t>
242 DETRAY_HOST_DEVICE constexpr enumerate_view(range_t &&rng, incr_t start)
243 : m_begin{detray::ranges::begin(rng), start},
244 m_end{detray::ranges::end(rng),
245 start + static_cast<incr_t>(rng.size())} {}
246
248 DETRAY_HOST_DEVICE
249 constexpr auto begin() -> iterator { return m_begin; }
250
252 DETRAY_HOST_DEVICE
253 constexpr auto begin() const -> iterator { return m_begin; }
254
256 DETRAY_HOST_DEVICE
257 constexpr auto end() const -> iterator { return m_end; }
258};
