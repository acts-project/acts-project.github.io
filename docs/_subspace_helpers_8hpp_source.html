36template <std::ranges::sized_range index_range_t>
37inline static bool checkSubspaceIndices(const index_range_t& indexRange,
38 std::size_t fullSize,
39 std::size_t subspaceSize) {
40 if (subspaceSize > fullSize) {
41 return false;
42 }
43 if (static_cast<std::size_t>(indexRange.size()) != subspaceSize) {
44 return false;
45 }
46 for (auto it = indexRange.begin(); it != indexRange.end();) {
47 auto index = *it;
48 if (index >= fullSize) {
49 return false;
50 }
51 ++it;
52 if (const auto tail = std::ranges::subrange(it, indexRange.end());
53 std::ranges::find(tail, index) != tail.end()) {
54 return false;
55 }
56 }
57 return true;
58}

67template <std::size_t FullSize>
68inline static SerializedSubspaceIndices serializeSubspaceIndices(
69 const SubspaceIndices<FullSize>& indices)
70 requires(FullSize <= 8)
71{
72 SerializedSubspaceIndices result = 0;
73 for (std::size_t i = 0; i < FullSize; ++i) {
74 result |= static_cast<SerializedSubspaceIndices>(indices[i] & 0xFF)
75 << (i * 8);
76 }
77 return result;
78}

87template <std::size_t FullSize>
88inline static SubspaceIndices<FullSize> deserializeSubspaceIndices(
89 SerializedSubspaceIndices serialized)
90 requires(FullSize <= 8)
91{
92 SubspaceIndices<FullSize> result;
93 for (std::size_t i = 0; i < FullSize; ++i) {
94 result[i] = static_cast<std::uint8_t>((serialized >> (i * 8)) & 0xFF);
95 }
96 return result;
97}

106template <typename Derived, std::size_t FullSize>
107class SubspaceHelperBase {
108 public:
109 static constexpr std::size_t kFullSize = FullSize;
110
111 using FullSquareMatrix = SquareMatrix<kFullSize>;
112
113 using size_type = std::size_t;
114
115 bool empty() const { return self().empty(); }
116 std::size_t size() const { return self().size(); }
117
118 auto operator[](std::size_t i) const { return self()[i]; }
119
120 auto begin() const { return self().begin(); }
121 auto end() const { return self().end(); }
122
123 bool contains(std::uint8_t index) const {
124 const auto r = std::ranges::subrange(begin(), end());
125 return std::ranges::find(r, index) != r.end();
126 }
127 std::size_t indexOf(std::uint8_t index) const {
128 const auto r = std::ranges::subrange(begin(), end());
129 const auto it = std::ranges::find(r, index);
130 return it != r.end()
131 ? static_cast<std::size_t>(std::ranges::distance(r.begin(), it))
132 : kFullSize;
133 }

135 template <typename EigenDerived>
136 Vector<kFullSize> expandVector(
137 const Eigen::DenseBase<EigenDerived>& vector) const {
138 Vector<kFullSize> result = Vector<kFullSize>::Zero();
139 for (auto [i, index] : enumerate(*this)) {
140 result(index) = vector(i);
141 }
142 return result;
143 }
144
145 template <typename EigenDerived>
146 FullSquareMatrix expandMatrix(
147 const Eigen::DenseBase<EigenDerived>& matrix) const {
148 FullSquareMatrix result = FullSquareMatrix::Zero();
149 for (auto [i, indexI] : enumerate(*this)) {
150 for (auto [j, indexJ] : enumerate(*this)) {
151 result(indexI, indexJ) = matrix(i, j);
152 }
153 }
154 return result;
155 }
156
157 FullSquareMatrix fullProjector() const {
158 FullSquareMatrix result = FullSquareMatrix::Zero();
159 for (auto [i, index] : enumerate(*this)) {
160 result(i, index) = 1;
161 }
162 return result;
163 }
164
165 FullSquareMatrix fullExpander() const {
166 FullSquareMatrix result = FullSquareMatrix::Zero();
167 for (auto [i, index] : enumerate(*this)) {
168 result(index, i) = 1;
169 }
170 return result;
171 }
172
173 ProjectorBitset projectorBitset() const
174 requires(kFullSize <= 8)
175 {
176 return matrixToBitset(fullProjector()).to_ullong();
177 }
178
179 private:
180 const Derived& self() const { return static_cast<const Derived&>(*this); }
181};

189template <std::size_t FullSize, typename index_t = std::uint8_t>
190class VariableSubspaceHelper
191 : public detail::SubspaceHelperBase<
192 VariableSubspaceHelper<FullSize, index_t>, FullSize> {
193 public:
195 static constexpr std::size_t kFullSize = FullSize;
196
198 using IndexType = index_t;
200 using Container = boost::container::static_vector<IndexType, FullSize>;
201
206 template <std::ranges::sized_range other_index_range_t>
207 explicit VariableSubspaceHelper(const other_index_range_t& indices) {
208 assert(checkSubspaceIndices(indices, kFullSize, indices.size()) &&
209 "Invalid indices");
210 m_indices.resize(indices.size());
211 std::ranges::transform(indices, m_indices.begin(), [](auto index) {
212 return static_cast<IndexType>(index);
213 });
214 }
215
218 bool empty() const { return m_indices.empty(); }
221 std::size_t size() const { return m_indices.size(); }
224 const Container& indices() const { return m_indices; }
225
229 IndexType operator[](std::size_t i) const { return m_indices[i]; }
230
233 auto begin() const { return m_indices.begin(); }
236 auto end() const { return m_indices.end(); }
237
238 private:
239 Container m_indices;
240};

430template <std::size_t kFullSize, typename Derived>
431SubspaceIndices<kFullSize> projectorToSubspaceIndices(
432 const Eigen::DenseBase<Derived>& projector) {
433 auto rows = static_cast<std::size_t>(projector.rows());
434 auto cols = static_cast<std::size_t>(projector.cols());
435 assert(cols == kFullSize && rows <= kFullSize && "Invalid projector size");
436 SubspaceIndices<kFullSize> result;
437 result.fill(kFullSize);
438 for (std::size_t i = 0; i < rows; ++i) {
439 for (std::size_t j = 0; j < cols; ++j) {
440 assert((projector(i, j) == 0 || projector(i, j) == 1) &&
441 "Invalid projector value");
442 if (projector(i, j) == 1) {
443 result[i] = j;
444 }
445 }
446 }
447 return result;
448}
