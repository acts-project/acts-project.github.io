66 DETRAY_HOST_DEVICE
67 inline void advance_track() {
68 auto& track = (*this)();
69 track.set_pos(track.pos() + track.dir() * this->step_size());
70
71 this->update_path_lengths(this->step_size());
72 }
73
75 DETRAY_HOST_DEVICE
76 inline void advance_jacobian() {
77 DETRAY_VERBOSE_HOST_DEVICE("Advance Jacobian");
78
79 // The step transport matrix in global coordinates
80 free_matrix<algebra_t> D = matrix::identity<free_matrix<algebra_t>>();
81
82 // d(x,y,z)/d(n_x,n_y,n_z)
83 matrix_type<3, 3> dxdn =
84 this->step_size() * matrix::identity<matrix_type<3, 3>>();
85 getter::set_block(D, dxdn, e_free_pos0, e_free_dir0);
86
89 this->transport_jacobian() = D * this->transport_jacobian();
90 }

113 DETRAY_HOST_DEVICE bool step(
114 const scalar_type dist_to_next, state& stepping,
115 const stepping::config& cfg, const bool /*unused*/ = true,
116 const material<scalar_type>* /*unused*/ = nullptr) const {
117 // In case of an overlap do nothing
118 if (math::fabs(dist_to_next) <= 1.f * unit<scalar_type>::um) {
119 stepping.run_inspector(cfg, "Step skipped (Overlap): ", dist_to_next);
120 return true;
121 }
122
123 // Straight line stepping: The distance given by the navigator is exact
124 stepping.set_step_size(dist_to_next);
125
126 DETRAY_VERBOSE_HOST_DEVICE("Take step: %f mm", stepping.step_size());
127
128 // Check constraints
129 if (const scalar_type max_step =
130 stepping.constraints().template size<>(stepping.direction());
131 math::fabs(stepping.step_size()) > math::fabs(max_step)) {
132 // Run inspection before step size is cut
133 stepping.run_inspector(cfg, "Before constraint: ", dist_to_next);
134
135 stepping.set_step_size(max_step);
136 }
137
138 // Update track state
139 stepping.advance_track();
140
141 // Advance jacobian transport
142 if (cfg.do_covariance_transport) {
143 stepping.advance_jacobian();
144 }
145
146 // Count the number of steps
147 stepping.count_trials();
148
149 // Run inspection if needed
150 stepping.run_inspector(cfg, "Step complete: ", dist_to_next);
151
152 return true;
153 }
154};
