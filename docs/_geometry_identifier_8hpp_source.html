201 Value m_value = 0;
202
204 static constexpr int extractShift(Value mask) {
205 // use compiler builtin to extract the number of trailing bits from the
206 // mask. the builtin should be available on all supported compilers.
207 // need unsigned long long version (...ll) to ensure std::uint64_t
208 // compatibility.
209 // WARNING undefined behaviour for mask == 0 which we should not have.
210 return __builtin_ctzll(mask);
211 }
212
213 constexpr static Value getMaxValue(Value mask) {
214 return mask >> extractShift(mask);
215 }
216
218 constexpr Value getBits(Value mask) const {
219 return (m_value & mask) >> extractShift(mask);
220 }
222 constexpr GeometryIdentifier& setBits(Value mask, Value id) {
223 if (id > getMaxValue(mask)) {
224 throw std::invalid_argument(
225 "Value " + std::to_string(id) + " exceeds maximum value " +
226 std::to_string(getMaxValue(mask)) + " for this field");
227 }
228
229 m_value = (m_value & ~mask) | ((id << extractShift(mask)) & mask);
230 // return *this here that we need to write fewer lines in the setXXX
231 // methods
232 return *this;
233 }
234
235 friend constexpr bool operator==(GeometryIdentifier lhs,
236 GeometryIdentifier rhs) {

238 }
239
240 friend constexpr bool operator<(GeometryIdentifier lhs,
241 GeometryIdentifier rhs) {
242 return lhs.m_value < rhs.m_value;
243 }
