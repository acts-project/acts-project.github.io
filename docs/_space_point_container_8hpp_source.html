675 static bool reservedColumn(const std::string &name) noexcept;
676
677 template <typename Holder>
678 MutableColumnProxy<typename Holder::Value> createColumnImpl(
679 const std::string &name) {
680 if (reservedColumn(name)) {
681 throw std::runtime_error("Column name is reserved: " + name);
682 }
683 if (hasColumn(name)) {
684 throw std::runtime_error("Column already exists: " + name);
685 }
686 auto holder = std::make_unique<Holder>();
687 holder->resize(size());
688 auto proxy = holder->proxy(*this);
689 m_allColumns.try_emplace(name, holder.get());
690 m_dynamicColumns.try_emplace(name, std::move(holder));
691 return proxy;
692 }
693
694 template <typename Holder, typename Self>
695 static auto columnImpl(Self &&self, const std::string &name) {
696 const auto it = self.m_allColumns.find(name);
697 if (it == self.m_allColumns.end()) {
698 throw std::runtime_error("Column not found: " + name);
699 }
700 auto &holder = dynamic_cast<Holder &>(*it->second);
701 return holder.proxy(self);
702 }
703
704 template <typename Holder>
705 MutableColumnProxy<typename Holder::Value> columnImpl(
706 const std::string &name) {
707 return columnImpl<Holder>(*this, name);
708 }
709
710 template <typename Holder>
711 ConstColumnProxy<typename Holder::Value> columnImpl(
712 const std::string &name) const {
713 return columnImpl<Holder>(*this, name);
714 }
