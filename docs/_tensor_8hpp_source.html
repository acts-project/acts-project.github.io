127template <Acts::Concepts::arithmetic T>
128class Tensor {
129 public:
131 using Shape = std::array<std::size_t, 2>;
132
137 static Tensor Create(Shape shape, const ExecutionContext &execContext) {
138 auto ptr = detail::createTensorMemory(shape[0] * shape[1] * sizeof(T),
139 execContext);
140 return Tensor(shape, std::move(ptr), execContext);
141 }
142
148 Tensor clone(const ExecutionContext &to) const {
149 auto clonedPtr = detail::cloneTensorMemory(m_ptr, nbytes(), m_device, to);
150 return Tensor(m_shape, std::move(clonedPtr), to);
151 }
152
155 T *data() { return static_cast<T *>(m_ptr.get()); }
156
159 const T *data() const { return static_cast<const T *>(m_ptr.get()); }
160
163 Shape shape() const { return m_shape; }
164
167 std::size_t size() const { return m_shape[0] * m_shape[1]; }
168
171 std::size_t nbytes() const { return size() * sizeof(T); }
172
175 Device device() const { return m_device; }
176
177 private:
178 Tensor(Shape shape, detail::TensorPtr ptr, const ExecutionContext &ctx)
179 : m_shape(shape), m_ptr(std::move(ptr)), m_device(ctx.device) {}
180
181 Shape m_shape{};
182 detail::TensorPtr m_ptr;
183 Device m_device{};
184};

197template <Acts::Concepts::arithmetic T>
198void checkMaskCompatibility(const Tensor<T> &tensor, const Tensor<bool> &mask,
199 std::size_t matchDim) {
200 if (tensor.shape()[matchDim] != mask.shape()[0]) {
201 throw std::invalid_argument(
202 "tensor shape[" + std::to_string(matchDim) +
203 "]=" + std::to_string(tensor.shape()[matchDim]) +
204 " does not match mask length " + std::to_string(mask.shape()[0]));
205 }
206 if (mask.shape()[1] != 1) {
207 throw std::invalid_argument("mask must have shape [N, 1], got [N, " +
208 std::to_string(mask.shape()[1]) + "]");
209 }
210 if (tensor.device() != mask.device()) {
211 throw std::invalid_argument(
212 "tensor and mask must reside on the same device");
213 }
214}
