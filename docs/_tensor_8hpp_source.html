119template <Acts::Concepts::arithmetic T>
120class Tensor {
121 public:
123 using Shape = std::array<std::size_t, 2>;
124
129 static Tensor Create(Shape shape, const ExecutionContext &execContext) {
130 auto ptr = detail::createTensorMemory(shape[0] * shape[1] * sizeof(T),
131 execContext);
132 return Tensor(shape, std::move(ptr), execContext);
133 }
134
140 Tensor clone(const ExecutionContext &to) const {
141 auto clonedPtr = detail::cloneTensorMemory(m_ptr, nbytes(), m_device, to);
142 return Tensor(m_shape, std::move(clonedPtr), to);
143 }
144
147 T *data() { return static_cast<T *>(m_ptr.get()); }
148
151 const T *data() const { return static_cast<const T *>(m_ptr.get()); }
152
155 Shape shape() const { return m_shape; }
156
159 std::size_t size() const { return m_shape[0] * m_shape[1]; }
160
163 std::size_t nbytes() const { return size() * sizeof(T); }
164
167 Device device() const { return m_device; }
168
169 private:
170 Tensor(Shape shape, detail::TensorPtr ptr, const ExecutionContext &ctx)
171 : m_shape(shape), m_ptr(std::move(ptr)), m_device(ctx.device) {}
172
173 Shape m_shape{};
174 detail::TensorPtr m_ptr;
175 Device m_device{};
176};

189template <Acts::Concepts::arithmetic T>
190void checkMaskCompatibility(const Tensor<T> &tensor, const Tensor<bool> &mask,
191 std::size_t matchDim) {
192 if (tensor.shape()[matchDim] != mask.shape()[0]) {
193 throw std::invalid_argument(
194 "tensor shape[" + std::to_string(matchDim) +
195 "]=" + std::to_string(tensor.shape()[matchDim]) +
196 " does not match mask length " + std::to_string(mask.shape()[0]));
197 }
198 if (mask.shape()[1] != 1) {
199 throw std::invalid_argument("mask must have shape [N, 1], got [N, " +
200 std::to_string(mask.shape()[1]) + "]");
201 }
202 if (tensor.device() != mask.device()) {
203 throw std::invalid_argument(
204 "tensor and mask must reside on the same device");
205 }
206}
