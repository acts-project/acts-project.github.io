175concept HasBarrelEndcapClassifier = requires(
176 const BackendT& backend, const typename BackendT::Element& element) {
177 { backend.isBarrel(element) } -> std::same_as<bool>;
178 { backend.isEndcap(element) } -> std::same_as<bool>;
179 { backend.isTracker(element) } -> std::same_as<bool>;

193template <typename BackendT>
194concept BlueprintBackend =
195 HasSurfaceFactory<BackendT> && HasLayerNameMember<BackendT> &&
196 requires(const typename BackendT::Config& cfg, const Acts::Logger& logger,
197 const BackendT& backend,
198 const typename BackendT::Element& element) {
199 BackendT{cfg, logger};
200 { BackendT::kIdentifier } -> std::convertible_to<std::string_view>;
201 { backend.world() } -> std::same_as<typename BackendT::Element>;
202 { backend.nameOf(element) } -> std::same_as<std::string>;
203 {
204 backend.children(element)
205 } -> std::same_as<std::vector<typename BackendT::Element>>;
206 { backend.parent(element) } -> std::same_as<typename BackendT::Element>;
207 { backend.isSensitive(element) } -> std::same_as<bool>;
208 requires requires(const typename BackendT::Element& a,
209 const typename BackendT::Element& b) {
210 { a == b } -> std::convertible_to<bool>;
211 };
212 };
