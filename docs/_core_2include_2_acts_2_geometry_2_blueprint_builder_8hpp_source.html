194template <typename BackendT>
195concept BlueprintBackend =
196 HasSurfaceFactory<BackendT> && HasLayerNameMember<BackendT> &&
197 requires(const typename BackendT::Config& cfg, const Acts::Logger& logger,
198 const BackendT& backend,
199 const typename BackendT::Element& element) {
200 BackendT{cfg, logger};
201 { BackendT::kIdentifier } -> std::convertible_to<std::string_view>;
202 { backend.world() } -> std::same_as<typename BackendT::Element>;
203 { backend.nameOf(element) } -> std::same_as<std::string>;
204 {
205 backend.children(element)
206 } -> std::same_as<std::vector<typename BackendT::Element>>;
207 { backend.parent(element) } -> std::same_as<typename BackendT::Element>;
208 { backend.isSensitive(element) } -> std::same_as<bool>;
209 requires requires(const typename BackendT::Element& a,
210 const typename BackendT::Element& b) {
211 { a == b } -> std::convertible_to<bool>;
212 };
213 };
