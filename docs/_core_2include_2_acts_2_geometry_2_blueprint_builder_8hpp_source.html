200template <typename BackendT>
201concept BlueprintBackend =
202 HasSurfaceFactory<BackendT> && HasLayerNameMember<BackendT> &&
203 requires(const typename BackendT::Config& cfg, const Acts::Logger& logger,
204 const BackendT& backend,
205 const typename BackendT::Element& element) {
206 BackendT{cfg, logger};
207 { BackendT::kIdentifier } -> std::convertible_to<std::string_view>;
208 { backend.world() } -> std::same_as<typename BackendT::Element>;
209 { backend.nameOf(element) } -> std::same_as<std::string>;
210 {
211 backend.children(element)
212 } -> std::same_as<std::vector<typename BackendT::Element>>;
213 { backend.parent(element) } -> std::same_as<typename BackendT::Element>;
214 { backend.isSensitive(element) } -> std::same_as<bool>;
215 requires requires(const typename BackendT::Element& a,
216 const typename BackendT::Element& b) {
217 { a == b } -> std::convertible_to<bool>;
218 };
