62template <concepts::scalar scalar_t>
63struct stepper_default_policy : public base_actor {
65 struct state {
67 scalar_t tol{std::numeric_limits<scalar_t>::epsilon()};
68 };
69
74 template <typename propagator_state_t>
75 DETRAY_HOST_DEVICE inline void operator()(
76 state &pol_state, propagator_state_t &propagation) const {
77 const auto &stepping = propagation.stepping();
78 auto &navigation = propagation.navigation();
79
80 // Not a severe change to track state expected
81 if (math::fabs(stepping.step_size()) <
82 math::fabs(
83 stepping.constraints().template size<>(stepping.direction())) -
84 pol_state.tol) {
85 // Re-evaluate only next candidate
86 navigation.set_high_trust();
87 }
88 // Step size hit a constraint - the track state was probably changed a
89 // lot
90 else {
91 // Re-evaluate all candidates
92 navigation.set_fair_trust();
93 }
94 }
95};

100template <concepts::scalar scalar_t>
101struct stepper_rk_policy : public base_actor {
103 struct state {
105 scalar_t m_threshold_fair_trust{0.05f};
107 scalar_t m_threshold_no_trust{0.1f};
108 };
109
114 template <typename propagator_state_t>
115 DETRAY_HOST_DEVICE inline void operator()(
116 const state &pol_state, propagator_state_t &propagation) const {
117 const auto &stepping = propagation.stepping();
118 auto &navigation = propagation.navigation();
119
120 // In case of an overlap, have navigator re-evaluate the next candidate
121 if (math::fabs(navigation()) <= 1.f * unit<float>::um) {
122 navigation.set_high_trust();
123 return;
124 }
125
126 // How strongly did the RKN algorithm reduce the step size?
127 const scalar_t rel_correction{(stepping.step_size() - navigation()) /
128 navigation()};
129
130 // Large correction to the stepsize - re-initialize the volume
131 if (rel_correction > pol_state.m_threshold_no_trust) {
132 navigation.set_no_trust();
133 }
134 // Medium correction - re-evaluate all current candidates
135 else if (rel_correction > pol_state.m_threshold_fair_trust) {
136 navigation.set_fair_trust();
137 } else {
138 // Small correction - re-evaluate only the next candidate
139 navigation.set_high_trust();
140 }
141 }
142};
