128 DETRAY_HOST_DEVICE constexpr auto full_sf_range() const ->
129 typename sf_link_type::index_type {
130 using idx_range_t = typename sf_link_type::index_type;
131
132 const auto& pt_range = sf_link<surface_id::e_portal>();
133 const auto& sen_range = sf_link<surface_id::e_sensitive>();
134 const auto& psv_range = sf_link<surface_id::e_passive>();
135
136 // Portal range is never empty
137 dindex min{detail::get<0>(pt_range)};
138 dindex max{detail::get<1>(pt_range)};
139
140 constexpr idx_range_t empty{};
141 if (sen_range != empty) {
142 min = detail::get<0>(sen_range) < min ? detail::get<0>(sen_range) : min;
143 max = detail::get<1>(sen_range) > max ? detail::get<1>(sen_range) : max;
144 }
145
146 if (psv_range != empty) {
147 min = detail::get<0>(psv_range) < min ? detail::get<0>(psv_range) : min;
148 max = detail::get<1>(psv_range) > max ? detail::get<1>(psv_range) : max;
149 }
150
151 return idx_range_t{min, max};
152 }

155 DETRAY_HOST_DEVICE constexpr dindex n_surfaces() const {
156 const auto sf_idx_range = full_sf_range();
157
158 assert(sf_idx_range[1] > sf_idx_range[0]);
159 return sf_idx_range[1] - sf_idx_range[0];

164 DETRAY_HOST_DEVICE constexpr dindex to_local_sf_index(dindex sf_idx) const {
165 auto full_range = full_sf_range();
166
167 assert(full_range[0] <= sf_idx);
168 assert(sf_idx < full_range[1]);
169
170 return sf_idx - full_range[0];
171 }

175 DETRAY_HOST_DEVICE constexpr dindex to_global_sf_index(dindex sf_idx) const {

192 template <surface_id id>
193 DETRAY_HOST auto update_sf_link(
194 const typename sf_link_type::index_type& other) noexcept -> void {
195 auto& rg = sf_link<id>();
196 // Range not set yet - initialize
197 if (constexpr typename sf_link_type::index_type empty{}; rg == empty) {
198 rg = other;
199 } else {
200 // Update upper border
201 assert(detail::get<1>(rg) == detail::get<0>(other));
202 rg.set_upper(other.upper());
203 }
204 }
205
214 template <surface_id id>
215 DETRAY_HOST auto update_sf_link(std::size_t shift,
216 std::size_t n_surfaces = 0) noexcept -> void {
217 auto& rg = sf_link<id>();
218 // Range not set yet - initialize
219 if (constexpr typename sf_link_type::index_type empty{}; rg == empty) {
220 rg = {0u, static_cast<dindex>(n_surfaces)};
221 }
222 // Update
223 rg.shift(static_cast<dindex>(shift));
224 }
