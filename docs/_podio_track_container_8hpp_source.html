115 template <bool EnsureConst, typename T>
116 static std::any component_impl(T& instance, Acts::HashedString key,
117 IndexType itrack) {
118 if constexpr (EnsureConst) {
119 static_assert(std::is_const_v<std::remove_reference_t<T>>,
120 "Is not const");
121 }
122
123 using namespace Acts::HashedStringLiteral;
124 auto track = instance.m_collection->at(itrack);
125 std::conditional_t<EnsureConst, const ActsPodioEdm::TrackInfo*,
126 ActsPodioEdm::TrackInfo*>
127 dataPtr;
128 if constexpr (EnsureConst) {
129 dataPtr = &track.getData();
130 } else {
131 dataPtr = &PodioUtil::getDataMutable(track);
132 }
133 auto& data = *dataPtr;
134 switch (key) {
135 case "tipIndex"_hash:
136 return &data.tipIndex;
137 case "stemIndex"_hash:
138 return &data.stemIndex;
139 case "params"_hash:
140 return data.parameters.data();
141 case "cov"_hash:
142 return data.covariance.data();
143 case "nMeasurements"_hash:
144 return &data.nMeasurements;
145 case "nHoles"_hash:
146 return &data.nHoles;
147 case "chi2"_hash:
148 return &data.chi2;
149 case "ndf"_hash:
150 return &data.ndf;
151 case "nOutliers"_hash:
152 return &data.nOutliers;
153 case "nSharedHits"_hash:
154 return &data.nSharedHits;
155 default:
156 auto it = instance.m_dynamic.find(key);
157 if (it == instance.m_dynamic.end()) {
158 throw std::runtime_error("Unable to handle this component");
159 }
160
161 std::conditional_t<EnsureConst,
162 const podio_detail::ConstDynamicColumnBase*,
163 podio_detail::DynamicColumnBase*>
164 col = it->second.get();
165 assert(col && "Dynamic column is null");
166 return col->get(itrack);
167 }
168 }

185 template <typename T>
186 static Acts::ParticleHypothesis particleHypothesis_impl(T& instance,
187 IndexType itrack) {
188 auto track = instance.m_collection->at(itrack);
189 const auto& src = track.getParticleHypothesis();
190 return Acts::ParticleHypothesis{static_cast<Acts::PdgParticle>(src.absPdg),
191 src.mass, Acts::AnyCharge{src.absQ}};
192 }
193
198 static void populateSurfaceBuffer(
199 const PodioUtil::ConversionHelper& helper,
200 const ActsPodioEdm::TrackCollection& collection,
201 std::vector<std::shared_ptr<const Acts::Surface>>& surfaces) noexcept {
202 surfaces.reserve(collection.size());
203 for (ActsPodioEdm::Track track : collection) {
204 surfaces.push_back(PodioUtil::convertSurfaceFromPodio(
205 helper, track.getReferenceSurface()));
206 }
207 }
208
210 std::reference_wrapper<const PodioUtil::ConversionHelper> m_helper;
212 std::vector<std::shared_ptr<const Acts::Surface>> m_surfaces;
213};

422 void releaseInto(podio::Frame& frame, const std::string& suffix = "")
423 requires(std::is_same_v<holder_t<ActsPodioEdm::TrackCollection>,
424 Acts::ValueHolder<ActsPodioEdm::TrackCollection>> ||
425 std::is_same_v<holder_t<ActsPodioEdm::TrackCollection>,
426 std::unique_ptr<ActsPodioEdm::TrackCollection>>)
427 {
428 std::string s = suffix;
429 if (!s.empty()) {
430 s = "_" + s;
431 }
432
433 if constexpr (std::is_same_v<
434 holder_t<ActsPodioEdm::TrackCollection>,
435 std::unique_ptr<ActsPodioEdm::TrackCollection>>) {
436 frame.put(std::move(m_collection), "tracks" + s);
437 } else {
438 frame.put(std::make_unique<ActsPodioEdm::TrackCollection>(
439 std::move(*m_collection)),
440 "tracks" + s);
441 }
442 m_surfaces.clear();
443
444 for (const auto& [key, col] : m_dynamic) {
445 col->releaseInto(frame, "tracks" + s + "_extra__");
446 }

593 static const ActsPodioEdm::TrackCollection* getTrackCollectionFromFrame(
594 const podio::Frame& frame, const std::string& suffix) {
595 std::string s = suffix.empty() ? suffix : "_" + suffix;
596 std::string tracksKey = "tracks" + s;
597
598 std::vector<std::string> available = frame.getAvailableCollections();
599 if (!Acts::rangeContainsValue(available, tracksKey)) {
600 throw std::runtime_error{"Track collection '" + tracksKey +
601 "' not found in frame"};
602 }
603
604 const auto* collection = frame.get(tracksKey);
605
606 const ActsPodioEdm::TrackCollection* d = nullptr;
607 if (const auto* casted =
608 dynamic_cast<const ActsPodioEdm::TrackCollection*>(collection);
609 casted != nullptr) {
610 d = casted;
611 } else {
612 throw std::runtime_error{"Unable to get collection " + tracksKey};
613 }
614
615 return d;
616 }
