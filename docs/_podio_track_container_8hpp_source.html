97 template <bool EnsureConst, typename T>
98 static std::any component_impl(T& instance, Acts::HashedString key,
99 IndexType itrack) {
100 using namespace Acts::HashedStringLiteral;
101 if constexpr (EnsureConst) {
102 static_assert(std::is_const_v<std::remove_reference_t<T>>,
103 "Is not const");
104 }
105
106 using namespace Acts::HashedStringLiteral;
107 auto track = instance.m_collection->at(itrack);
108 std::conditional_t<EnsureConst, const ActsPodioEdm::TrackInfo*,
109 ActsPodioEdm::TrackInfo*>
110 dataPtr;
111 if constexpr (EnsureConst) {
112 dataPtr = &track.getData();
113 } else {
114 dataPtr = &PodioUtil::getDataMutable(track);
115 }
116 auto& data = *dataPtr;
117 switch (key) {
118 case "tipIndex"_hash:
119 return &data.tipIndex;
120 case "stemIndex"_hash:
121 return &data.stemIndex;
122 case "params"_hash:
123 return data.parameters.data();
124 case "cov"_hash:
125 return data.covariance.data();
126 case "nMeasurements"_hash:
127 return &data.nMeasurements;
128 case "nHoles"_hash:
129 return &data.nHoles;
130 case "chi2"_hash:
131 return &data.chi2;
132 case "ndf"_hash:
133 return &data.ndf;
134 case "nOutliers"_hash:
135 return &data.nOutliers;
136 case "nSharedHits"_hash:
137 return &data.nSharedHits;
138 default:
139 auto it = instance.m_dynamic.find(key);
140 if (it == instance.m_dynamic.end()) {
141 throw std::runtime_error("Unable to handle this component");
142 }
143
144 std::conditional_t<EnsureConst,
145 const podio_detail::ConstDynamicColumnBase*,
146 podio_detail::DynamicColumnBase*>
147 col = it->second.get();
148 assert(col && "Dynamic column is null");
149 return col->get(itrack);
150 }
151 }
152
156 template <typename T>
157 static auto dynamicKeys_impl(T& instance) {
158 using column_type =
159 typename decltype(instance.m_dynamic)::mapped_type::element_type;
160 return Acts::detail::DynamicKeyRange<column_type>{
161 instance.m_dynamic.begin(), instance.m_dynamic.end()};
162 }
163
168 template <typename T>
169 static Acts::ParticleHypothesis particleHypothesis_impl(T& instance,
170 IndexType itrack) {
171 auto track = instance.m_collection->at(itrack);
172 const auto& src = track.getParticleHypothesis();
173 return Acts::ParticleHypothesis{static_cast<Acts::PdgParticle>(src.absPdg),
174 src.mass, Acts::AnyCharge{src.absQ}};
175 }
176
181 static void populateSurfaceBuffer(
182 const PodioUtil::ConversionHelper& helper,
183 const ActsPodioEdm::TrackCollection& collection,
184 std::vector<std::shared_ptr<const Acts::Surface>>& surfaces) noexcept {
185 surfaces.reserve(collection.size());
186 for (ActsPodioEdm::Track track : collection) {
187 surfaces.push_back(PodioUtil::convertSurfaceFromPodio(
188 helper, track.getReferenceSurface()));
189 }
190 }
191
193 std::reference_wrapper<const PodioUtil::ConversionHelper> m_helper;
195 std::vector<std::shared_ptr<const Acts::Surface>> m_surfaces;
196};

437class ConstPodioTrackContainer : public PodioTrackContainerBase {
438 public:
442 ConstPodioTrackContainer(const PodioUtil::ConversionHelper& helper,
443 const ActsPodioEdm::TrackCollection& collection)
444 : PodioTrackContainerBase{helper}, m_collection{&collection} {
445 // Not much we can do to recover dynamic columns here
446 populateSurfaceBuffer(m_helper, *m_collection, m_surfaces);
447 }
448
453 ConstPodioTrackContainer(const PodioUtil::ConversionHelper& helper,
454 const podio::Frame& frame,
455 const std::string& suffix = "")
456 : PodioTrackContainerBase{helper} {
457 std::string s = suffix.empty() ? suffix : "_" + suffix;
458 std::string tracksKey = "tracks" + s;
459
460 std::vector<std::string> available = frame.getAvailableCollections();
461 if (!Acts::rangeContainsValue(available, tracksKey)) {
462 throw std::runtime_error{"Track collection '" + tracksKey +
463 "' not found in frame"};
464 }
465
466 const auto* collection = frame.get(tracksKey);
467
468 if (const auto* d =
469 dynamic_cast<const ActsPodioEdm::TrackCollection*>(collection);
470 d != nullptr) {
471 m_collection = d;
472 } else {
473 throw std::runtime_error{"Unable to get collection " + tracksKey};
474 }
475
476 populateSurfaceBuffer(m_helper, *m_collection, m_surfaces);
477
478 podio_detail::recoverDynamicColumns(frame, tracksKey, m_dynamic);
479 }
