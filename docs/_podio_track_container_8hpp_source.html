112 template <bool EnsureConst, typename T>
113 static std::any component_impl(T& instance, Acts::HashedString key,
114 IndexType itrack) {
115 if constexpr (EnsureConst) {
116 static_assert(std::is_const_v<std::remove_reference_t<T>>,
117 "Is not const");
118 }
119
120 using namespace Acts::HashedStringLiteral;
121 auto track = instance.m_collection->at(itrack);
122 std::conditional_t<EnsureConst, const ActsPodioEdm::TrackInfo*,
123 ActsPodioEdm::TrackInfo*>
124 dataPtr;
125 if constexpr (EnsureConst) {
126 dataPtr = &track.getData();
127 } else {
128 dataPtr = &PodioUtil::getDataMutable(track);
129 }
130 auto& data = *dataPtr;
131 switch (key) {
132 case "tipIndex"_hash:
133 return &data.tipIndex;
134 case "stemIndex"_hash:
135 return &data.stemIndex;
136 case "params"_hash:
137 return data.parameters.data();
138 case "cov"_hash:
139 return data.covariance.data();
140 case "nMeasurements"_hash:
141 return &data.nMeasurements;
142 case "nHoles"_hash:
143 return &data.nHoles;
144 case "chi2"_hash:
145 return &data.chi2;
146 case "ndf"_hash:
147 return &data.ndf;
148 case "nOutliers"_hash:
149 return &data.nOutliers;
150 case "nSharedHits"_hash:
151 return &data.nSharedHits;
152 default:
153 auto it = instance.m_dynamic.find(key);
154 if (it == instance.m_dynamic.end()) {
155 throw std::runtime_error("Unable to handle this component");
156 }
157
158 std::conditional_t<EnsureConst,
159 const podio_detail::ConstDynamicColumnBase*,
160 podio_detail::DynamicColumnBase*>
161 col = it->second.get();
162 assert(col && "Dynamic column is null");
163 return col->get(itrack);
164 }
165 }

174 return Acts::detail::DynamicKeyRange<column_type>{

182 template <typename T>
183 static Acts::ParticleHypothesis particleHypothesis_impl(T& instance,
184 IndexType itrack) {
185 auto track = instance.m_collection->at(itrack);
186 const auto& src = track.getParticleHypothesis();
187 return Acts::ParticleHypothesis{static_cast<Acts::PdgParticle>(src.absPdg),
188 src.mass, Acts::ChargeHypothesis{src.absQ}};
189 }
190
195 static void populateSurfaceBuffer(
196 const PodioUtil::ConversionHelper& helper,
197 const ActsPodioEdm::TrackCollection& collection,
198 std::vector<std::shared_ptr<const Acts::Surface>>& surfaces) noexcept {
199 surfaces.reserve(collection.size());
200 for (ActsPodioEdm::Track track : collection) {
201 surfaces.push_back(PodioUtil::convertSurfaceFromPodio(
202 helper, track.getReferenceSurface()));
203 }
204 }
205
207 std::reference_wrapper<const PodioUtil::ConversionHelper> m_helper;
209 std::vector<std::shared_ptr<const Acts::Surface>> m_surfaces;
210};

419 void releaseInto(podio::Frame& frame, const std::string& suffix = "")
420 requires(std::is_same_v<holder_t<ActsPodioEdm::TrackCollection>,
421 Acts::ValueHolder<ActsPodioEdm::TrackCollection>> ||

431 holder_t<ActsPodioEdm::TrackCollection>,
432 std::unique_ptr<ActsPodioEdm::TrackCollection>>) {
433 frame.put(std::move(m_collection), "tracks" + s);
434 } else {
435 frame.put(std::make_unique<ActsPodioEdm::TrackCollection>(
436 std::move(*m_collection)),
437 "tracks" + s);
438 }
439 m_surfaces.clear();
440
441 for (const auto& [key, col] : m_dynamic) {
442 col->releaseInto(frame, "tracks" + s + "_extra__");
443 }

590 static const ActsPodioEdm::TrackCollection* getTrackCollectionFromFrame(
591 const podio::Frame& frame, const std::string& suffix) {
592 std::string s = suffix.empty() ? suffix : "_" + suffix;
593 std::string tracksKey = "tracks" + s;
594
595 std::vector<std::string> available = frame.getAvailableCollections();
596 if (!Acts::rangeContainsValue(available, tracksKey)) {
597 throw std::runtime_error{"Track collection '" + tracksKey +
598 "' not found in frame"};
599 }
600
601 const auto* collection = frame.get(tracksKey);
602
603 const ActsPodioEdm::TrackCollection* d = nullptr;
604 if (const auto* casted =
605 dynamic_cast<const ActsPodioEdm::TrackCollection*>(collection);
606 casted != nullptr) {
607 d = casted;
608 } else {
609 throw std::runtime_error{"Unable to get collection " + tracksKey};
610 }
611
612 return d;
613 }
