129 template <typename T>
130 static constexpr bool has_impl(T& instance, Acts::HashedString key,
131 Acts::TrackIndexType istate) {
132 constexpr auto kInvalid = Acts::kTrackIndexInvalid;
133 using namespace Acts::HashedStringLiteral;
134 auto trackState = instance.m_collection->at(istate);
135 const auto& data = trackState.getData();
136 switch (key) {
137 case "predicted"_hash:
138 return data.ipredicted != kInvalid;
139 case "filtered"_hash:
140 return data.ifiltered != kInvalid;
141 case "smoothed"_hash:
142 return data.ismoothed != kInvalid;
143 case "calibrated"_hash:
144 return data.measdim != kInvalid;
145 case "calibratedCov"_hash:
146 return data.measdim != kInvalid;
147 case "jacobian"_hash:
148 return data.ijacobian != kInvalid;
149 case "projector"_hash:
150 return data.hasProjector;
151 case "uncalibratedSourceLink"_hash:
152 return data.uncalibratedIdentifier != PodioUtil::kNoIdentifier;
153 case "previous"_hash:
154 case "next"_hash:
155 case "measdim"_hash:
156 case "referenceSurface"_hash:
157 case "chi2"_hash:
158 case "pathLength"_hash:
159 case "typeFlags"_hash:
160 return true;
161 default:
162 return instance.m_dynamic.contains(key);
163 }
164
165 return false;
166 }

173 template <bool EnsureConst, typename T>
174 static std::any component_impl(T& instance, Acts::HashedString key,
175 Acts::TrackIndexType istate) {
176 if constexpr (EnsureConst) {
177 static_assert(std::is_const_v<std::remove_reference_t<T>>,
178 "Is not const");
179 }
180 using namespace Acts::HashedStringLiteral;
181 auto trackState = instance.m_collection->at(istate);
182 std::conditional_t<EnsureConst, const ActsPodioEdm::TrackStateInfo*,
183 ActsPodioEdm::TrackStateInfo*>
184 dataPtr;
185 if constexpr (EnsureConst) {
186 dataPtr = &trackState.getData();
187 } else {
188 dataPtr = &PodioUtil::getDataMutable(trackState);
189 }
190 auto& data = *dataPtr;
191 switch (key) {
192 case "previous"_hash:
193 return &data.previous;
194 case "next"_hash:
195 return &data.next;
196 case "predicted"_hash:
197 return &data.ipredicted;
198 case "filtered"_hash:
199 return &data.ifiltered;
200 case "smoothed"_hash:
201 return &data.ismoothed;
202 case "projector"_hash:
203 return &data.projector;
204 case "measdim"_hash:
205 return &data.measdim;
206 case "chi2"_hash:
207 return &data.chi2;
208 case "pathLength"_hash:
209 return &data.pathLength;
210 case "typeFlags"_hash:
211 return &data.typeFlags;
212 default:
213 auto it = instance.m_dynamic.find(key);
214 if (it == instance.m_dynamic.end()) {
215 throw std::runtime_error("Unable to handle this component");
216 }
217 std::conditional_t<EnsureConst,
218 const podio_detail::ConstDynamicColumnBase*,
219 podio_detail::DynamicColumnBase*>
220 col = it->second.get();
221 assert(col && "Dynamic column is null");
222 return col->get(istate);
223 }
224 }

258 const PodioUtil::ConversionHelper& helper,
259 const ActsPodioEdm::TrackStateCollection& collection,
260 std::vector<std::shared_ptr<const Acts::Surface>>& surfaces) noexcept {

649 IndexType addTrackState_impl(
650 Acts::TrackStatePropMask mask = Acts::TrackStatePropMask::All,
651 Acts::TrackIndexType iprevious = Acts::kTrackIndexInvalid) {
652 auto trackState = m_collection->create();
653 auto& data = PodioUtil::getDataMutable(trackState);
654 data.previous = iprevious;
655 data.ipredicted = kInvalid;
656 data.ifiltered = kInvalid;
657 data.ismoothed = kInvalid;
658 data.ijacobian = kInvalid;
659
660 PodioUtil::getReferenceSurfaceMutable(trackState).surfaceType =
661 PodioUtil::kNoSurface;
662
663 using enum Acts::TrackStatePropMask;
664
665 if (ACTS_CHECK_BIT(mask, Predicted)) {
666 m_params->create();
667 data.ipredicted = m_params->size() - 1;
668 }
669 if (ACTS_CHECK_BIT(mask, Filtered)) {
670 m_params->create();
671 data.ifiltered = m_params->size() - 1;
672 }
673 if (ACTS_CHECK_BIT(mask, Smoothed)) {
674 m_params->create();
675 data.ismoothed = m_params->size() - 1;
676 }
677 if (ACTS_CHECK_BIT(mask, Jacobian)) {
678 m_jacs->create();
679 data.ijacobian = m_jacs->size() - 1;
680 }
681 data.measdim = kInvalid;
682 data.hasProjector = false;
683 if (ACTS_CHECK_BIT(mask, Calibrated)) {
684 data.hasProjector = true;
685 }
686 m_surfaces.emplace_back();
687
688 data.uncalibratedIdentifier = PodioUtil::kNoIdentifier;
689 assert(m_collection->size() == m_surfaces.size() &&
690 "Inconsistent surface buffer");
691
692 for (const auto& [key, vec] : m_dynamic) {
693 vec->add();
694 }
695
696 return m_collection->size() - 1;
697 }

702 void addTrackStateComponents_impl(IndexType istate,
703 Acts::TrackStatePropMask mask) {
704 auto& data = PodioUtil::getDataMutable(m_collection->at(istate));
705
706 using enum Acts::TrackStatePropMask;
707
708 if (ACTS_CHECK_BIT(mask, Predicted) && data.ipredicted == kInvalid) {
709 m_params->create();
710 data.ipredicted = m_params->size() - 1;
711 }
712
713 if (ACTS_CHECK_BIT(mask, Filtered) && data.ifiltered == kInvalid) {
714 m_params->create();
715 data.ifiltered = m_params->size() - 1;
716 }
717
718 if (ACTS_CHECK_BIT(mask, Smoothed) && data.ismoothed == kInvalid) {
719 m_params->create();
720 data.ismoothed = m_params->size() - 1;
721 }
722
723 if (ACTS_CHECK_BIT(mask, Jacobian) && data.ijacobian == kInvalid) {
724 m_jacs->create();
725 data.ijacobian = m_jacs->size() - 1;
726 }
727
728 if (ACTS_CHECK_BIT(mask, Calibrated) && !data.hasProjector) {
729 data.hasProjector = true;
730 }
731 }

738 void shareFrom_impl(Acts::TrackIndexType iself, Acts::TrackIndexType iother,
739 Acts::TrackStatePropMask shareSource,
740 Acts::TrackStatePropMask shareTarget) {
741 auto& self = PodioUtil::getDataMutable(m_collection->at(iself));
742 auto& other = PodioUtil::getDataMutable(m_collection->at(iother));
743
744 assert(ACTS_CHECK_BIT(this->getTrackState(iother).getMask(), shareSource) &&
745 "Source has incompatible allocation");
746
747 using enum Acts::TrackStatePropMask;
748
749 IndexType sourceIndex{kInvalid};
750 switch (shareSource) {
751 case Predicted:
752 sourceIndex = other.ipredicted;
753 break;
754 case Filtered:
755 sourceIndex = other.ifiltered;
756 break;
757 case Smoothed:
758 sourceIndex = other.ismoothed;
759 break;
760 case Jacobian:
761 sourceIndex = other.ijacobian;
762 break;
763 default:
764 throw std::domain_error{"Unable to share this component"};
765 }
766
767 assert(sourceIndex != kInvalid);
768
769 switch (shareTarget) {
770 case Predicted:
771 assert(shareSource != Jacobian);
772 self.ipredicted = sourceIndex;
773 break;
774 case Filtered:
775 assert(shareSource != Jacobian);
776 self.ifiltered = sourceIndex;
777 break;
778 case Smoothed:
779 assert(shareSource != Jacobian);
780 self.ismoothed = sourceIndex;
781 break;
782 case Jacobian:
783 assert(shareSource == Jacobian);
784 self.ijacobian = sourceIndex;
785 break;
786 default:
787 throw std::domain_error{"Unable to share this component"};
788 }
789 }
