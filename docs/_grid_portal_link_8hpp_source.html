63 template <AxisConcept axis_t>
64 static std::unique_ptr<GridPortalLinkT<axis_t>> make(
65 const std::shared_ptr<RegularSurface>& surface, AxisDirection direction,
66 axis_t&& axis) {
67 using enum AxisDirection;
68 if (dynamic_cast<const CylinderSurface*>(surface.get()) != nullptr) {
69 if (direction != AxisZ && direction != AxisRPhi) {
70 throw std::invalid_argument{"Invalid binning direction"};
71 }
72 } else if (dynamic_cast<const DiscSurface*>(surface.get()) != nullptr &&
73 direction != AxisR && direction != AxisPhi) {
74 throw std::invalid_argument{"Invalid binning direction"};
75 } else if (dynamic_cast<const PlaneSurface*>(surface.get()) != nullptr &&
76 direction != AxisX && direction != AxisY) {
77 throw std::invalid_argument{"Invalid binning direction"};
78 }
79
80 return std::make_unique<GridPortalLinkT<axis_t>>(
81 surface, direction, std::forward<axis_t>(axis));
82 }

459template <typename... Axes>
460 requires(sizeof...(Axes) <= 2)
461class GridPortalLinkT : public GridPortalLink {
462 public:
464 using GridType = Grid<const TrackingVolume*, Axes...>;
465
467 static constexpr std::size_t DIM = sizeof...(Axes);
468
475 GridPortalLinkT(std::shared_ptr<RegularSurface> surface,
476 AxisDirection direction, Axes&&... axes)
477 : GridPortalLink(std::move(surface), direction),
478 m_grid(std::tuple{std::move(axes)...}) {
479 using enum AxisDirection;
480
481 // prevent open axis from having a nullptr edge from under/overflow bins
482 for (const IAxis* axis : m_grid.axes()) {
483 if (axis->getBoundaryType() == AxisBoundaryType::Open) {
484 throw std::invalid_argument(
485 "GridPortalLink: open axis boundary type is not supported");
486 }
487 }
488
489 if (const auto* cylinder =
490 dynamic_cast<const CylinderSurface*>(m_surface.get())) {
491 checkConsistency(m_grid, direction, *cylinder);
492
493 if (direction == AxisRPhi) {
494 m_projection = &projection<CylinderSurface, AxisRPhi>;
495 } else if (direction == AxisZ) {
496 m_projection = &projection<CylinderSurface, AxisZ>;
497 } else {
498 throw std::invalid_argument{"Invalid binning direction"};
499 }
500
501 } else if (const auto* disc =
502 dynamic_cast<const DiscSurface*>(m_surface.get())) {
503 checkConsistency(m_grid, direction, *disc);
504
505 if (direction == AxisR) {
506 m_projection = &projection<DiscSurface, AxisR>;
507 } else if (direction == AxisDirection::AxisPhi) {
508 m_projection = &projection<DiscSurface, AxisPhi>;
509 } else {
510 throw std::invalid_argument{"Invalid binning direction"};
511 }
512 } else if (const auto* plane =
513 dynamic_cast<const PlaneSurface*>(m_surface.get())) {
514 checkConsistency(m_grid, direction, *plane);
515
516 if (direction == AxisX) {
517 m_projection = &projection<PlaneSurface, AxisX>;
518 } else if (direction == AxisDirection::AxisY) {
519 m_projection = &projection<PlaneSurface, AxisY>;
520 } else {
521 throw std::invalid_argument{"Invalid binning direction"};
522 }
523
524 } else {
525 throw std::logic_error{"Surface type is not supported"};
526 }
527 }

551 std::unique_ptr<GridPortalLink> extendTo2d(
552 const IAxis* other) const override {
553 if constexpr (DIM == 2) {
554 return std::make_unique<GridPortalLinkT<Axes...>>(*this);
555 } else {
556 if (auto cylinder =
557 std::dynamic_pointer_cast<CylinderSurface>(m_surface)) {
558 return extendTo2dImpl(cylinder, other);
559 } else if (auto disc =
560 std::dynamic_pointer_cast<DiscSurface>(m_surface)) {
561 return extendTo2dImpl(disc, other);
562 } else if (auto plane =
563 std::dynamic_pointer_cast<PlaneSurface>(m_surface)) {
564 return extendTo2dImpl(plane, other);
565 } else {
566 throw std::logic_error{
567 "Surface type is not supported (this should not happen)"};
568 }
569 }
570 }

574 void setVolume(TrackingVolume* volume) override {
575 auto loc = m_grid.multiAxis().getNBins();
576 if constexpr (GridType::DIM == 1) {
577 for (std::size_t i = 1; i <= loc[0]; i++) {
578 m_grid.atLocalBins({i}) = volume;
579 }
580 } else {
581 for (std::size_t i = 1; i <= loc[0]; i++) {
582 for (std::size_t j = 1; j <= loc[1]; j++) {
583 m_grid.atLocalBins({i, j}) = volume;
584 }
585 }
586 }

595 Result<const TrackingVolume*> resolveVolume(
596 const GeometryContext& gctx, const Vector3& position,
597 double tolerance = s_onSurfaceTolerance) const override {
598 auto res = m_surface->globalToLocal(gctx, position, tolerance);
599 if (!res.ok()) {
600 return res.error();
601 }
602
603 const Vector2& local = *res;
604 return resolveVolume(gctx, local, tolerance);
605 }
606
615 Result<const TrackingVolume*> resolveVolume(
616 const GeometryContext& /*gctx*/, const Vector2& position,
617 double tolerance = s_onSurfaceTolerance) const override {
618 if (surface().insideBounds(position, BoundaryTolerance::None())) {
619 return m_grid.atPosition(m_projection(position));
620 }
621 if (!surface().insideBounds(
622 position, BoundaryTolerance::AbsoluteEuclidean(tolerance))) {
623 return Result<const TrackingVolume*>::failure(
624 PortalError::PositionOutsideLinkBounds);
625 }
626 const SurfaceBounds& bounds = surface().bounds();
627 const Vector2 onBounds =
628 bounds.closestPoint(position, bounds.boundToCartesianMetric(position));
629 return m_grid.atPosition(m_projection(onBounds));
630 }
631
632 private:
635 template <class surface_t, AxisDirection direction>
636 static Vector<DIM> projection(const Vector2& position) {
637 using enum AxisDirection;
638 if constexpr (DIM == 2) {
639 return position;
640 } else {
641 if constexpr (std::is_same_v<surface_t, CylinderSurface>) {
642 static_assert(direction == AxisRPhi || direction == AxisZ,
643 "Invalid binning direction");
644
645 if constexpr (direction == AxisRPhi) {
646 return Vector<1>{position[0]};
647 } else if constexpr (direction == AxisZ) {
648 return Vector<1>{position[1]};
649 }
650 } else if constexpr (std::is_same_v<surface_t, DiscSurface>) {
651 static_assert(direction == AxisR || direction == AxisPhi,
652 "Invalid binning direction");
653
654 if constexpr (direction == AxisR) {
655 return Vector<1>{position[0]};
656 } else if constexpr (direction == AxisPhi) {
657 return Vector<1>{position[1]};
658 }
659 } else if constexpr (std::is_same_v<surface_t, PlaneSurface>) {
660 static_assert(direction == AxisX || direction == AxisY,
661 "Invalid binning direction");
662
663 if constexpr (direction == AxisX) {
664 return Vector<1>{position[0]};
665 } else if constexpr (direction == AxisY) {
666 return Vector<1>{position[1]};
667 }
668 }
669 }
670 }

672 GridType m_grid;
673
675 Vector<DIM> (*m_projection)(const Vector2& position);
