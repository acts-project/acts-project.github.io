37template <detray::concepts::detector detector_t,
38 typename DIM = std::integral_constant<std::size_t, 2u>>
39class material_map_reader final : public reader_interface<detector_t> {
40 using material_reader_t = homogeneous_material_reader<detector_t>;
41
42 public:
44 static constexpr std::size_t dim{DIM()};
45
47 using payload_type =
48 detector_grids_payload<surface_material_payload, io::material_id>;
49
51 using bin_index_type = axis::multi_bin<dim>;
52
54 static constexpr std::string_view s_tag = "material_maps";
55
57 std::string_view tag() const override { return s_tag; }
58
63 void from_payload(detector_builder<typename detector_t::metadata,
64 volume_builder> &det_builder,
65 const detector_payload &det_data) const override {
66 DETRAY_VERBOSE_HOST("Reading payload object...");
67
68 using scalar_t = dscalar<typename detector_t::algebra_type>;
69 using mat_factory_t = material_map_factory<detector_t, bin_index_type>;
70 using mat_data_t = typename mat_factory_t::data_type;
71 using mat_id = typename detector_t::material::id;
72
73 // Map the io::shape_id in the payload to the shape types known by
74 // the detector under construction
75 using mat_registry_t =
76 io::detail::filtered_material_map_registry<detector_t>;
77 static_assert((!types::contains<mat_registry_t, io::detail::unknown_type> &&
78 types::size<mat_registry_t> > 0) ||
79 (types::contains<mat_registry_t, io::detail::unknown_type> &&
80 types::size<mat_registry_t> > 1));
81
82 if (!det_data.material_maps.has_value()) {
83 std::string err_str{"No data in material maps payload"};
84 DETRAY_FATAL_HOST(err_str);
85 throw std::invalid_argument(err_str);
86 }
87
88 const payload_type &grids_data = *det_data.material_maps;
89
90 DETRAY_VERBOSE_HOST("Converting material grids for "
91 << grids_data.grids.size() << " volumes");
92 // Convert the material volume by volume
93 for (const auto &[vol_idx, mat_grids] : grids_data.grids) {
94 if (!det_builder.has_volume(vol_idx)) {
95 std::stringstream err_stream;
96 err_stream << "Volume " << vol_idx << ": "
97 << "Cannot build material map for volume "
98 << "(volume not registered in detector builder)";
99 DETRAY_FATAL_HOST(err_stream.str());
100 throw std::invalid_argument(err_stream.str());
101 }
102
103 DETRAY_VERBOSE_HOST("Reading material maps in volume: "
104 << det_builder[static_cast<dindex>(vol_idx)]->name());
105
106 // Decorate the current volume builder with material maps
107 auto vm_builder =
108 det_builder.template decorate<material_map_builder<detector_t, dim>>(
109 static_cast<dindex>(vol_idx));
110
111 DETRAY_VERBOSE_HOST("-> Found " << mat_grids.size()
112 << " material grids:");
113
114 // Add the material data to the factory
115 auto mat_factory =
116 std::make_shared<material_map_factory<detector_t, bin_index_type>>();
117
118 // Convert the material grid of each surface
119 for (const auto &[idx, grid_data] : detray::views::enumerate(mat_grids)) {
120 DETRAY_VERBOSE_HOST("Reading material map payload #" << idx << "...");
121 // Get the number of bins per axis
122 std::vector<std::size_t> n_bins{};
123 for (const auto &axis_data : grid_data.axes) {
124 n_bins.push_back(axis_data.bins);
125 }
126
127 DETRAY_VERBOSE_HOST("-> Belongs to "
128 << (dim == 2 ? "surface" : "volume") << " = "
129 << grid_data.owner_link.link);
130
131 mat_id map_id =
132 from_payload_impl<mat_registry_t>(grid_data.grid_link.type);
133 DETRAY_VERBOSE_HOST("-> Type id: " << map_id);
134
135 // Material map id from payload could be matched to detector store ids
136 if (map_id == mat_id::e_none) {
137 std::stringstream err_str{};
138 err_str << "Material map ID " << grid_data.grid_link.type
139 << " in payload does not match any material type in "
140 "detector ("
141 << detray::types::get_name<typename detector_t::metadata>()
142 << "). Check the detector metadata!\nMaterial store:\n"
143 << detray::types::get_name<typename detector_t::material>();
144 DETRAY_FATAL_HOST(err_str.str());
145 throw std::invalid_argument(err_str.str());
146 }
147
148 DETRAY_VERBOSE_HOST("-> Reading axis bins: Dims = " << dim);
149 for ([[maybe_unused]] const auto &[i, count] :
150 detray::views::enumerate(n_bins)) {
151 DETRAY_VERBOSE_HOST("--> Axis " << i << ": " << count << " bins");
152 }
153 // Get the axis spans
154 DETRAY_VERBOSE_HOST("-> Reading axis spans:");
155 std::vector<std::vector<scalar_t>> axis_spans = {};
156 for (const auto &[i, axis_data] :
157 detray::views::enumerate(grid_data.axes)) {
158 axis_spans.push_back({static_cast<scalar_t>(axis_data.edges.front()),
159 static_cast<scalar_t>(axis_data.edges.back())});
160 DETRAY_VERBOSE_HOST("--> Axis " << i << " ["
161 << axis_spans.back().at(0) << ", "
162 << axis_spans.back().at(1) << "]");
163 }
164
165 // Get the local bin indices and the material parametrization
166 DETRAY_VERBOSE_HOST("-> Reading bin content...");
167
168 std::vector<bin_index_type> loc_bins{};
169 mat_data_t mat_data{
170 detail::basic_converter::from_payload(grid_data.owner_link)};
171 DETRAY_VERBOSE_HOST("--> Found " << grid_data.bins.size() << " bins");
172 for (const auto &bin_data : grid_data.bins) {
173 assert(dim == bin_data.loc_index.size() &&
174 "Dimension of local bin indices in input file does "
175 "not match material grid dimension");
176
177 // The local bin indices for the bin to be filled
178 bin_index_type mbin;
179 for (const auto &[i, bin_idx] :
180 detray::views::enumerate(bin_data.loc_index)) {
181 mbin[i] = bin_idx;
182 }
183 loc_bins.push_back(std::move(mbin));
184
185 // Add the material slab per bin
186 for (const auto &slab_data : bin_data.content) {
187 mat_data.append(
188 material_reader_t::template from_payload_impl<scalar_t>(
189 slab_data));
190 }
191 }
192
193 DETRAY_VERBOSE_HOST("Finished reading data for material map #" << idx);
194 DETRAY_DEBUG_HOST(mat_data);
195
196 DETRAY_VERBOSE_HOST("Add material data to factory...");
197 mat_factory->add_material(map_id, std::move(mat_data),
198 std::move(n_bins), std::move(axis_spans),
199 std::move(loc_bins));
200 }
201
202 // Add the material maps to the volume
203 DETRAY_VERBOSE_HOST("Add the material maps to the volume:");
204 vm_builder->add_surfaces(mat_factory);
205 }
206 }
207
208 private:
210 template <typename mat_registry_t, std::size_t I = 0u>
211 static typename detector_t::material::id from_payload_impl(
212 io::material_id type_id) {
213 // Get the next mask shape type
214 using frame_t = types::at<mat_registry_t, I>;
215 using algebra_t = typename detector_t::algebra_type;
216
217 // Material id of map data found
218 if constexpr (!std::is_same_v<frame_t, io::detail::unknown_type> &&
219 ::detray::concepts::coordinate_frame<frame_t>) {
220 // Get the corresponding material id for this detector
221 constexpr auto mat_id{
222 types::id<io::material_registry<algebra_t>, frame_t>};
223 if (type_id == mat_id) {
224 using mat_frame_registry_t = io::detail::mat_frame_registry<detector_t>;
225
226 constexpr std::size_t mapped_idx{
227 types::position<mat_frame_registry_t, frame_t>};
228
229 return types::id_cast<typename detector_t::material,
230 mat_frame_registry_t::original_index(mapped_idx)>;
231 }
232 }
233 // Test next material type id
234 if constexpr (I < types::size<mat_registry_t> - 1u) {
235 return from_payload_impl<mat_registry_t, I + 1u>(type_id);
236 } else {
237 return detector_t::material::id::e_none;
238 }
239 }
240};
