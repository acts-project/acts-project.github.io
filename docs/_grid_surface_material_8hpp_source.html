69struct IndexedMaterialAccessor : public IGridMaterialAccessor {
71 using grid_value_type = std::size_t;
72
75 explicit IndexedMaterialAccessor(std::vector<MaterialSlab>&& mmaterial)
76 : IGridMaterialAccessor(), material(std::move(mmaterial)) {}
77
79 std::vector<MaterialSlab> material;
86 template <typename grid_type>
87 inline const MaterialSlab& slab(
88 const grid_type& grid, const typename grid_type::point_t& point) const
89 requires(std::is_same_v<typename grid_type::value_type, grid_value_type>)
90 {
91 std::size_t index = grid.atPosition(point);
92 return material[index];
93 }
94
98 template <typename grid_type>
99 void scale(grid_type& /*grid*/, double scale) {
100 for (auto& m : material) {
101 m.scaleThickness(static_cast<float>(scale));
102 }
103 }
104};

108struct GloballyIndexedMaterialAccessor : public IGridMaterialAccessor {
112 explicit GloballyIndexedMaterialAccessor(
113 std::shared_ptr<std::vector<MaterialSlab>> gMaterial, bool shared = false)
114 : IGridMaterialAccessor(),
115 globalMaterial(std::move(gMaterial)),
116 sharedEntries(shared) {}
117
119 using grid_value_type = std::size_t;
120
122 std::shared_ptr<std::vector<MaterialSlab>> globalMaterial = nullptr;
123
128 bool sharedEntries = false;
129
138 template <typename grid_type>
139 inline const MaterialSlab& slab(
140 const grid_type& grid, const typename grid_type::point_t& point) const {
141 auto index = grid.atPosition(point);
142 return (*globalMaterial)[index];
143 }
144
154 template <typename grid_type>
155 void scale(grid_type& grid, double scale) {
156 if (sharedEntries) {
157 throw std::invalid_argument(
158 "GloballyIndexedMaterialAccessor: shared entry scaling is not "
159 "supported.");
160 }
161 // Loop through the grid bins, get the indices and scale the material
162 for (std::size_t ib = 0; ib < grid.size(); ++ib) {
163 auto index = grid.at(ib);
164 (*globalMaterial)[index].scaleThickness(static_cast<float>(scale));
165 }
166 }
167};

214class GridSurfaceMaterialT

217 public:
220 using BoundToGridLocalDelegate =
221 OwningDelegate<typename grid_t::point_t(const Vector2&),
222 GridAccess::IBoundToGridLocal>;
223
226 using GlobalToGridLocalDelegate =
227 OwningDelegate<typename grid_t::point_t(const Vector3&),
228 GridAccess::IGlobalToGridLocal>;
229
231 using grid_type = grid_t;
232
234 using material_accessor_type = material_accessor_t;
235
242 GridSurfaceMaterialT(grid_type&& grid,
243 material_accessor_type&& materialAccessor,
244 BoundToGridLocalDelegate boundToGridLocal,
245 GlobalToGridLocalDelegate globalToGridLocal)
246 : m_grid(std::move(grid)),
247 m_materialAccessor(std::move(materialAccessor)),
248 m_globalToGridLocal(std::move(globalToGridLocal)),
249 m_boundToGridLocal(std::move(boundToGridLocal)) {
250 if (!m_globalToGridLocal.connected()) {
251 throw std::invalid_argument(
252 "GridSurfaceMaterialT: GlobalToGridLocalDelegate is not connected.");
253 }
254 if (!m_boundToGridLocal.connected()) {
255 throw std::invalid_argument(
256 "GridSurfaceMaterialT: BoundToGridLocalDelegate is not connected.");
257 }
258 }
259
261 const MaterialSlab& materialSlab(const Vector2& lp) const final {
262 return m_materialAccessor.slab(m_grid, m_boundToGridLocal(lp));
263 }
264
266 std::vector<AxisDirection> localAxisDirections() const final { return {}; }
267
269 [[deprecated(
270 "Use materialSlab(const Vector2& lp) with a prior "
271 "Surface::globalToLocal() call instead")]] const MaterialSlab&
272 materialSlab(const Vector3& gp) const final {
273 return m_materialAccessor.slab(m_grid, m_globalToGridLocal(gp));
274 }
275
276 using ISurfaceMaterial::materialSlab;
