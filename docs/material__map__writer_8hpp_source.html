56 template <class detector_t>
57 static payload_type to_payload(
58 const detector_t& det, const typename detector_t::name_map& /*unused*/) {
59 using algebra_t = typename detector_t::algebra_type;
60 using material_t = material_slab<typename detector_t::scalar_type>;
61
62 payload_type grids_data;
63
64 for (const auto& vol_desc : det.volumes()) {
65 // Volume local surface indices
66 dindex offset{dindex_invalid};
67
69 auto vol = tracking_volume{det, vol_desc};
70 for (const auto& sf_desc : vol.surfaces()) {
71 if (sf_desc.index() < offset) {
72 offset = sf_desc.index();
73 }
74
75 const auto& mat_link = sf_desc.material();
76 // Don't look at empty links
77 if (mat_link.is_invalid() ||
78 mat_link.id() == detector_t::material::id::e_none) {
79 continue;
80 }
81
82 // Do not attempt to convert homogeneous material
83 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
84 if (mat_link.id() == detector_t::material::id::e_material_slab) {
85 continue;
86 }
87 }
88 if constexpr (detray::concepts::has_material_rods<detector_t>) {
89 if (mat_link.id() == detector_t::material::id::e_material_rod) {
90 continue;
91 }
92 }
93
94 // How to convert a material slab in the grid
95 auto mat_converter = [&sf_desc](const material_t& mat) {
96 return mat_writer_t::to_payload<algebra_t>(mat, sf_desc.index());
97 };
98
99 // Generate the payload
100 grid_writer_t::to_payload(det.material_store(), mat_link,
101 vol_desc.index(), sf_desc.index() - offset,
102 grids_data, mat_converter);
103 }
104 }
105
106 return grids_data;
107 }
108};
