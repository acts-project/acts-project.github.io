71 static payload_type to_payload_impl(
72 const detector_t& det, const typename detector_t::name_map& /*unused*/) {
73 using algebra_t = typename detector_t::algebra_type;
74 using material_t = material_slab<typename detector_t::scalar_type>;
75
76 payload_type grids_data;
77
78 for (const auto& vol_desc : det.volumes()) {
79 // Volume local surface indices
80 dindex offset{dindex_invalid};
81
83 auto vol = tracking_volume{det, vol_desc};
84 for (const auto& sf_desc : vol.surfaces()) {
85 if (sf_desc.index() < offset) {
86 offset = sf_desc.index();
87 }
88
89 const auto& mat_link = sf_desc.material();
90 // Don't look at empty links
91 if (mat_link.is_invalid() ||
92 mat_link.id() == detector_t::material::id::e_none) {
93 continue;
94 }
95
96 // Do not attempt to convert homogeneous material
97 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
98 if (mat_link.id() == detector_t::material::id::e_material_slab) {
99 continue;
100 }
101 }
102 if constexpr (detray::concepts::has_material_rods<detector_t>) {
103 if (mat_link.id() == detector_t::material::id::e_material_rod) {
104 continue;
105 }
106 }
107
108 // How to convert a material slab in the grid
109 auto mat_converter = [&sf_desc](const material_t& mat) {
110 return mat_writer_t::template to_payload_impl<algebra_t>(
111 mat, sf_desc.index());
112 };
113
114 // Generate the payload
115 grid_writer_t::to_payload_impl(
116 det.material_store(), mat_link, vol_desc.index(),
117 sf_desc.index() - offset, grids_data, mat_converter);
118 }
119 }
120
121 return grids_data;
122 }
123};
