180 Particle &correctEnergy(double delta) {
181 const auto newEnergy = std::hypot(m_mass, m_absMomentum) + delta;
182 if (newEnergy <= m_mass) {
183 m_absMomentum = 0.;
184 } else {
185 m_absMomentum = Acts::fastCathetus(newEnergy, m_mass);
186 }
187 return *this;
188 }

239 Acts::Vector4 mom4;

241 mom4[Acts::eMom0] = m_absMomentum * m_direction[Acts::ePos0];
242 mom4[Acts::eMom1] = m_absMomentum * m_direction[Acts::ePos1];
243 mom4[Acts::eMom2] = m_absMomentum * m_direction[Acts::ePos2];
244 mom4[Acts::eEnergy] = energy();

333 Acts::Result<Acts::BoundTrackParameters> boundParameters(
334 const Acts::GeometryContext &gctx) const {
335 if (!hasReferenceSurface()) {
336 return Acts::Result<Acts::BoundTrackParameters>::failure(
337 std::error_code());
338 }
339 Acts::Result<Acts::Vector2> localResult =
340 m_referenceSurface->globalToLocal(gctx, position(), direction());
341 if (!localResult.ok()) {
342 return localResult.error();
343 }
344 Acts::BoundVector params;
345 params << localResult.value(), phi(), theta(), qOverP(), time();
346 return Acts::BoundTrackParameters(referenceSurface()->getSharedPtr(),
347 params, std::nullopt, hypothesis());
348 }
