189template <typename view_impl_t>
190class view_interface : public base_view {
193 DETRAY_HOST_DEVICE
194 constexpr auto cast_impl() -> view_impl_t& {
195 return static_cast<view_impl_t&>(*this);
196 }
197 DETRAY_HOST_DEVICE
198 constexpr auto cast_impl() const -> const view_impl_t& {
199 return static_cast<const view_impl_t&>(*this);
200 }
202
203 public:
207 template <detray::ranges::forward_range R = view_impl_t>
208 DETRAY_HOST_DEVICE constexpr auto empty() const -> bool {
209 return (detray::ranges::begin(cast_impl()) ==
210 detray::ranges::end(cast_impl()));
211 }
212
215 DETRAY_HOST_DEVICE
216 constexpr explicit operator bool() const {
217 return !detray::ranges::empty(cast_impl());
218 }
219
223 DETRAY_HOST_DEVICE
224 constexpr auto data() const {
225 return empty() ? nullptr
226 : std::addressof(*(detray::ranges::begin(cast_impl())));
227 }
228
232 DETRAY_HOST_DEVICE
233 constexpr auto data() {
234 return empty() ? nullptr : &(*(detray::ranges::begin(cast_impl())));
235 }
236
240 template <detray::ranges::forward_range R = view_impl_t>
241 DETRAY_HOST_DEVICE constexpr auto size() const {
242 return static_cast<dindex>(detray::ranges::distance(
243 detray::ranges::begin(cast_impl()), detray::ranges::end(cast_impl())));
244 }
245
249 template <detray::ranges::forward_range R = view_impl_t>
250 DETRAY_HOST_DEVICE constexpr decltype(auto) front() const {
251 const auto bg = detray::ranges::begin(cast_impl());
252 assert(!empty());
253 return *bg;
254 }
255
259 template <detray::ranges::forward_range R = view_impl_t>
260 DETRAY_HOST_DEVICE constexpr decltype(auto) front() {
261 const auto bg = detray::ranges::begin(cast_impl());
262 assert(!empty());
263 return *bg;
264 }
265
269 template <detray::ranges::bidirectional_range R = view_impl_t>
270 DETRAY_HOST_DEVICE constexpr decltype(auto) back() const {
271 auto sentinel = detray::ranges::end(cast_impl());
272 assert(!empty());
273 return *(--sentinel);
274 }
275
279 template <detray::ranges::bidirectional_range R = view_impl_t>
280 DETRAY_HOST_DEVICE constexpr decltype(auto) back() {
281 auto sentinel = detray::ranges::end(cast_impl());
282 assert(!empty());
283 return *(--sentinel);
284 }

291 template <detray::ranges::random_access_range R = view_impl_t>
292 DETRAY_HOST_DEVICE constexpr decltype(auto) operator[](const dindex i) const {
293 // Call 'begin()' directly here to make CUDA happy
294 return (
295 cast_impl()
296 .begin())[static_cast<detray::ranges::range_difference_t<R>>(i)];
