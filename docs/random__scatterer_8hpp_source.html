82 struct kernel {
92 template <typename mat_group_t, typename index_t>
93 DETRAY_HOST_DEVICE inline bool operator()(
94 [[maybe_unused]] const mat_group_t& material_group,
95 [[maybe_unused]] const index_t& mat_index,
96 [[maybe_unused]] typename random_scatterer::state& s,
97 [[maybe_unused]] const pdg_particle<scalar_type>& ptc,
98 [[maybe_unused]] const bound_track_parameters<algebra_t>& bound_params,
99 [[maybe_unused]] const scalar_type cos_inc_angle,
100 [[maybe_unused]] const scalar_type approach) const {
101 using material_t = typename mat_group_t::value_type;
102
103 if constexpr (concepts::surface_material<material_t>) {
104 const scalar_type qop = bound_params.qop();
105
106 const auto mat = detail::material_accessor::get(
107 material_group, mat_index, bound_params.bound_local());
108
109 // return early in case of zero thickness
110 if (mat.thickness() <= std::numeric_limits<scalar_type>::epsilon()) {
111 return false;
112 }
113
114 const scalar_type path_segment{
115 mat.path_segment(cos_inc_angle, approach)};
116
117 const detail::relativistic_quantities rq(ptc, qop);
118
119 // Energy Loss
120 if (s.do_energy_loss) {
121 s.e_loss_mpv = interaction_type().compute_energy_loss_landau(
122 path_segment, mat.get_material(), ptc, rq);
123
124 s.e_loss_sigma = interaction_type().compute_energy_loss_landau_sigma(
125 path_segment, mat.get_material(), ptc, rq);
126 }
127
128 // Scattering angle
129 if (s.do_multiple_scattering) {
130 // @todo: use momentum before or after energy loss in
131 // backward mode?
132 s.projected_scattering_angle =
133 interaction_type().compute_multiple_scattering_theta0(
134 mat.path_segment_in_X0(cos_inc_angle, approach), ptc, rq);
135 }
136
137 return true;
138 } else {
139 // For non-pointwise material interactions, do nothing
140 return false;
141 }
142 }
143 };

150 template <typename propagator_state_t>
151 DETRAY_HOST inline void operator()(
152 state& simulator_state, propagator_state_t& prop_state,
153 parameter_transporter_result<algebra_t>& res) const {
154 // @Todo: Make context part of propagation state
155 using detector_type = typename propagator_state_t::detector_type;
156 using geo_context_type = typename detector_type::geometry_context;
157
158 auto& navigation = prop_state.navigation();
159
160 if (!navigation.encountered_sf_material()) {
161 return;
162 }
163
164 auto& stepping = prop_state.stepping();
165 const auto& ptc = stepping.particle_hypothesis();
166 auto& bound_params = res.destination_params();
167 const auto sf = navigation.current_surface();
168 const scalar_type cos_inc_angle{cos_angle(geo_context_type{}, sf,
169 bound_params.dir(),
170 bound_params.bound_local())};
171
172 const bool success = sf.template visit_material<kernel>(
173 simulator_state, ptc, bound_params, cos_inc_angle,
174 bound_params.bound_local()[0]);
175
176 if (success) {
177 // Get the new momentum
178 const auto new_mom = attenuate(
179 simulator_state.e_loss_mpv, simulator_state.e_loss_sigma, ptc.mass(),
180 bound_params.p(ptc.charge()), simulator_state.generator);
181
182 // Update Qop
183 bound_params.set_qop(ptc.charge() / new_mom);
184
185 // Get the new direction from random scattering
186 const auto new_dir = scatter(bound_params.dir(),
187 simulator_state.projected_scattering_angle,
188 simulator_state.generator);
189
190 // Update Phi and Theta
191 bound_params.set_phi(vector::phi(new_dir));
192 bound_params.set_theta(vector::theta(new_dir));
193 // Signal parameter update
194 res.status = actor::status::e_success;
195
196 // Flag renavigation of the current candidate
197 prop_state.navigation().set_high_trust();
198 }
199 }

208 template <typename generator_t>
209 DETRAY_HOST inline scalar_type attenuate(const scalar_type mpv,
210 const scalar_type sigma,
211 const scalar_type m0,
212 const scalar_type p0,
213 generator_t& generator) const {
214 // Get the random energy loss
215 // @todo tune the scale parameters (e_loss_mpv and e_loss_sigma)
216 const auto e_loss =
217 landau_distribution<scalar_type>{}(generator, mpv, sigma);
218
219 // E = sqrt(m^2 + p^2)
220 const auto energy = math::sqrt(m0 * m0 + p0 * p0);
221 const auto new_energy = energy - e_loss;
222
223 auto p2 = new_energy * new_energy - m0 * m0;
224
225 // To avoid divergence
226 if (p2 < 0) {
227 p2 = 1.f * unit<scalar_type>::eV * unit<scalar_type>::eV;
228 }
229
230 // p = sqrt(E^2 - m^2)
231 return math::sqrt(p2);
232 }

240 template <typename generator_t>
241 DETRAY_HOST inline vector3_type scatter(
242 const vector3_type& dir, const scalar_type projected_scattering_angle,
243 generator_t& generator) const {
244 const auto scattering_angle =
245 constant<scalar_type>::sqrt2 * projected_scattering_angle;
246
247 return scattering_helper<algebra_t>()(dir, scattering_angle, generator);
248 }
