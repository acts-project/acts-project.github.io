54// Print the values of an enum by identifier
55#define ENUM_PRINT(x) \
56 case x: \
57 os << #x; \
58 break

60DETRAY_HOST inline std::ostream& operator<<(std::ostream& os, bounds b) {
61 switch (b) {
62 using enum bounds;
63 ENUM_PRINT(e_open);
64 ENUM_PRINT(e_closed);
65 ENUM_PRINT(e_circular);
66 }
67 return os;
68}

70DETRAY_HOST inline std::ostream& operator<<(std::ostream& os, label l) {
