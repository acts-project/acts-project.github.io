70 template <concepts::scalar scalar_t, concepts::point point_t>
71 DETRAY_HOST_DEVICE inline scalar_t min_dist_to_boundary(
72 const bounds_type<scalar_t> &bounds, const point_t &loc_p) const {

86 template <concepts::algebra algebra_t>
87 DETRAY_HOST_DEVICE constexpr result_type<dbool<algebra_t>> check_boundaries(
88 const bounds_type<dscalar<algebra_t>> &bounds,
89 const dtransform3D<algebra_t> &trf, const dpoint3D<algebra_t> &glob_p,
90 const dscalar<algebra_t> tol =
91 std::numeric_limits<dscalar<algebra_t>>::epsilon(),
92 const dscalar<algebra_t> /*edge_tol*/ = 0.f) const {
93 // Rotate to the local cartesian frame
94 const dpoint3D<algebra_t> loc_p =
95 cartesian3D<algebra_t>::global_to_local(trf, glob_p, {});
96
97 // Only check the radius
98 return check_boundaries(
99 bounds,
100 dpoint2D<algebra_t>{vector::perp(loc_p),
101 static_cast<dscalar<algebra_t>>(0.f)},
102 tol);
103 }

114 template <concepts::scalar scalar_t, concepts::point point_t>
115 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
116 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
117 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
118 const scalar_t /*edge_tol*/ = 0.f) const {
119 return ((loc_p[0] + tol) >= bounds[e_inner_r] &&
120 loc_p[0] <= (bounds[e_outer_r] + tol));
121 }

140 template <concepts::scalar scalar_t>
141 DETRAY_HOST_DEVICE constexpr scalar_t area(
142 const bounds_type<scalar_t> &bounds) const {
143 return (bounds[e_outer_r] * bounds[e_outer_r] -
144 bounds[e_inner_r] * bounds[e_inner_r]) *
145 constant<scalar_t>::pi;
146 }

154 template <concepts::scalar scalar_t>
155 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
156 const bounds_type<scalar_t> &bounds,
157 const bounds_type<scalar_t> &o_bounds) const {
158 bounds_type<scalar_t> new_bounds{};
159
160 new_bounds[e_inner_r] = math::min(bounds[e_inner_r], o_bounds[e_inner_r]);
161 new_bounds[e_outer_r] = math::max(bounds[e_outer_r], o_bounds[e_outer_r]);
162
163 return new_bounds;
164 }

176 DETRAY_HOST_DEVICE inline darray<dscalar<algebra_t>, 6> local_min_bounds(

212 template <concepts::scalar scalar_t>
213 DETRAY_HOST constexpr bool check_consistency(
214 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
215 constexpr auto tol{10.f * std::numeric_limits<scalar_t>::epsilon()};
216
217 if (math::signbit(bounds[e_inner_r]) || bounds[e_outer_r] < tol) {
218 os << "DETRAY ERROR (HOST): Radius must be in the range [0, "
219 "numeric_max)"
220 << std::endl;
221 return false;
222 }
223 if (bounds[e_inner_r] >= bounds[e_outer_r] ||
224 math::fabs(bounds[e_inner_r] - bounds[e_outer_r]) < tol) {
225 os << "DETRAY ERROR (HOST): Inner radius must be smaller outer "
226 "radius.";
227 return false;
228 }
229
230 return true;
231 }
