28template <concepts::detector detector_t, typename view_t>
29auto portal(const typename detector_t::geometry_context& context,
30 const detector_t& detector,
31 const detray::geometry::surface<detector_t>& d_portal,
32 const view_t& view,
33 const styling::portal_style& style =
34 styling::tableau_colorblind::portal_style,
35 bool hide_links = false, bool hide_material = false) {
36 assert(d_portal.is_portal());
37
38 using point3_container_t = std::vector<typename detector_t::point3_type>;
39 using p_portal_t = actsvg::proto::portal<point3_container_t>;
40 using p_surface_t = actsvg::proto::surface<point3_container_t>;
41
42 const auto p_surfaces = svgtools::conversion::surface(
43 context, detector, d_portal, view, style._surface_style, hide_material);
44
45 std::vector<typename p_portal_t::link> pt_links{};
46 const bool display_links{!hide_links &&
47 svgtools::utils::is_not_world_portal(d_portal)};
48 if (display_links) {
49 pt_links = svgtools::conversion::links(context, detector, d_portal);
50 }
51
52 std::vector<p_portal_t> p_portals{};
53 for (std::size_t i = 0u; i < p_surfaces.size(); ++i) {
54 const auto& p_sub_portal = p_surfaces.at(i);
55
56 auto& p_portal = p_portals.emplace_back();
57 p_portal._name =
58 "portal_" + std::to_string(d_portal.index()) + "_" + std::to_string(i);
59 p_portal._surface = p_sub_portal;
60 p_portal._surface._sf_type = p_surface_t::sf_type::e_portal;
61
62 if (display_links) {
63 p_portal._volume_links = {pt_links.at(i)};
64 }
65
66 svgtools::styling::apply_style(p_portal, style);
67 }
68
69 return p_portals;
70}
