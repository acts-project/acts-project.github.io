80 DETRAY_HOST_DEVICE
81 bool operator==(const bound_parameters_vector& rhs) const {
82 for (unsigned int i = 0u; i < e_bound_size; i++) {
83 const auto lhs_val = getter::element(m_vector, i, 0u);

217 const scalar_type sinTheta{math::sin(theta)};

219 return getter::element(m_vector, e_bound_qoverp, 0u) / sinTheta;

233 DETRAY_HOST_DEVICE
234 scalar_type p(const scalar_type q) const {

236 assert(q * qop() > 0.f);

242 DETRAY_HOST_DEVICE
243 vector3_type mom(const scalar_type q) const { return p(q) * dir(); }

247 DETRAY_HOST_DEVICE
248 scalar_type pT(const scalar_type q) const {
249 assert(qop() != 0.f);
250 assert(q * qop() > 0.f);
251 return math::fabs(q / qop() * vector::perp(dir()));
252 }
253
256 DETRAY_HOST_DEVICE
257 scalar_type pz(const scalar_type q) const {
258 assert(qop() != 0.f);
259 assert(q * qop() > 0.f);
260 return math::fabs(q / qop() * dir()[2]);
261 }

266 DETRAY_HOST_DEVICE
267 constexpr bool is_invalid(const bool do_check = true) const {
268 if (!do_check) {
269 return false;
270 }
271 bool inv_elem{false};
272 bool is_all_zero{true};
273 for (std::size_t i = 0u; i < e_bound_size; ++i) {
274 inv_elem = inv_elem || !math::isfinite(getter::element(m_vector, i, 0u));
275 is_all_zero =
276 is_all_zero && (math::fabs(getter::element(m_vector, i, 0u)) == 0.f);
277 }
278
279 return (inv_elem || is_all_zero);
280 }

337 DETRAY_HOST_DEVICE
338 bool operator==(const bound_track_parameters& rhs) const {
339 if (m_identifier != rhs.surface_link()) {
340 return false;
341 }
342
343 if (!base_type::operator==(rhs)) {
344 return false;
345 }
346
347 for (unsigned int i = 0u; i < e_bound_size; i++) {
348 for (unsigned int j = 0u; j < e_bound_size; j++) {
349 const auto lhs_val = getter::element(m_covariance, i, j);
350 const auto rhs_val = getter::element(rhs.covariance(), i, j);
351
352 if (math::fabs(lhs_val - rhs_val) >
353 std::numeric_limits<scalar_type>::epsilon()) {
354 return false;
355 }
356 }
357 }
358
359 return true;
360 }

432template <concepts::value value_t, concepts::algebra algebra_t>
433DETRAY_HOST_DEVICE constexpr decltype(auto) cast_to(
434 const bound_parameters_vector<algebra_t>& param_vec) {
435 using new_algebra_t = dalgebra<algebra_t, value_t>;
436
437 if constexpr (std::same_as<new_algebra_t, algebra_t>) {
438 return param_vec;

446template <concepts::value value_t, concepts::algebra algebra_t>
447DETRAY_HOST_DEVICE constexpr decltype(auto) cast_to(
448 const bound_track_parameters<algebra_t>& param) {
449 using new_algebra_t = dalgebra<algebra_t, value_t>;
450
451 if constexpr (std::same_as<new_algebra_t, algebra_t>) {
452 return param;

454 const bound_parameters_vector<algebra_t>& param_vec = param;
455 return bound_track_parameters<new_algebra_t>{
456 param.surface_link(), ::detray::algebra::cast_to<value_t>(param_vec),
457 ::detray::algebra::cast_to<value_t>(param.covariance())};
