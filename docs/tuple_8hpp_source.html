33template <typename T, typename... Ts>
34struct tuple<T, Ts...> {
35 // Deliberately not value-initializing: that would require every element to
36 // be default constructible, which is not a requirement of this tuple type.
37 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-member-init)
39 constexpr tuple() = default;
40
43 constexpr tuple(const tuple &o)
44 requires(std::is_copy_constructible_v<T> &&
45 (std::is_copy_constructible_v<Ts> && ...))
46 = default;
47
52 template <typename U, typename... Us>
53 requires std::is_constructible_v<T, U &&> &&
54 std::is_constructible_v<tuple<Ts...>, Us &&...>
55 DETRAY_HOST_DEVICE explicit constexpr tuple(const tuple<U, Us...> &o)
56 : v(o.v), r(o.r) {}
57
60 constexpr tuple(tuple &&o) noexcept
61 requires(std::is_move_constructible_v<T> &&
62 (std::is_move_constructible_v<Ts> && ...))
63 = default;
64
69 template <typename U, typename... Us>
70 requires std::is_constructible_v<T, U &&> &&
71 std::is_constructible_v<tuple<Ts...>, Us &&...>
72 DETRAY_HOST_DEVICE explicit constexpr tuple(tuple<U, Us...> &&o)
73 : v(std::move(o.v)), r(std::move(o.r)) {}
74
80 template <typename U, typename... Us>
81 requires std::is_constructible_v<T, U &&> &&
82 std::is_constructible_v<tuple<Ts...>, Us &&...> &&
83 (!(std::is_same_v<tuple, U> ||
84 (std::is_same_v<tuple, Us> || ...)))
85 DETRAY_HOST_DEVICE explicit constexpr tuple(U &&_v, Us &&..._r)
86 : v(std::forward<U>(_v)), r(std::forward<Us>(_r)...) {}
87
88 constexpr ~tuple() noexcept = default;
89
93 constexpr tuple &operator=(const tuple &other)
94 requires(std::is_copy_assignable_v<T> &&
95 (std::is_copy_assignable_v<Ts> && ...))
96 = default;

103 template <typename U, typename... Us>
104 DETRAY_HOST_DEVICE constexpr tuple &operator=(const tuple<U, Us...> &other)
105 requires(std::is_assignable_v<T &, const U &> &&
106 (std::is_assignable_v<Ts &, const Us &> && ...))
107 {
108 v = other.v;
109 r = other.r;
110 return *this;
111 }
112
116 constexpr tuple &operator=(tuple &&other) noexcept
117 requires(std::is_move_assignable_v<T> &&
118 (std::is_move_assignable_v<Ts> && ...))
119 = default;
120
126 template <typename U, typename... Us>
127 DETRAY_HOST_DEVICE constexpr tuple &operator=(tuple<U, Us...> &&other)
128 requires(std::is_assignable_v<T &, U> &&
129 (std::is_assignable_v<Ts &, Us> && ...))
130 {
131 v = std::move(other.v);
132 r = std::move(other.r);
133 return *this;
134 }
135
136 T v;
137 // Value-initialization would require every tail element to be default
138 // constructible, which is not a requirement of this tuple type.
140 tuple<Ts...> r; // NOLINT(cppcoreguidelines-pro-type-member-init)
141};

151template <std::size_t I, typename... Ts>
152DETRAY_HOST_DEVICE const auto &get(const detray::tuple<Ts...> &t) noexcept {
153 static_assert(I < sizeof...(Ts),
154 "Attempt to access index greater than tuple size.");

183template <typename U, typename T, typename... Ts>
184DETRAY_HOST_DEVICE const auto &get(const detray::tuple<T, Ts...> &t) noexcept {
185 if constexpr (std::is_same_v<U, T>) {
186 return t.v;
187 } else if constexpr (sizeof...(Ts) > 0) {
188 static_assert((std::is_same_v<U, Ts> || ...), "Type not found in tuple.");
189 return ::detray::get<U, Ts...>(t.r);
190 }
