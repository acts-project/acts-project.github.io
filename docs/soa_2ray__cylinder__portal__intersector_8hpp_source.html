75 template <typename mask_t, concepts::algebra other_algebra_t>
76 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
77 const trajectory_type<other_algebra_t> &ray,
78 const transform3_type & /*trf*/, const mask_t &mask,
79 const scalar_type overstep_tol = 0.f) const {
80 const scalar_type r = mask[mask_t::shape::e_r];
81
82 const auto &pos = ray.pos();
83 const auto &dir = ray.dir();
84
85 const value_type rd_perp_2{dir[0] * dir[0] + dir[1] * dir[1]};
86
87 // The ray is parallel to all/any cylinder axes (z-axis)...
88 if (rd_perp_2 < std::numeric_limits<value_type>::epsilon()) [[unlikely]] {
89 return {};
90 }
91
92 // ...otherwise, two solutions should exist, if the descriminator is
93 // greater than zero
94 const scalar_type ro_perp_2 = pos[0] * pos[0] + pos[1] * pos[1];
95 const scalar_type rad_diff = r * r - ro_perp_2;
96
97 const scalar_type rd_perp_inv_2 = 1.f / rd_perp_2;
98 const scalar_type k = -rd_perp_inv_2 * (pos[0] * dir[0] + pos[1] * dir[1]);
99 const scalar_type discr = rd_perp_inv_2 * rad_diff + k * k;
100
101 // No intersection found for any cylinder
102 if (detray::detail::all_of(discr < 0.f)) [[unlikely]] {
103 return {};
104 }
105
106 const scalar_type sqrt_discr = math::sqrt(discr);
107 const scalar_type s1 = k + sqrt_discr;
108 const scalar_type s2 = k - sqrt_discr;
109
110 // Take the nearest solution in every lane
111 auto is_smaller_sol = math::fabs(s1) < math::fabs(s2);
112
113 scalar_type path = 0.f;
114 path(is_smaller_sol) = s1;
115 path(!is_smaller_sol) = s2;
116
117 // If any of the the near solutions is outside the overstepping
118 // tolerance, take the far solution (if it exists)
119 if (const auto outside_overstep_tol = path < overstep_tol;
120 detray::detail::any_of(outside_overstep_tol) &&
121 detray::detail::all_of((discr > 0.f) || !outside_overstep_tol)) {
122 is_smaller_sol =
123 (math::fabs(s1) >= math::fabs(s2)) && outside_overstep_tol;
124 path(is_smaller_sol) = s1;
125 path(!is_smaller_sol) = s2;
126 }
127
128 point2_type loc;
129 loc[0] = 0.f;
130 loc[1] = pos[2] + path * dir[2];
131
132 return {path, loc};
133 }
134};
