102 template <typename smoother_t = MbfSmoother, TrackProxyConcept track_proxy_t>
103 Result<void> joinInPlace(const GeometryContext& gctx, track_proxy_t& first,
104 typename track_proxy_t::TrackStateProxy sharedState,
105 track_proxy_t& second,
106 const smoother_t& smoother = smoother_t{}) const {
107 if (first.stemIndex() == kTrackIndexInvalid) {
108 first.linkForward();
109 }
110
111 if (second.tipIndex() == kTrackIndexInvalid) {
112 // An empty second pass can still carry the estimate at the target.
113 const auto tipIndex = first.tipIndex();
114 const auto stemIndex = first.stemIndex();
115 first.copyFromWithoutStates(second);
116 first.tipIndex() = tipIndex;
117 first.stemIndex() = stemIndex;
118 return Result<void>::success();
119 }
120
121 const auto tipIndex = first.tipIndex();
122 auto res = joinStates(gctx, first, sharedState, second, smoother);
123 if (!res.ok()) {
124 return res.error();
125 }
126 first.copyFromWithoutStates(second);
127 first.tipIndex() = tipIndex;
128 first.stemIndex() = second.stemIndex();
129
130 return Result<void>::success();
131 }

154 template <typename visitor_t, typename smoother_t = MbfSmoother,
155 TrackProxyConcept track_proxy_t>
156 requires std::invocable<visitor_t&, track_proxy_t&>
157 Result<void> joinVisit(const GeometryContext& gctx, track_proxy_t& first,
158 typename track_proxy_t::TrackStateProxy sharedState,
159 track_proxy_t& second, visitor_t&& visitor,
160 const smoother_t& smoother = smoother_t{}) const {
161 auto& trajectory = first.container().trackStateContainer();
162
163 // what the join changes on the first tracklet
164 struct Smoothed {
165 TrackIndexType index{};
166 TrackIndexType next{};
167 std::optional<std::pair<BoundVector, BoundMatrix>> estimate;
168 };
169 std::vector<Smoothed> smoothed;
170 for (const auto& ts : first.trackStatesReversed()) {
171 Smoothed& entry = smoothed.emplace_back(
172 ts.index(),
173 ts.template component<TrackIndexType>(detail_tp::kNextKey),
174 std::nullopt);
175 if (ts.hasSmoothed()) {
176 entry.estimate.emplace(ts.smoothed(), ts.smoothedCovariance());
177 }
178 }
179 const TrackIndexType stemIndex = first.stemIndex();
180 const TrackIndexType previous = sharedState.previous();
181 std::optional<BoundMatrix> jacobian;
182 if (sharedState.hasJacobian()) {
183 jacobian = sharedState.jacobian();
184 }
185 auto undo = [&] {
186 for (const auto& [index, next, estimate] : smoothed) {
187 auto ts = trajectory.getTrackState(index);
188 ts.template component<TrackIndexType>(detail_tp::kNextKey) = next;
189 if (estimate.has_value()) {
190 ts.smoothed() = estimate->first;
191 ts.smoothedCovariance() = estimate->second;
192 } else if (ts.hasSmoothed()) {
193 ts.unset(TrackStatePropMask::Smoothed);
194 }
195 }
196 sharedState.previous() = previous;
197 if (jacobian.has_value()) {
198 sharedState.jacobian() = *jacobian;
199 }
200 first.stemIndex() = stemIndex;
201 };
202 const detail::ScopeExit restore(undo);
203
204 if (first.stemIndex() == kTrackIndexInvalid) {
205 first.linkForward();
206 }
207 if (second.tipIndex() == kTrackIndexInvalid) {
208 second.tipIndex() = first.tipIndex();
209 second.stemIndex() = first.stemIndex();
210 } else {
211 auto res = joinStates(gctx, first, sharedState, second, smoother);
212 if (!res.ok()) {
213 return res.error();
214 }
215 second.tipIndex() = first.tipIndex();
216 }
217 visitor(second);
218 return Result<void>::success();
219 }

235 template <typename smoother_t = MbfSmoother,
236 TrackContainerFrontend track_container_t,
237 TrackProxyConcept track_proxy_t,
238 TrackStateProxyConcept track_state_proxy_t>
239 Result<typename track_container_t::TrackProxy> joinCopy(
240 const GeometryContext& gctx, track_container_t& out,
241 const track_proxy_t& first, const track_state_proxy_t& sharedState,
242 const track_proxy_t& second,
243 const smoother_t& smoother = smoother_t{}) const {
244 using TrackProxy = typename track_container_t::TrackProxy;
245 using TrackStateProxy = typename track_container_t::TrackStateProxy;
246
247 // position of the shared state, counted from the tip
248 auto firstStates = first.trackStatesReversed();
249 auto sharedIt = std::ranges::find_if(firstStates, [&](const auto& ts) {
250 return ts.index() == sharedState.index();
251 });
252 if (sharedIt == firstStates.end()) {
253 ACTS_DEBUG("Shared state " << sharedState.index()
254 << " is not on the first tracklet");
255 return Result<TrackProxy>::failure(KalmanFitterError::SmoothFailed);
256 }
257
258 TrackProxy firstCopy = out.makeTrack();
259 firstCopy.copyFrom(first);
260 TrackProxy secondCopy = out.makeTrack();
261 secondCopy.copyFrom(second);
262
263 auto copyStates = firstCopy.trackStatesReversed();
264 TrackStateProxy sharedCopy = *std::ranges::next(
265 copyStates.begin(),
266 std::ranges::distance(firstStates.begin(), sharedIt));
267
268 auto res = joinInPlace(gctx, firstCopy, sharedCopy, secondCopy, smoother);
269 // the second copy is a shell now, its states are in the joined chain
270 out.removeTrack(secondCopy.index());
271 if (!res.ok()) {
272 out.removeTrack(firstCopy.index());
273 return Result<TrackProxy>::failure(res.error());
274 }
275 return firstCopy;
276 }

282 template <typename smoother_t, TrackProxyConcept track_proxy_t>
283 Result<void> joinStates(const GeometryContext& gctx, track_proxy_t& first,
284 typename track_proxy_t::TrackStateProxy sharedState,
285 track_proxy_t& second,
286 const smoother_t& smoother) const {
287 auto& trajectory = first.container().trackStateContainer();
288
289 if (!m_cfg.smoothSecond && !m_cfg.conditionFirst) {
290 detail::linkTrackletBehind(sharedState, second, m_cfg.invertJacobians);
291 return Result<void>::success();
292 }
293 if (!sharedState.hasSmoothed()) {
294 ACTS_DEBUG("Shared state " << sharedState.index()
295 << " has no smoothed parameters");
296 return KalmanFitterError::SmoothFailed;
297 }
298
299 BoundVector sharedMean = sharedState.smoothed();
300 BoundMatrix sharedCov = sharedState.smoothedCovariance();
301
302 if (m_cfg.smoothSecond) {
303 auto res = smoother(gctx, trajectory, second.tipIndex(), logger());
304 if (!res.ok()) {
305 ACTS_DEBUG("Smoothing of the second tracklet failed: "
306 << res.error().message());
307 return res.error();
308 }
309 // the first state of the second tracklet was predicted from the shared
310 // state
311 auto secondStart = second.outermostTrackState();
312 for (auto ts : second.trackStatesReversed()) {
313 secondStart = ts;
314 }
315 auto prior = detail::smoothPriorNode(AnyConstTrackStateProxy{secondStart},
316 sharedMean, sharedCov, logger());
317 if (!prior.ok()) {
318 return prior.error();
319 }
320 std::tie(sharedMean, sharedCov) = *prior;
321 }
322
323 if (m_cfg.conditionFirst) {
324 auto res = detail::conditionSmoothedStatesOnSharedState(
325 trajectory, sharedState.index(), first.tipIndex(), sharedMean,
326 sharedCov, logger());
327 if (!res.ok()) {
328 ACTS_DEBUG("Conditioning of the first tracklet failed: "
329 << res.error().message());
330 return res.error();
331 }
332 } else {
333 detail::setSmoothedWithoutSharing(AnyMutableTrackStateProxy{sharedState},
334 sharedMean, sharedCov);
335 }
336
337 detail::linkTrackletBehind(sharedState, second, m_cfg.invertJacobians);
338 return Result<void>::success();
339 }
340
341 const Logger& logger() const { return *m_logger; }
342
343 Config m_cfg;
344 std::unique_ptr<const Logger> m_logger;
