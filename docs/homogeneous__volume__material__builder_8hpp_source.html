29template <typename detector_t>
30class homogeneous_volume_material_builder final
31 : public volume_decorator<detector_t> {
32 static_assert(concepts::detector<detector_t>);
33
34 public:
36 using scalar_type = dscalar<typename detector_t::algebra_type>;
37
40 DETRAY_HOST
41 explicit homogeneous_volume_material_builder(
42 std::unique_ptr<volume_builder_interface<detector_t>> vol_builder)
43 : volume_decorator<detector_t>(std::move(vol_builder)) {}
44
48 DETRAY_HOST
49 void set_material(material<scalar_type> mat) { m_volume_material = mat; }
50
56 DETRAY_HOST
57 auto build(detector_t &det, const volume_builder_options &opt,
58 typename detector_t::geometry_context ctx = {}) ->
59 typename detector_t::volume_type * override {
60 // Call the underlying volume builder(s)
61 typename detector_t::volume_type *vol =
62 volume_decorator<detector_t>::build(det, opt, ctx);
63
64 // Nothing left to do
65 if (m_volume_material == detray::vacuum<scalar_type>{}) {
66 DETRAY_WARN_HOST("Volume " << this->vol_index()
67 << " has vacuum material");
68
69 return vol;
70 }
71
72 constexpr auto material_id{detector_t::material::id::e_raw_material};
73
74 // Update the volume material link
75 dindex coll_size{det.material_store().template size<material_id>()};
76 vol->set_material(material_id, coll_size);
77
78 // Append the material
79 det._materials.template push_back<material_id>(m_volume_material);
80
81 // Give the volume to the next decorator
82 return vol;
83 }
84
85 private:
86 // Material for this volume
87 material<scalar_type> m_volume_material{};
88};
