99 virtual AnyLocalBins getNBinsAny() const {

156 virtual bool isInsideAny(const AnyPoint& point) const {
157 if (point.size() != getNAxes()) {
158 throw std::invalid_argument("Invalid number of coordinates");
159 }
160 for (std::size_t i = 0; i < point.size(); ++i) {
161 const IAxis& axis = getAxis(i);
162 if (!axis.isInside(point[i])) {
163 return false;
164 }
165 }
166 return true;
167 }

210 class iterator {
211 public:
213 using value_type = const IAxis;
215 using difference_type = std::ptrdiff_t;
217 using pointer = const IAxis*;
219 using reference = const IAxis&;
220
222 using iterator_category = std::random_access_iterator_tag;
224 using iterator_concept = std::random_access_iterator_tag;
225
226 constexpr iterator() noexcept = default;
230 constexpr iterator(const IMultiAxis& multiAxis, std::size_t index) noexcept
231 : m_multiAxis(&multiAxis), m_index(index) {}
232
235 constexpr reference operator*() const {
236 return m_multiAxis->getAxis(m_index);
237 }
238
240 constexpr iterator& operator++() noexcept {
241 ++m_index;
242 return *this;
243 }
244
246 constexpr iterator operator++(int) noexcept {
247 auto tmp = *this;
248 ++(*this);
249 return tmp;
250 }
251
253 constexpr iterator& operator--() noexcept {
254 --m_index;
255 return *this;
256 }
257
259 constexpr iterator operator--(int) noexcept {
260 auto tmp = *this;
261 --(*this);
262 return tmp;
263 }
264
267 constexpr iterator& operator+=(difference_type n) noexcept {
268 m_index += static_cast<std::size_t>(n);
269 return *this;
270 }
271
274 constexpr iterator& operator-=(difference_type n) noexcept {
275 m_index -= static_cast<std::size_t>(n);
276 return *this;
277 }
278
279 private:
280 const IMultiAxis* m_multiAxis{};
281 std::size_t m_index{};
282
283 friend constexpr iterator operator+(iterator it,
284 difference_type n) noexcept {
285 return it += n;
286 }
287
288 friend constexpr iterator operator+(difference_type n,
289 iterator it) noexcept {
290 return it += n;
291 }
292
293 friend constexpr iterator operator-(iterator it,
294 difference_type n) noexcept {
295 return it -= n;
296 }
297
298 friend constexpr difference_type operator-(const iterator& lhs,
299 const iterator& rhs) noexcept {
300 return static_cast<difference_type>(lhs.m_index - rhs.m_index);
301 }
302
303 friend constexpr auto operator<=>(const iterator& a,
304 const iterator& b) noexcept {
305 return a.m_index <=> b.m_index;
306 }
307
308 friend constexpr bool operator==(const iterator& a,
309 const iterator& b) noexcept {
310 return a.m_index == b.m_index;
311 }
312 };
313
315 iterator begin() const { return iterator(*this, 0); }
316
318 iterator end() const { return iterator(*this, getNAxes()); }
319
320 protected:
323 virtual void toStream(std::ostream& os) const {
324 for (std::size_t i = 0; i < getNAxes(); ++i) {
325 os << getAxis(i);
326 if (i < getNAxes() - 1) {
327 os << ", ";
328 }
329 }
330 }
331
332 private:
338 friend bool operator==(const IMultiAxis& lhs, const IMultiAxis& rhs) {
339 if (lhs.getNAxes() != rhs.getNAxes()) {
340 return false;
341 }
342 return std::ranges::equal(lhs, rhs);
343 }
344
349 friend std::ostream& operator<<(std::ostream& os,
350 const IMultiAxis& multiAxis) {
351 multiAxis.toStream(os);
352 return os;
353 }
