94 template <typename track_t, typename nav_state_t, typename context_t>
95 DETRAY_HOST_DEVICE DETRAY_INLINE constexpr bool update(
96 const track_t &track, nav_state_t &navigation,
97 const navigation::config &cfg, const context_t &ctx) const {
98 assert(navigation.is_alive());
99 assert(!track.is_invalid());
100
101 // Update was completely successful (most likely case)
102 if (navigation.trust_level() == navigation::trust_level::e_full) {
103 DETRAY_VERBOSE_HOST_DEVICE(
104 "-> Full trust, nothing left to do: dist to next %f mm",
105 navigation());
106 return false;
107 }
108
109 // Candidates are re-evaluated based on the current trust level.
110 // Should result in 'full trust'
111 constexpr const navigator_impl_t navigation_impl{};
112 bool is_init = navigation_impl.update_impl(track, navigation, cfg, ctx);
113
114 // Otherwise: if we encountered a portal, perform volume switch
115 if (navigation.is_on_portal()) {
116 navigation::volume_switch(track, navigation, cfg, ctx);
117 is_init = true;
118
119 // Reached end of detector
120 if (!navigation.is_alive()) {
121 return false;
122 }
123 }
124 // If no trust could be restored during the update, try to rescue the
125 // navigation stream by re-initializing with loose tolerances
126 if (navigation.trust_level() != navigation::trust_level::e_full ||
127 navigation.cache_exhausted()) {
128 is_init = true;
129 navigation::init_loose_cfg(track, navigation, cfg, ctx);
130
131 navigation.run_inspector(cfg, track.pos(), track.dir(), "Re-init: ");
132 }
133
134 DETRAY_VERBOSE_HOST("Status: " << navigation.status() << " (vol.:"
135 << navigation.volume() << ", next sf.: "
136 << navigation.next_surface().index() << ")");
137 if (navigation.is_on_surface()) {
138 DETRAY_VERBOSE_HOST("-> Current surface: "
139 << navigation.current_surface().index()
140 << ", has material: " << std::boolalpha
141 << navigation.current_surface().has_material()
142 << std::noboolalpha);
143 }
144 DETRAY_VERBOSE_HOST_DEVICE("Update complete: dist to next %f mm",
145 navigation());
146
147 return is_init;
148 }
149};
