846template <bool reverse, bool read_only>
847class AnyTrackStateRange {
848 using ProxyType = AnyTrackStateProxy<read_only>;
849
850 public:
853 struct Iterator {
855 std::optional<ProxyType> proxy;
856
857 using iterator_category = std::forward_iterator_tag;
858 using value_type = ProxyType;
859 using difference_type = std::ptrdiff_t;
860 using pointer = void;
861 using reference = void;
862
865 Iterator& operator++() {
866 if (!proxy) {
867 return *this;
868 }
869 if constexpr (reverse) {
870 if (proxy->hasPrevious()) {
871 proxy = proxy->withIndex(proxy->previous());
872 } else {
873 proxy = std::nullopt;
874 }
875 } else {
876 TrackIndexType next =
877 proxy->template component<TrackIndexType, hashString("next")>();
878 if (next != kTrackIndexInvalid) {
879 proxy = proxy->withIndex(next);
880 } else {
881 proxy = std::nullopt;
882 }
883 }
884 return *this;
885 }
