61 template <concepts::algebra other_algebra_t>
62 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
63 const trajectory_type<other_algebra_t> &ray, const transform3_type &trf,
64 const scalar_type /*overstep_tol*/ = 0.f) const {
65 // line direction
66 const vector3_type &sz = trf.z();
67
68 // line center
69 const point3_type &st = trf.translation();
70
71 // Broadcast ray data
72 const auto &pos = ray.pos();
73 const auto &dir = ray.dir();
74 const vector3_type ro{pos[0], pos[1], pos[2]};
75 const vector3_type rd{dir[0], dir[1], dir[2]};
76
77 // Projection of line to track direction
78 const scalar_type zd = vector::dot(sz, rd);
79
80 const scalar_type denom = 1.f - (zd * zd);
81
82 // Case for wire is parallel to track
83 if (detray::detail::all_of(denom < 1e-5f)) {
84 return {};
85 }
86
87 // vector from track position to line center
88 const vector3_type t2l = st - ro;
89
90 // t2l projection on line direction
91 const scalar_type t2l_on_line = vector::dot(t2l, sz);
92
93 // t2l projection on track direction
94 const scalar_type t2l_on_track = vector::dot(t2l, rd);
95
96 // path length to the point of closest approach on the track
97 scalar_type s = (t2l_on_track - t2l_on_line * zd) / denom;
98
99 // point of closest approach on the track
100 point3_type glob_pos = ro + s * rd;
101
102 return {s, glob_pos};
103 }
104};
