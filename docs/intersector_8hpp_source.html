123 const mask_t &mask, const transform3_type &trf,
124 const intersection::config &cfg = {},

136 template <typename surface_descr_t, typename mask_t>
137 DETRAY_HOST_DEVICE inline decltype(auto) operator()(
138 const detail::helix<algebra_t> &h, const surface_descr_t &sf,
139 const mask_t &mask, const transform3_type &trf,
140 const intersection::config &cfg = {},
141 const scalar_type /*unused*/ = 0.f) const {
142 return helix_intersector_type{}(h, sf, mask, trf, cfg);
143 }
144
145 private:
147 DETRAY_HOST_DEVICE
148 constexpr result_type to_result_type(
149 const ray_intersector_type::result_type &ray_results) const {
150 result_type result;
151 if constexpr (n_solutions > 1) {
152 for (std::size_t i = 0u; i < n_solutions; ++i) {
153 result[i] = typename result_type::value_type{ray_results[i]};
154 }
155 } else {
156 result = result_type{ray_results};
157 }
158 return result;
159 }
160};
