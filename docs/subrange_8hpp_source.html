117 template <detray::ranges::range deduced_range_t>

119 DETRAY_HOST_DEVICE constexpr explicit subrange(deduced_range_t &&range)
120 : base_type{detray::ranges::begin(range), detray::ranges::end(range)} {}

126 template <detray::ranges::range deduced_range_t, concepts::index index_t>
127 requires std::is_convertible_v<
128 index_t, detray::ranges::range_difference_t<deduced_range_t>>
129 DETRAY_HOST_DEVICE constexpr subrange(deduced_range_t &&range, index_t pos)
130 : base_type{detray::ranges::next(detray::ranges::begin(range),
131 static_cast<difference_t>(pos)),
132 detray::ranges::next(
133 detray::ranges::begin(range),
134 static_cast<difference_t>(pos) + difference_t{1})} {}
135
139 template <detray::ranges::range deduced_range_t,
140 concepts::interval deduced_index_range_t>
141 DETRAY_HOST_DEVICE constexpr subrange(deduced_range_t &&range,
142 const deduced_index_range_t &pos)
143 : base_type{get_itr<0>(range, pos), get_itr<1>(range, pos)} {}
144
148 template <detray::ranges::range deduced_range_t>
149 DETRAY_HOST_DEVICE constexpr auto operator()(deduced_range_t &&range) {
150 return detail::subrange_view<deduced_range_t>{get_itr<0>(range, m_pos),
151 get_itr<1>(range, m_pos)};
152 }
153
154 private:
160 template <std::size_t I, detray::ranges::range deduced_range_t,
161 concepts::interval deduced_index_range_t>
162 DETRAY_HOST_DEVICE constexpr auto get_itr(deduced_range_t &&range,
163 const deduced_index_range_t &pos) {
164 return detray::ranges::next(
165 detray::ranges::begin(std::forward<deduced_range_t>(range)),
166 static_cast<difference_t>(detray::detail::get<I>(pos)));
167 }
168
170 index_range_t m_pos{};
171};
