65 template <typename propagator_state_t>
66 DETRAY_HOST_DEVICE void operator()(state& tracer_state,
67 propagator_state_t& prop_state) const {
68 tracer_state.m_steps.push_back(collect_data(prop_state));
69 }

76 template <typename propagator_state_t>
77 DETRAY_HOST_DEVICE void operator()(
78 state& tracer_state, propagator_state_t& prop_state,
79 const parameter_transporter_result<algebra_t>& res) const {
80 const auto& navigation = prop_state.navigation();
81 assert(navigation.is_on_surface());
82
83 // If the state has already been collected by the other call operator,
84 // update the bound track parameters
85 if (!tracer_state.m_steps.empty() &&
86 navigation.geometry_identifier() ==
87 tracer_state.m_steps.back().identifier()) {
88 tracer_state.m_steps.back().bound_param() = res.destination_params();
89 } else {
90 tracer_state.m_steps.push_back(
91 collect_data(prop_state, res.destination_params()));
92 }
93 }
94
95 private:
97 template <typename propagator_state_t>
98 DETRAY_HOST_DEVICE step_data_t collect_data(
99 propagator_state_t& prop_state,
100 const bound_track_parameters<algebra_t>& bound_param = {}) const {
101 const auto& navigation = prop_state.navigation();
102 const auto& stepping = prop_state.stepping();
103
104 const auto geo_id{navigation.is_on_surface()
105 ? navigation.geometry_identifier()
106 : geometry::identifier{}};
107
108 return {stepping.step_size(),
109 stepping.path_length(),
110 stepping.n_total_trials(),
111 navigation.direction(),
112 geo_id,
113 stepping(),
114 bound_param,
115 stepping.transport_jacobian()};
116 }
117};
