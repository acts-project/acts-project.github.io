69template <detray::concepts::detector detector_t>
70inline auto read_intersection2D(const std::string &file_name) {
71 using algebra_t = typename detector_t::algebra_type;
72 using scalar_t = dscalar<algebra_t>;
73 using surface_t = typename detector_t::surface_type;
74 using nav_link_t = typename surface_t::navigation_link;
75 using mask_link_t = typename surface_t::mask_link;
76 using material_link_t = typename surface_t::material_link;
77 using mask_id_t = typename detector_t::masks::id;
78 using material_id_t = typename detector_t::material::id;
79
80 using intersection_t =
81 detray::intersection2D<surface_t, algebra_t, intersection::contains_pos>;
82
83 dfe::NamedTupleCsvReader<io::csv::intersection2D> inters_reader(file_name);
84
85 io::csv::intersection2D inters_data{};
86 std::vector<std::vector<intersection_t>> intersections_per_track;
87
88 // Read the data
89 while (inters_reader.read(inters_data)) {
90 // Add new intersection to correct track
91 dindex trk_index{inters_data.track_id};
92 while (intersections_per_track.size() <= trk_index) {
93 intersections_per_track.push_back({});
94 }
95
96 // Read the intersection
97 intersection_t inters{};
98
99 mask_link_t mask_link{};
100 mask_link.set_id(static_cast<mask_id_t>(inters_data.mask_id));
101 if constexpr (detray::concepts::interval<
102 typename mask_link_t::index_type>) {
103 mask_link.set_index({inters_data.mask_index, inters_data.n_masks});
104 } else {
105 mask_link.set_index(inters_data.mask_index);
106 }
107
108 material_link_t material_link{
109 static_cast<material_id_t>(inters_data.material_id),
110 inters_data.material_index};
111 inters.set_surface({inters_data.transform_index, mask_link, material_link,
112 dindex_invalid, surface_id::e_unknown});
113 inters.surface().set_identifier(
114 geometry::identifier{inters_data.identifier});
115 inters.set_local({static_cast<scalar_t>(inters_data.loc_0),
116 static_cast<scalar_t>(inters_data.loc_1), 0.f});
117 inters.set_path(static_cast<scalar_t>(inters_data.path));
118 inters.set_volume_link(static_cast<nav_link_t>(inters_data.volume_link));
119 inters.set_direction(static_cast<bool>(inters_data.direction));
120 inters.set_status(static_cast<intersection::status>(inters_data.status));
121
122 // Add to collection
123 intersections_per_track[trk_index].push_back(inters);
124 }
125
126 // Check the result
127 if (intersections_per_track.empty()) {
128 throw std::invalid_argument(
129 "ERROR: csv reader: Failed to read intersection data");
130 }
131
132 return intersections_per_track;
133}

136template <typename intersection_t>
137inline void write_intersection2D(
138 const std::string &file_name,
139 const std::vector<std::vector<intersection_t>> &intersections_per_track,
140 const bool replace = true) {
141 using sf_desc_t = decltype(intersections_per_track.at(0).at(0).surface());
142 using mask_link_t = typename sf_desc_t::mask_link;
143
144 // Don't write over existing data
145 std::string inters_file_name{file_name};
146 if (!replace && io::file_exists(file_name)) {
147 inters_file_name = io::alt_file_name(file_name);
148 } else {
149 // Make sure the output directories exit
150 io::create_path(std::filesystem::path{inters_file_name}.parent_path());
151 }
152
153 dfe::NamedTupleCsvWriter<io::csv::intersection2D> inters_writer(
154 inters_file_name);
155
156 for (const auto &[track_idx, intersections] :
157 detray::views::enumerate(intersections_per_track)) {
158 // Skip empty traces
159 if (intersections.empty()) {
160 continue;
161 }
162
163 for (const auto &inters : intersections) {
164 io::csv::intersection2D inters_data{};
165
166 inters_data.track_id = track_idx;
167 inters_data.identifier = inters.surface().identifier().value();
168 inters_data.type =
169 static_cast<unsigned int>(inters.surface().identifier().id());
170 inters_data.transform_index = inters.surface().transform();
171 inters_data.mask_id =
172 static_cast<unsigned int>(inters.surface().mask().id());
173 const auto mask_index = inters.surface().mask().index();
174 if constexpr (detray::concepts::interval<
175 typename mask_link_t::index_type>) {
176 inters_data.mask_index = static_cast<unsigned int>(mask_index.lower());
177 inters_data.n_masks = static_cast<unsigned int>(mask_index.size());
178 } else {
179 inters_data.mask_index = static_cast<unsigned int>(mask_index);
180 inters_data.n_masks = 1u;
181 }
182 inters_data.material_id =
183 static_cast<unsigned int>(inters.surface().material().id());
184 inters_data.material_index = inters.surface().material().index();
185 inters_data.loc_0 = inters.local()[0];
186 inters_data.loc_1 = inters.local()[1];
187 inters_data.path = inters.path();
188 inters_data.volume_link = inters.volume_link();
189 inters_data.direction = static_cast<int>(inters.is_along());
190 inters_data.status = static_cast<int>(inters.status());
191
192 inters_writer.append(inters_data);
193 }
194 }
195}
