40 enum boundaries : unsigned int {
41 e_half_length_0 = 0u,
42 e_half_length_1 = 1u,
43 e_half_length_2 = 2u,
44 e_divisor = 3u, // 1 / (2 * bounds[e_half_length_2])
45 e_size = 4u,

72 template <concepts::scalar scalar_t, concepts::point point_t>
73 DETRAY_HOST_DEVICE inline scalar_t min_dist_to_boundary(
74 const bounds_type<scalar_t> &bounds, const point_t &loc_p) const {
75 // Minimum distance between loc point and line between corner points
76 const scalar_t d_y = -2.f * bounds[e_half_length_2];
77 const scalar_t d_x = bounds[e_half_length_1] - bounds[e_half_length_0];
78
79 const scalar_t denom = math::sqrt(d_y * d_y + d_x * d_x);
80 const scalar_t dist_to_line =
81 math::fabs(d_y * loc_p[0] - d_x * loc_p[1] -
82 bounds[e_half_length_2] *
83 (bounds[e_half_length_1] + bounds[e_half_length_0])) /
84 denom;
85
86 return math::min(dist_to_line, math::fabs(math::fabs(loc_p[1]) -
87 bounds[e_half_length_2]));
88 }

101 const bounds_type<dscalar<algebra_t>> &bounds,

123 template <concepts::scalar scalar_t, concepts::point point_t>
124 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
125 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
126 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
127 const scalar_t edge_tol = 0.f) const {
128 const scalar_t rel_y =
129 (bounds[e_half_length_2] + loc_p[1]) * bounds[e_divisor];
130 const scalar_t bound_x{
131 bounds[e_half_length_0] +
132 rel_y * (bounds[e_half_length_1] - bounds[e_half_length_0])};
133
134 auto inside_mask{(math::fabs(loc_p[0]) <= (bound_x + tol) &&
135 math::fabs(loc_p[1]) <= (bounds[e_half_length_2] + tol))};
136
137 decltype(inside_mask) inside_edge{false};
138 if (detail::any_of(edge_tol > 0.f)) {
139 const scalar_t full_tol{tol + edge_tol};
140
141 inside_edge =
142 (math::fabs(loc_p[0]) <= (bound_x + full_tol) &&
143 math::fabs(loc_p[1]) <= (bounds[e_half_length_2] + full_tol));
144 }
145
146 return result_type<decltype(inside_mask)>{inside_mask, inside_edge};
147 }

167 DETRAY_HOST_DEVICE constexpr scalar_t area(
168 const bounds_type<scalar_t> &bounds) const {

179 template <concepts::scalar scalar_t>
180 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
181 const bounds_type<scalar_t> &bounds,
182 const bounds_type<scalar_t> &o_bounds) const {
183 bounds_type<scalar_t> new_bounds{};
184
185 new_bounds[e_half_length_0] =
186 math::max(bounds[e_half_length_0], o_bounds[e_half_length_0]);
187 new_bounds[e_half_length_1] =
188 math::max(bounds[e_half_length_1], o_bounds[e_half_length_1]);
189 new_bounds[e_half_length_2] =
190 math::max(bounds[e_half_length_2], o_bounds[e_half_length_2]);
191
192 // Recalculate the memoized divisor
193 new_bounds[e_divisor] = 1.f / (2.f * new_bounds[e_half_length_2]);
194
195 return new_bounds;
196 }

207 template <concepts::algebra algebra_t>
208 DETRAY_HOST_DEVICE inline darray<dscalar<algebra_t>, 6> local_min_bounds(
209 const bounds_type<dscalar<algebra_t>> &bounds,
210 const dscalar<algebra_t> env =
211 std::numeric_limits<dscalar<algebra_t>>::epsilon()) const {
212 using scalar_t = dscalar<algebra_t>;
213
214 assert(env > 0.f);
215 const scalar_t x_bound{(bounds[e_half_length_0] > bounds[e_half_length_1]
216 ? bounds[e_half_length_0]
217 : bounds[e_half_length_1]) +
218 env};
219 const scalar_t y_bound{bounds[e_half_length_2] + env};
220 return {-x_bound, -y_bound, -env, x_bound, y_bound, env};
221 }

225 template <concepts::algebra algebra_t>
226 DETRAY_HOST_DEVICE dpoint3D<algebra_t> centroid(
227 const bounds_type<dscalar<algebra_t>> &bounds) const {
228 using scalar_t = dscalar<algebra_t>;
229
230 const scalar_t h_2{bounds[e_half_length_2]};
231 const scalar_t a_2{bounds[e_half_length_1]};
232 const scalar_t b_2{bounds[e_half_length_0]};
233
234 const scalar_t y{2.f * h_2 * (2.f * a_2 + b_2) * 1.f / (3.f * (a_2 + b_2))};
235
236 return {static_cast<scalar_t>(0), y - h_2, static_cast<scalar_t>(0)};
237 }

244 template <concepts::algebra algebra_t>
245 DETRAY_HOST dvector<dpoint3D<algebra_t>> vertices(
246 const bounds_type<dscalar<algebra_t>> &bounds, dindex /*ignored*/) const {
247 using point3_t = dpoint3D<algebra_t>;
248
249 // left hand lower corner
250 point3_t lh_lc{-bounds[e_half_length_0], -bounds[e_half_length_2],
251 static_cast<dscalar<algebra_t>>(0.f)};
252 // right hand lower corner
253 point3_t rh_lc{bounds[e_half_length_0], -bounds[e_half_length_2],
254 static_cast<dscalar<algebra_t>>(0.f)};
255 // right hand upper corner
256 point3_t rh_uc{bounds[e_half_length_1], bounds[e_half_length_2],
257 static_cast<dscalar<algebra_t>>(0.f)};
258 // left hand upper corner
259 point3_t lh_uc{-bounds[e_half_length_1], bounds[e_half_length_2],
260 static_cast<dscalar<algebra_t>>(0.f)};
261
262 // Return the confining vertices
263 return {lh_lc, rh_lc, rh_uc, lh_uc};
264 }

272 template <concepts::scalar scalar_t>
273 DETRAY_HOST constexpr bool check_consistency(
274 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
275 constexpr auto tol{10.f * std::numeric_limits<scalar_t>::epsilon()};
276
277 if (bounds[e_half_length_0] < tol || bounds[e_half_length_1] < tol) {
278 os << "DETRAY ERROR (HOST): Half length in x must be in the range "
279 "(0, "
280 "numeric_max)"
281 << std::endl;
282 return false;
283 }
284 if (bounds[e_half_length_2] < tol) {
285 os << "DETRAY ERROR (HOST): Half length in y must be in the range "
286 "(0, "
287 "numeric_max)"
288 << std::endl;
289 return false;
290 }
291
292 if (const auto div{1.f / (2.f * bounds[e_half_length_2])};
293 math::fabs(bounds[e_divisor] - div) > tol) {
294 os << "DETRAY ERROR (HOST): Divisor incorrect. Should be: " << div
295 << std::endl;
296 return false;
297 }
298
299 return true;
300 }
