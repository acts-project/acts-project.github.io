40inline std::ostream& operator<<(std::ostream& os, const JetLabel& label) {
41 switch (label) {
42 case JetLabel::Unknown:
43 os << "Unknown";
44 break;
45 case JetLabel::BJet:
46 os << "BJet";
47 break;
48 case JetLabel::CJet:
49 os << "CJet";
50 break;

53 break;

55 os << "InvalidJetLabel";
