207 // Backwards compatibility: the pre-split fitter state and the entry points
208 // taking it. Both are adapters on top of the interface above, which they
209 // reproduce exactly: the state is decomposed into a problem and a cache on
210 // the way in and recomposed from them on the way out.
211 //
212 // Only the entry points carry the deprecation attribute. The types are
213 // documented as deprecated but not attributed, so that naming them in the
214 // adapter's own declarations does not warn -- a caller reaching the legacy
215 // interface has to go through fit() or addVtxToFit() anyway.

255 void addVertexToMultiMap(Vertex& vtx) {
256 for (auto trk : vtxInfoMap[&vtx].trackLinks) {
257 trackToVerticesMultiMap.emplace(trk, &vtx);
258 }
259 }

263 void removeVertexFromMultiMap(const Vertex& vtx) {
264 for (auto iter = trackToVerticesMultiMap.begin();
265 iter != trackToVerticesMultiMap.end();) {
266 if (iter->second == &vtx) {
267 iter = trackToVerticesMultiMap.erase(iter);
268 } else {
269 ++iter;
270 }
271 }
272 }

278 Result<void> removeVertexFromCollection(Vertex& vtxToRemove,
279 const Logger& logger) {
280 auto it = std::ranges::find(vertexCollection, &vtxToRemove);
281 // Check if the value was found before erasing
282 if (it == vertexCollection.end()) {
283 ACTS_ERROR("vtxToRemove is not part of vertexCollection.");
284 return VertexingError::ElementNotFound;
285 }
286 // Erase the element if found
287 vertexCollection.erase(it);
288 return {};
289 }
