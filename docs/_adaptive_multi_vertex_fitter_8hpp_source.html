342 void removeVertexFromMultiMap(const Vertex& vtx) {
343 for (auto iter = trackToVerticesMultiMap.begin();
344 iter != trackToVerticesMultiMap.end();) {
345 if (iter->second == &vtx) {
346 iter = trackToVerticesMultiMap.erase(iter);
347 } else {
348 ++iter;
349 }
350 }
351 }

357 Result<void> removeVertexFromCollection(Vertex& vtxToRemove,
358 const Logger& logger) {
359 auto it = std::ranges::find(vertexCollection, &vtxToRemove);
360 // Check if the value was found before erasing
361 if (it == vertexCollection.end()) {
362 ACTS_ERROR("vtxToRemove is not part of vertexCollection.");
363 return VertexingError::ElementNotFound;
364 }
365 // Erase the element if found
366 vertexCollection.erase(it);
367 return {};
368 }
