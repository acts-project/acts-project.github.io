80 template <typename propagator_state_t, typename stepper_t,
81 typename navigator_t, typename... Args>
82 Result<void> act(propagator_state_t& state, const stepper_t& stepper,
83 const navigator_t& navigator, State& actorState,
84 Args&&... args) const {
85 Result<void> result = Result<void>::success();
86 std::apply(
87 [&](const actors_t&... actor) {
88 // Stop at the first actor that fails
89 static_cast<void>(
90 ((result = detail::callActorAct(
91 actor, stateOf<actors_t>(actorState), state, stepper,
92 navigator, std::forward<Args>(args)...))
93 .ok() &&
94 ...));
95 },
96 m_actors);
97 return result;
98 }
