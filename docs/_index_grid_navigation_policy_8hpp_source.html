44template <typename GridType>
45class IndexGridNavigationPolicy : public INavigationPolicy {
46 public:
48 using IndexGridType = IndexGrid<GridType>;
49
58 explicit IndexGridNavigationPolicy(const GeometryContext& gctx,
59 const TrackingVolume& volume,
60 const Logger& logger,
61 const IndexGridNavigationConfig& config,
62 const IndexGridType& grid)
63 : m_cfg(config), m_volume(volume), m_indexGrid(grid) {
64 ACTS_VERBOSE("Constructing IndexGridNavigationPolicy for volume '"
65 << m_volume.volumeName() << "'");
66
67 // Fill the grid with the surfaces from the volume
68 IndexGridFiller filler{m_cfg.binExpansion, m_cfg.referenceExpansion,
69 getDefaultLogger("IndexGridFiller", logger.level())};
70 const auto& surfaces = m_volume.surfaces();
71 // Fill the grid with surfaces
72 std::vector<std::shared_ptr<const Surface>> surfacePtrs = {};
73 surfacePtrs.reserve(surfaces.size());
74 for (const auto& surface : surfaces) {
75 surfacePtrs.push_back(surface.getSharedPtr());
76 }
77 filler.fill(gctx, m_indexGrid, surfacePtrs, *m_cfg.referenceGenerator);
78 }
79
86 void initializeCandidates(const GeometryContext& gctx,
87 const NavigationArguments& args,
88 NavigationPolicyState& state,
89 AppendOnlyNavigationStream& stream,
90 const Logger& logger) const {
91 static_cast<void>(state); // unused
92 ACTS_VERBOSE(
93 "IndexGridNavigationPolicy: candidates initialization for volume '"
94 << m_volume.volumeName() << "'");
95
96 // Nominal or intersected position
97 Vector3 position = args.position;
98 if (m_cfg.surface) {
99 auto multiIntersection = m_cfg.surface->intersect(
100 gctx, position, args.direction, BoundaryTolerance::Infinite());
101 position = multiIntersection.closestForward().position();
102 }
103
104 // The indexing is guaranteed
105 const auto& surfaces = m_volume.surfaces();
106 const auto& indices =
107 m_indexGrid.grid.atPosition(GridAccessHelpers::castPosition<GridType>(
108 m_indexGrid.toLocalFrame(gctx) * position, m_indexGrid.casts));
109 // Fill the navigation stream with the container
110 for (const auto& idx : indices) {
111 stream.addSurfaceCandidate(surfaces[idx], args.tolerance);
112 }
113 }
114
117 void connect(NavigationDelegate& delegate) const override {
118 connectDefault<IndexGridNavigationPolicy>(delegate);
119 }
120
123 const IndexGridType& indexGrid() const { return m_indexGrid; }
124
125 private:
126 // Keep the config
127 IndexGridNavigationConfig m_cfg;
128 // The tracking volume
129 const TrackingVolume& m_volume;
130 // The index grid
131 IndexGridType m_indexGrid;
132};
