182template <typename T>
183std::string demangle_type_name() {
184#if defined(__clang__)
185 constexpr std::string_view prefix{"[T = "};
186 constexpr std::string_view suffix{"]"};
187 constexpr std::string_view function{__PRETTY_FUNCTION__};
188#elif defined(__GNUC__)
189 constexpr std::string_view prefix{"with T = "};
190 constexpr std::string_view suffix{"; "};
191 constexpr std::string_view function{__PRETTY_FUNCTION__};
192#elif defined(_MSC_VER)
193 constexpr std::string_view prefix{"get_type_name<"};
194 constexpr std::string_view suffix{">(void)"};
195 constexpr std::string_view function{__FUNCSIG__};
196#else
197 std::string type_name{typeid(T).name()};
198 type_name += " ";
199 std::string_view function{type_name};
200 constexpr std::string_view prefix{""};
201 constexpr std::string_view suffix{" "};
202#endif
203
204 const std::size_t start{function.find(prefix) + prefix.size()};
205 const std::size_t end{function.find(suffix)};
206 const std::size_t length{end - start};
207
208 return std::string{function.substr(start, length)};
209}

213template <typename T>
214std::string get_name(bool full = false) {
215 std::string tp_str{""};
216 try {
217 tp_str = detray::types::demangle_type_name<T>();
218 } catch (...) {
219 return "unknown";
220 }
221
222 if (tp_str.empty()) {
223 return "unknown";
224 }
225
226 if (full) {
227 return tp_str;
228 }
229
230 // Remove template parameter list by removing everything behind the first
231 // occurrence of '<'
232 auto tokens = detray::utils::split_at_delim(tp_str, '<');
233 tp_str = tokens.front();
234
235 tokens.clear();
236
237 // Strip the namespaces by removing everything before the last occurrence of
238 // ':'
239 tokens = detray::utils::split_at_delim(tp_str, ':');
240 return tokens.back();
241}

286template <typename orig_list_t, class type_selector, std::size_t I = 0u,
287 typename... Fs>
288DETRAY_HOST_DEVICE consteval auto filtered_list(
289 const list<Fs...>& filtered = list<>{}) {
290 static_assert(sizeof...(Fs) <= I, "Can only map down to list with ");
291
292 // The current list of mapped types
293 using list_t = list<Fs...>;
294
295 if constexpr (I == size<orig_list_t>) {
296 return filtered;
297 } else {
298 // Map the current type corresponding to position 'I'
299 // to the new type according to the given type_selector
300 using next_type = at<orig_list_t, I>;
301 using mapped_t = typename type_selector::template type<next_type>;
302
303 // Don't add the same type multiple times
304 if constexpr (contains<list_t, mapped_t>) {
305 return filtered_list<orig_list_t, type_selector, I + 1u>(filtered);
306 } else {
307 return filtered_list<orig_list_t, type_selector, I + 1u>(
308 push_back<list_t, mapped_t>{});
309 }
310 }
311}

325template <typename orig_list_t, class type_selector, std::size_t I = 0u,
326 typename... Fs>
327DETRAY_HOST_DEVICE consteval auto filtered_indices(
328 const list<Fs...>& filtered,
329 std::array<dindex, size<orig_list_t>> idx_array = {0}) {
330 // The current list of mapped types
331 using list_t = list<Fs...>;
332
333 if constexpr (I == size<orig_list_t>) {
334 return idx_array;
335 } else {
336 // Map the current type corresponding to position 'I'
337 // to the new type according to the given type_selector
338 using next_type = at<orig_list_t, I>;
339 using mapped_t = typename type_selector::template type<next_type>;
340
341 // If the type is already mapped, recover the index
342 if constexpr (contains<list_t, mapped_t>) {
343 idx_array[I] = position<list_t, mapped_t>;
344
345 return filtered_indices<orig_list_t, type_selector, I + 1u>(filtered,
346 idx_array);
347 } else {
348 idx_array[I] = sizeof...(Fs);
349
350 return filtered_indices<orig_list_t, type_selector, I + 1u>(
351 push_back<list_t, mapped_t>{}, idx_array);
352 }
353 }
