64template <concepts::scalar scalar_t>
65DETRAY_HOST_DEVICE constexpr pdg_particle<scalar_t> charge_conjugation(
66 const pdg_particle<scalar_t>& ptc) {
67 return (ptc.charge() != 0)
68 ? detray::pdg_particle<scalar_t>{-ptc.pdg_num(), ptc.mass(),
69 -ptc.charge()}
70 : ptc;
71}

77template <concepts::scalar scalar_t, typename track_t>
78DETRAY_HOST_DEVICE constexpr pdg_particle<scalar_t> update_particle_hypothesis(
79 const pdg_particle<scalar_t>& ptc, const track_t& params) {
80 return (ptc.charge() * params.qop() > 0.f) ? ptc : charge_conjugation(ptc);
81}
