24struct GaussianComponent {
25 double weight = 0;
26 double mean = 0;
27 double var = 0;
28};
29
32inline GaussianComponent transformComponent(const GaussianComponent &cmp) {
33 GaussianComponent transformed_cmp;
34 transformed_cmp.weight = std::log(cmp.weight) - std::log(1 - cmp.weight);
35 transformed_cmp.mean = std::log(cmp.mean) - std::log(1 - cmp.mean);
36 transformed_cmp.var = std::log(cmp.var);
37 return transformed_cmp;
38}
39
42inline GaussianComponent inverseTransformComponent(
43 const GaussianComponent &transformed_cmp) {
44 GaussianComponent cmp;
45 cmp.weight = 1 / (1 + std::exp(-transformed_cmp.weight));
46 cmp.mean = 1 / (1 + std::exp(-transformed_cmp.mean));
47 cmp.var = std::exp(transformed_cmp.var);
48 return cmp;
49}

86class BetheHeitlerApproxSingleCmp final : public BetheHeitlerApprox {
87 public:
90 std::size_t maxComponents() const override { return 1; }
91
96 bool validXOverX0(const double xOverX0) const override {
97 return xOverX0 < 0.002;
98 }
99
106 std::span<Component> mixture(
107 const double xOverX0, const std::span<Component> mixture) const override {
108 mixture[0].weight = 1.0;
109
110 const double c = xOverX0 / std::numbers::ln2;
111 mixture[0].mean = std::pow(2, -c);
112 mixture[0].var = std::pow(3, -c) - std::pow(4, -c);
113
114 return mixture;
115 }
