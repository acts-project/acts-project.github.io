25struct GaussianComponent {
26 double weight = 0;
27 double mean = 0;
28 double var = 0;
29};
31
34inline GaussianComponent transformComponent(const GaussianComponent &cmp) {
35 GaussianComponent transformed_cmp;
36 transformed_cmp.weight = std::log(cmp.weight) - std::log(1 - cmp.weight);
37 transformed_cmp.mean = std::log(cmp.mean) - std::log(1 - cmp.mean);
38 transformed_cmp.var = std::log(cmp.var);
39 return transformed_cmp;
40}
41
44inline GaussianComponent inverseTransformComponent(
45 const GaussianComponent &transformed_cmp) {
46 GaussianComponent cmp;
47 cmp.weight = 1 / (1 + std::exp(-transformed_cmp.weight));
48 cmp.mean = 1 / (1 + std::exp(-transformed_cmp.mean));
49 cmp.var = std::exp(transformed_cmp.var);
50 return cmp;
51}

90class BetheHeitlerApproxSingleCmp final : public BetheHeitlerApprox {
91 public:
94 std::size_t maxComponents() const override { return 1; }

100 bool validXOverX0(const double xOverX0) const override {
101 return xOverX0 < 0.002;
102 }
103
110 std::span<Component> mixture(
111 const double xOverX0, const std::span<Component> mixture) const override {
112 mixture[0].weight = 1.0;
113
115 const double c = xOverX0 / std::numbers::ln2;
116 mixture[0].mean = std::pow(2, -c);
117 mixture[0].var = std::pow(3, -c) - std::pow(4, -c);
119
120 return mixture;
121 }
