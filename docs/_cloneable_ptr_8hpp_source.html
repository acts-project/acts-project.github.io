26template <typename T>

44 explicit CloneablePtr(std::unique_ptr<T> ptr)

66 CloneablePtr(const CloneablePtr& other)

73 CloneablePtr& operator=(const CloneablePtr& other) {
74 if (this != &other) {
75 m_ptr = other.m_ptr ? other.m_cloner(*other.m_ptr) : nullptr;
76 m_cloner = other.m_cloner;
77 }
78 return *this;
79 }

97 T* operator->() const { return m_ptr.get(); }
98
100 explicit operator bool() const { return m_ptr != nullptr; }
101
103 friend bool operator==(const CloneablePtr& lhs, std::nullptr_t) {

105 }
106
109 T* get() const { return m_ptr.get(); }
110
113 T* release() {
114 m_cloner = nullptr;

116 }
117
120 void reset(T* ptr = nullptr) {
121 m_ptr.reset(ptr);
122 if (m_ptr == nullptr) {
123 m_cloner = nullptr;
124 }
125 }
126
127 private:
