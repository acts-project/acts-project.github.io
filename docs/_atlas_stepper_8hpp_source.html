493 void update(State& state, const Vector3& uposition, const Vector3& udirection,
494 double qop, double time) const {
495 // update the vector
496 state.pVector[0] = uposition[0];
497 state.pVector[1] = uposition[1];
498 state.pVector[2] = uposition[2];
499 state.pVector[3] = time;
500 state.pVector[4] = udirection[0];
501 state.pVector[5] = udirection[1];
502 state.pVector[6] = udirection[2];
503 // The scaled q/p Jacobian column requires a nonzero divisor on transport.
504 constexpr double minQOverP = 1e-15;
505 state.pVector[7] =
506 state.particleHypothesis.absoluteCharge() == 0.
507 ? qop
508 : std::copysign(std::max(std::abs(qop), minQOverP), qop);
509 state.newField = true;
510 state.derivativeReady = false;
511 }

520 Jacobian transportToCurvilinear(State& state) const {
521 if (!state.cov.has_value()) {
522 return Jacobian::Identity();
523 }
524
525 const auto surface =
526 CurvilinearSurface(position(state), direction(state)).surface();
527 return transportToBound(state, *surface).value();
528 }
529
540 Result<Jacobian> transportToBound(
541 State& state, const Surface& surface,
542 const FreeToBoundCorrection& /*freeToBoundCorrection*/ =
543 FreeToBoundCorrection(false)) const {
544 if (!surface.isOnSurface(state.options.geoContext, position(state),
545 direction(state), BoundaryTolerance::Infinite())) {
546 return Result<Jacobian>::failure(
547 SurfaceError::GlobalPositionNotOnSurface);
548 }
549
550 if (!state.cov.has_value()) {
551 return Result<Jacobian>::success(Jacobian::Identity());
552 }
553
554 auto prepared = prepareDerivatives(state);
555 if (!prepared.ok()) {
556 return Result<Jacobian>::failure(prepared.error());
557 }
558
559 const Vector3 pos = position(state);
560 const Vector3 dir = direction(state);
561 BoundToFreeMatrix transportJacobian =
562 Eigen::Map<const BoundToFreeMatrix>(state.pVector.data() + eFreeSize);
563 // Undo the kernel's q/p scaling of the position and direction derivatives.
564 const double inverseQOverP = qOverP(state) == 0. ? 0. : 1. / qOverP(state);
565 transportJacobian.block<3, 1>(eFreePos0, eBoundQOverP) *= inverseQOverP;
566 transportJacobian.block<3, 1>(eFreeDir0, eBoundQOverP) *= inverseQOverP;
567
568 FreeVector pathDerivative = FreeVector::Zero();
569 pathDerivative.segment<3>(eFreePos0) = dir;
570 pathDerivative[eFreeTime] = state.pVector[59];
571 pathDerivative.segment<3>(eFreeDir0) =
572 Eigen::Map<const Vector3>(state.pVector.data() + 56);
573 const FreeToPathMatrix freeToPath =
574 surface.freeToPathDerivative(state.options.geoContext, pos, dir);
575 const FreeToBoundMatrix freeToBound =
576 surface.freeToBoundJacobian(state.options.geoContext, pos, dir);
577 const Jacobian jac =
578 freeToBound *
579 (transportJacobian + pathDerivative * (freeToPath * transportJacobian));
580 *state.cov = jac * *state.cov * jac.transpose();
581
582 resetJacobian(state, surface);
583
584 return Result<Jacobian>::success(jac);
585 }

599 Result<double> step(State& state, Direction propDir,
600 const IVolumeMaterial* material) const {
601 static_cast<void>(material);
602
603 // Keep the scalar notation of the original ATLAS kernel.
604 auto h = state.stepSize.value() * propDir;
605 const bool withJacobian = state.cov.has_value();
606 if (!std::isfinite(h)) {
607 return EigenStepperError::StepInvalid;
608 }
609 // Skip a zero step: it is the identity, and h = 0 raises an FPE in the
610 // 2 / h below
611 if (h == 0.) {
612 return 0.;
613 }
614
615 double* R = &(state.pVector[0]); // Coordinates
616 double* A = &(state.pVector[4]); // Directions
617 double* sA = &(state.pVector[56]);
618 // A neutral particle must not bend in a magnetic field.
619 const double halfQOverP = charge(state) == 0. ? 0. : 0.5 * qOverP(state);
620 Vector3 f0, f;
621
622 // if new field is required get it
623 if (state.newField) {
624 const Vector3 pos(R[0], R[1], R[2]);
625 // This is sd.B_first in EigenStepper
626 auto fRes = getField(state, pos);
627 if (!fRes.ok()) {
628 return fRes.error();
629 }
630 f0 = *fRes;
631 } else {
632 f0 = state.field;
633 }
634
635 const auto calcStepSizeScaling = [&](const double errorEstimate) -> double {
636 // For details about these values see ATL-SOFT-PUB-2009-001
637 constexpr double lower = 0.25;
638 constexpr double upper = 4.0;
639 if (errorEstimate == 0.) {
640 return upper;
641 }
642 // Fourth-order controller; two square roots avoid a general pow call.
643 double x =
644 std::sqrt(std::sqrt(state.options.stepTolerance / errorEstimate));
645
646 return std::clamp(x, lower, upper);
647 };
648
649 const auto isErrorTolerable = [&](const double errorEstimate) {
650 // For details about these values see ATL-SOFT-PUB-2009-001
651 constexpr double marginFactor = 4.0;
652
653 return errorEstimate <= marginFactor * state.options.stepTolerance;
654 };
655
656 double errorEstimate = 0;
657 double initialStepSize = h;
658
659 std::uint32_t nStepTrials = 0;
660 while (true) {
661 ++state.statistics.nAttemptedSteps;
662
663 // PS2 is h/(2*momentum) in EigenStepper
664 double S3 = (1. / 3.) * h, S4 = .25 * h, PS2 = halfQOverP * h;
665
666 // First point
667 //
668 // H0 is (h/(2*momentum) * sd.B_first) in EigenStepper
669 double H0[3] = {f0[0] * PS2, f0[1] * PS2, f0[2] * PS2};
670 // { A0, B0, C0 } is (h/2 * sd.k1) in EigenStepper
671 double A0 = A[1] * H0[2] - A[2] * H0[1];
672 double B0 = A[2] * H0[0] - A[0] * H0[2];
673 double C0 = A[0] * H0[1] - A[1] * H0[0];
674 // { A2, B2, C2 } is (h/2 * sd.k1 + direction) in EigenStepper
675 double A2 = A0 + A[0];
676 double B2 = B0 + A[1];
677 double C2 = C0 + A[2];
678 // { A1, B1, C1 } is (h/2 * sd.k1 + 2*direction) in EigenStepper
679 double A1 = A2 + A[0];
680 double B1 = B2 + A[1];
681 double C1 = C2 + A[2];
682
683 // Second point
684 const Vector3 middlePosition(R[0] + A1 * S4, R[1] + B1 * S4,
685 R[2] + C1 * S4);
686 // This is sd.B_middle in EigenStepper
687 auto middleFieldResult = getField(state, middlePosition);
688 if (!middleFieldResult.ok()) {
689 return middleFieldResult.error();
690 }
691 f = *middleFieldResult;
692
693 // H1 is (h/(2*momentum) * sd.B_middle) in EigenStepper
694 double H1[3] = {f[0] * PS2, f[1] * PS2, f[2] * PS2};
695 // { A3, B3, C3 } is (direction + h/2 * sd.k2) in EigenStepper
696 double A3 = (A[0] + B2 * H1[2]) - C2 * H1[1];
697 double B3 = (A[1] + C2 * H1[0]) - A2 * H1[2];
698 double C3 = (A[2] + A2 * H1[1]) - B2 * H1[0];
699 // { A4, B4, C4 } is (direction + h/2 * sd.k3) in EigenStepper
700 double A4 = (A[0] + B3 * H1[2]) - C3 * H1[1];
701 double B4 = (A[1] + C3 * H1[0]) - A3 * H1[2];
702 double C4 = (A[2] + A3 * H1[1]) - B3 * H1[0];
703 // { A5, B5, C5 } is (direction + h * sd.k3) in EigenStepper
704 double A5 = 2. * A4 - A[0];
705 double B5 = 2. * B4 - A[1];
706 double C5 = 2. * C4 - A[2];
707
708 // Last point
709 const Vector3 endPosition(R[0] + h * A4, R[1] + h * B4, R[2] + h * C4);
710 // This is sd.B_last in Eigen stepper
711 auto endFieldResult = getField(state, endPosition);
712 if (!endFieldResult.ok()) {
713 return endFieldResult.error();
714 }
715 f = *endFieldResult;
716
717 // H2 is (h/(2*momentum) * sd.B_last) in EigenStepper
718 double H2[3] = {f[0] * PS2, f[1] * PS2, f[2] * PS2};
719 // { A6, B6, C6 } is (h/2 * sd.k4) in EigenStepper
720 double A6 = B5 * H2[2] - C5 * H2[1];
721 double B6 = C5 * H2[0] - A5 * H2[2];
722 double C6 = A5 * H2[1] - B5 * H2[0];
723
724 // Test approximation quality on give step and possible step reduction
725 //
726 // This is (h2 * (sd.k1 - sd.k2 - sd.k3 + sd.k4).template lpNorm<1>())
727 // in EigenStepper
728 errorEstimate =
729 2. * std::abs(h) *
730 (std::abs((A1 + A6) - (A3 + A4)) + std::abs((B1 + B6) - (B3 + B4)) +
731 std::abs((C1 + C6) - (C3 + C4)));
732 if (!std::isfinite(errorEstimate)) {
733 return EigenStepperError::StepInvalid;
734 }
735 if (!isErrorTolerable(errorEstimate)) {
736 ++state.statistics.nRejectedSteps;
737 const double stepSizeScaling = calcStepSizeScaling(errorEstimate);
738 h *= stepSizeScaling;
739 // neutralize the sign of h again
740 state.stepSize.setAccuracy(h * propDir);
741 if (h == 0. || std::abs(h) < std::abs(state.options.stepSizeCutOff)) {
742 return EigenStepperError::StepSizeStalled;
743 }
744 if (++nStepTrials > state.options.maxRungeKuttaStepTrials) {
745 return EigenStepperError::StepSizeAdjustmentFailed;
746 }
747 continue;
748 }
749
750 // Parameters calculation
751 //
752 double A00 = A[0], A11 = A[1], A22 = A[2];
753
754 A[0] = 2. * A3 + (A0 + A5 + A6);
755 A[1] = 2. * B3 + (B0 + B5 + B6);
756 A[2] = 2. * C3 + (C0 + C5 + C6);
757
758 double D = (A[0] * A[0] + A[1] * A[1]) + (A[2] * A[2] - 9.);
759 double Sl = 2. / h;
760 D = (1. / 3.) - ((1. / 648.) * D) * (12. - D);
761
762 R[0] += (A2 + A3 + A4) * S3;
763 R[1] += (B2 + B3 + B4) * S3;
764 R[2] += (C2 + C3 + C4) * S3;
765 A[0] *= D;
766 A[1] *= D;
767 A[2] *= D;
768 sA[0] = A6 * Sl;
769 sA[1] = B6 * Sl;
770 sA[2] = C6 * Sl;
771
772 double mass = particleHypothesis(state).mass();
773 double momentum = absoluteMomentum(state);
774
775 // Evaluate the time propagation
776 const double mOverP = mass / momentum;
777 double dtds = fastHypot(1., mOverP);
778 state.pVector[3] += h * dtds;
779 state.pVector[59] = dtds;
780 state.field = f;
781 state.newField = false;
782 state.derivativeReady = true;
783
784 if (withJacobian) {
785 // d(t)/d(q/p) = h m^2 (q/p) / (q^2 dt/ds), with the q^2 folded into p
786 // via p = |q| / |q/p|.
787 double dtdl = particleHypothesis(state).hasMomentumHypothesis()
788 ? 0.
789 : h * mOverP * mOverP / (qOverP(state) * dtds);
790 state.pVector[43] += dtdl;
791
792 // Jacobian calculation
793 //
794 double* d2A = &state.pVector[28];
795 double* d3A = &state.pVector[36];
796 double* d4A = &state.pVector[44];
797 double d2A0 = H0[2] * d2A[1] - H0[1] * d2A[2];
798 double d2B0 = H0[0] * d2A[2] - H0[2] * d2A[0];
799 double d2C0 = H0[1] * d2A[0] - H0[0] * d2A[1];
800 double d3A0 = H0[2] * d3A[1] - H0[1] * d3A[2];
801 double d3B0 = H0[0] * d3A[2] - H0[2] * d3A[0];
802 double d3C0 = H0[1] * d3A[0] - H0[0] * d3A[1];
803 double d4A0 = (A0 + H0[2] * d4A[1]) - H0[1] * d4A[2];
804 double d4B0 = (B0 + H0[0] * d4A[2]) - H0[2] * d4A[0];
805 double d4C0 = (C0 + H0[1] * d4A[0]) - H0[0] * d4A[1];
806 double d2A2 = d2A0 + d2A[0];
807 double d2B2 = d2B0 + d2A[1];
808 double d2C2 = d2C0 + d2A[2];
809 double d3A2 = d3A0 + d3A[0];
810 double d3B2 = d3B0 + d3A[1];
811 double d3C2 = d3C0 + d3A[2];
812 double d4A2 = d4A0 + d4A[0];
813 double d4B2 = d4B0 + d4A[1];
814 double d4C2 = d4C0 + d4A[2];
815 double d0 = d4A[0] - A00;
816 double d1 = d4A[1] - A11;
817 double d2 = d4A[2] - A22;
818 double d2A3 = (d2A[0] + d2B2 * H1[2]) - d2C2 * H1[1];
819 double d2B3 = (d2A[1] + d2C2 * H1[0]) - d2A2 * H1[2];
820 double d2C3 = (d2A[2] + d2A2 * H1[1]) - d2B2 * H1[0];
821 double d3A3 = (d3A[0] + d3B2 * H1[2]) - d3C2 * H1[1];
822 double d3B3 = (d3A[1] + d3C2 * H1[0]) - d3A2 * H1[2];
823 double d3C3 = (d3A[2] + d3A2 * H1[1]) - d3B2 * H1[0];
824 double d4A3 = ((A3 + d0) + d4B2 * H1[2]) - d4C2 * H1[1];
825 double d4B3 = ((B3 + d1) + d4C2 * H1[0]) - d4A2 * H1[2];
826 double d4C3 = ((C3 + d2) + d4A2 * H1[1]) - d4B2 * H1[0];
827 double d2A4 = (d2A[0] + d2B3 * H1[2]) - d2C3 * H1[1];
828 double d2B4 = (d2A[1] + d2C3 * H1[0]) - d2A3 * H1[2];
829 double d2C4 = (d2A[2] + d2A3 * H1[1]) - d2B3 * H1[0];
830 double d3A4 = (d3A[0] + d3B3 * H1[2]) - d3C3 * H1[1];
831 double d3B4 = (d3A[1] + d3C3 * H1[0]) - d3A3 * H1[2];
832 double d3C4 = (d3A[2] + d3A3 * H1[1]) - d3B3 * H1[0];
833 double d4A4 = ((A4 + d0) + d4B3 * H1[2]) - d4C3 * H1[1];
834 double d4B4 = ((B4 + d1) + d4C3 * H1[0]) - d4A3 * H1[2];
835 double d4C4 = ((C4 + d2) + d4A3 * H1[1]) - d4B3 * H1[0];
836 double d2A5 = 2. * d2A4 - d2A[0];
837 double d2B5 = 2. * d2B4 - d2A[1];
838 double d2C5 = 2. * d2C4 - d2A[2];
839 double d3A5 = 2. * d3A4 - d3A[0];
840 double d3B5 = 2. * d3B4 - d3A[1];
841 double d3C5 = 2. * d3C4 - d3A[2];
842 double d4A5 = 2. * d4A4 - d4A[0];
843 double d4B5 = 2. * d4B4 - d4A[1];
844 double d4C5 = 2. * d4C4 - d4A[2];
845 double d2A6 = d2B5 * H2[2] - d2C5 * H2[1];
846 double d2B6 = d2C5 * H2[0] - d2A5 * H2[2];
847 double d2C6 = d2A5 * H2[1] - d2B5 * H2[0];
848 double d3A6 = d3B5 * H2[2] - d3C5 * H2[1];
849 double d3B6 = d3C5 * H2[0] - d3A5 * H2[2];
850 double d3C6 = d3A5 * H2[1] - d3B5 * H2[0];
851 double d4A6 = d4B5 * H2[2] - d4C5 * H2[1];
852 double d4B6 = d4C5 * H2[0] - d4A5 * H2[2];
853 double d4C6 = d4A5 * H2[1] - d4B5 * H2[0];
854
855 double* dR = &state.pVector[24];
856 dR[0] += (d2A2 + d2A3 + d2A4) * S3;
857 dR[1] += (d2B2 + d2B3 + d2B4) * S3;
858 dR[2] += (d2C2 + d2C3 + d2C4) * S3;
859 d2A[0] = ((d2A0 + 2. * d2A3) + (d2A5 + d2A6)) * (1. / 3.);
860 d2A[1] = ((d2B0 + 2. * d2B3) + (d2B5 + d2B6)) * (1. / 3.);
861 d2A[2] = ((d2C0 + 2. * d2C3) + (d2C5 + d2C6)) * (1. / 3.);
862
863 dR = &state.pVector[32];
864 dR[0] += (d3A2 + d3A3 + d3A4) * S3;
865 dR[1] += (d3B2 + d3B3 + d3B4) * S3;
866 dR[2] += (d3C2 + d3C3 + d3C4) * S3;
867 d3A[0] = ((d3A0 + 2. * d3A3) + (d3A5 + d3A6)) * (1. / 3.);
868 d3A[1] = ((d3B0 + 2. * d3B3) + (d3B5 + d3B6)) * (1. / 3.);
869 d3A[2] = ((d3C0 + 2. * d3C3) + (d3C5 + d3C6)) * (1. / 3.);
870
871 dR = &state.pVector[40];
872 dR[0] += (d4A2 + d4A3 + d4A4) * S3;
873 dR[1] += (d4B2 + d4B3 + d4B4) * S3;
874 dR[2] += (d4C2 + d4C3 + d4C4) * S3;
875 d4A[0] = ((d4A0 + 2. * d4A3) + (d4A5 + d4A6 + A6)) * (1. / 3.);
876 d4A[1] = ((d4B0 + 2. * d4B3) + (d4B5 + d4B6 + B6)) * (1. / 3.);
877 d4A[2] = ((d4C0 + 2. * d4C3) + (d4C5 + d4C6 + C6)) * (1. / 3.);
878 }
879
880 break;
881 }
882
883 state.pathAccumulated += h;
884 ++state.nSteps;
885
886 ++state.statistics.nSuccessfulSteps;
887 if (propDir != Direction::fromScalarZeroAsPositive(initialStepSize)) {
888 ++state.statistics.nReverseSteps;
889 }
890 state.statistics.pathLength += h;
891 state.statistics.absolutePathLength += std::abs(h);
892
893 const double stepSizeScaling = calcStepSizeScaling(errorEstimate);
894 const double nextAccuracy = std::abs(h * stepSizeScaling);
895 const double previousAccuracy = std::abs(state.stepSize.accuracy());
896 const double initialStepLength = std::abs(initialStepSize);
897 if (nextAccuracy < initialStepLength || nextAccuracy > previousAccuracy) {
898 state.stepSize.setAccuracy(nextAccuracy);
899 }
900
901 return h;
902 }
903
904 private:
906 void resetJacobian(State& state, const Surface& surface) const {
907 std::ranges::fill(state.pVector.begin() + eFreeSize, state.pVector.end(),
908 0.);
909 if (state.cov.has_value()) {
910 Eigen::Map<BoundToFreeMatrix>(state.pVector.data() + eFreeSize) =
911 surface.boundToFreeJacobian(state.options.geoContext, position(state),
912 direction(state));
913 }
914 state.derivativeReady = false;
915 }
916
918 Result<void> prepareDerivatives(State& state) const {
919 if (state.derivativeReady) {
920 return Result<void>::success();
921 }
922 auto field = getField(state, position(state));
923 if (!field.ok()) {
924 return field.error();
925 }
926 const double curvature = charge(state) == 0. ? 0. : qOverP(state);
927 Eigen::Map<Vector3>(state.pVector.data() + 56) =
928 curvature * direction(state).cross(*field);
929 const double mOverP =
930 particleHypothesis(state).mass() / absoluteMomentum(state);
931 state.pVector[59] = fastHypot(1., mOverP);
932 state.derivativeReady = true;
933 return Result<void>::success();
934 }
935
936 std::shared_ptr<const MagneticFieldProvider> m_bField;
937
939 double m_overstepLimit = 100 * UnitConstants::um;
940};
