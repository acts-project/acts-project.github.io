40struct MaxMomentumComponent {
41 template <typename component_range_t>
42 auto operator()(const component_range_t& cmps) const {
43 return std::ranges::max_element(cmps, [&](const auto& a, const auto& b) {
44 return std::abs(a.state.pars[eFreeQOverP]) >
45 std::abs(b.state.pars[eFreeQOverP]);
46 });
47 }
48};
49
50struct MaxWeightComponent {
51 template <typename component_range_t>
52 auto operator()(const component_range_t& cmps) {
53 return std::ranges::max_element(cmps, [&](const auto& a, const auto& b) {
54 return a.weight < b.weight;
55 });
56 }
57};

59template <typename component_chooser_t>
60struct SingleComponentReducer {
61 template <typename stepper_state_t>
62 static Vector3 position(const stepper_state_t& s) {
63 return component_chooser_t{}(s.components)
64 ->state.pars.template segment<3>(eFreePos0);
65 }
66
67 template <typename stepper_state_t>
68 static Vector3 direction(const stepper_state_t& s) {
69 return component_chooser_t{}(s.components)
70 ->state.pars.template segment<3>(eFreeDir0);
71 }
72
73 template <typename stepper_state_t>
74 static double qOverP(const stepper_state_t& s) {
75 const auto cmp = component_chooser_t{}(s.components);
76 return cmp->state.pars[eFreeQOverP];
77 }
78
79 template <typename stepper_state_t>
80 static double absoluteMomentum(const stepper_state_t& s) {
81 const auto cmp = component_chooser_t{}(s.components);
82 return s.particleHypothesis.extractMomentum(cmp->state.pars[eFreeQOverP]);
83 }
84
85 template <typename stepper_state_t>
86 static Vector3 momentum(const stepper_state_t& s) {
87 const auto cmp = component_chooser_t{}(s.components);
88 return s.particleHypothesis.extractMomentum(cmp->state.pars[eFreeQOverP]) *
89 cmp->state.pars.template segment<3>(eFreeDir0);
90 }
91
92 template <typename stepper_state_t>
93 static double charge(const stepper_state_t& s) {
94 const auto cmp = component_chooser_t{}(s.components);
95 return s.particleHypothesis.extractCharge(cmp->state.pars[eFreeQOverP]);
96 }
97
98 template <typename stepper_state_t>
99 static double time(const stepper_state_t& s) {
100 return component_chooser_t{}(s.components)->state.pars[eFreeTime];

102

274 void initialize(State& state, const BoundParameters& par) const {
275 if (par.empty()) {
276 throw std::invalid_argument(
277 "Cannot construct MultiEigenStepperLoop::State with empty "
278 "multi-component parameters");
279 }
280
281 state.particleHypothesis = par.particleHypothesis();
282
283 const auto surface = par.referenceSurface().getSharedPtr();
284
285 state.components.clear();
286 state.components.reserve(par.size());
287
288 for (std::size_t i = 0; i < par.size(); ++i) {
289 const auto& [weight, singlePars] = par[i];
290 auto& cmp = state.components.emplace_back(
291 m_singleStepper.makeState(state.options), weight,
292 IntersectionStatus::onSurface);
293 m_singleStepper.initialize(cmp.state, singlePars);
294 }
295
296 state.covTransport = par.hasCovariance();
297 }

532 IntersectionStatus updateSurfaceStatus(
533 State& state, const Surface& surface, std::uint8_t index,
534 Direction navDir, const BoundaryTolerance& boundaryTolerance,
535 double surfaceTolerance, ConstrainedStep::Type stype,
536 const Logger& logger = getDummyLogger()) const {
537 using Status = IntersectionStatus;
538
539 std::array<int, 3> counts = {0, 0, 0};
540
541 for (auto& component : state.components) {
542 component.status = detail::updateSingleSurfaceStatus<SingleStepper>(
543 m_singleStepper, component.state, surface, index, navDir,
544 boundaryTolerance, surfaceTolerance, stype, logger);
545 ++counts[static_cast<std::size_t>(component.status)];
546 }
547
548 // If at least one component is on a surface, we can remove all missed
549 // components before the step. If not, we must keep them for the case that
550 // all components miss and we need to retarget
551 if (counts[static_cast<std::size_t>(Status::onSurface)] > 0) {
552 removeMissedComponents(state);
553 reweightComponents(state);
554 }
555
556 ACTS_VERBOSE("Component status wrt "
557 << surface.geometryId() << " at {"
558 << surface.center(state.options.geoContext).transpose()
559 << "}:\t" << [&]() {
560 std::stringstream ss;
561 for (auto& component : state.components) {
562 ss << component.status << "\t";
563 }
564 return ss.str();
565 }());
566
567 // Switch on stepCounter if one or more components reached a surface, but
568 // some are still in progress of reaching the surface
569 if (!state.stepCounterAfterFirstComponentOnSurface &&
570 counts[static_cast<std::size_t>(Status::onSurface)] > 0 &&
571 counts[static_cast<std::size_t>(Status::reachable)] > 0) {
572 state.stepCounterAfterFirstComponentOnSurface = 0;
573 ACTS_VERBOSE("started stepCounterAfterFirstComponentOnSurface");
574 }
575
576 // If there are no components onSurface, but the counter is switched on
577 // (e.g., if the navigator changes the target surface), we need to switch it
578 // off again
579 if (state.stepCounterAfterFirstComponentOnSurface &&
580 counts[static_cast<std::size_t>(Status::onSurface)] == 0) {
581 state.stepCounterAfterFirstComponentOnSurface.reset();
582 ACTS_VERBOSE("switch off stepCounterAfterFirstComponentOnSurface");
583 }
584
585 // This is a 'any_of' criterium. As long as any of the components has a
586 // certain state, this determines the total state (in the order of a
587 // somewhat importance)
588 if (counts[static_cast<std::size_t>(Status::reachable)] > 0) {
589 return Status::reachable;
590 } else if (counts[static_cast<std::size_t>(Status::onSurface)] > 0) {
591 state.stepCounterAfterFirstComponentOnSurface.reset();
592 return Status::onSurface;
593 } else {
594 return Status::unreachable;
595 }
596 }

609 template <typename object_intersection_t>
610 void updateStepSize(State& state, const object_intersection_t& oIntersection,
611 Direction direction, ConstrainedStep::Type stype) const {
612 const Surface& surface = *oIntersection.object();
613
614 for (auto& component : state.components) {
615 auto intersection = surface.intersect(
616 component.state.options.geoContext,
617 m_singleStepper.position(component.state),
618 direction * m_singleStepper.direction(component.state),
619 BoundaryTolerance::None())[oIntersection.index()];
620
621 m_singleStepper.updateStepSize(component.state, intersection, direction,
622 stype);
623 }
624 }
