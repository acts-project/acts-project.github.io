227 [[nodiscard]] Result<void> initialize(
228 State& state, const NavigatorInitializeArguments& args) const {
229 ACTS_VERBOSE("Initialize. Surface sequence for navigation:");
230 for (const Surface* surface : state.options.externalSurfaces) {
231 ACTS_VERBOSE(surface->geometryId()
232 << " - "
233 << surface->center(state.options.geoContext).transpose());
234 }
235
236 state.direction = args.propagationDirection;
237 ACTS_VERBOSE("Navigation direction is " << args.propagationDirection);
238
239 state.startSurface = args.startSurface;
240 state.targetSurface = args.targetSurface;
241
242 // We set the current surface to the start surface
243 state.currentSurface = state.startSurface;
244 if (state.currentSurface != nullptr) {
245 ACTS_VERBOSE("Current surface set to start surface "
246 << state.currentSurface->geometryId());
247 } else {
248 ACTS_VERBOSE("Current surface set to nullptr");
249 }
250
251 // Find initial index.
252 auto found =
253 std::ranges::find(state.options.externalSurfaces, state.startSurface);
254
255 if (found != state.options.externalSurfaces.end()) {
256 // The index should be the index before the start surface, depending on
257 // the direction
258 state.surfaceIndex =
259 std::distance(state.options.externalSurfaces.begin(), found);
260 state.surfaceIndex += state.direction == Direction::Backward() ? 1 : -1;
261 } else {
262 ACTS_DEBUG(
263 "Did not find the start surface in the sequence. Assuming it is not "
264 "part of the sequence. Trusting the correctness of the input "
265 "sequence. Resetting the surface index.");
266 state.resetSurfaceIndex();
267 }
268
269 state.navigationBreak = false;
270
271 return Result<void>::success();
272 }

284 NavigationTarget nextTarget(State& state, const Vector3& position,
285 const Vector3& direction) const {
286 // Navigator target always resets the current surface
287 state.currentSurface = nullptr;
288
289 if (state.navigationBreak) {
290 return NavigationTarget::None();
291 }
292
293 ACTS_VERBOSE("DirectNavigator::nextTarget");
294
295 // Move the sequence to the next surface
296 state.nextSurface();
297
298 while (!state.endOfSurfaces()) {
299 ACTS_VERBOSE("Next surface candidate is "
300 << state.navSurface().geometryId() << ". "
301 << state.remainingSurfaces() << " out of "
302 << state.options.externalSurfaces.size()
303 << " surfaces remain to try.");
304
305 // Establish & update the surface status
306 // TODO we do not know the intersection index - passing the closer one
307 const Surface& surface = state.navSurface();
308 const double farLimit = std::numeric_limits<double>::max();
309 const NavigationTarget target = chooseIntersection(
310 state.options.geoContext, surface, position, direction,
311 BoundaryTolerance::Infinite(), state.options.nearLimit, farLimit,
312 state.options.surfaceTolerance);
313 if (target.isValid()) {
314 return target;
315 }
316
317 ACTS_VERBOSE("No valid intersection found with surface "
318 << surface.geometryId() << ", trying next surface.");
319 state.nextSurface();
320 }
321
322 ACTS_VERBOSE("End of surfaces reached, navigation break.");
323 state.navigationBreak = true;
324 return NavigationTarget::None();
325 }

374 private:
375 NavigationTarget chooseIntersection(
376 const GeometryContext& gctx, const Surface& surface,
377 const Vector3& position, const Vector3& direction,
378 const BoundaryTolerance& boundaryTolerance, double nearLimit,
379 double farLimit, double tolerance) const {
380 auto intersections = surface.intersect(gctx, position, direction,
381 boundaryTolerance, tolerance);
382
383 for (auto [intersectionIndex, intersection] :
384 Acts::enumerate(intersections)) {
385 if (intersection.isValid() &&
386 detail::checkPathLength(intersection.pathLength(), nearLimit,
387 farLimit, logger())) {
388 return NavigationTarget(intersection, intersectionIndex, surface,
389 boundaryTolerance);
390 }
391 }
392
393 return NavigationTarget::None();
394 }
395
396 const Logger& logger() const { return *m_logger; }
