222 [[nodiscard]] Result<void> initialize(State& state, const Vector3& position,
223 const Vector3& direction,
224 Direction propagationDirection) const {
225 static_cast<void>(position);
226 static_cast<void>(direction);
227
228 ACTS_VERBOSE("Initialize. Surface sequence for navigation:");
229 for (const Surface* surface : state.options.externalSurfaces) {
230 ACTS_VERBOSE(surface->geometryId()
231 << " - "
232 << surface->center(state.options.geoContext).transpose());
233 }
234
235 state.direction = propagationDirection;
236 ACTS_VERBOSE("Navigation direction is " << propagationDirection);
237
238 // We set the current surface to the start surface
239 state.currentSurface = state.options.startSurface;
240 if (state.currentSurface != nullptr) {
241 ACTS_VERBOSE("Current surface set to start surface "
242 << state.currentSurface->geometryId());
243 } else {
244 ACTS_VERBOSE("Current surface set to nullptr");
245 }
246
247 // Find initial index.
248 auto found = std::ranges::find(state.options.externalSurfaces,
249 state.options.startSurface);
250
251 if (found != state.options.externalSurfaces.end()) {
252 // The index should be the index before the start surface, depending on
253 // the direction
254 state.surfaceIndex =
255 std::distance(state.options.externalSurfaces.begin(), found);
256 state.surfaceIndex += state.direction == Direction::Backward() ? 1 : -1;
257 } else {
258 ACTS_DEBUG(
259 "Did not find the start surface in the sequence. Assuming it is not "
260 "part of the sequence. Trusting the correctness of the input "
261 "sequence. Resetting the surface index.");
262 state.resetSurfaceIndex();
263 }
264
265 state.navigationBreak = false;
266
267 return Result<void>::success();
268 }

280 NavigationTarget nextTarget(State& state, const Vector3& position,
281 const Vector3& direction) const {
282 // Navigator target always resets the current surface
283 state.currentSurface = nullptr;
284
285 if (state.navigationBreak) {
286 return NavigationTarget::None();
287 }
288
289 ACTS_VERBOSE("DirectNavigator::nextTarget");
290
291 // Move the sequence to the next surface
292 state.nextSurface();
293
294 while (!state.endOfSurfaces()) {
295 ACTS_VERBOSE("Next surface candidate is "
296 << state.navSurface().geometryId() << ". "
297 << state.remainingSurfaces() << " out of "
298 << state.options.externalSurfaces.size()
299 << " surfaces remain to try.");
300
301 // Establish & update the surface status
302 // TODO we do not know the intersection index - passing the closer one
303 const Surface& surface = state.navSurface();
304 const double farLimit = std::numeric_limits<double>::max();
305 const NavigationTarget target = chooseIntersection(
306 state.options.geoContext, surface, position, direction,
307 BoundaryTolerance::Infinite(), state.options.nearLimit, farLimit,
308 state.options.surfaceTolerance);
309 if (target.isValid()) {
310 return target;
311 }
312
313 ACTS_VERBOSE("No valid intersection found with surface "
314 << surface.geometryId() << ", trying next surface.");
315 state.nextSurface();
316 }
317
318 ACTS_VERBOSE("End of surfaces reached, navigation break.");
319 state.navigationBreak = true;
320 return NavigationTarget::None();
321 }

370 private:
371 NavigationTarget chooseIntersection(
372 const GeometryContext& gctx, const Surface& surface,
373 const Vector3& position, const Vector3& direction,
374 const BoundaryTolerance& boundaryTolerance, double nearLimit,
375 double farLimit, double tolerance) const {
376 auto intersections = surface.intersect(gctx, position, direction,
377 boundaryTolerance, tolerance);
378
379 for (auto [intersectionIndex, intersection] :
380 Acts::enumerate(intersections)) {
381 if (intersection.isValid() &&
382 detail::checkPathLength(intersection.pathLength(), nearLimit,
383 farLimit, logger())) {
384 return NavigationTarget(intersection, intersectionIndex, surface,
385 boundaryTolerance);
386 }
387 }
388
389 return NavigationTarget::None();
390 }
