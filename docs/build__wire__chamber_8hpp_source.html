209template <concepts::algebra algebra_t, typename wire_shape_t>
210inline auto build_wire_chamber(
211 vecmem::memory_resource &resource,
212 wire_chamber_config<dscalar<algebra_t>, wire_shape_t> &cfg) {
213 using builder_t =
214 detector_builder<wire_chamber_metadata<algebra_t>, volume_builder>;
215 using detector_t = typename builder_t::detector_type;
216 using scalar_t = dscalar<typename detector_t::algebra_type>;
217
218 // Wire chamber detector builder
219 builder_t det_builder;
220 det_builder.set_name("wire_chamber");
221
222 // Geometry context object
223 typename detector_t::geometry_context gctx{};
224
225 // Navigation link when leaving the detector
226 using nav_link_t = typename detector_t::surface_type::navigation_link;
227 constexpr auto leaving_world{detail::invalid_value<nav_link_t>()};
228 const scalar_t inner_rad{cfg.first_layer_inner_radius()};
229 const scalar_t cell_size{cfg.cell_size()};
230
231 // Prepare grid building
232 constexpr auto grid_id{
233 detector_t::accel::id::e_surface_concentric_cylinder2D_grid};
234 using cyl_grid_t = types::get<typename detector_t::accel, grid_id>;
235 using loc_bin_idx_t = typename cyl_grid_t::loc_bin_index;
236 static_assert(cyl_grid_t::dim == 2);
237 using grid_builder_t =
238 grid_builder<detector_t, cyl_grid_t, detray::fill_by_pos>;
239
240 // Binning of the grid
241 axis::multi_bin_range<cyl_grid_t::dim> bin_range{};
242 // Min, max bin indices per axis
243 bin_range[static_cast<std::size_t>(axis::label::e_rphi)] = {0, 100};
244 bin_range[static_cast<std::size_t>(axis::label::e_cyl_z)] = {0, 1};
245
246 // Spans of the grid axes
247 std::vector<scalar_t> sf_grid_spans{-constant<scalar_t>::pi,
248 constant<scalar_t>::pi, -cfg.half_z(),
249 cfg.half_z()};
250
251 constexpr unsigned int bin_capacity{3u};
252 std::vector<std::pair<loc_bin_idx_t, dindex>> capacities{};
253 capacities.reserve(
254 static_cast<std::size_t>(bin_range[0][1] * bin_range[1][1]));
255
256 //
257 // Build empty inner volume, where silicon subdetectors would sit
258 //
259 auto inner_v_builder = det_builder.new_volume(volume_id::e_cylinder);
260 inner_v_builder->add_volume_placement(/*identity*/);
261 const dindex inner_vol_idx{inner_v_builder->vol_index()};
262 inner_v_builder->set_name("inner_vol_" + std::to_string(inner_vol_idx));
263
264 // Configure the portal factory
265 // TODO: Add material maps that model the silicon detector budget
266 cylinder_portal_config<scalar_t> inner_pt_cfg{};
267 inner_pt_cfg.do_autofit(false)
268 .fixed_half_length(cfg.half_z())
269 .fixed_inner_radius(0.f)
270 .fixed_outer_radius(inner_rad)
271 // No inner subdetectors present -> don't build inner portal
272 .build_inner(false)
273 .link_north(inner_vol_idx + 1u)
274 .link_south(leaving_world)
275 .link_east(leaving_world)
276 .link_west(leaving_world);
277
278 auto inner_pt_factory =
279 std::make_shared<cylinder_portal_generator<detector_t>>(inner_pt_cfg);
280 inner_v_builder->add_surfaces(inner_pt_factory, gctx);
281
282 //
283 // Build layer volumes
284 //
285
286 // Configure the homogeneous material generation
287 auto &mat_cfg = cfg.material_config();
288 // Only build material on sensitive surfaces
289 constexpr auto vac{vacuum<scalar_t>{}};
290 mat_cfg.portal_material(vac).passive_material(vac);
291 mat_cfg.sensitive_material(tungsten<scalar_t>{}).thickness(cfg.mat_radius());
292
293 for (unsigned int i_lay = 0; i_lay < cfg.n_layers(); i_lay++) {
294 // New volume for layer
295 auto v_builder = det_builder.new_volume(volume_id::e_cylinder);
296 auto vm_builder =
297 det_builder.template decorate<homogeneous_material_builder<detector_t>>(
298 v_builder);
299
300 const dindex vol_idx{vm_builder->vol_index()};
301 vm_builder->set_name("layer_vol_" + std::to_string(vol_idx));
302
303 // The barrel volumes are centered at the origin
304 vm_builder->add_volume_placement(/*identity*/);
305
306 // The maximal inner and outer radius of the volume
307 const scalar_t inner_layer_rad =
308 inner_rad + static_cast<scalar_t>(i_lay) * 2.f * cell_size;
309 const scalar_t outer_layer_rad =
310 inner_rad + static_cast<scalar_t>(i_lay + 1u) * 2.f * cell_size;
311
312 // Configure the wire layer factory for this layer
313 auto &layer_cfg = cfg.layer_config();
314 layer_cfg.inner_layer_radius(inner_layer_rad).half_length(cfg.half_z());
315
316 const scalar_t sign = (i_lay % 2 == 0) ? 1 : -1;
317 layer_cfg.stereo_angle(sign * math::fabs(layer_cfg.stereo_angle()));
318
319 // Configure the portal factory
320 cylinder_portal_config<scalar_t> layer_portal_cfg{};
321 // Limit to maximum valid link
322 auto link_north{i_lay == cfg.n_layers() - 1u ? leaving_world
323 : vol_idx + 1u};
324
325 layer_portal_cfg.do_autofit(false)
326 .fixed_half_length(cfg.half_z())
327 .fixed_inner_radius(inner_layer_rad)
328 .fixed_outer_radius(outer_layer_rad)
329 // Link the volume portals to its neighbors
330 .link_north(link_north)
331 .link_south(vol_idx - 1u)
332 .link_east(leaving_world)
333 .link_west(leaving_world);
334
335 // Register sensitive wires together with material
336 auto wire_factory =
337 std::make_unique<wire_layer_generator<detector_t, wire_shape_t>>(
338 layer_cfg);
339 auto wire_mat_factory =
340 std::make_shared<homogeneous_material_generator<detector_t>>(
341 std::move(wire_factory), mat_cfg);
342
343 // Register portals
344 auto portal_mat_factory =
345 std::make_shared<cylinder_portal_generator<detector_t>>(
346 layer_portal_cfg);
347
348 // Add surfaces and material to volume builder
349 vm_builder->add_surfaces(portal_mat_factory);
350 vm_builder->add_surfaces(wire_mat_factory, gctx);
351
352 // Add a cylinder grid to every barrel layer
353 auto vgr_builder =
354 det_builder.template decorate<grid_builder_t>(vm_builder);
355
356 // Determine bin capacities
357 capacities.clear();
358 auto bin_indexer2D = axis::detail::get_bin_indexer(
359 bin_range, std::make_integer_sequence<std::size_t, cyl_grid_t::dim>{});
360 for (const auto [bin_idx0, bin_idx1] : bin_indexer2D) {
361 // @Todo: fine-tune capacity
362 loc_bin_idx_t loc_bin{static_cast<unsigned int>(bin_idx0),
363 static_cast<unsigned int>(bin_idx1)};
364 capacities.emplace_back(loc_bin, bin_capacity);
365 }
366
367 vgr_builder->set_type(detector_t::geo_obj_ids::e_sensitive);
368 vgr_builder->init_grid(sf_grid_spans,
369 {static_cast<std::size_t>(bin_range[0][1]),
370 static_cast<std::size_t>(bin_range[1][1])},
371 capacities);
372 }
373
374 // Build and return the detector and fill name map
375 typename detector_t::name_map name_map{};
376 auto det = det_builder.build(resource, cfg.builder_options(), name_map);
377
378 if (cfg.do_check()) {
379 const bool verbose_check{false};
380 detray::detail::check_consistency(det, verbose_check, name_map);
381 }
382
383 DETRAY_DEBUG_HOST("\n" << detray::utils::print_detector(det, name_map));
384
385 return std::make_pair(std::move(det), std::move(name_map));
386}
