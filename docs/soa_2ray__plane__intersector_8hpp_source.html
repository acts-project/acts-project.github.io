62 template <concepts::algebra other_algebra_t>
63 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
64 const trajectory_type<other_algebra_t> &ray, const transform3_type &trf,
65 const scalar_type /*overstep_tol*/ = 0.f) const {
66 // Retrieve the surface normal & translation (context resolved)
67 const vector3_type &sn = trf.z();
68 const vector3_type &st = trf.translation();
69
70 // Broadcast ray data
71 const auto &pos = ray.pos();
72 const auto &dir = ray.dir();
73 const vector3_type ro{pos[0], pos[1], pos[2]};
74 const vector3_type rd{dir[0], dir[1], dir[2]};
75
76 const scalar_type denom = vector::dot(rd, sn);
77 const vector3_type diff = st - ro;
78 const scalar_type s = vector::dot(sn, diff) / denom;
79
80 const point3_type glob_pos = ro + s * rd;
81
82 return {s, glob_pos};
83 }
84};
