46template <bool reverse, typename trajectory_t, std::size_t M, bool ReadOnly>
47class TrackStateRange {
48 using ProxyType = TrackStateProxy<trajectory_t, M, ReadOnly>;
49 using IndexType = typename ProxyType::IndexType;
50 static constexpr IndexType kInvalid = ProxyType::kInvalid;
51
52 public:
55 struct Iterator {
56 std::optional<ProxyType> proxy;
57
58 using iterator_category = std::forward_iterator_tag;
59 using value_type = ProxyType;
60 using difference_type = std::ptrdiff_t;
61 using pointer = void;
62 using reference = void;
63
64 Iterator& operator++() {
65 if (!proxy) {
66 return *this;
67 }
68 if constexpr (reverse) {
69 if (proxy->hasPrevious()) {
70 proxy = proxy->trajectory().getTrackState(proxy->previous());
71 return *this;
72 } else {
73 proxy = std::nullopt;
74 return *this;
75 }
76 } else {
77 IndexType next =
78 proxy->template component<IndexType, hashString("next")>();
79 if (next != kInvalid) {
80 proxy = proxy->trajectory().getTrackState(next);
81 return *this;
82 } else {
83 proxy = std::nullopt;
84 return *this;
85 }
86 }
87 }
88
89 Iterator operator++(int) {
90 Iterator tmp(*this);
91 operator++();
92 return tmp;
93 }
94
95 bool operator==(const Iterator& other) const {
96 if (!proxy && !other.proxy) {
97 return true;
98 }
99 if (proxy && other.proxy) {
100 return proxy->index() == other.proxy->index();
101 }
102 return false;
103 }
104
105 ProxyType operator*() const { return *proxy; }
106 ProxyType operator*() { return *proxy; }
107 };
108
109 explicit TrackStateRange(ProxyType _begin) : m_begin{_begin} {}
110 TrackStateRange() : m_begin{std::nullopt} {}
111
112 Iterator begin() { return Iterator{m_begin}; }
113 Iterator end() { return Iterator{std::nullopt}; }
114
115 Iterator cbegin() const { return Iterator{m_begin}; }
116 Iterator cend() const { return Iterator{std::nullopt}; }
117
118 private:
119 Iterator m_begin;
120};

284 template <typename F>
285 void applyBackwards(IndexType iendpoint, F&& callable)
286 requires(!ReadOnly) && detail_lt::VisitorConcept<F, TrackStateProxy>
287 {
288 if (iendpoint == kInvalid) {
289 throw std::runtime_error(
290 "Cannot apply backwards with kInvalid as endpoint");
291 }
292
293 while (true) {
294 auto ts = getTrackState(iendpoint);
295 if constexpr (std::is_same_v<std::invoke_result_t<F, TrackStateProxy>,
296 bool>) {
297 bool proceed = callable(ts);
298 // this point has no parent and ends the trajectory, or a break was
299 // requested
300 if (!proceed || !ts.hasPrevious()) {
301 break;
302 }
303 } else {
304 callable(ts);
305 // this point has no parent and ends the trajectory
306 if (!ts.hasPrevious()) {
307 break;
308 }
309 }
310 iendpoint = ts.previous();
311 }
312 }
