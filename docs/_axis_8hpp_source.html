20// This object can be iterated to produce up to two sequences of integer
21// indices, corresponding to the half-open integer ranges [begin1, end1[ and
22// [begin2, end2[.
23//
24// The goal is to emulate the effect of enumerating a range of neighbor
25// indices on an axis (which may go out of bounds and wrap around since we
26// have AxisBoundaryType::Closed), inserting them into an std::vector, and
27// discarding duplicates, without paying the price of duplicate removal
28// and dynamic memory allocation in hot magnetic field interpolation code.
29//
31class NeighborHoodIndices {
32 public:
33 NeighborHoodIndices() = default;
34
38 NeighborHoodIndices(std::size_t begin, std::size_t end)
39 : m_begin1(begin), m_end1(end), m_begin2(end), m_end2(end) {}
40
46 NeighborHoodIndices(std::size_t begin1, std::size_t end1, std::size_t begin2,
47 std::size_t end2)
48 : m_begin1(begin1), m_end1(end1), m_begin2(begin2), m_end2(end2) {}
49
51 class iterator {
52 public:
53 iterator() = default;
54
57 explicit iterator(std::size_t current)
58 : m_current(current), m_wrapped(true) {}
59
64 iterator(std::size_t begin1, std::size_t end1, std::size_t begin2)
65 : m_current(begin1),
66 m_end1(end1),
67 m_begin2(begin2),
68 m_wrapped(begin1 == begin2) {}
69
72 std::size_t operator*() const { return m_current; }
73
76 iterator& operator++() {
77 ++m_current;
78 if (m_current == m_end1) {
79 m_current = m_begin2;
80 m_wrapped = true;
81 }
82 return *this;
83 }
84
88 bool operator==(const iterator& it) const {
89 return (m_current == it.m_current) && (m_wrapped == it.m_wrapped);
90 }
91
92 private:
93 std::size_t m_current = 0, m_end1 = 0, m_begin2 = 0;
94 bool m_wrapped = false;
95 };
96
99 iterator begin() const { return iterator(m_begin1, m_end1, m_begin2); }
100
103 iterator end() const { return iterator(m_end2); }
104
107 std::size_t size() const { return (m_end1 - m_begin1) + (m_end2 - m_begin2); }
108
111 std::vector<std::size_t> collect() const {
112 std::vector<std::size_t> result;
113 result.reserve(this->size());
114 for (std::size_t idx : *this) {
115 result.push_back(idx);
116 }
117 return result;
118 }
119
120 private:
121 std::size_t m_begin1 = 0, m_end1 = 0, m_begin2 = 0, m_end2 = 0;
122};

205 NeighborHoodIndices neighborHoodIndices(std::size_t idx,
206 std::pair<int, int> sizes = {-1,
207 1}) const
208 requires(bdt == AxisBoundaryType::Open)
209 {
210 constexpr int min = 0;
211 const int max = getNBins() + 1;
212 const int itmin = std::clamp(static_cast<int>(idx + sizes.first), min, max);
213 const int itmax =
214 std::clamp(static_cast<int>(idx + sizes.second), min, max);
215 return NeighborHoodIndices(itmin, itmax + 1);

227 NeighborHoodIndices neighborHoodIndices(std::size_t idx,
228 std::pair<int, int> sizes = {-1,
229 1}) const
230 requires(bdt == AxisBoundaryType::Bound)
231 {
232 if (idx <= 0 || idx >= (getNBins() + 1)) {
233 return NeighborHoodIndices();
234 }
235 constexpr int min = 1;
236 const int max = getNBins();
237 const int itmin = std::clamp(static_cast<int>(idx) + sizes.first, min, max);
238 const int itmax =
239 std::clamp(static_cast<int>(idx) + sizes.second, min, max);
240 return NeighborHoodIndices(itmin, itmax + 1);
241 }
242
252 NeighborHoodIndices neighborHoodIndices(std::size_t idx,
253 std::pair<int, int> sizes = {-1,
254 1}) const
255 requires(bdt == AxisBoundaryType::Closed)
256 {
257 // Handle invalid indices
258 if (idx <= 0 || idx >= (getNBins() + 1)) {
259 return NeighborHoodIndices();
260 }
261
262 // Handle corner case where user requests more neighbours than the number
263 // of bins on the axis. All bins are returned in this case.
264
265 const int max = getNBins();
266 sizes.first = std::clamp(sizes.first, -max, max);
267 sizes.second = std::clamp(sizes.second, -max, max);
268 if (std::abs(sizes.first - sizes.second) >= max) {
269 sizes.first = 1 - idx;
270 sizes.second = max - idx;
271 }
272
273 // If the entire index range is not covered, we must wrap the range of
274 // targeted neighbor indices into the range of valid bin indices. This may
275 // split the range of neighbor indices in two parts:
276 //
277 // Before wraparound - [ XXXXX]XXX
278 // After wraparound - [ XXXX XXXX ]
279 //
280 const int itmin = idx + sizes.first;
281 const int itmax = idx + sizes.second;
282 const std::size_t itfirst = wrapBin(itmin);
283 const std::size_t itlast = wrapBin(itmax);
284 if (itfirst <= itlast) {
285 return NeighborHoodIndices(itfirst, itlast + 1);
286 } else {
287 return NeighborHoodIndices(itfirst, max + 1, 1, itlast + 1);
288 }
289 }

301 }

309 std::size_t wrapBin(int bin) const

321 std::size_t wrapBin(int bin) const
322 requires(bdt == AxisBoundaryType::Closed)
323 {
324 const int w = getNBins();

326 // return int(bin<1)*w - int(bin>w)*w + bin;
327 }

360 return getMin() + (bin - 1) * getBinWidth();
361 }
362
373 double getBinUpperBound(std::size_t bin) const {
374 return getMin() + bin * getBinWidth();
375 }
376
384 double getBinCenter(std::size_t bin) const {
385 return getMin() + (bin - 0.5) * getBinWidth();
386 }
387
391 double getMax() const final { return m_max; }
392
396 double getMin() const final { return m_min; }
397
401 std::size_t getNBins() const final { return m_bins; }
402
411 bool isInside(double x) const { return (m_min <= x) && (x < m_max); }
412
415 std::vector<double> getBinEdges() const final {
416 std::vector<double> binEdges;
417 for (std::size_t i = 1; i <= m_bins; i++) {
418 binEdges.push_back(getBinLowerBound(i));
419 }
420 binEdges.push_back(getBinUpperBound(m_bins));
421 return binEdges;
422 }
423
424 friend std::ostream& operator<<(std::ostream& os, const Axis& axis) {
425 os << "Axis<Equidistant, " << bdt << ">(";
426 os << axis.m_min << ", ";
427 os << axis.m_max << ", ";
428 os << axis.m_bins << ")";
429 return os;
430 }
431
432 protected:
433 void toStream(std::ostream& os) const final { os << *this; }
434
435 private:
437 double m_min{};
439 double m_max{};
441 double m_width{};
443 std::size_t m_bins{};
444};
