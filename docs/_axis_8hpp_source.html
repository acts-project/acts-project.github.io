149 NeighborHoodIndices neighborHoodIndices(std::size_t idx,
150 std::pair<int, int> sizes = {-1,
151 1}) const
152 requires(bdt == AxisBoundaryType::Closed)
153 {
154 // Handle invalid indices
155 if (idx <= 0 || idx >= (getNBins() + 1)) {
156 return NeighborHoodIndices();
157 }
158
159 // Handle corner case where user requests more neighbours than the number
160 // of bins on the axis. All bins are returned in this case.
161
162 const int max = getNBins();
163 sizes.first = std::clamp(sizes.first, -max, max);
164 sizes.second = std::clamp(sizes.second, -max, max);
165 if (std::abs(sizes.first - sizes.second) >= max) {
166 sizes.first = 1 - idx;
167 sizes.second = max - idx;
168 }
169
170 // If the entire index range is not covered, we must wrap the range of
171 // targeted neighbor indices into the range of valid bin indices. This may
172 // split the range of neighbor indices in two parts:
173 //
174 // Before wraparound - [ XXXXX]XXX
175 // After wraparound - [ XXXX XXXX ]
176 //
177 const int itmin = idx + sizes.first;
178 const int itmax = idx + sizes.second;
179 const std::size_t itfirst = wrapBin(itmin);
180 const std::size_t itlast = wrapBin(itmax);
181 if (itfirst <= itlast) {
182 return NeighborHoodIndices(itfirst, itlast + 1);
183 } else {
184 return NeighborHoodIndices(itfirst, max + 1, 1, itlast + 1);
185 }
186 }
