150 NeighborHoodIndices neighborHoodIndices(std::size_t idx,
151 std::pair<int, int> sizes = {-1,
152 1}) const
153 requires(bdt == AxisBoundaryType::Closed)
154 {
155 // Handle invalid indices
156 if (idx <= 0 || idx >= (getNBins() + 1)) {
157 return NeighborHoodIndices();
158 }
159
160 // Handle corner case where user requests more neighbours than the number
161 // of bins on the axis. All bins are returned in this case.
162
163 const int max = static_cast<int>(getNBins());
164 sizes.first = std::clamp(sizes.first, -max, max);
165 sizes.second = std::clamp(sizes.second, -max, max);
166 if (std::abs(sizes.first - sizes.second) >= max) {
167 sizes.first = 1 - static_cast<int>(idx);
168 sizes.second = max - static_cast<int>(idx);
169 }
170
171 // If the entire index range is not covered, we must wrap the range of
172 // targeted neighbor indices into the range of valid bin indices. This may
173 // split the range of neighbor indices in two parts:
174 //
175 // Before wraparound - [ XXXXX]XXX
176 // After wraparound - [ XXXX XXXX ]
177 //
178 const int itmin = static_cast<int>(idx) + sizes.first;
179 const int itmax = static_cast<int>(idx) + sizes.second;
180 const std::size_t itfirst = wrapBin(itmin);
181 const std::size_t itlast = wrapBin(itmax);
182 if (itfirst <= itlast) {
183 return NeighborHoodIndices(itfirst, itlast + 1);
184 } else {
185 return NeighborHoodIndices(itfirst, static_cast<std::size_t>(max + 1), 1,
186 itlast + 1);
187 }
188 }
