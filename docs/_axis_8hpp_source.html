131 NeighborHoodIndices neighborHoodIndices(std::size_t idx,
132 std::pair<int, int> sizes = {-1,
133 1}) const
134 requires(bdt == AxisBoundaryType::Closed)
135 {
136 // Handle invalid indices
137 if (idx <= 0 || idx >= (getNBins() + 1)) {
138 return NeighborHoodIndices();
139 }
140
141 // Handle corner case where user requests more neighbours than the number
142 // of bins on the axis. All bins are returned in this case.
143
144 const int max = getNBins();
145 sizes.first = std::clamp(sizes.first, -max, max);
146 sizes.second = std::clamp(sizes.second, -max, max);
147 if (std::abs(sizes.first - sizes.second) >= max) {
148 sizes.first = 1 - idx;
149 sizes.second = max - idx;
150 }
151
152 // If the entire index range is not covered, we must wrap the range of
153 // targeted neighbor indices into the range of valid bin indices. This may
154 // split the range of neighbor indices in two parts:
155 //
156 // Before wraparound - [ XXXXX]XXX
157 // After wraparound - [ XXXX XXXX ]
158 //
159 const int itmin = idx + sizes.first;
160 const int itmax = idx + sizes.second;
161 const std::size_t itfirst = wrapBin(itmin);
162 const std::size_t itlast = wrapBin(itmax);
163 if (itfirst <= itlast) {
164 return NeighborHoodIndices(itfirst, itlast + 1);
165 } else {
166 return NeighborHoodIndices(itfirst, max + 1, 1, itlast + 1);
167 }
168 }

194 std::size_t wrapBin(int bin) const

263
270 bool isInside(double x) const { return (m_min <= x) && (x < m_max); }
271
274 std::vector<double> getBinEdges() const final {
275 std::vector<double> binEdges;
276 for (std::size_t i = 1; i <= m_bins; i++) {
277 binEdges.push_back(getBinLowerBound(i));
278 }
279 binEdges.push_back(getBinUpperBound(m_bins));
280 return binEdges;
281 }
