73 template <typename internal_jacobian_t = free_matrix_type>
74 struct state {
76 using internal_jacobian_type = internal_jacobian_t;
77
80 DETRAY_HOST_DEVICE
81 explicit state(const free_track_parameters_type &free_params)
82 : m_track(free_params) {
83 assert(!m_track.is_invalid());
84
85 // HACK: When the overload resolution for the transport Jacobian
86 // type is resolved, turn this into a default member
87 // initialization.
88 if constexpr (concepts::transport_jacobian<internal_jacobian_type>) {
89 m_jac_transport = internal_jacobian_type::identity();
90 } else {
91 m_jac_transport = matrix::identity<internal_jacobian_type>();

100 template <concepts::detector detector_t>
101 DETRAY_HOST_DEVICE state(const bound_track_parameters_type &bound_params,
102 const detector_t &det,
103 const typename detector_t::geometry_context &ctx) {
104 assert(!bound_params.is_invalid());
105 assert(!bound_params.surface_link().is_invalid());
106
107 // Departure surface
108 const auto sf = tracking_surface{det, bound_params.surface_link()};
109
110 // Set free track parameters for stepping/navigation
111 m_track = sf.bound_to_free_vector(ctx, bound_params);
112
113 assert(!m_track.is_invalid());
114
115 // HACK: When the overload resolution for the transport Jacobian
116 // type is resolved, turn this into a default member
117 // initialization.
118 if constexpr (concepts::transport_jacobian<internal_jacobian_type>) {
119 m_jac_transport = internal_jacobian_type::identity();
120 } else {
121 m_jac_transport = matrix::identity<internal_jacobian_type>();
122 }
123 }
124
126 DETRAY_HOST_DEVICE
127 free_track_parameters_type &operator()() { return m_track; }
128
130 DETRAY_HOST_DEVICE
131 const free_track_parameters_type &operator()() const { return m_track; }
132
135 DETRAY_HOST_DEVICE
136 inline step::direction direction() const {
137 return m_step_size >= 0.f ? step::direction::e_forward
138 : step::direction::e_backward;
139 }
140
142 DETRAY_HOST_DEVICE inline void count_trials() { ++m_n_total_trials; }
143
146 DETRAY_HOST_DEVICE inline std::size_t n_total_trials() const {
147 return m_n_total_trials;
148 }
149
152 DETRAY_HOST_DEVICE
153 inline void set_step_size(const scalar_type step) {
154 assert(math::fabs(step) >= 1e-4f * unit<scalar_type>::mm);
155 m_step_size = step;
156 }
157
159 DETRAY_HOST_DEVICE
160 inline scalar_type step_size() const { return m_step_size; }
161
163 DETRAY_HOST_DEVICE
164 inline scalar_type path_length() const { return m_path_length; }
165
167 DETRAY_HOST_DEVICE
168 inline scalar_type abs_path_length() const { return m_abs_path_length; }
169
172 DETRAY_HOST_DEVICE inline void update_path_lengths(scalar_type seg) {
173 m_path_length += seg;
174 m_abs_path_length += math::fabs(seg);
175 }
176
180 template <step::constraint type = step::constraint::e_actor>
181 DETRAY_HOST_DEVICE inline void set_constraint(scalar_type step_size) {
182 m_constraint.template set<type>(step_size);
183 }
184
186 DETRAY_HOST_DEVICE
187 inline const constraint_t &constraints() const { return m_constraint; }
188
190 template <step::constraint type = step::constraint::e_actor>
191 DETRAY_HOST_DEVICE inline void release_step() {
192 m_constraint.template release<type>();
193 }
194
196 DETRAY_HOST_DEVICE
197 const auto &particle_hypothesis() const { return m_ptc; }
198
201 DETRAY_HOST_DEVICE
202 inline void set_particle(const pdg_particle<scalar_type> &ptc) {
203 m_ptc = ptc;
204 }
205
208 DETRAY_HOST_DEVICE
209 inline free_matrix_type transport_jacobian() const
210 requires(!std::same_as<free_matrix_type, internal_jacobian_type>)
211 {
212 return m_jac_transport.operator dmatrix<algebra_t, 8, 8>();
213 }
214
216 DETRAY_HOST_DEVICE
217 inline const free_matrix_type &transport_jacobian() const
218 requires std::same_as<free_matrix_type, internal_jacobian_type>
219 {
220 return m_jac_transport;
221 }

224 DETRAY_HOST_DEVICE
225 inline free_matrix_type &transport_jacobian()
226 requires std::same_as<free_matrix_type, internal_jacobian_type>
227 {
228 return m_jac_transport;
229 }
230
232 DETRAY_HOST_DEVICE
233 inline internal_jacobian_type &internal_transport_jacobian() {
234 return m_jac_transport;
235 }
236
238 DETRAY_HOST_DEVICE
239 inline const internal_jacobian_type &internal_transport_jacobian() const {
240 return m_jac_transport;
241 }
242
244 DETRAY_HOST_DEVICE
245 inline void reset_transport_jacobian() {
246 // HACK: When the overload resolution for the transport Jacobian
247 // type is resolved, remove this conditional logic.
248 if constexpr (concepts::transport_jacobian<internal_jacobian_type>) {
249 m_jac_transport = internal_jacobian_type::identity();
250 } else {
251 m_jac_transport = matrix::identity<internal_jacobian_type>();
252 }
253 }
254
256 DETRAY_HOST_DEVICE
257 inline typename policy_t::state &policy_state() { return m_policy_state; }
258
260 DETRAY_HOST
261 constexpr auto &inspector() { return m_inspector; }
262
267 DETRAY_HOST_DEVICE
268 inline void run_inspector([[maybe_unused]] const stepping::config &cfg,
269 [[maybe_unused]] const char *message,
270 [[maybe_unused]] const scalar_type dist) {
271 if constexpr (!std::is_same_v<inspector_t, stepping::void_inspector>) {
272 m_inspector(*this, cfg, message, dist);
273 }
274
275 DETRAY_DEBUG_HOST("" << message << "\n"
276 << detray::stepping::print_state(*this, dist));
277 }
278
279 private:
281 internal_jacobian_type m_jac_transport;
282
284 free_track_parameters_type m_track;
285
287 pdg_particle<scalar_type> m_ptc = muon<scalar_type>();
288
290 std::size_t m_n_total_trials{0u};
291
293 scalar_type m_step_size{0.f};
294
296 scalar_type m_path_length{0.f};
297
299 scalar_type m_abs_path_length{0.f};
300
302 DETRAY_NO_UNIQUE_ADDRESS constraint_t m_constraint = {};
303
305 DETRAY_NO_UNIQUE_ADDRESS typename policy_t::state m_policy_state = {};
306
308 DETRAY_NO_UNIQUE_ADDRESS inspector_type m_inspector;
309 };
