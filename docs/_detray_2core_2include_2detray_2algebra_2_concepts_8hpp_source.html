39template <typename T>
40concept scalar =
41 !traits::is_matrix<T> && !traits::is_vector<T> && requires(T a, T b) {
42 { a + b } -> std::convertible_to<T>;
43 { a - b } -> std::convertible_to<T>;
44 { a * b } -> std::convertible_to<T>;
45 { a / b } -> std::convertible_to<T>;
46 };

100template <typename MA, typename MB>
101concept matrix_compatible =
102 matrix<MA> && matrix<MB> &&
103 std::convertible_to<traits::index_t<MA>, traits::index_t<MB>> &&
104 std::convertible_to<traits::index_t<MB>, traits::index_t<MA>>;
105
106template <typename MA, typename MB>
107concept matrix_multipliable =
108 matrix_compatible<MA, MB> && (traits::columns<MA> == traits::rows<MB>) &&
109 requires(traits::scalar_t<MA> sa, traits::scalar_t<MB> sb) {
110 { (sa * sb) + (sa * sb) };
111 };
112
113template <typename MA, typename MB, typename MC>
114concept matrix_multipliable_into =
115 matrix_multipliable<MA, MB> && matrix_compatible<MA, MC> &&
116 matrix_compatible<MB, MC> && (traits::rows<MC> == traits::rows<MA>) &&
117 (traits::columns<MC> == traits::columns<MB>) &&
118 requires(traits::scalar_t<MA> sa, traits::scalar_t<MB> sb,
119 traits::scalar_t<MC>& sc) {
120 { sc += (sa * sb) };
121 };

125template <typename T>
126concept transform3D = requires(T trf) {
127 // Local type definitions
128 requires scalar<typename T::scalar_type>;
129 requires vector2D<typename T::vector2>;
130 requires vector3D<typename T::vector3>;
131 requires point2D<typename T::point2>;
132 requires point3D<typename T::point3>;
133
134 // Methods
135 trf.rotation();
136 trf.translation();
137 trf.point_to_global(typename T::vector2());
138 trf.point_to_global(typename T::vector3());
139 trf.point_to_local(typename T::vector3());
140 trf.vector_to_global(typename T::vector2());
141 trf.vector_to_global(typename T::vector3());
142 trf.vector_to_local(typename T::vector3());
143};
