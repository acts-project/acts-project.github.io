237 void resetAfterVolumeSwitch() {
238 resetAfterLayerSwitch();

240 navLayers.clear();
241 navLayerIndex.reset();
242 navBoundaries.clear();
243 navBoundaryIndex.reset();
244 navCandidates.clear();
245 navCandidateIndex.reset();
246
247 currentLayer = nullptr;
248
249 policyStateManager.reset();

253 void resetForRenavigation() {
254 resetAfterVolumeSwitch();
255
256 currentVolume = nullptr;
257 currentSurface = nullptr;
258
259 navigationBreak = false;
260 navigationStage = Stage::initial;
261
262 // Set the surface reached switches back to false
263 std::ranges::for_each(freeCandidates,
264 [](std::pair<const Surface*, bool>& freeSurface) {
265 freeSurface.second = false;
266 });
267
268 stream.reset();
269 }
