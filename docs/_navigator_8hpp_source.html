276 void resetAfterVolumeSwitch() {
277 resetAfterLayerSwitch();
278
279 navLayers.clear();
280 navLayerIndex.reset();
281 navBoundaries.clear();
282 navBoundaryIndex.reset();
283 navCandidates.clear();
284 navCandidateIndex.reset();
285
286 currentLayer = nullptr;
287 }
288
290 void reset() {
291 resetAfterVolumeSwitch();
292
293 currentVolume = nullptr;
294 currentSurface = nullptr;
295
296 navigationBreak = false;
297 navigationStage = Stage::initial;
298 // Set the surface reached switches back to false
299 std::ranges::for_each(freeCandidates,
300 [](std::pair<const Surface*, bool>& freeSurface) {
301 freeSurface.second = false;
302 });
303 }
304 };
