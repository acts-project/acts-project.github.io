241 void resetAfterVolumeSwitch() {
242 resetAfterLayerSwitch();
243
244 navLayers.clear();
245 navLayerIndex.reset();
246 navBoundaries.clear();
247 navBoundaryIndex.reset();
248
249 currentLayer = nullptr;
250
251 policyStateManager.reset();
252 policyStateIsDefault = true;
253 resetStream();
254 }
255
260 void resetForRenavigation() {
261 resetAfterVolumeSwitch();
262
263 currentVolume = nullptr;
264 currentSurface = nullptr;
265
266 navigationBreak = false;
267 navigationStage = Stage::initial;
268
269 // Set the surface reached switches back to false
270 std::ranges::for_each(freeCandidates,
271 [](std::pair<const Surface*, bool>& freeSurface) {
272 freeSurface.second = false;
273 });
274
275 stream.reset(false);
276 }
277
278 void resetStream() { stream.reset(options.keepUnreachedExternal); }
279
284 void resetForInitialization() {
285 resetForRenavigation();
286
287 startVolume = nullptr;
288 startLayer = nullptr;
289 startSurface = nullptr;
290 targetSurface = nullptr;
291 }
