65class input_file_converter_interface : public input_converter_interface {
66 public:
69 explicit input_file_converter_interface(const std::string& ext)
70 : m_file_extension{ext} {};
71
73 std::string_view file_extension() const { return m_file_extension; }
74
78 virtual void add_from_file(io::file_handle& file,
79 detector_payload& payload) = 0;
80
84 void add_to_payload(const std::any& input_data,
85 detector_payload& payload) override {
86 const std::string file_name = get_file_name(input_data);
87 io::file_handle file{file_name, std::ios_base::in | std::ios_base::binary};
88
89 add_from_file(file, payload);
90 };
91
92 private:
94 std::string get_file_name(const std::any& input_data) {
95 // Read input data from file
96 std::string file_name{""};
97 try {
98 file_name = std::any_cast<std::string>(input_data);
99 } catch (const std::bad_any_cast& e) {
100 DETRAY_FATAL_HOST(
101 "Unknown input data type in "
102 << m_file_extension
103 << " file converter! Should be 'std::string' (file name)");
104 throw;
105 }
106
107 return file_name;
108 }
109
111 std::string m_file_extension;
112};
