720 static bool reservedColumn(const std::string &name) noexcept;
721
722 template <typename Holder>
723 MutableColumnProxy<typename Holder::Value> createColumnImpl(
724 const std::string &name) {
725 if (reservedColumn(name)) {
726 throw std::runtime_error("Column name is reserved: " + name);
727 }
728 if (hasColumn(name)) {
729 throw std::runtime_error("Column already exists: " + name);
730 }
731 auto holder = std::make_unique<Holder>();
732 holder->resize(size());
733 auto proxy = holder->proxy(*this);
734 m_allColumns.try_emplace(name, holder.get());
735 m_dynamicColumns.try_emplace(name, std::move(holder));
736 return proxy;
737 }
738
739 template <typename Holder, typename Self>
740 static auto columnImpl(Self &&self, const std::string &name) {
741 const auto it = self.m_allColumns.find(name);
742 if (it == self.m_allColumns.end()) {
743 throw std::runtime_error("Column not found: " + name);
744 }
745 auto &holder = dynamic_cast<Holder &>(*it->second);
746 return holder.proxy(self);
747 }
748
749 template <typename Holder>
750 MutableColumnProxy<typename Holder::Value> columnImpl(
751 const std::string &name) {
752 return columnImpl<Holder>(*this, name);
753 }
754
755 template <typename Holder>
756 ConstColumnProxy<typename Holder::Value> columnImpl(
757 const std::string &name) const {
758 return columnImpl<Holder>(*this, name);
759 }
