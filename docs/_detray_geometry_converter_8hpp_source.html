57template <typename metadata_t>
58std::pair<std::shared_ptr<detray::detector<metadata_t>>, detray::name_map>
59toDetray(vecmem::memory_resource& mr, const Acts::GeometryContext& gctx,
60 const Acts::TrackingGeometry& trackingGeometry,
61 const std::string& beampipeVolumeName,
62 const std::string& detectorName = "",
63 Acts::Logging::Level logLevel = Acts::Logging::INFO,
64 bool convertMaterial = true, bool convertSurfaceGrids = true) {
65 auto localLogger =
66 Acts::getDefaultLogger("DetrayGeometryConverter", logLevel);
67 auto payloadLogger = localLogger->clone("DetrayPayloadConverter");
68 ACTS_LOCAL_LOGGER(std::move(localLogger));
69
70 // ── Convert TrackingGeometry → detray payloads ──────────────────────────
71 DetrayPayloadConverter::Config convCfg;
72
73 ACTS_INFO("Looking for beampipe volume: " << beampipeVolumeName);
74
75 // Find beampipe volume
76 trackingGeometry.apply(
77 [&beampipeVolumeName, &convCfg](const Acts::TrackingVolume& volume) {
78 if (volume.volumeName() == beampipeVolumeName) {
79 convCfg.beampipeVolume = &volume;
80 }
81 });
82
83 if (convCfg.beampipeVolume == nullptr) {
84 ACTS_WARNING("DetrayGeometryProvider: beampipe volume '"
85 << beampipeVolumeName << "' not found");
86 }
87
88 DetrayPayloadConverter converter(convCfg, std::move(payloadLogger));
89
90 auto payloads = converter.convertTrackingGeometry(gctx, trackingGeometry);
91
92 // ── Build detray detector from payloads ──────────────────────────────────
93 using detector_t = detray::detector<metadata_t>;
94
95 detray::detector_builder<metadata_t> detectorBuilder{};
96
97 detray::io::geometry_reader::from_payload<detector_t>(detectorBuilder,
98 *payloads.detector);
99
100 if (convertMaterial) {
101 detray::io::homogeneous_material_reader::from_payload<detector_t>(
102 detectorBuilder, *payloads.homogeneousMaterial);
103
104 detray::io::material_map_reader<std::integral_constant<std::size_t, 2>>::
105 from_payload<detector_t>(detectorBuilder,
106 std::move(*payloads.materialGrids));
107 }
108
109 if (convertSurfaceGrids) {
110 detray::io::surface_grid_reader<typename detector_t::surface_type,
111 std::integral_constant<std::size_t, 0>,
112 std::integral_constant<std::size_t, 2>>::
113 template from_payload<detector_t>(detectorBuilder,
114 *payloads.surfaceGrids);
115 }
116
117 if (!detectorName.empty()) {
118 detectorBuilder.set_name(detectorName);
119 } else if (payloads.names.contains(0)) {
120 detectorBuilder.set_name(payloads.names.at(0));
121 }
122
123 detray::name_map names{};
124 auto det = std::make_shared<detector_t>(detectorBuilder.build(mr, names));
125
126 return {std::move(det), std::move(names)};
127}

136template <typename detector_t>
137std::unordered_map<detray::geometry::identifier, Acts::GeometryIdentifier>
138buildDetrayToActsMap(const detector_t& detrayDetector,
139 Acts::Logging::Level logLevel = Acts::Logging::INFO) {
140 auto localLogger =
141 Acts::getDefaultLogger("DetrayGeometryConverter", logLevel);
142 ACTS_LOCAL_LOGGER(std::move(localLogger));
143 // ── Build detray→Acts geometry ID map
144 // ───────────────────────────────────
145 std::unordered_map<detray::geometry::identifier, Acts::GeometryIdentifier>
146 detrayToActsMap;
147
148 for (const auto& surface : detrayDetector.surfaces()) {
149 // surface.source is the Acts GeometryIdentifier encoded as uint64
150 const Acts::GeometryIdentifier actsId(surface.source);
151 if (actsId.sensitive() == 0) {
152 continue; // skip portals and passives
153 }
154 detrayToActsMap[surface.identifier()] = actsId;
155 }
156
157 ACTS_INFO("DetrayGeometryProvider: built detray→Acts map with "
158 << detrayToActsMap.size() << " sensitive surfaces");
159 return detrayToActsMap;
160}
