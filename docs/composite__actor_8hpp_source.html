35template <concepts::actor principal_actor_t = base_actor,
36 concepts::actor... observers>

44 using actor_type = principal_actor_t;
46 using state = typename actor_type::state;
48 using result = typename actor_type::result;

70 template <typename actor_states_t, typename propagator_state_t,
71 typename subj_result_t = typename actor::empty_result>
72 DETRAY_HOST_DEVICE void operator()(actor_states_t &states,
73 propagator_state_t &p_state,
74 subj_result_t &&subject_res = {}) const {
75 // State of the primary actor that is implement by this composite actor
76 auto &actor_state = detail::get<typename actor_type::state &>(states);
77
78 // Do your own work ...
79 // Two cases: This is a simple actor or observing actor (pass on its
80 // subject's state)
81 result res{};
82 if constexpr (std::same_as<subj_result_t, actor::empty_result>) {
83 res = actor_type::operator()(actor_state, p_state);
84 } else {
85 res = actor_type::operator()(actor_state, p_state,
86 std::forward<subj_result_t>(subject_res));
87 }
88
89 // ... then run the observers on the new result
90 if (res.status == actor::status::e_notify) {
91 notify(states, p_state, res,
92 std::make_index_sequence<sizeof...(observers)>{});
93 }
94 }
95
96 private:
103 template <concepts::actor observer_t, typename actor_states_t,
104 typename propagator_state_t>
105 DETRAY_HOST_DEVICE inline void notify(const observer_t &observer,
106 actor_states_t &states,
107 propagator_state_t &p_state,
108 result &res) const {
109 // Two cases: observer is a simple actor or a composite actor
110 if constexpr (!concepts::composite_actor<observer_t>) {
111 // No actor state defined (empty)
112 if constexpr (std::same_as<typename observer_t::state,
113 detray::base_actor::state>) {
114 observer(p_state, res);
115 } else {
116 observer(detail::get<typename observer_t::state &>(states), p_state,
117 res);
118 }
119 } else {
120 observer(states, p_state, res);
121 }
122 }
123
132 template <std::size_t... indices, typename actor_states_t,
133 typename propagator_state_t>
134 DETRAY_HOST_DEVICE inline void notify(
135 actor_states_t &states, propagator_state_t &p_state, result &res,
136 std::index_sequence<indices...> /*ids*/) const {
137 (notify(detail::get<indices>(m_observers), states, p_state, res), ...);
138 }
139
141 DETRAY_NO_UNIQUE_ADDRESS dtuple<observers...> m_observers = {};
142};
