37template <concepts::metadata metadata_t,
38 template <typename> class volume_builder_t = volume_builder,
39 template <typename...> class volume_data_t = std::vector>
40class detector_builder {
41 public:
42 // Cannot use the host::detector type, because the friendship relation with
43 // the volume builders cannot be resolved through the type alias
45 using detector_type = detail::detector<metadata_t, host_container_types>;
47 using algebra_type = typename detector_type::algebra_type;
49 using scalar_type = dscalar<algebra_type>;
50
51 DETRAY_HOST detector_builder() { DETRAY_VERBOSE_HOST("New builder created"); }
52
55 DETRAY_HOST void set_name(std::string det_name) {
56 m_detector_name = std::move(det_name);
57 DETRAY_VERBOSE_HOST("Set detector name: " << m_detector_name);
58 }
59
61 DETRAY_HOST std::string_view name() const { return m_detector_name; }
62
68 template <typename... Args>
69 DETRAY_HOST auto new_volume(const volume_id id, Args&&... args)
70 -> volume_builder_interface<detector_type>* {
71 DETRAY_VERBOSE_HOST("Adding new volume: " << m_volumes.size());
72
73 m_volumes.push_back(std::make_unique<volume_builder_t<detector_type>>(
74 id, static_cast<dindex>(m_volumes.size()),
75 std::forward<Args>(args)...));
76
77 return m_volumes.back().get();
78 }
79
81 DETRAY_HOST auto n_volumes() const -> dindex {
82 return static_cast<dindex>(m_volumes.size());
83 }
84
88 DETRAY_HOST bool has_volume(const std::size_t volume_idx) const {
89 return volume_idx < m_volumes.size();
90 }
91
96 template <class builder_t>
97 DETRAY_HOST auto decorate(dindex volume_idx) -> builder_t* {
98 assert(has_volume(volume_idx));
99
100 m_volumes[volume_idx] =
101 std::make_unique<builder_t>(std::move(m_volumes[volume_idx]));
102
103 // Always works, we set it as this type in the line above
104 return dynamic_cast<builder_t*>(m_volumes[volume_idx].get());
105 }
106
110 template <class builder_t>
111 DETRAY_HOST auto decorate(
112 const volume_builder_interface<detector_type>* v_builder) -> builder_t* {
113 assert(v_builder != nullptr);
114
115 return decorate<builder_t>(v_builder->vol_index());
116 }
117
121 DETRAY_HOST
122 auto operator[](dindex volume_idx)
123 -> volume_builder_interface<detector_type>* {
124 return m_volumes[volume_idx].get();
125 }

132 DETRAY_HOST
133 auto build(vecmem::memory_resource& resource,
134 const volume_builder_options& opt) -> detector_type {
135 DETRAY_INFO_HOST("Building detector: \"" << name() << "\"... ");
136 DETRAY_INFO_HOST("-> type: " << DETRAY_TYPENAME(metadata_t));
137
138 detector_type det{resource};
139
140 DETRAY_VERBOSE_HOST("Have " << m_volumes.size()
141 << " configured volume builders");
142 DETRAY_VERBOSE_HOST("Start building the volumes...");
143 for (auto& vol_builder : m_volumes) {
144 DETRAY_VERBOSE_HOST("-> Build: " << vol_builder->name());
145 vol_builder->build(det, opt);
146 }
147
148 // TODO: Make fully generic for more volume accelerator types
149
150 // TODO: Add sorting, data deduplication etc. here later...
151
152 DETRAY_INFO_HOST("-> Built " << det.volumes().size() << " volumes");
153 DETRAY_INFO_HOST("-> Built " << det.surfaces().size() << " surfaces:");
154 DETRAY_INFO_HOST("--> portals: " << detray::n_portals(det));
155 DETRAY_INFO_HOST("--> sensitives: " << detray::n_sensitives(det));
156 DETRAY_INFO_HOST("--> passives: " << detray::n_passives(det));
157
158 if constexpr (detray::concepts::has_surface_grids<detector_type>) {
159 DETRAY_INFO_HOST("-> Built " << detray::n_surface_grids(det)
160 << " surface grids");
161 }
162
163 if constexpr (detray::concepts::has_material_maps<detector_type>) {
164 DETRAY_INFO_HOST("-> Built " << detray::n_material_maps(det)
165 << " material maps");
166 }
167
168 if constexpr (detray::concepts::has_homogeneous_material<detector_type>) {
169 DETRAY_INFO_HOST("-> Built homogeneous material:");
170 DETRAY_INFO_HOST("--> slabs: " << detray::n_material_slabs(det));
171 DETRAY_INFO_HOST("--> rods: " << detray::n_material_rods(det));
172 }
173 DETRAY_INFO_HOST("Detector building complete: " << name());
174
175 return det;
176 }
177
184 DETRAY_HOST
185 auto build(vecmem::memory_resource& resource,
186 const volume_builder_options& opt,
187 typename detector_type::name_map& name_map) -> detector_type {
188 DETRAY_VERBOSE_HOST("detray: filling names for detector " << name());
189
190 assert(name_map.empty());
191
192 // By convention the name of the detector is at position 0
193 name_map.set_detector_name(m_detector_name);
194
195 for (auto& vol_builder : m_volumes) {
196 name_map.emplace(vol_builder->vol_index(),
197 std::string{vol_builder->name()});
198 }
199
200 return build(resource, opt);
201 }
202
203 private:
205 std::string m_detector_name{"detray_detector"};
207 volume_data_t<std::unique_ptr<volume_builder_interface<detector_type>>>
208 m_volumes{};
209};
