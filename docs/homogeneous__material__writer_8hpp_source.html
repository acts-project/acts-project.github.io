44 template <class detector_t>
45 static homogeneous_material_header_payload to_header_payload(
46 const detector_t& det) {
47 homogeneous_material_header_payload header_data;
48
49 using algebra_t = typename detector_t::algebra_type;
50 using material_id_t = surface_material_payload::mat_type;
51
52 auto count_surface_with_material_type = [&det](material_id_t target_type) {
53 std::size_t count{0u};
54 for (const auto& [sf_idx, sf_desc] :
55 detray::views::enumerate(det.surfaces())) {
56 const auto sf = geometry::surface{det, sf_desc};
57
58 if (surface_material_payload mslp =
59 sf.has_material() ? sf.template visit_material<
60 get_material_payload<algebra_t>>(sf_idx)
61 : surface_material_payload{};
62 mslp.type == target_type) {
63 count++;
64 }
65 }
66
67 return count;
68 };
69
70 const auto& materials = det.material_store();
71
72 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
73 header_data.n_slabs =
74 materials.template size<detector_t::material::id::e_material_slab>();
75 header_data.n_slab_surfaces =
76 count_surface_with_material_type(material_id_t::slab);
77 }
78 header_data.n_rods = 0u;
79 if constexpr (detray::concepts::has_material_rods<detector_t>) {
80 header_data.n_rods =
81 materials.template size<detector_t::material::id::e_material_rod>();
82 header_data.n_rod_surfaces =
83 count_surface_with_material_type(material_id_t::rod);
84 }
85
86 return header_data;
87 }

94 template <class detector_t>
95 static payload_type to_payload(
96 const detector_t& det, const typename detector_t::name_map& /*unused*/) {
97 payload_type dm_data;
98 dm_data.volumes.reserve(det.volumes().size());
99
100 for (const auto& vol : det.volumes()) {
101 dm_data.volumes.push_back(to_payload(vol, det));
102 }
103
104 return dm_data;
105 }

113 template <class detector_t>
114 static material_volume_payload to_payload(
115 const typename detector_t::volume_type& vol_desc, const detector_t& det) {
116 using material_type = surface_material_payload::mat_type;
117
118 material_volume_payload mv_data;
119 mv_data.volume_link = detail::basic_converter::to_payload(vol_desc.index());
120
121 // Return early if the stores for homogeneous materials are empty
122 using mat_id = typename detector_t::material::id;
123 using algebra_t = typename detector_t::algebra_type;
124
125 bool contains_rods{true};
126 bool contains_slabs{true};
127 if constexpr (detray::concepts::has_material_rods<detector_t>) {
128 if (det.material_store().template empty<mat_id::e_material_rod>()) {
129 contains_rods = false;
130 }
131 }
132 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
133 if (det.material_store().template empty<mat_id::e_material_slab>()) {
134 contains_slabs = false;
135 }
136 }
137 if (!contains_rods && !contains_slabs) {
138 return mv_data;
139 }
140
141 // Find all surfaces that belong to the volume and count them
142 std::size_t sf_idx{0u};
143 std::size_t slab_idx{0u};
144 std::size_t rod_idx{0u};
145 auto vol = tracking_volume{det, vol_desc};
146 for (const auto& sf_desc : vol.surfaces()) {
147 // Convert material slabs and rods
148 const auto sf = geometry::surface{det, sf_desc};
149
150 if (surface_material_payload mslp =
151 sf.has_material()
152 ? sf.template visit_material<get_material_payload<algebra_t>>(
153 sf_idx)
154 : surface_material_payload{};
155 mslp.type == material_type::slab) {
156 mslp.index_in_coll = slab_idx++;
157 mv_data.surface_mat.push_back(mslp);
158 } else if (mslp.type == material_type::rod) {
159 mslp.index_in_coll = rod_idx++;
160 mv_data.surface_mat.push_back(mslp);
161 } else { /* material maps are handled by another writer */
162 }
163 ++sf_idx;
164 }
165
166 return mv_data;
167 }

173 template <detray::concepts::material_params material_t>
174 static material_param_payload to_payload(const material_t& mat) {
175 material_param_payload mat_data;
176
177 mat_data.params = {mat.X0(),
178 mat.L0(),
179 mat.Ar(),
180 mat.Z(),
181 mat.mass_density(),
182 mat.molar_density(),
183 static_cast<io::scalar>(mat.state())};
184 return mat_data;
185 }

193 template <detray::concepts::algebra algebra_t,
194 detray::concepts::homogeneous_material material_t>
195 static surface_material_payload to_payload(const material_t& mat,
196 std::size_t sf_idx) {
197 surface_material_payload mat_data;
198
199 mat_data.type = io::detail::get_id<algebra_t, material_t>();
200 mat_data.surface = detail::basic_converter::to_payload(sf_idx);
201 mat_data.thickness = mat.thickness();
202 mat_data.mat = to_payload(mat.get_material());
203
204 return mat_data;
205 }
206
207 private:
209 template <detray::concepts::algebra algebra_t>
210 struct get_material_payload {
211 template <typename material_group_t, typename index_t>
212 inline auto operator()(const material_group_t& material_group,
213 const index_t& index,
214 [[maybe_unused]] std::size_t sf_index) const {
215 using material_t = typename material_group_t::value_type;
216
217 if constexpr (detray::concepts::material_slab<material_t> ||
218 detray::concepts::material_rod<material_t>) {
219 return homogeneous_material_writer::to_payload<algebra_t>(
220 material_group[index], sf_index);
221 } else {
222 return surface_material_payload{};
223 }
224 }
225 };
226};
