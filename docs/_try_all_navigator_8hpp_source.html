241 [[nodiscard]] Result<void> initialize(State& state, const Vector3& position,
242 const Vector3& direction,
243 Direction propagationDirection) const {
244 static_cast<void>(propagationDirection);
245
246 ACTS_VERBOSE("initialize");
247
248 state.startSurface = state.options.startSurface;
249 state.targetSurface = state.options.targetSurface;
250
251 const TrackingVolume* startVolume = nullptr;
252
253 if (state.startSurface != nullptr &&
254 state.startSurface->associatedLayer() != nullptr) {
255 ACTS_VERBOSE(
256 "Fast start initialization through association from Surface.");
257 const auto* startLayer = state.startSurface->associatedLayer();
258 startVolume = startLayer->trackingVolume();
259 } else {
260 ACTS_VERBOSE("Slow start initialization through search.");
261 ACTS_VERBOSE("Starting from position " << toString(position)
262 << " and direction "
263 << toString(direction));
264 startVolume = m_cfg.trackingGeometry->lowestTrackingVolume(
265 state.options.geoContext, position);
266 }
267
268 // Initialize current volume, layer and surface
269 {
270 state.currentVolume = startVolume;
271 if (state.currentVolume != nullptr) {
272 ACTS_VERBOSE(volInfo(state) << "Start volume resolved.");
273 } else {
274 ACTS_DEBUG("Start volume not resolved.");
275 state.navigationBreak = true;
276 return NavigatorError::NoStartVolume;
277 }
278
279 state.currentSurface = state.startSurface;
280 if (state.currentSurface != nullptr) {
281 ACTS_VERBOSE(volInfo(state) << "Current surface set to start surface "
282 << state.currentSurface->geometryId());
283 } else {
284 ACTS_VERBOSE(volInfo(state) << "No start surface set.");
285 }
286 }
287
288 // Initialize navigation candidates for the start volume
289 reinitializeCandidates(state);
290
291 state.lastPosition.reset();
292
293 return Result<void>::success();
294 }

307 NavigationTarget nextTarget(State& state, const Vector3& position,
308 const Vector3& direction) const {
309 // Navigator preStep always resets the current surface
310 state.currentSurface = nullptr;
311
312 // Check if the navigator is inactive
313 if (state.navigationBreak) {
314 return NavigationTarget::None();
315 }
316
317 ACTS_VERBOSE(volInfo(state) << "nextTarget");
318
319 if (state.lastPosition.has_value() && !state.hasTargetsBehind()) {
320 ACTS_VERBOSE(volInfo(state) << "Evaluate blind step");
321
322 const Vector3 stepStart = state.lastPosition.value();
323 state.lastPosition.reset();
324 const Vector3 stepEnd = position;
325 const Vector3 step = stepEnd - stepStart;
326 const double stepDistance = step.norm();
327
328 ACTS_VERBOSE("- from: " << stepStart.transpose());
329 ACTS_VERBOSE("- to: " << stepEnd.transpose());
330 ACTS_VERBOSE("- distance: " << stepDistance);
331
332 if (stepDistance < std::numeric_limits<double>::epsilon()) {
333 ACTS_DEBUG(volInfo(state) << "Step distance is zero: " << stepDistance
334 << ". Retry to resolve the next target.");
335 return nextTarget(state, position, direction);
336 }
337
338 const Vector3 stepDirection = step.normalized();
339
340 const double nearLimit = -stepDistance + state.options.surfaceTolerance;
341 const double farLimit = 0;
342
343 state.activeTargetsBehind =
344 resolveTargets(state, stepEnd, stepDirection, nearLimit, farLimit);
345 state.activeTargetBehindIndex = -1;
346
347 ACTS_VERBOSE(volInfo(state)
348 << "Found " << state.activeTargetsBehind.size()
349 << " intersections behind");
350
351 for (const auto& target : state.activeTargetsBehind) {
352 ACTS_VERBOSE("Found target behind " << target.surface().geometryId());
353 }
354 }
355
356 // Prioritize targets behind the current position
357 ++state.activeTargetBehindIndex;
358 if (state.hasTargetsBehind()) {
359 ACTS_VERBOSE(volInfo(state) << "Handle active candidates behind");
360
361 ACTS_VERBOSE(volInfo(state)
362 << (state.activeTargetsBehind.size() -
363 state.activeTargetBehindIndex)
364 << " out of " << state.activeTargetsBehind.size()
365 << " surfaces remain to try.");
366
367 const NavigationTarget& nextTarget = state.activeTargetBehind();
368
369 ACTS_VERBOSE(volInfo(state) << "Next target behind selected: "
370 << nextTarget.surface().geometryId());
371
372 return nextTarget;
373 }
374
375 // No more targets behind, now try to find targets ahead as usual
376
377 state.lastPosition = position;
378 state.activeTargetsBehind.clear();
379 state.activeTargetBehindIndex = -1;
380
381 ACTS_VERBOSE(volInfo(state)
382 << "No targets behind, try to find targets ahead");
383
384 const double nearLimit = state.options.nearLimit;
385 const double farLimit = state.options.farLimit;
386
387 state.activeTargetsAhead =
388 resolveTargets(state, position, direction, nearLimit, farLimit);
389
390 NavigationTarget nextTarget = NavigationTarget::None();
391
392 for (const auto& target : state.activeTargetsAhead) {
393 const Intersection3D& intersection = target.intersection();
394
395 if (intersection.status() == IntersectionStatus::onSurface) {
396 ACTS_ERROR(volInfo(state)
397 << "We are on surface " << target.surface().geometryId()
398 << " before trying to reach it. This should not happen. "
399 "Good luck.");
400 continue;
401 }
402
403 if (intersection.status() == IntersectionStatus::reachable) {
404 nextTarget = target;
405 break;
406 }
407 }
408
409 if (nextTarget.isNone()) {
410 ACTS_VERBOSE(volInfo(state)
411 << "No target ahead found. Step blindly forward.");
412 } else {
413 ACTS_VERBOSE(volInfo(state) << "Next target ahead selected: "
414 << nextTarget.surface().geometryId());
415 }
416
417 return nextTarget;
418 }

450 void handleSurfaceReached(State& state, const Vector3& position,
451 const Vector3& direction,
452 const Surface& /*surface*/) const {
453 // Check if the navigator is inactive
454 if (state.navigationBreak) {
455 return;
456 }
457
458 ACTS_VERBOSE(volInfo(state) << "handleSurfaceReached");
459
460 const std::vector<NavigationTarget>& currentTargets =
461 state.currentTargets();
462
463 if (currentTargets.empty()) {
464 ACTS_VERBOSE(volInfo(state) << "No current target set.");
465 return;
466 }
467
468 assert(state.currentSurface == nullptr && "Current surface must be reset.");
469
470 // handle multiple surface intersections due to increased bounds
471
472 std::vector<NavigationTarget> hitTargets;
473
474 for (const auto& target : currentTargets) {
475 const std::uint8_t index = target.intersectionIndex();
476 const Surface& surface = target.surface();
477 const BoundaryTolerance boundaryTolerance = BoundaryTolerance::None();
478
479 const Intersection3D intersection =
480 surface
481 .intersect(state.options.geoContext, position, direction,
482 boundaryTolerance, state.options.surfaceTolerance)
483 .at(index);
484
485 if (intersection.status() == IntersectionStatus::onSurface) {
486 hitTargets.emplace_back(target);
487 }
488 }
489
490 ACTS_VERBOSE(volInfo(state)
491 << "Found " << hitTargets.size()
492 << " intersections on surface with bounds check.");
493
494 // reset stored targets
495 state.lastPosition.reset();
496 state.activeTargetsAhead.clear();
497 state.activeTargetsBehind.clear();
498 state.activeTargetBehindIndex = -1;
499
500 if (hitTargets.empty()) {
501 ACTS_VERBOSE(volInfo(state) << "No hit targets found.");
502 return;
503 }
504
505 if (hitTargets.size() > 1) {
506 ACTS_VERBOSE(volInfo(state)
507 << "Only using first intersection within bounds.");
508 }
509
510 // we can only handle a single surface hit so we pick the first one
511 const NavigationTarget& target = hitTargets.front();
512 const Surface& surface = target.surface();
513
514 ACTS_VERBOSE(volInfo(state) << "Surface " << surface.geometryId()
515 << " successfully hit, storing it.");
516 state.currentSurface = &surface;
517
518 if (target.isSurfaceTarget()) {
519 ACTS_VERBOSE(volInfo(state) << "This is a surface");
520 } else if (target.isLayerTarget()) {
521 ACTS_VERBOSE(volInfo(state) << "This is a layer");
522 } else if (target.isPortalTarget()) {
523 ACTS_VERBOSE(volInfo(state)
524 << "This is a boundary. Reinitialize navigation");
525
526 const BoundarySurface& boundary = target.boundarySurface();
527
528 state.currentVolume = boundary.attachedVolume(state.options.geoContext,
529 position, direction);
530
531 ACTS_VERBOSE(volInfo(state) << "Switched volume");
532
533 reinitializeCandidates(state);
534 } else {
535 ACTS_ERROR(volInfo(state) << "Unknown intersection type");
536 }
537 }

578 std::vector<NavigationTarget> resolveTargets(State& state,
579 const Vector3& position,
580 const Vector3& direction,
581 double nearLimit,
582 double farLimit) const {
583 std::vector<NavigationTarget> targets;
584
585 // Find intersections with all candidates
586 for (const auto& candidate : state.navigationCandidates) {
587 auto intersections =
588 candidate.intersect(state.options.geoContext, position, direction,
589 state.options.surfaceTolerance);
590 for (auto [intersectionIndex, intersection] :
591 Acts::enumerate(intersections)) {
592 // exclude invalid intersections
593 if (!intersection.isValid() ||
594 !detail::checkPathLength(intersection.pathLength(), nearLimit,
595 farLimit)) {
596 continue;
597 }
598 // store candidate
599 targets.emplace_back(candidate.target(intersection, intersectionIndex));
600 }
601 }
602
603 std::ranges::sort(targets, NavigationTarget::pathLengthOrder);
604
605 return targets;
606 }
