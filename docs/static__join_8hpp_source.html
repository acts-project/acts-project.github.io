44template <std::size_t I, std::input_iterator range_itr_t>
45struct static_join_view
46 : public detray::ranges::view_interface<static_join_view<I, range_itr_t>> {
48 using iterator_coll_t = darray<range_itr_t, I>;
50 using iterator_t =
51 detray::ranges::detail::static_join_iterator<iterator_coll_t>;
53 using value_t = std::iter_value_t<iterator_t>;
54
56 constexpr static_join_view() = default;
57
61 template <detray::ranges::range... ranges_t>
62 requires((sizeof...(ranges_t) > 1) ||
63 ((!std::same_as<static_join_view, ranges_t>) && ...))
64 DETRAY_HOST_DEVICE constexpr explicit static_join_view(ranges_t &&...ranges)
65 : m_begins{detray::ranges::begin(std::forward<ranges_t>(ranges))...},
66 m_ends{detray::ranges::end(std::forward<ranges_t>(ranges))...} {}
67
71 template <detray::ranges::range... ranges_t>
72 DETRAY_HOST_DEVICE constexpr explicit static_join_view(
73 const ranges_t &...ranges)
74 : m_begins{detray::ranges::cbegin(ranges)...},
75 m_ends{detray::ranges::cend(ranges)...} {}
76
78 DETRAY_HOST_DEVICE
79 constexpr auto begin() const -> iterator_t { return {m_begins, m_ends}; }
80
82 DETRAY_HOST_DEVICE
83 constexpr auto end() const -> iterator_t {
84 // Build a joined itr from the last value in the iterator collection
85 return {m_begins, m_ends, detray::detail::get<I - 1>(m_ends), I - 1};
86 }
87
90 DETRAY_HOST_DEVICE
91 constexpr auto data() const -> const value_t * {
92 return &(*(detray::detail::get<0>(m_begins())));
93 }
94
97 DETRAY_HOST_DEVICE
98 constexpr auto data() -> value_t * {
99 return &(*(detray::detail::get<0>(m_begins())));
100 }
101
103 DETRAY_HOST_DEVICE
104 constexpr auto size() const noexcept -> std::size_t {
105 std::size_t size{0};
106 for (std::size_t i{0}; i < I; ++i) {
107 const range_itr_t &begin = m_begins[i];
108 const range_itr_t &end = m_ends[i];
109 size += static_cast<std::size_t>(detray::ranges::distance(begin, end));
110 }
111 return size;
112 }
113
115 iterator_coll_t m_begins{};
116 iterator_coll_t m_ends{};
117};

160template <detray::ranges::range iterator_coll_t>
161 requires std::input_iterator<detray::detail::get_value_t<iterator_coll_t>>
162struct static_join_iterator {
164 using iterator_t = detray::detail::get_value_t<iterator_coll_t>;
165
167 using difference_type = std::iter_difference_t<iterator_t>;
169 using value_type = std::iter_value_t<iterator_t>;
171 using pointer = typename std::iterator_traits<iterator_t>::pointer;
173 using reference = std::iter_reference_t<iterator_t>;
175 using iterator_category =
176 typename std::iterator_traits<iterator_t>::iterator_category;
177
179 constexpr static_join_iterator() = default;
180
184 DETRAY_HOST_DEVICE
185 constexpr static_join_iterator(const iterator_coll_t &begins,
186 const iterator_coll_t &ends)
187 : m_begins(&begins), m_ends(&ends), m_iter{(*m_begins)[0]} {}
188
194 DETRAY_HOST_DEVICE
195 constexpr static_join_iterator(const iterator_coll_t &begins,
196 const iterator_coll_t &ends,
197 iterator_t current, const std::size_t i)
198 : m_begins(&begins), m_ends(&ends), m_iter{current}, m_idx{i} {}
199
203 DETRAY_HOST_DEVICE constexpr auto operator++() -> static_join_iterator & {
204 ++m_iter;
205 // Switch to next range in the collection
206 if (constexpr std::size_t max_idx{
207 sizeof(iterator_coll_t) / sizeof(iterator_t) - 1u};
208 (m_iter == (*m_ends)[m_idx]) && (m_idx < max_idx)) {
209 ++m_idx;
210 m_iter = (*m_begins)[m_idx];
211 }
212 return *this;
213 }
214
217 DETRAY_HOST_DEVICE constexpr auto operator++(int) -> static_join_iterator {
218 auto tmp(*this);
219 ++(*this);
220 return tmp;
221 }
222
223
227 DETRAY_HOST_DEVICE constexpr auto operator--() -> static_join_iterator &
228 requires std::bidirectional_iterator<iterator_t>
229 {
230 if (m_iter != (*m_begins)[m_idx]) {
231 // Normal case
232 --m_iter;
233 } else if (m_idx > 0u) {
234 // Iterator has reached last valid position in this range during the
235 // previous decrement. Now go to the end of the previous range
236 --m_idx;
237 m_iter = detray::ranges::prev((*m_ends)[m_idx]);
238 }
239 return *this;
240 }
241
242
245 DETRAY_HOST_DEVICE constexpr auto operator--(int) -> static_join_iterator
246 requires std::bidirectional_iterator<iterator_t>
247 {
248 auto tmp(*this);
249 ++(*this);
250 return tmp;
251 }
252
254 DETRAY_HOST_DEVICE
255 constexpr auto operator*() const -> decltype(auto) { return *m_iter; }
256
260 DETRAY_HOST_DEVICE constexpr auto operator+=(const difference_type j)
261 -> static_join_iterator &
262 requires std::random_access_iterator<iterator_t>
263 {
264 // walk through join to catch the switch between intermediate ranges
265 if (difference_type i{j}; i >= difference_type{0}) {
266 while (i--) {
267 ++(*this);
268 }
269 } else {
270 while (i++) {
271 --(*this);
272 }
273 }
274 return *this;
275 }
276
280 DETRAY_HOST_DEVICE constexpr auto operator-=(const difference_type j)
281 -> static_join_iterator &
282 requires std::random_access_iterator<iterator_t>
283 {
284 m_iter += (-j);
285 return *this;
286 }
287
291 DETRAY_HOST_DEVICE constexpr decltype(auto) operator[](
292 const difference_type i) const
293 requires std::random_access_iterator<iterator_t>
294 {
295 difference_type offset{i - (m_iter - (*m_begins)[0])};
296 return *(*this + offset);
297 }
298
299 private:
301 DETRAY_HOST_DEVICE friend constexpr bool operator==(
302 const static_join_iterator &lhs, const static_join_iterator &rhs) {
303 return (lhs.m_iter == rhs.m_iter);
304 }
305
307 DETRAY_HOST_DEVICE friend constexpr auto operator<=>(
308 const static_join_iterator &lhs, const static_join_iterator &rhs)
309 requires std::random_access_iterator<iterator_t>
310 {
311#if defined(__apple_build_version__)
312 const auto l{lhs.m_iter};
313 const auto r{rhs.m_iter};
314 if (l < r || (l == r && l < r)) {
315 return std::strong_ordering::less;
316 }
317 if (l > r || (l == r && l > r)) {
318 return std::strong_ordering::greater;
319 }
320 return std::strong_ordering::equivalent;
321#else
322 return (lhs.m_iter <=> rhs.m_iter);
323#endif
324 }
325
327 DETRAY_HOST_DEVICE friend constexpr auto operator+(
328 const static_join_iterator &itr, const difference_type j)
329 -> static_join_iterator
330 requires std::random_access_iterator<iterator_t>
331 {
332 static_join_iterator<iterator_coll_t> tmp(itr);
333 // walk through join to catch the switch between intermediate ranges
334 if (difference_type i{j}; i >= 0) {
335 while (i--) {
336 ++tmp;
337 }
338 } else {
339 while (i++) {
340 --tmp;
341 }
342 }
343 return tmp;
344 }

347 DETRAY_HOST_DEVICE friend constexpr auto operator+(
348 const difference_type j, const static_join_iterator &itr)
349 -> static_join_iterator
350 requires std::random_access_iterator<iterator_t>
351 {
352 return itr + j;
353 }
354
356 DETRAY_HOST_DEVICE friend constexpr auto operator-(
357 const static_join_iterator &itr, const difference_type j)
358 -> static_join_iterator
359 requires std::random_access_iterator<iterator_t>
360 {
361 return itr + (-j);
362 }
363
365 DETRAY_HOST_DEVICE friend constexpr auto operator-(
366 const static_join_iterator &lhs, const static_join_iterator &rhs)
367 -> difference_type
368 requires std::random_access_iterator<iterator_t>
369 {
370 const static_join_iterator l{lhs};
371 const static_join_iterator r{rhs};
372
373 if (l.m_idx == r.m_idx) {
374 return l.m_iter - r.m_iter;
375 }
376 if (l.m_idx < r.m_idx) {
377 // Negative distance
378 difference_type diff{l.m_iter - (*l.m_ends)[l.m_idx]};
379 for (std::size_t i{l.m_idx + 1u}; i < r.m_idx; ++i) {
380 diff += (*l.m_begins)[i] - (*l.m_ends)[i];
381 }
382 diff += (*r.m_begins)[l.m_idx] - r.m_iter;
383 return diff;
384 } else {
385 // Positive distance
386 difference_type diff{l.m_iter - (*l.m_begins)[l.m_idx]};
387 for (std::size_t i{l.m_idx - 1u}; i > r.m_idx; --i) {
388 diff += (*l.m_ends)[i] - (*l.m_begins)[i];
389 }
390 diff += (*r.m_ends)[r.m_idx] - r.m_iter;
391 return diff;
392 }
393 }
394
396 const iterator_coll_t *m_begins{nullptr};
397 const iterator_coll_t *m_ends{nullptr};
399 iterator_t m_iter{};
401 std::size_t m_idx{0};
402};
