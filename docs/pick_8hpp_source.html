50 struct iterator {
51 using itr_value_t = std::iter_value_t<range_itr_t>;
52 using itr_ref_t = std::iter_reference_t<range_itr_t>;
53 using itr_ptr_t = typename std::iterator_traits<range_itr_t>::pointer;
54
55 using difference_type = std::iter_difference_t<range_itr_t>;
56 using value_type = std::pair<std::iter_value_t<sequence_itr_t>, itr_ref_t>;
57 using pointer = value_type *;
58 using reference = value_type;
59 using iterator_category =
60 typename std::iterator_traits<sequence_itr_t>::iterator_category;
61
62 static_assert(std::is_convertible_v<index_t, difference_type>,
63 "Given sequence cannot be "
64 "used to index elements of range.");

67 constexpr iterator() = default;

71 constexpr iterator(range_itr_t rng_itr, range_itr_t rng_begin,
72 sequence_itr_t sq_itr, sequence_itr_t sq_end)
73 : m_range_iter{rng_itr},
74 m_range_begin{rng_begin},
75 m_seq_iter{sq_itr},
76 m_seq_end{sq_end} {}

80 DETRAY_HOST_DEVICE constexpr auto operator++() -> iterator & {
81 ++m_seq_iter;
82 if (m_seq_iter != m_seq_end) {
83 m_range_iter =
84 m_range_begin + static_cast<difference_type>(*m_seq_iter);
85 }
86 return *this;
87 }

89 DETRAY_HOST_DEVICE constexpr auto operator++(int) -> iterator {

93 }

98 DETRAY_HOST_DEVICE constexpr auto operator--() -> iterator &
99 requires std::bidirectional_iterator<range_itr_t>
100 {
101 m_range_iter =
102 m_range_begin + static_cast<difference_type>(*(--m_seq_iter));
103 return *this;
104 }
105
106 DETRAY_HOST_DEVICE constexpr auto operator--(int) -> iterator
107 requires std::bidirectional_iterator<range_itr_t>
108 {
109 auto tmp(*this);
110 --(*this);
111 return tmp;
112 }
113
115 DETRAY_HOST_DEVICE auto operator*() const {
116 return value_type(*m_seq_iter, *m_range_iter);
117 }
118
121 DETRAY_HOST_DEVICE constexpr auto operator+=(const difference_type j)
122 -> iterator &
123 requires std::random_access_iterator<range_itr_t>
124 {
125 detray::ranges::advance(m_seq_iter, j);
126 m_range_begin += static_cast<difference_type>(*m_seq_iter);
127 return *this;
128 }
129
132 DETRAY_HOST_DEVICE constexpr auto operator-=(const difference_type j)
133 -> iterator &
134 requires std::random_access_iterator<range_itr_t> &&
135 std::random_access_iterator<sequence_itr_t>
136 {
137 return *this += -j;
138 }
139
141 DETRAY_HOST_DEVICE constexpr auto operator[](const difference_type i) const
142 requires std::random_access_iterator<range_itr_t> &&
143 std::random_access_iterator<sequence_itr_t>
144 {
145 const auto index{m_seq_iter[i]};
146 return value_type(index, m_range_begin[index]);
147 }

149 private:
151 DETRAY_HOST_DEVICE
152 friend constexpr auto operator==(const iterator &lhs, const iterator &rhs)
153 -> bool {
154 return (lhs.m_seq_iter == rhs.m_seq_iter);
155 }

158 DETRAY_HOST_DEVICE
159 friend constexpr auto operator<=>(const iterator &lhs, const iterator &rhs)
160 requires detray::ranges::random_access_iterator<sequence_itr_t>
161 {
162#if defined(__apple_build_version__)
163 const auto l{lhs.m_seq_iter};
164 const auto r{rhs.m_seq_iter};
165 if (l < r || (l == r && l < r)) {
166 return std::strong_ordering::less;
167 }
168 if (l > r || (l == r && l > r)) {
169 return std::strong_ordering::greater;
170 }
171 return std::strong_ordering::equivalent;
172#else
173 return (lhs.m_seq_iter <=> rhs.m_seq_iter);
174#endif
175 }

188 itr.m_range_begin, seq_iter, itr.m_seq_end};
189 }

195 DETRAY_HOST_DEVICE friend constexpr auto operator+(const difference_type j,
196 const iterator &itr)

200 return itr + j;
201 }

207 DETRAY_HOST_DEVICE friend constexpr auto operator-(const iterator &itr,
208 const difference_type j)

210 requires std::random_access_iterator<range_itr_t>
211 {
212 return itr + (-j);
213 }

216 DETRAY_HOST_DEVICE friend constexpr auto operator-(const iterator &lhs,
217 const iterator &rhs)
218 -> difference_type
219 requires std::random_access_iterator<range_itr_t> &&
220 std::random_access_iterator<sequence_itr_t>
221 {
222 return lhs.m_seq_iter - rhs.m_seq_iter;
223 }

363 template <detray::ranges::range range_t>
364 requires(std::same_as<sequence_t, std::remove_cvref_t<sequence_t>> &&
365 !std::is_pointer_v<sequence_t>)
366 DETRAY_HOST_DEVICE constexpr auto operator()(range_t &&rng) && {
367 using itr_t = detray::ranges::iterator_t<std::remove_reference_t<range_t>>;
368 return detray::ranges::pick_view<itr_t, sequence_itr_t>(
369 std::forward<range_t>(rng), std::move(m_seq));
370 }
371
375 template <detray::ranges::range range_t>
376 requires(!std::same_as<sequence_t, std::remove_cvref_t<sequence_t>> ||
377 std::is_pointer_v<sequence_t>)
378 DETRAY_HOST_DEVICE constexpr auto operator()(range_t &&rng) {
379 using itr_t = detray::ranges::iterator_t<std::remove_reference_t<range_t>>;
380 return detray::ranges::pick_view<itr_t, sequence_itr_t>(
381 std::forward<range_t>(rng), m_seq);
382 }

387 DETRAY_HOST_DEVICE
388 pick &operator=(const pick &other) {
389 base_type::operator=(other);
390 m_seq = other.m_seq;
391 return *this;
392 }
393
394 private:
395 sequence_t m_seq{};
396};
