108class NavigationPolicyStateManager {
109 public:
115 template <typename T, typename... Args>
116 T& pushState(Args&&... args) {
117 static_assert(
118 sizeof(T) <= kNavigationPolicyStateSbo,
119 "Navigation policy state does not fit the small buffer and would be "
120 "heap-allocated on the navigation hot path. Increase "
121 "kNavigationPolicyStateSbo to accommodate it.");
122 // Reserve a typical stack depth on first use only. This keeps the stack
123 // from reallocating during ordinary navigation, while not allocating at all
124 // for navigators that never push a state (e.g. the Gen1 navigator, which
125 // carries this manager but does not use it). Once cleared by reset() the
126 // capacity is retained, so this stays a no-op afterwards.
127 if (m_stateStack.empty()) {
128 m_stateStack.reserve(16);
129 }
130 NavigationPolicyStateStorage& state = m_stateStack.emplace_back();
131 return state.emplace<T>(std::forward<Args>(args)...);
132 }
133
134 friend class Navigator;
135
138 NavigationPolicyState currentState() {
139 if (m_stateStack.empty()) {
140 return {}; // Empty state as sentinel
141 }
142 return NavigationPolicyState{*this, m_stateStack.size() - 1};
143 }
144
147 void popState() {
148 if (m_stateStack.empty()) {
149 throw std::runtime_error(
150 "NavigationPolicyStateManager: Attempt to pop from empty stack");
151 }
152 m_stateStack.pop_back();
153 }
154
156 void reset() { m_stateStack.clear(); }
157
158 private:
159 friend class NavigationPolicyState;
160
161 // We might want to extend this to a stack
162 std::vector<NavigationPolicyStateStorage> m_stateStack;
163};

182namespace detail {
183template <typename T>
184concept HasOldInitializeCandidates = requires {
185 requires requires(T policy, const GeometryContext& gctx,
186 const NavigationArguments& args,
187 AppendOnlyNavigationStream& stream, const Logger& logger) {
188 policy.initializeCandidates(gctx, args, stream, logger);
189 };
190};
191
192template <typename T>
193concept HasNewInitializeCandidates = requires {
194 requires requires(T policy, const GeometryContext& gctx,
195 const NavigationArguments& args,
196 NavigationPolicyState& state,
197 AppendOnlyNavigationStream& stream, const Logger& logger) {
198 policy.initializeCandidates(gctx, args, state, stream, logger);
199 };
200};
201
202template <detail::HasOldInitializeCandidates T>
203void oldToNewSignatureAdapter(const void* instance, const GeometryContext& gctx,
204 const NavigationArguments& args,
205 NavigationPolicyState& /*state*/,
206 AppendOnlyNavigationStream& stream,
207 const Logger& logger) {
208 const auto* policy = static_cast<const T*>(instance);
209 policy->initializeCandidates(gctx, args, stream, logger);
210}
211} // namespace detail

216template <typename T>
217concept NavigationPolicyConcept = requires {
218 requires std::is_base_of_v<INavigationPolicy, T>;
219
220 // Require either of the signatures to allow backwards compatibility
221 requires(detail::HasOldInitializeCandidates<T> ||
222 detail::HasNewInitializeCandidates<T>);
223};

273 struct EmptyState {};

280 virtual void createState([[maybe_unused]] const GeometryContext& gctx,
281 [[maybe_unused]] const NavigationArguments& args,
282 NavigationPolicyStateManager& stateManager,
283 const Logger& logger) const {
284 ACTS_VERBOSE(

288
307 void initializeStatelessCache(const GeometryContext& gctx,
308 const Logger& logger = getDummyLogger()) {
309 m_stateless = false;
310 // Create the policy state once into a scratch manager and observe whether
311 // the result is the default sentinel. Per the contract above, the
312 // defaultness must not depend on the arguments, so probing with arbitrary
313 // ones is valid.
314 NavigationPolicyStateManager scratchManager;
315 NavigationArguments args{.position = Vector3::Zero(),
316 .direction = Vector3::UnitZ()};
317 createState(gctx, args, scratchManager, logger);
318 m_stateless = scratchManager.currentState().isDefault();
319 }
320
324 bool isStateless() const { return m_stateless; }
325
329 virtual void popState(NavigationPolicyStateManager& stateManager,
330 const Logger& logger) const {
331 // By default, we didn't push anything, so we don't need to poop anything
332 ACTS_VERBOSE("Default navigation policy pop state. (pops empty state)");
333 stateManager.popState(); // This will be correct regardless of if a derived
334 // class pushed a concrete state type that's
335 // different from the default EmptyState.
336 }
337
338 protected:
344 template <NavigationPolicyConcept T>
345 void connectDefault(NavigationDelegate& delegate) const {
346 // This cannot be a concept because we use it in CRTP below
347 const auto* self = static_cast<const T*>(this);
348
349 if constexpr (detail::HasNewInitializeCandidates<T>) {
350 delegate.template connect<&T::initializeCandidates>(self);
351 } else {
352 // @TODO: Remove navigation policy signature compatibility eventually
353 delegate.connect(&detail::oldToNewSignatureAdapter<T>, self);
354 }
355 }
356
357 private:
361 bool m_stateless = false;
362};

364inline bool NavigationPolicyState::isDefault() const {
365 if (empty()) {
366 return true; // no state at all -> nothing to validate
367 }
368 const NavigationPolicyStateStorage& stored = payload();
369 return !stored || stored.is<INavigationPolicy::EmptyState>();
370}
