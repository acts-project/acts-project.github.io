46template <concepts::algebra algebra_t, concepts::point point_t>
47struct intersection_point<algebra_t, point_t, !intersection::contains_pos> {

74template <concepts::algebra algebra_t, concepts::point point_t>
75struct intersection_point<algebra_t, point_t, intersection::contains_pos>
76 : public intersection_point<algebra_t, point_t,
77 !intersection::contains_pos> {
78 private:
79 using base_type =
80 intersection_point<algebra_t, point_t, !intersection::contains_pos>;
81
82 public:
84 using scalar_type = dscalar<algebra_t>;
86 using point_type = point_t;

89 constexpr intersection_point() = default;
90
93 DETRAY_HOST_DEVICE
94 explicit constexpr intersection_point(const base_type ip) : base_type{ip} {}
95
99 DETRAY_HOST_DEVICE
100 constexpr intersection_point(const scalar_type p, const point_type &pnt)
101 : base_type{p}, point{pnt} {}
102
104 static consteval bool contains_pos() { return intersection::contains_pos; }
105
107 point_type point{init_point()};
108
110 DETRAY_HOST
111 friend std::ostream &operator<<(std::ostream &out_stream,
112 const intersection_point &ip) {
113 using base_t =
114 intersection_point<algebra_t, point_t, !intersection::contains_pos>;
115
116 out_stream << static_cast<base_t>(ip);
117
118 out_stream << ", point [" << ip.point[0] << ", " << ip.point[1];
119
120 if constexpr (std::same_as<point_t, dpoint3D<algebra_t>>) {
121 out_stream << ", " << ip.point[2] << "]";
122 } else {
123 out_stream << "]";
124 }
125
126 return out_stream;
127 }
128
129 private:
131 DETRAY_HOST_DEVICE
132 constexpr point_t init_point() const {
133 constexpr auto inv{detail::invalid_value<dvalue<algebra_t>>()};
134 if constexpr (std::same_as<point_t, dpoint2D<algebra_t>>) {
135 return {inv, inv};
136 } else if constexpr (std::same_as<point_t, dpoint3D<algebra_t>>) {
137 return {inv, inv, inv};
138 } else {
139 assert(false);
140 return {};
141 }
142 }

425 DETRAY_HOST
426 friend std::ostream &operator<<(std::ostream &out_stream,
427 const intersection2D &is) {
428 out_stream << is.m_ip << ", "
429 << ", surface: " << is.surface().identifier()
430 << ", type: " << static_cast<int>(is.surface().mask().id())
431 << ", links to vol:" << is.volume_link() << ")";
432
433 if (is.is_inside()) {
434 out_stream << ", status: inside";
435 } else if (is.is_edge()) {
436 out_stream << ", status: edge";
437 } else {
438 out_stream << ", status: outside";
439 }
440 if constexpr (std::is_scalar_v<bool_type>) {
441 out_stream << (is.is_along() ? ", direction: along"
442 : ", direction: opposite");
443 } else {
444 out_stream << ", status: " << is.status();
445 out_stream << ", direction: " << is.is_along();
446 }
447
448 return out_stream;
449 }
