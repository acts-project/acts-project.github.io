57 template <typename propagator_state_t, typename stepper_t,
58 typename navigator_t>
59 Result<void> act(propagator_state_t& state, const stepper_t& stepper,
60 const navigator_t& navigator, result_type& result,
61 const Logger& logger) const {
62 if (state.stage == PropagatorStage::postPropagation) {
63 return Result<void>::success();
64 }
65
66 // Do nothing if nothing is what is requested.
67 if (!(multipleScattering || energyLoss || recordInteractions)) {
68 return Result<void>::success();
69 }
70
71 // Handle surface material
72
73 // Note that start and target surface conditions are handled in the
74 // interaction code
75 const Surface* surface = navigator.currentSurface(state.navigation);
76
77 // We only have material interactions if there is potential material
78 if (surface != nullptr && surface->surfaceMaterial()) {
79 ACTS_VERBOSE("MaterialInteractor | " << "Found material on surface "
80 << surface->geometryId());
81
82 const Result<MaterialSlab> slabResult = detail::evaluateMaterialSlab(
83 state, stepper, *surface,
84 detail::determineMaterialUpdateMode(state, navigator,
85 MaterialUpdateMode::FullUpdate));
86 if (!slabResult.ok()) {
87 ACTS_DEBUG("MaterialInteractor | "
88 << "Failed to evaluate material slab: "
89 << slabResult.error());
90 return Result<void>::failure(slabResult.error());
91 }
92 const MaterialSlab& slab = *slabResult;
93
94 // Determine the effective traversed material and its properties
95 // Material exists but it's not real, i.e. vacuum; there is nothing to do
96 if (!slab.isVacuum()) {
97 // To integrate process noise, we need to transport
98 // the covariance to the current position in space
99 if (state.stepping.covTransport) {
100 stepper.transportCovarianceToCurvilinear(state.stepping);
101 }
102
103 const double initialMomentum = stepper.absoluteMomentum(state.stepping);
104
105 // Apply the material interactions
106 const detail::PointwiseMaterialEffects effects =
107 detail::performMaterialInteraction(state, stepper, slab,
108 noiseUpdateMode,
109 multipleScattering, energyLoss);
110
111 if (energyLoss) {
112 using namespace UnitLiterals;
113
114 const ParticleHypothesis& particleHypothesis =
115 stepper.particleHypothesis(state.stepping);
116 const PdgParticle absPdg = particleHypothesis.absolutePdg();
117 const double mass = particleHypothesis.mass();
118 const double momentum = stepper.absoluteMomentum(state.stepping);
119
120 ACTS_VERBOSE("MaterialInteractor | "
121 << slab << " absPdg=" << absPdg
122 << " mass=" << mass / 1_MeV << "MeV"
123 << " momentum=" << momentum / 1_GeV << "GeV"
124 << " energyloss=" << effects.eLoss / 1_MeV << "MeV");
125 }
126
127 // Record the result
128 recordResult(state, stepper, navigator, slab, initialMomentum, effects,
129 result);
130 }
131 }
132
133 // Handle volume material
134
135 // In case of Volume material update the result of the previous step
136 if (!result.materialInteractions.empty() &&
137 !result.materialInteractions.back().volume.empty() &&
138 result.materialInteractions.back().updatedVolumeStep == false) {
139 updateResult(state, stepper, result);
140 }
141
142 const TrackingVolume* volume = navigator.currentVolume(state.navigation);
143
144 // We only have material interactions if there is potential material
145 if (volume && volume->hasMaterial()) {
146 ACTS_VERBOSE("MaterialInteractor | " << "Found material in volume "
147 << volume->geometryId());
148
149 // Prepare relevant input particle properties
150 detail::VolumeMaterialInteraction interaction(volume, state, stepper);
151 // Determine the effective traversed material and its properties
152 // Material exists but it's not real, i.e. vacuum; there is nothing to do
153 if (interaction.evaluateMaterialSlab(state, navigator)) {
154 // Record the result
155 recordResult(interaction, result);
156 }
157 }
158
159 return Result<void>::success();
160 }

162 private:
176 template <typename propagator_state_t, typename stepper_t,
177 typename navigator_t>
178 void recordResult(const propagator_state_t& state, const stepper_t& stepper,
179 const navigator_t& navigator, const MaterialSlab& slab,
180 double initialMomentum,
181 const detail::PointwiseMaterialEffects& effects,
182 result_type& result) const {
183 result.materialInX0 += slab.thicknessInX0();
184 result.materialInL0 += slab.thicknessInL0();
185
186 // Record the interaction if requested
187 if (!recordInteractions) {
188 return;
189 }
190
191 const Vector3 position = stepper.position(state.stepping);
192 const double time = stepper.time(state.stepping);
193 const Vector3 direction = stepper.direction(state.stepping);
194 const double finalMomentum = stepper.absoluteMomentum(state.stepping);
195
196 const Surface* surface = navigator.currentSurface(state.navigation);
197 const double pathCorrection =
198 surface->pathCorrection(state.options.geoContext, position, direction);
199
200 MaterialInteraction mi;
201 mi.position = position;
202 mi.time = time;
203 mi.direction = direction;
204 mi.deltaP = initialMomentum - finalMomentum;
205 mi.sigmaPhi2 = effects.variancePhi;
206 mi.sigmaTheta2 = effects.varianceTheta;
207 mi.sigmaQoP2 = effects.varianceQoverP;
208 mi.surface = surface;
209 mi.volume = InteractionVolume();
210 mi.pathCorrection = pathCorrection;
211 mi.materialSlab = slab;
212 result.materialInteractions.push_back(std::move(mi));
213 }
214
219 void recordResult(const detail::VolumeMaterialInteraction& interaction,
220 result_type& result) const {
221 // Record the interaction if requested
222 if (!recordInteractions) {
223 return;
224 }
225
226 MaterialInteraction mi;
227 mi.position = interaction.pos;
228 mi.time = interaction.time;
229 mi.direction = interaction.dir;
230 mi.surface = nullptr;
231 mi.volume = interaction.volume;
232 mi.pathCorrection = interaction.pathCorrection;
233 mi.materialSlab = interaction.slab;
234 result.materialInteractions.push_back(std::move(mi));
235 }

242 template <typename propagator_state_t, typename stepper_t>
243 void updateResult(propagator_state_t& state, const stepper_t& stepper,
244 result_type& result) const {
245 // Update the previous interaction if requested
246 if (!recordInteractions) {
247 return;
248 }
249
250 Vector3 shift = stepper.position(state.stepping) -
251 result.materialInteractions.back().position;
252 double momentum = stepper.direction(state.stepping).norm();
253 result.materialInteractions.back().deltaP =
254 momentum - result.materialInteractions.back().direction.norm();
255 result.materialInteractions.back().materialSlab.scaleThickness(
256 shift.norm());
257 result.materialInteractions.back().updatedVolumeStep = true;
258 result.materialInX0 +=
259 result.materialInteractions.back().materialSlab.thicknessInX0();
260 result.materialInL0 +=
261 result.materialInteractions.back().materialSlab.thicknessInL0();
262 }
263};
