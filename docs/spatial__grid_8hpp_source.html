114 DETRAY_HOST_DEVICE query_type
115 local_point_of_intersection(const detray::detail::ray<algebra_t> &tangential,
116 const dtransform3D<algebra_t> &trf) const {
117 // Intersect the (virtual) reference surface of the grid to find
118 // the correct bin
119 using intersector_t =
120 ray_intersector_impl<frame_t, algebra_t, intersection::contains_pos>;
121
122 constexpr intersector_t intersector{};
123 typename intersector_t::result_type result{};
124 constexpr auto overstep_tol{
125 -std::numeric_limits<dscalar<algebra_t>>::max()};
126
127 // The cylinder intersector requires the radius from the mask
128 if constexpr (concepts::cylindrical<frame_t>) {
129 result = intersector.point_of_intersection(tangential, trf, m_mask,
130 overstep_tol);
131 } else {
132 result = intersector.point_of_intersection(tangential, trf, overstep_tol);
133 }
134
135 // Retrieve the closest intersection point and check its valid
136 typename intersector_t::point_type intr_point;
137 bool is_inside{true};
138 if constexpr (intersector_t::n_solutions == 1) {
139 // Intersection with the grid was found
140 if (detray::detail::all_of(result.is_valid())) [[likely]] {
141 intr_point = result.point;
142 } else {
143 is_inside = false;
144 }
145 } else {
146 // Closest intersection with the grid was found
147 if (detray::detail::all_of(result[0].is_valid())) [[likely]] {
148 intr_point = result[0].point;
149 } else {
150 is_inside = false;
151 }
152 }
153 // Use ray origin as query position
154 if (!is_inside) [[unlikely]] {
155 // In case the intersection results come in local coordinates
156 if constexpr (std::same_as<decltype(intr_point), query_type>) {
157 intr_point =
158 frame_t::global_to_local(trf, tangential.pos(), tangential.dir());
159 } else {
160 intr_point = tangential.pos();
161 }
162 }
163
164 // Most intersectors return global positions -> project to grid axes
165 if constexpr (std::same_as<frame_t, concentric_cylindrical2D<algebra_t>>) {
166 // The concentric cylinder intersector only returns the z-pos
167 return {vector::phi(tangential.pos(result.path)), intr_point[1]};
168 } else if constexpr (std::same_as<decltype(intr_point), query_type>) {
169 return intr_point;
170 } else {
171 // Project the intersection point into the local grid frame
172 return this->project(trf, intr_point, tangential.dir());
173 }
174 }

185 template <concepts::arithmetic window_size_t>
186 DETRAY_HOST_DEVICE auto search(
187 const query_type &p,
188 const search_window<window_size_t, 2> &win_size) const {
189 DETRAY_DEBUG_HOST(" -> lookup pos: " << p);
190 DETRAY_DEBUG_HOST(" -> search window: [" << win_size[0] << ", "
191 << win_size[1] << "]");
192
193 // Return iterable over bins in the search window
194 auto bin_ranges = this->axes().bin_ranges(p, win_size);
195 auto search_area = axis::detail::bin_view(*this, bin_ranges);
196
197 // Join the respective bins to a single iteration
198 return detray::views::join(std::move(search_area));
199 }

208 template <concepts::detector detector_t, typename track_t,
209 concepts::arithmetic window_size_t>
210 DETRAY_HOST_DEVICE auto search(
211 const detector_t &det, const typename detector_t::volume_type &volume,
212 const track_t &track, const search_window<window_size_t, 2> &win_size,
213 const typename detector_t::geometry_context &ctx) const {
214 // Placement of the grid (same as volume)
215 const auto &trf = det.transform_store().at(volume.transform(), ctx);
216
217 query_type loc_pos{};
218 // For 2-dimensional grids, project the track position along the track
219 // direction to find the optimal grid bin
220 if constexpr (base_grid::dim == 2) {
221 if constexpr (concepts::cylindrical<frame_t>) {
222 DETRAY_DEBUG_HOST("2D spatial grid: " << m_mask);
223 } else {
224 DETRAY_DEBUG_HOST("2D spatial grid: " << DETRAY_TYPENAME(frame_t));
225 }
226
227 // Tangential to the current track parameters
228 const detray::detail::ray<algebra_t> tangential{track};
229
230 // Intersect the (virtual) reference surface of the grid to find
231 // the correct bin
232 loc_pos = local_point_of_intersection(tangential, trf);
233 } else {
234 loc_pos = this->project(trf, track.pos(), track.dir());
235 }
236
237 // Grid lookup
238 return search(loc_pos, win_size);
239 }
240
241 private:
243 template <typename... Args>
244 DETRAY_HOST_DEVICE mask_type get_mask_from_axes(Args &&...mask_values) {
245 constexpr auto inv_vol_link{detray::detail::invalid_value<std::uint8_t>()};
246
248 if constexpr (concepts::cylindrical<frame_t>) {
249 static_assert(sizeof...(Args) == 1,
250 "Spatial cylinder grid: Only the cylinder radius "
251 "needs to be provided externally");
252
253 constexpr auto inf{std::numeric_limits<dscalar<algebra_t>>::max()};
254 return mask_type{inv_vol_link, mask_values..., -inf, inf};
255 } else {
257 return {};
258 }
259 }
260
262 mask_type m_mask{};
263};
