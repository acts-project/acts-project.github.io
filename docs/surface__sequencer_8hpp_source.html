80 template <typename propagator_state_t>
81 requires std::same_as<
82 sf_descriptor_t,
83 typename propagator_state_t::detector_type::surface_type>
84 DETRAY_HOST_DEVICE void operator()(state& actor_state,
85 propagator_state_t& propagation) const {
86 const auto& navigation = propagation.navigation();
87
88 DETRAY_VERBOSE_HOST_DEVICE(
89 "Actor: Checking for next surface in sequence...");
90
91 if (actor_state.collect_only_material_surfaces() &&
92 navigation.is_on_sensitive()) {
93 return;
94 } else if (!(navigation.is_on_sensitive() ||
95 navigation.encountered_sf_material())) {
96 return;
97 }
98
99 if (actor_state.sequence().size() == actor_state.sequence().capacity()) {
100 if (actor_state.sequence().capacity() == 0) {
101 DETRAY_VERBOSE_HOST_DEVICE(
102 "Actor: No capacity to store surface candidate - skipping");
103 } else {
104 DETRAY_VERBOSE_HOST_DEVICE("Actor: Sequence overflow!");
105 actor_state.set_overflow();
106 propagation.navigation().exit();
107 propagation.heartbeat(false);
108 }
109 return;
110 }
111
112 const auto& sf_desc = std::as_const(navigation).current().surface();
113 assert(!sf_desc.identifier().is_invalid());
114
115 actor_state.sequence().push_back(sf_desc);
116 DETRAY_VERBOSE_HOST("Actor: Added surface: " << sf_desc);
117 }
118};
