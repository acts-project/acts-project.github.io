104template <>
105void configure_options<detray::navigation::config>(
106 const boost::program_options::variables_map &vm,
107 detray::navigation::config &cfg) {
108 // Local navigation search window
109 if (vm.count("search_window") != 0u) {
110 const auto window = vm["search_window"].as<std::vector<dindex>>();
111 if (window.size() == 2u) {
112 cfg.search_window = {window[0], window[1]};
113 } else {
114 throw std::invalid_argument(
115 "Incorrect surface grid search window. Please provide two "
116 "integer distances.");
117 }
118 }
119 // Overstepping tolerance
120 if (!vm["min_mask_tolerance"].defaulted()) {
121 const float mask_tol{vm["min_mask_tolerance"].as<float>()};
122 assert(mask_tol >= 0.f);
123
124 cfg.intersection.min_mask_tolerance = mask_tol * unit<float>::mm;
125 }
126 if (!vm["max_mask_tolerance"].defaulted()) {
127 const float mask_tol{vm["max_mask_tolerance"].as<float>()};
128 assert(mask_tol >= 0.f);
129
130 cfg.intersection.max_mask_tolerance = mask_tol * unit<float>::mm;
131 }
132 if (!vm["mask_tolerance_scalor"].defaulted()) {
133 const float mask_tol_scalor{vm["mask_tolerance_scalor"].as<float>()};
134 assert(mask_tol_scalor >= 0.f);
135
136 cfg.intersection.mask_tolerance_scalor = mask_tol_scalor;
137 }
138 if (!vm["overstep_tolerance"].defaulted()) {
139 const float overstep_tol{vm["overstep_tolerance"].as<float>()};
140 assert(overstep_tol <= 0.f);
141
142 cfg.intersection.overstep_tolerance = overstep_tol * unit<float>::um;
143 }
144 if (!vm["path_tolerance"].defaulted()) {
145 const float path_tol{vm["path_tolerance"].as<float>()};
146 assert(path_tol >= 0.f);
147
148 cfg.intersection.path_tolerance = path_tol * unit<float>::um;
149 }
150 cfg.estimate_scattering_noise = false;
151 if (vm.count("estimate_scattering_noise") != 0u) {
152 cfg.estimate_scattering_noise = true;
153
154 if (!vm["n_scattering_stddev"].defaulted()) {
155 const int n_stddev{vm["n_scattering_stddev"].as<int>()};
156 assert(n_stddev >= 0);
157
158 cfg.n_scattering_stddev = n_stddev;
159 }
160
161 if (!vm["accumulated_error"].defaulted()) {
162 const float err{vm["accumulated_error"].as<float>()};
163 assert(err >= 0.f);
164
165 cfg.accumulated_error = err;
166 }
167 } else {
168 if (!vm["n_scattering_stddev"].defaulted()) {
169 throw std::invalid_argument(
170 "Option 'n_scattering_stddev' cannot not be configured unless "
171 "'estimate_scattering_noise' is activated");
172 }
173 if (!vm["accumulated_error"].defaulted()) {
174 throw std::invalid_argument(
175 "Option 'accumulated_error' cannot not be configured unless "
176 "'estimate_scattering_noise' is activated");
177 }
178 }
179}

182template <>
183void configure_options<detray::stepping::config>(
184 const boost::program_options::variables_map &vm,
185 detray::stepping::config &cfg) {
186 // Overstepping tolerance
187 if (!vm["minimum_stepsize"].defaulted()) {
188 const float min_step{vm["minimum_stepsize"].as<float>()};
189 assert(min_step >= 0.f);
190
191 cfg.min_stepsize = min_step * unit<float>::mm;
192 }
193 if (!vm["step_contraint"].defaulted()) {
194 const float constraint{vm["step_contraint"].as<float>()};
195 assert(constraint >= 0.f);
196
197 cfg.step_constraint = constraint * unit<float>::mm;
198 }
199 if (!vm["rk-tolerance"].defaulted()) {
200 const float err_tol{vm["rk-tolerance"].as<float>()};
201 assert(err_tol >= 0.f);
202
203 cfg.rk_error_tol = err_tol * unit<float>::mm;
204 }
205 if (!vm["path_limit"].defaulted()) {
206 const float path_limit{vm["path_limit"].as<float>()};
207 assert(path_limit > 0.f);
208
209 cfg.path_limit = path_limit * unit<float>::m;
210 }
211 cfg.do_covariance_transport = false;
212 if (vm.count("covariance_transport") != 0u) {
213 cfg.do_covariance_transport = true;
214 }
215 if (vm.count("mean_eloss") != 0u) {
216 cfg.use_mean_loss = true;
217 }
218 if (vm.count("eloss_gradient") != 0u) {
219 cfg.use_eloss_gradient = true;
220 }
221 if (vm.count("bfield_gradient") != 0u) {
222 cfg.use_field_gradient = true;
223 }
224}
