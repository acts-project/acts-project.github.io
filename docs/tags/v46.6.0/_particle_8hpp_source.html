195 Particle &loseEnergy(double delta, SimulationOutcome stoppedOutcome =
196 SimulationOutcome::Alive) {
197 const double newEnergy = energy() - delta;
198 if (newEnergy <= m_mass) {
199 if (stoppedOutcome == SimulationOutcome::Alive) {
200 throw std::invalid_argument(
201 "Energy loss cannot exceed the current energy of the particle if "
202 "the particle is to remain alive.");
203 }
204 killParticle(stoppedOutcome);
205 } else {
206 m_absMomentum = Acts::fastCathetus(newEnergy, m_mass);
207 }
208 return *this;
209 }

262 mom4[Acts::eMom0] = m_absMomentum * m_direction[Acts::ePos0];
263 mom4[Acts::eMom1] = m_absMomentum * m_direction[Acts::ePos1];
264 mom4[Acts::eMom2] = m_absMomentum * m_direction[Acts::ePos2];
265 mom4[Acts::eEnergy] = energy();

354 Acts::Result<Acts::BoundTrackParameters> boundParameters(
355 const Acts::GeometryContext &gctx) const {
356 if (!hasReferenceSurface()) {
357 return Acts::Result<Acts::BoundTrackParameters>::failure(
358 std::error_code());
359 }
360 Acts::Result<Acts::Vector2> localResult =
361 m_referenceSurface->globalToLocal(gctx, position(), direction());
362 if (!localResult.ok()) {
363 return localResult.error();
364 }
365 Acts::BoundVector params;
366 params << localResult.value(), phi(), theta(), qOverP(), time();
367 return Acts::BoundTrackParameters(referenceSurface()->getSharedPtr(),
368 params, std::nullopt, hypothesis());
369 }
