101 template <typename metadata_t>
102 DetrayGeometry<metadata_t> convert(
103 vecmem::memory_resource& mr, const Acts::GeometryContext& gctx,
104 const std::shared_ptr<const Acts::TrackingGeometry>& trackingGeometry,
105 const std::string& detectorName = "") const {
106 using detector_t = detray::detector<metadata_t>;
107
108 if (trackingGeometry == nullptr) {
109 throw std::invalid_argument(
110 "DetrayGeometryConverter: trackingGeometry must not be null");
111 }
112
113 // ── Convert TrackingGeometry → detray payloads ────────────────────────
114 auto payloads = m_cfg.payloadConverter->convertTrackingGeometry(
115 gctx, *trackingGeometry);
116
117 // ── Build detray detector from payloads ───────────────────────────────
118 detray::detector_builder<metadata_t> detectorBuilder{};
119
120 detray::io::geometry_reader::from_payload<detector_t>(detectorBuilder,
121 *payloads.detector);
122
123 if (m_cfg.convertMaterial) {
124 if (!payloads.homogeneousMaterial) {
125 ACTS_DEBUG("No homogeneous material payload found, skipping");
126 } else {
127 const auto& materialVolumes = payloads.homogeneousMaterial->volumes;
128 ACTS_DEBUG("Found homogeneous material payload with "
129 << materialVolumes.size() << " volumes");
130
131 auto hasMaterialType =
132 [&materialVolumes](detray::io::material_id type) {
133 return std::ranges::any_of(
134 materialVolumes, [type](const auto& volume) {
135 return std::ranges::any_of(
136 volume.surface_mat,
137 [type](const auto& sm) { return sm.type == type; });
138 });
139 };
140
141 if (hasMaterialType(detray::io::material_id::slab) &&
142 !detray::concepts::has_material_slabs<detector_t>) {
143 throw std::invalid_argument(
144 "DetrayGeometryConverter: the tracking geometry contains "
145 "homogeneous material slabs, but the target Detray metadata "
146 "type does not support material slabs");
147 }
148 if (hasMaterialType(detray::io::material_id::rod) &&
149 !detray::concepts::has_material_rods<detector_t>) {
150 throw std::invalid_argument(
151 "DetrayGeometryConverter: the tracking geometry contains "
152 "homogeneous material rods, but the target Detray metadata "
153 "type does not support material rods");
154 }
155
156 if constexpr (detray::concepts::has_material_slabs<detector_t> ||
157 detray::concepts::has_material_rods<detector_t>) {
158 detray::io::homogeneous_material_reader::from_payload<detector_t>(
159 detectorBuilder, *payloads.homogeneousMaterial);
160 }
161 }
162
163 detray::io::material_map_reader<std::integral_constant<std::size_t, 2>>::
164 from_payload<detector_t>(detectorBuilder,
165 std::move(*payloads.materialGrids));
166 }
167
168 if (m_cfg.convertSurfaceGrids) {
169 detray::io::surface_grid_reader<typename detector_t::surface_type,
170 std::integral_constant<std::size_t, 0>,
171 std::integral_constant<std::size_t, 2>>::
172 template from_payload<detector_t>(detectorBuilder,
173 *payloads.surfaceGrids);
174 }
175
176 if (!detectorName.empty()) {
177 detectorBuilder.set_name(detectorName);
178 } else if (payloads.names.contains(0)) {
179 detectorBuilder.set_name(payloads.names.at(0));
180 }
181
182 DetrayGeometry<metadata_t> result{};
183 result.detector =
184 std::make_shared<detector_t>(detectorBuilder.build(mr, result.names));
185
186 return result;
187 }
188
189 private:
190 Config m_cfg;
191
192 const Acts::Logger& logger() const { return *m_logger; }
193 std::unique_ptr<const Acts::Logger> m_logger;
194};
