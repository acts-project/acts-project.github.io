135 template <typename callable_t>
136 decltype(auto) visit(const callable_t& callable) const {
137 auto switchOnType =
138 [this, &callable]<AxisBoundaryType bdt>(AxisBoundaryTypeTag<bdt>) {
139 switch (getType()) {
140 using enum AxisType;
141 case Equidistant:
142 return callable(
143 dynamic_cast<const Axis<AxisType::Equidistant, bdt>&>(*this));
144 case Variable:
145 return callable(
146 dynamic_cast<const Axis<AxisType::Variable, bdt>&>(*this));
147 default:
148 throw std::logic_error("Unknown axis type");
149 }
150 };
151
152 switch (getBoundaryType()) {
153 using enum AxisBoundaryType;
154 case Open:
155 return switchOnType(AxisOpen);
156 case Bound:
157 return switchOnType(AxisBound);
158 case Closed:
159 return switchOnType(AxisClosed);
160 default:
161 throw std::logic_error("Unknown axis type");
162 }
163 }
