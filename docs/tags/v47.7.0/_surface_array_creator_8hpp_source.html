243 static bool isSurfaceEquivalent(const GeometryContext& gctx,
244 AxisDirection aDir, const Surface* a,
245 const Surface* b) {
246 using namespace UnitLiterals;
247 using VectorHelpers::perp;
248
249 if (aDir == AxisDirection::AxisPhi) {
250 // Take the two binning positions
251 Vector3 pos1 = a->referencePosition(gctx, AxisDirection::AxisR);
252 Vector3 pos2 = b->referencePosition(gctx, AxisDirection::AxisR);
253
254 // Project them on the (x, y) plane, where Phi angles are calculated
255 auto proj1 = pos1.head<2>(), proj2 = pos2.head<2>();
256
257 // Basic dot and cross products identities give us the cosine and sine
258 // of these angles, time the squared vector norm
259 auto cos_dPhi_n2 = proj1.dot(proj2);
260 auto sin_dPhi_n2 = proj1.x() * proj2.y() - proj2.x() * proj1.y();
261
262 // ...so by injecting them into atan2, we get the angle between them
263 auto dPhi = std::atan2(sin_dPhi_n2, cos_dPhi_n2);
264 return std::abs(dPhi) < std::numbers::pi / 180.;
265 }
266
267 if (aDir == AxisDirection::AxisZ) {
268 return (std::abs(a->referencePosition(gctx, AxisDirection::AxisR).z() -
269 b->referencePosition(gctx, AxisDirection::AxisR).z()) <
270 1_um);
271 }
272
273 if (aDir == AxisDirection::AxisR) {
274 return (std::abs(perp(a->referencePosition(gctx, AxisDirection::AxisR)) -
275 perp(b->referencePosition(gctx, AxisDirection::AxisR))) <
276 1_um);
277 }
278
279 return false;
280 }
