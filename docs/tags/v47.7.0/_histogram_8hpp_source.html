64class Histogram {
65 public:
71 Histogram(std::string name, std::string title,
72 const std::array<AxisVariant, Dim>& axes)
73 : m_name(std::move(name)),
74 m_title(std::move(title)),

76 std::vector<AxisVariant>(axes.begin(), axes.end()))) {}

99 void fill(const std::array<double, Dim>& values) {
100 std::apply([this](auto... v) { m_hist(v...); }, std::tuple_cat(values));
101 }

111 void setBinContent(const std::array<int, Dim>& indices, double content) {
112 std::apply(
113 [&](auto... i) {
114 m_hist.at(i...) =
115 boost::histogram::accumulators::weighted_sum<double>(content);
116 },
117 indices);
118 }
119
127 void setBin(const std::array<int, Dim>& indices, double content,
128 double error) {
129 std::apply(
130 [&](auto... i) {
131 m_hist.at(i...) =
132 boost::histogram::accumulators::weighted_sum<double>(
133 content, error * error);
134 },
135 indices);
136 }
137
143 double binContent(const std::array<int, Dim>& indices) const {
144 return std::apply([&](auto... i) { return m_hist.at(i...).value(); },
145 indices);
146 }
147
153 double binError(const std::array<int, Dim>& indices) const {
154 return std::apply(
155 [&](auto... i) { return std::sqrt(m_hist.at(i...).variance()); },
156 indices);
157 }
158
161 const std::string& name() const { return m_name; }
162
165 const std::string& title() const { return m_title; }

169 static constexpr std::size_t rank() { return Dim; }

173 const BoostHist& histogram() const { return m_hist; }

186 Histogram<1> sliceLastAxis(const std::array<int, Dim - 1>& outerBins) const {
187 const auto& lastAxis = m_hist.axis(Dim - 1);
188
189 assert(std::ranges::all_of(std::views::iota(std::size_t{0}, Dim - 1),
190 [&](std::size_t d) {
191 return outerBins[d] >= 0 &&
192 outerBins[d] < m_hist.axis(d).size();
193 }) &&
194 "outer bin index out of range");
195
196 std::string sliceName = m_name + "_slice";
197 for (const int bin : outerBins) {
198 sliceName += "_" + std::to_string(bin);
199 }
200
201 std::array<AxisVariant, 1> axes = {lastAxis};
202 Histogram<1> slice(std::move(sliceName), m_title, axes);
203
204 for (int k = 0; k < lastAxis.size(); ++k) {
205 std::array<int, Dim> indices{};
206 std::ranges::copy(outerBins, indices.begin());
207 indices[Dim - 1] = k;
208 slice.setBin({k}, binContent(indices), binError(indices));
209 }
210
211 return slice;
212 }
213
221 template <std::integral... Ints>
222 requires(sizeof...(Ints) + 1 == Dim)
223 Histogram<1> sliceLastAxis(Ints... outerBins) const {
224 return sliceLastAxis(
225 std::array<int, Dim - 1>{static_cast<int>(outerBins)...});
226 }

234 double totalContent() const {
235 double total = 0;
236 for (auto&& bin : boost::histogram::indexed(m_hist)) {
237 total += (*bin).value();
238 }
239 return total;
240 }
241
242 private:
243 std::string m_name;
244 std::string m_title;
245
246 BoostHist m_hist;
247};

263template <std::size_t Dim>

273 ProfileHistogram(std::string name, std::string title,
274 const std::array<AxisVariant, Dim>& axes,
275 std::string sampleAxisTitle,
276 Range1D<double> sampleRange = {})
277 : m_name(std::move(name)),
278 m_title(std::move(title)),
279 m_sampleAxisTitle(std::move(sampleAxisTitle)),
280 m_sampleRange(sampleRange),
281 m_hist(boost::histogram::make_profile(axes.begin(), axes.end())) {}
282
287 void fill(const std::array<double, Dim>& values, double sample) {
288 if (!m_sampleRange.contains(sample)) {
289 return;
290 }
291
292 std::apply(
293 [&](auto... v) { m_hist(v..., boost::histogram::sample(sample)); },
294 std::tuple_cat(values));
295 }
