313 template <typename propagator_state_t, typename stepper_t,
314 typename navigator_t>
315 Result<void> act(propagator_state_t& state, const stepper_t& stepper,
316 const navigator_t& navigator, result_type& result,
317 const Logger& /*logger*/) const {
318 ACTS_VERBOSE("CKF Actor called");
319
320 assert(result.trackStates && "No MultiTrajectory set");
321
322 if (state.stage == PropagatorStage::prePropagation &&
323 skipPrePropagationUpdate) {
324 ACTS_VERBOSE("Skip pre-propagation update (first surface)");
325 return Result<void>::success();
326 }
327 if (state.stage == PropagatorStage::postPropagation) {
328 ACTS_VERBOSE("Skip post-propagation action");
329 return Result<void>::success();
330 }
331
332 ACTS_VERBOSE("CombinatorialKalmanFilter step");
333
334 assert(!result.activeBranches.empty() && "No active branches");
335 assert(!result.finished && "Should never reach this when finished");
336
337 // Initialize path limit reached aborter
338 if (result.pathLimitReached.internalLimit ==
339 std::numeric_limits<double>::max()) {
340 detail::setupLoopProtection(state, stepper, result.pathLimitReached,
341 true, logger());
342 }
343
344 // Update:
345 // - Waiting for a current surface
346 if (const Surface* surface = navigator.currentSurface(state.navigation);
347 surface != nullptr) {
348 // There are three scenarios:
349 // 1) The surface is in the measurement map
350 // -> Select source links
351 // -> Perform the kalman update for selected non-outlier source links
352 // -> Add track states in multitrajectory. Multiple states mean branch
353 // splitting.
354 // -> Call branch stopper to justify each branch
355 // -> If there is non-outlier state, update stepper information
356 // 2) The surface is not in the measurement map but with material or is
357 // an active surface
358 // -> Add a hole or passive material state in multitrajectory
359 // -> Call branch stopper to justify the branch
360 // 3) The surface is neither in the measurement map nor with material
361 // -> Do nothing
362 ACTS_VERBOSE("Perform filter step");
363 auto res = filter(*surface, state, stepper, navigator, result);
364 if (!res.ok()) {
365 ACTS_DEBUG("Error in filter: " << res.error().message());
366 return res.error();
367 }
368
369 if (result.finished) {
370 ACTS_VERBOSE("CKF Actor returns after filter step");
371 return Result<void>::success();
372 }
373 }
374
375 assert(!result.activeBranches.empty() && "No active branches");
376
377 const bool isEndOfWorldReached =
378 endOfWorldReached.checkAbort(state, stepper, navigator, logger());
379 const bool isVolumeConstraintReached = volumeConstraintAborter.checkAbort(
380 state, stepper, navigator, logger());
381 const bool isPathLimitReached = result.pathLimitReached.checkAbort(
382 state, stepper, navigator, logger());
383 const bool isTargetReached =
384 targetReached.checkAbort(state, stepper, navigator, logger());
385 if (isEndOfWorldReached || isVolumeConstraintReached ||
386 isPathLimitReached || isTargetReached) {
387 if (isEndOfWorldReached) {
388 ACTS_VERBOSE("End of world reached");
389 } else if (isVolumeConstraintReached) {
390 ACTS_VERBOSE("Volume constraint reached");
391 } else if (isPathLimitReached) {
392 ACTS_VERBOSE("Path limit reached");
393 } else if (isTargetReached) {
394 ACTS_VERBOSE("Target surface reached");
395
396 // Bind the parameter to the target surface
397 auto res = stepper.boundState(state.stepping, *targetReached.surface);
398 if (!res.ok()) {
399 ACTS_DEBUG("Error while acquiring bound state for target surface: "
400 << res.error() << " " << res.error().message());
401 return res.error();
402 }
403
404 const auto& [boundParams, jacobian, pathLength] = *res;
405 auto currentBranch = result.activeBranches.back();
406 // Assign the fitted parameters
407 if constexpr (!IsMultiStepper) {
408 currentBranch.parameters() = boundParams.parameters();
409 currentBranch.covariance() = *boundParams.covariance();
410 } else {
411 const auto singleParams = boundParams.merge(brem.mergeMethod);
412 currentBranch.parameters() = singleParams.parameters();
413 currentBranch.covariance() = *singleParams.covariance();
414 }
415 currentBranch.setReferenceSurface(
416 boundParams.referenceSurface().getSharedPtr());
417
418 stepper.releaseStepSize(state.stepping,
419 ConstrainedStep::Type::Navigator);
420 }
421
422 // Record the active branch and remove it from the list
423 storeLastActiveBranch(result);
424 result.activeBranches.pop_back();
425
426 // Reset propagation state to track state at next active branch
427 auto resetRes = reset(state, stepper, navigator, result);
428 if (!resetRes.ok()) {
429 return resetRes.error();
430 }
431 }
432
433 return Result<void>::success();
434 }

439 const navigator_t& /*navigator*/, const result_type& result,
440 const Logger& /*logger*/) const {
441 return result.finished;

454 template <typename propagator_state_t, typename stepper_t,
455 typename navigator_t>
456 Result<void> reset(propagator_state_t& state, const stepper_t& stepper,
457 const navigator_t& navigator,
458 result_type& result) const {
459 if (result.activeBranches.empty()) {
460 ACTS_VERBOSE("Stop CKF with " << result.collectedTracks.size()
461 << " found tracks");
462 result.finished = true;
463
464 return Result<void>::success();
465 }
466
467 auto currentBranch = result.activeBranches.back();
468 auto currentState = currentBranch.outermostTrackState();
469
470 ACTS_VERBOSE("Propagation jumps to branch with tip = "
471 << currentBranch.tipIndex());
472
473 // Reset the stepping state
474 if constexpr (!IsMultiStepper) {
475 stepper.initialize(state.stepping, currentState.filtered(),
476 currentState.filteredCovariance(),
477 stepper.particleHypothesis(state.stepping),
478 currentState.referenceSurface());
479 } else {
480 const MultiComponentBoundTrackParameters multiBoundParameters(
481 currentState.referenceSurface().getSharedPtr(),
482 currentState.filtered(), currentState.filteredCovariance(),
483 stepper.particleHypothesis(state.stepping));
484
485 stepper.initialize(state.stepping, multiBoundParameters);
486 }
487
488 // Reset the navigation state
489 // Set targetSurface to nullptr for forward filtering
490 state.navigation.options.startSurface = &currentState.referenceSurface();
491 state.navigation.options.targetSurface = nullptr;
492 auto navInitRes = navigator.initialize(
493 state.navigation, stepper.position(state.stepping),
494 stepper.direction(state.stepping), state.options.direction);
495 if (!navInitRes.ok()) {
496 ACTS_DEBUG("Navigation initialization failed: " << navInitRes.error());
497 return navInitRes.error();
498 }
499
500 // No Kalman filtering for the starting surface, but still need
501 // to consider the material effects here
502 const Result<void> materialRes = performMaterialInteraction(
503 state, stepper, currentState.referenceSurface(),
504 detail::determineMaterialUpdateMode(state, navigator,
505 MaterialUpdateMode::PostUpdate));
506 if (!materialRes.ok()) {
507 ACTS_DEBUG("Material interaction failed during reset: "
508 << materialRes.error().message());
509 return materialRes.error();
510 }
511
512 // Set path limit based on loop protection
513 detail::setupLoopProtection(state, stepper, result.pathLimitReached, true,
514 logger());
515
516 // Set path limit based on target surface
517 targetReached.checkAbort(state, stepper, navigator, logger());
518
519 return Result<void>::success();
520 }

536 template <typename propagator_state_t, typename stepper_t,
537 typename navigator_t>
538 Result<void> filter(const Surface& surface, propagator_state_t& state,
539 const stepper_t& stepper, const navigator_t& navigator,
540 result_type& result) const {
541 using PM = TrackStatePropMask;
542
543 const bool isSensitive = surface.isSensitive();
544 const bool hasMaterial = surface.hasMaterial();
545 const bool isMaterialOnly = hasMaterial && !isSensitive;
546 bool expectMeasurements = isSensitive;
547
548 if (isSensitive) {
549 ACTS_VERBOSE("Measurement surface " << surface.geometryId()
550 << " detected.");
551 } else if (isMaterialOnly) {
552 ACTS_VERBOSE("Material surface " << surface.geometryId()
553 << " detected.");
554 } else {
555 ACTS_VERBOSE("Passive surface " << surface.geometryId()
556 << " detected.");
557 return Result<void>::success();
558 }
559
560 // Transport the covariance to the surface
561 if (isMaterialOnly) {
562 stepper.transportCovarianceToCurvilinear(state.stepping);
563 } else {
564 stepper.transportCovarianceToBound(state.stepping, surface);
565 }
566
567 // Update state and stepper with pre material effects
568 const Result<void> materialPreRes = performMaterialInteraction(
569 state, stepper, surface,
570 detail::determineMaterialUpdateMode(state, navigator,
571 MaterialUpdateMode::PreUpdate));
572 if (!materialPreRes.ok()) {
573 ACTS_DEBUG("Material interaction failed during reset: "
574 << materialPreRes.error().message());
575 return materialPreRes.error();
576 }
577
578 // Bind the transported state to the current surface
579 auto boundStateRes = [&]() -> Result<SingleBoundState> {
580 if constexpr (!IsMultiStepper) {
581 auto res = stepper.boundState(state.stepping, surface, false);
582 if (!res.ok()) {
583 return res.error();
584 }
585 auto& [boundParams, jacobian, pathLength] = *res;
586 boundParams.covariance() = state.stepping.cov;
587 return res;
588 } else {
589 // This triggers a second covariance transport which is wasteful in
590 // terms of compute. But the multi stepper might filter bound
591 // components and we cannot easily match stepper components to bound
592 // parameter components right now to make use of the same track as
593 // above.
594 // TODO this should be revisited
595 auto res = stepper.boundState(state.stepping, surface, true);
596 if (!res.ok()) {
597 return res.error();
598 }
599 auto [multiBoundParams, jacobian, pathLength] = *res;
600 const auto singleParams = multiBoundParams.merge(brem.mergeMethod);
601 return Result<SingleBoundState>::success(
602 {singleParams, jacobian, pathLength});
603 }
604 }();
605 if (!boundStateRes.ok()) {
606 return boundStateRes.error();
607 }
608 auto& boundState = *boundStateRes;
609
610 auto currentBranch = result.activeBranches.back();
611 TrackIndexType prevTip = currentBranch.tipIndex();
612
613 using TrackStatesResult = Result<CkfTypes::BranchVector<TrackIndexType>>;
614 TrackStatesResult tsRes = TrackStatesResult::success({});
615 if (isSensitive) {
616 // extend trajectory with measurements associated to the current surface
617 // which may create extra trajectory branches if more than one
618 // measurement is selected.
619 tsRes = extensions.createTrackStates(
620 state.geoContext, *calibrationContextPtr, surface, boundState,
621 prevTip, result.trackStateCandidates, *result.trackStates,
622 logger());
623 }
624
625 if (tsRes.ok() && !(*tsRes).empty()) {
626 const CkfTypes::BranchVector<TrackIndexType>& newTrackStateList =
627 *tsRes;
628 const Result<unsigned int> procRes =
629 processNewTrackStates(state, stepper, newTrackStateList, result);
630 if (!procRes.ok()) {
631 ACTS_DEBUG("Processing of selected track states failed: "
632 << procRes.error().message());
633 return procRes.error();
634 }
635 unsigned int nBranchesOnSurface = *procRes;
636
637 if (nBranchesOnSurface == 0) {
638 ACTS_VERBOSE("All branches on surface " << surface.geometryId()
639 << " have been stopped");
640
641 Result<void> resetRes = reset(state, stepper, navigator, result);
642 if (!resetRes.ok()) {
643 return resetRes.error();
644 }
645
646 return Result<void>::success();
647 }
648
649 // `currentBranch` is invalidated after `processNewTrackStates`
650 currentBranch = result.activeBranches.back();
651 prevTip = currentBranch.tipIndex();
652 } else {
653 if (!tsRes.ok()) {
654 if (static_cast<CombinatorialKalmanFilterError>(
655 tsRes.error().value()) ==
656 CombinatorialKalmanFilterError::NoMeasurementExpected) {
657 // recoverable error returned by track state creator
658 expectMeasurements = false;
659 } else {
660 ACTS_DEBUG("Track state creation failed on surface "
661 << surface.geometryId() << ": " << tsRes.error());
662 return tsRes.error();
663 }
664 }
665
666 if (expectMeasurements) {
667 ACTS_VERBOSE("Detected hole after measurement selection on surface "
668 << surface.geometryId());
669 }
670
671 const auto stateMask = PM::Predicted | PM::Jacobian;
672
673 // Add a hole or material track state to the multitrajectory
674 TrackIndexType currentTip =
675 addNonSourcelinkState(stateMask, boundState, result, isSensitive,
676 expectMeasurements, prevTip);
677 currentBranch.tipIndex() = currentTip;
678 auto currentState = currentBranch.outermostTrackState();
679 if (expectMeasurements) {
680 currentBranch.nHoles()++;
681 }
682
683 BranchStopperResult branchStopperResult =
684 extensions.branchStopper(currentBranch, currentState);
685
686 // Check the branch
687 if (branchStopperResult == BranchStopperResult::Continue) {
688 // Remembered the active branch and its state
689 } else {
690 // No branch on this surface
691 if (branchStopperResult == BranchStopperResult::StopAndKeep) {
692 storeLastActiveBranch(result);
693 }
694 // Remove the branch from list
695 result.activeBranches.pop_back();
696
697 // Branch on the surface has been stopped - reset
698 ACTS_VERBOSE("Branch on surface " << surface.geometryId()
699 << " has been stopped");
700
701 Result<void> resetRes = reset(state, stepper, navigator, result);
702 if (!resetRes.ok()) {
703 return resetRes.error();
704 }
705
706 return Result<void>::success();
707 }
708 }
709
710 auto currentState = currentBranch.outermostTrackState();
711
712 if (currentState.typeFlags().isOutlier()) {
713 // We don't need to update the stepper given an outlier state
714 ACTS_VERBOSE("Outlier state detected on surface "
715 << surface.geometryId());
716 } else if (currentState.typeFlags().isMeasurement()) {
717 // If there are measurement track states on this surface
718 // Update stepping state using filtered parameters of last track
719 // state on this surface
720 updateStepper(state, stepper, currentState);
721 ACTS_VERBOSE("Stepping state is updated with filtered parameter:");
722 ACTS_VERBOSE("-> " << currentState.filtered().transpose()
723 << " of track state with tip = "
724 << currentState.index());
725 }
726
727 // Update state and stepper with post material effects
728 const Result<void> materialPostRes = performMaterialInteraction(
729 state, stepper, surface,
730 detail::determineMaterialUpdateMode(state, navigator,
731 MaterialUpdateMode::PostUpdate));
732 if (!materialPostRes.ok()) {
733 ACTS_DEBUG("Material interaction failed during post-update: "
734 << materialPostRes.error().message());
735 return materialPostRes.error();
736 }
737
738 return Result<void>::success();
739 }

754 template <typename propagator_state_t, typename stepper_t>
755 Result<unsigned int> processNewTrackStates(
756 propagator_state_t& state, const stepper_t& stepper,
757 const CkfTypes::BranchVector<TrackIndexType>& newTrackStateList,
758 result_type& result) const {
759 using PM = TrackStatePropMask;
760
761 unsigned int nBranchesOnSurface = 0;
762
763 auto rootBranch = result.activeBranches.back();
764
765 // Build the new branches by forking the root branch. Reverse the order
766 // to process the best candidate first
767 CkfTypes::BranchVector<TrackProxy> newBranches;
768 for (auto it = newTrackStateList.rbegin(); it != newTrackStateList.rend();
769 ++it) {
770 // Keep the root branch as the first branch, make a copy for the
771 // others
772 auto shallowCopy = [&] {
773 auto sc = rootBranch.container().makeTrack();
774 sc.copyFromShallow(rootBranch);
775 return sc;
776 };
777 auto newBranch =
778 (it == newTrackStateList.rbegin()) ? rootBranch : shallowCopy();
779 newBranch.tipIndex() = *it;
780 newBranches.push_back(newBranch);
781 }
782
783 // Remove the root branch
784 result.activeBranches.pop_back();
785
786 // Update and select from the new branches
787 for (TrackProxy newBranch : newBranches) {
788 auto trackState = newBranch.outermostTrackState();
789 TrackStateTypeMap typeFlags = trackState.typeFlags();
790
791 if (typeFlags.isOutlier()) {
792 // No Kalman update for outlier
793 // Set the filtered parameter index to be the same with predicted
794 // parameter
795 trackState.shareFrom(PM::Predicted, PM::Filtered);
796 // Increment number of outliers
797 newBranch.nOutliers()++;
798 } else if (typeFlags.isMeasurement()) {
799 // Kalman update
800 const auto updateRes = kalmanUpdate(state, stepper, trackState);
801 if (!updateRes.ok()) {
802 ACTS_DEBUG("Update step failed: " << updateRes.error().message());
803 return updateRes.error();
804 }
805 ACTS_VERBOSE("Appended measurement track state with tip = "
806 << newBranch.tipIndex());
807 // Increment number of measurements
808 newBranch.nMeasurements()++;
809 newBranch.nDoF() += trackState.calibratedSize();
810 newBranch.chi2() += trackState.chi2();
811 } else {
812 ACTS_WARNING("Cannot handle this track state flags");
813 continue;
814 }
815
816 result.activeBranches.push_back(newBranch);
817
818 BranchStopperResult branchStopperResult =
819 extensions.branchStopper(newBranch, trackState);
820
821 // Check if need to stop this branch
822 if (branchStopperResult == BranchStopperResult::Continue) {
823 // Record the number of branches on surface
824 nBranchesOnSurface++;
825 } else {
826 // Record the number of stopped branches
827 if (branchStopperResult == BranchStopperResult::StopAndKeep) {
828 storeLastActiveBranch(result);
829 }
830 // Remove the branch from list
831 result.activeBranches.pop_back();
832 }
833 }
834
835 return nBranchesOnSurface;
836 }

848 TrackIndexType addNonSourcelinkState(TrackStatePropMask stateMask,
849 const BoundState& boundState,
850 result_type& result, bool isSensitive,
851 bool expectMeasurements,
852 TrackIndexType prevTip) const {
853 using PM = TrackStatePropMask;
854
855 // Add a track state
856 auto trackStateProxy =
857 result.trackStates->makeTrackState(stateMask, prevTip);
858 ACTS_VERBOSE("Create "
859 << (isSensitive
860 ? (expectMeasurements ? "Hole"
861 : "noMeasurementExpected")
862 : "Material")
863 << " output track state #" << trackStateProxy.index()
864 << " with mask: " << stateMask);
865
866 const auto& [boundParams, jacobian, pathLength] = boundState;
867 // Fill the track state

910 template <typename propagator_state_t, typename stepper_t>
911 Result<void> performMaterialInteraction(
912 propagator_state_t& state, const stepper_t& stepper,
913 const Surface& surface, MaterialUpdateMode updateMode) const {
914 if (!surface.hasMaterial()) {
915 return Result<void>::success();
916 }
917
918 if constexpr (!IsMultiStepper) {
919 const Result<detail::PointwiseMaterialEffects> materialInteractionRes =
920 detail::performMaterialInteraction(
921 state, stepper, surface, updateMode, NoiseUpdateMode::addNoise,
922 multipleScattering, energyLoss, logger());
923 if (!materialInteractionRes.ok()) {
924 ACTS_DEBUG("Material interaction failed during reset: "
925 << materialInteractionRes.error().message());
926 return materialInteractionRes.error();
927 }
928
929 return Result<void>::success();
930 } else {
931 if (ACTS_CHECK_BIT(updateMode, MaterialUpdateMode::PostUpdate)) {
932 brem.betheHeitlerCache->clear();
933 brem.componentCache->clear();
934 std::size_t nInvalidBetheHeitler = 0;
935 double maxPathXOverX0 = 0;
936
937 for (auto cmp : stepper.componentIterable(state.stepping)) {
938 const auto boundParamsRes = transformFreeToBoundParameters(
939 cmp.state().pars, surface, state.options.geoContext);
940 if (!boundParamsRes.ok()) {
941 ACTS_DEBUG(
942 "Failed to transform free to bound parameters for "
943 "Bethe-Heitler approximation: "
944 << boundParamsRes.error());
945 continue;
946 }
947
948 const BoundTrackParameters bound(
949 surface.getSharedPtr(), *boundParamsRes, cmp.state().cov,
950 stepper.particleHypothesis(state.stepping));
951
952 detail::Gsf::applyBetheHeitler(
953 state.options.geoContext, surface, state.options.direction,
954 bound, cmp.weight(), *brem.betheHeitlerApprox,
955 *brem.betheHeitlerCache, brem.weightCutoff,
956 *brem.componentCache, nInvalidBetheHeitler, maxPathXOverX0,
957 logger());
958 }
959
960 if (brem.componentCache->empty()) {
961 ACTS_WARNING(
962 "No components left after applying energy loss. "
963 "Is the weight cutoff "
964 << brem.weightCutoff << " too high?");
965 ACTS_WARNING("Return to propagator without applying energy loss");
966 return Result<void>::success();
967 }
968
969 // reduce component number
970 const auto finalCmpNumber =
971 std::min(static_cast<std::size_t>(stepper.maxComponents),
972 brem.maxComponents);
973 extensions.mixtureReducer(*brem.componentCache, finalCmpNumber,
974 surface);
975
976 detail::Gsf::removeLowWeightComponents(*brem.componentCache,
977 brem.weightCutoff);
978
979 detail::Gsf::updateStepper(state, stepper, surface,
980 *brem.componentCache);
981 }
982
983 detail::Gsf::applyMultipleScattering(state, stepper, surface,
984 updateMode, logger());
985
986 return Result<void>::success();
987 }
988 }

990 template <typename propagator_state_t, typename stepper_t>
991 Result<void> kalmanUpdate(propagator_state_t& state,
992 const stepper_t& stepper,
993 TrackStateProxy& trackState) const {
994 if constexpr (!IsMultiStepper) {
995 return extensions.updater(state.geoContext, trackState, *updaterLogger);
996 } else {
997 using PrtProjector = detail::Gsf::MultiTrajectoryProjector<
998 detail::Gsf::StatesType::ePredicted, TrackStateContainerBackend>;
999 using FltProjector = detail::Gsf::MultiTrajectoryProjector<
1000 detail::Gsf::StatesType::eFiltered, TrackStateContainerBackend>;
1001
1002 const auto& surface = trackState.referenceSurface();
1003
1004 auto& temporaryStates = *brem.temporaryStates;
1005 temporaryStates.clear();
1006
1007 for (auto cmp : stepper.componentIterable(state.stepping)) {
1008 auto& singleState = cmp.singleState(state).stepping;
1009 const auto& singleStepper = cmp.singleStepper(stepper);
1010
1011 TrackStatePropMask mask =
1012 TrackStatePropMask::Predicted | TrackStatePropMask::Filtered |
1013 TrackStatePropMask::Jacobian | TrackStatePropMask::Calibrated;
1014 TrackStateProxy trackStateProxy =
1015 temporaryStates.traj.makeTrackState(mask, kTrackIndexInvalid);
1016
1017 // TODO call calibrator again?
1018
1019 trackStateProxy.setReferenceSurface(surface.getSharedPtr());
1020 // Bind the transported state to the current surface
1021 auto res = singleStepper.boundState(singleState, surface);
1022 if (!res.ok()) {
1023 ACTS_ERROR("Propagate to surface " << surface.geometryId()
1024 << " failed: " << res.error());
1025 return res.error();
1026 }
1027 const auto& [boundParams, jacobian, pathLength] = *res;
1028
1029 // Fill the track state
1030 trackStateProxy.predicted() = boundParams.parameters();
1031 trackStateProxy.predictedCovariance() = *boundParams.covariance();
1032 trackStateProxy.allocateCalibrated(trackState.calibratedSize());
1033 trackStateProxy.setProjectorSubspaceIndices(
1034 trackState.projectorSubspaceIndices());
1035 trackStateProxy.effectiveCalibrated() =
1036 trackState.effectiveCalibrated();
1037 trackStateProxy.effectiveCalibratedCovariance() =
1038 trackState.effectiveCalibratedCovariance();
1039
1040 const auto updateRes = extensions.updater(
1041 state.geoContext, trackStateProxy, *updaterLogger);
1042 if (!updateRes.ok()) {
1043 return updateRes.error();
1044 }
1045
1046 temporaryStates.tips.push_back(trackStateProxy.index());
1047 temporaryStates.weights[trackStateProxy.index()] = cmp.weight();
1048 }
1049
1050 detail::Gsf::computePosteriorWeights(temporaryStates.traj,
1051 temporaryStates.tips,
1052 temporaryStates.weights);
1053
1054 detail::Gsf::normalizeWeights(temporaryStates.tips,
1055 [&](auto idx) -> double& {
1056 return temporaryStates.weights.at(idx);
1057 });
1058
1059 const auto [prtMean, prtCov] = detail::Gsf::mergeGaussianMixture(
1060 temporaryStates.tips,
1061 PrtProjector{temporaryStates.traj, temporaryStates.weights},
1062 surface, brem.mergeMethod);
1063 trackState.predicted() = prtMean;
1064 trackState.predictedCovariance() = prtCov;
1065
1066 const auto [fltMean, fltCov] = detail::Gsf::mergeGaussianMixture(
1067 temporaryStates.tips,
1068 FltProjector{temporaryStates.traj, temporaryStates.weights},
1069 surface, brem.mergeMethod);
1070 trackState.filtered() = fltMean;
1071 trackState.filteredCovariance() = fltCov;
1072
1073 return Result<void>::success();
1074 }
1075 }
