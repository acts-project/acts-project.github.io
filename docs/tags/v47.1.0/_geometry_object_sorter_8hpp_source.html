22template <class T>
23class ObjectSorterT {
24 public:
28 explicit ObjectSorterT(AxisDirection aDir) : m_sortingDirection(aDir) {}
29
36 bool operator()(T one, T two) const {
37 using VectorHelpers::eta;
38 using VectorHelpers::perp;
39 using VectorHelpers::phi;
40 using enum AxisDirection;
41 switch (m_sortingDirection) {
42 // compare on x
43 case AxisX: {
44 return one.x() < two.x();
45 }
46 // compare on y
47 case AxisY: {
48 return one.y() < two.y();
49 }
50 // compare on z
51 case AxisZ: {
52 return one.z() < two.z();
53 }
54 // compare on r
55 case AxisR: {
56 return perp(one) < perp(two);
57 }
58 // compare on phi
59 case AxisPhi: {
60 return phi(one) < phi(two);
61 }
62 // compare on eta
63 case AxisEta: {
64 return eta(one) < eta(two);
65 }
66 // default for the moment
67 default: {
68 return one.norm() < two.norm();
69 }
70 }
71 }
72
75 AxisDirection sortingDirection() const { return m_sortingDirection; }

77 private:
78 AxisDirection m_sortingDirection;
79};

82template <class T>
83class DistanceSorterT {
84 public:
89 DistanceSorterT(AxisDirection aDir, Vector3 reference)
90 : m_sortingDirection(aDir),
91 m_reference(reference),
92 m_refR(VectorHelpers::perp(reference)),
93 m_refPhi(VectorHelpers::phi(reference)),
94 m_refEta(VectorHelpers::eta(reference)) {}
95
101 bool operator()(T one, T two) const {
102 using Acts::VectorHelpers::eta;
103 using Acts::VectorHelpers::perp;
104 using Acts::VectorHelpers::phi;
105 // switch the sorting value
106 // - AxisX, AxisY, AxisZ, AxisR, AxisPhi, AxisRPhi, AxisTheta, AxisEta
107 switch (m_sortingDirection) {
108 // compare on diff x
109 case AxisDirection::AxisX: {
110 double diffOneX = one.x() - m_reference.x();
111 double diffTwoX = two.x() - m_reference.x();
112 return std::abs(diffOneX) < std::abs(diffTwoX);
113 }
114 // compare on diff y
115 case AxisDirection::AxisY: {
116 double diffOneY = one.y() - m_reference.y();
117 double diffTwoY = two.y() - m_reference.y();
118 return std::abs(diffOneY) < std::abs(diffTwoY);
119 }
120 // compare on diff z
121 case AxisDirection::AxisZ: {
122 double diffOneZ = one.z() - m_reference.z();
123 double diffTwoZ = two.z() - m_reference.z();
124 return std::abs(diffOneZ) < std::abs(diffTwoZ);
125 }
126 // compare on r
127 case AxisDirection::AxisR: {
128 double diffOneR = perp(one) - m_refR;
129 double diffTwoR = perp(two) - m_refR;
130 return std::abs(diffOneR) < std::abs(diffTwoR);
131 }
132 // compare on phi /// @todo add cyclic value
133 case AxisDirection::AxisPhi: {
134 double diffOnePhi = phi(one) - m_refPhi;
135 double diffTwoPhi = phi(two) - m_refPhi;
136 return std::abs(diffOnePhi) < std::abs(diffTwoPhi);
137 }
138 // compare on eta
139 case AxisDirection::AxisEta: {
140 double diffOneEta = eta(one) - m_refEta;
141 double diffTwoEta = eta(two) - m_refEta;
142 return std::abs(diffOneEta) < std::abs(diffTwoEta);
143 }
144 // default for the moment
145 default: {
146 T diffOne(one - m_reference);
147 T diffTwo(two - m_reference);
148 return diffOne.mag2() < diffTwo.mag2();
149 }
150 }
151 }
152
153 private:
154 AxisDirection m_sortingDirection;
155 T m_reference;
156 double m_refR;
157 double m_refPhi;
158 double m_refEta;
159};

162template <class T>
163class GeometryObjectSorterT {
164 public:
170 GeometryObjectSorterT(const GeometryContext& gctx, AxisDirection aDir,
171 std::shared_ptr<const Transform3> transform = nullptr)
172 : m_context(gctx),
173 m_objectSorter(aDir),
174 m_transform(std::move(transform)) {}
175
182 bool operator()(const T& one, const T& two) const {
183 // get the pos one / pos two
184 Vector3 posOne = m_transform
185 ? m_transform->inverse() *
186 one->referencePosition(
187 m_context, m_objectSorter.sortingDirection())
188 : one->referencePosition(
189 m_context, m_objectSorter.sortingDirection());
190 Vector3 posTwo = m_transform
191 ? m_transform->inverse() *
192 two->referencePosition(
193 m_context, m_objectSorter.sortingDirection())
194 : two->referencePosition(
195 m_context, m_objectSorter.sortingDirection());
196 // now call the distance sorter
197 return m_objectSorter.operator()(posOne, posTwo);
198 }
199
200 protected:
202 std::reference_wrapper<const GeometryContext> m_context;
204 ObjectSorterT<Vector3> m_objectSorter;
206 std::shared_ptr<const Transform3> m_transform;
207};
