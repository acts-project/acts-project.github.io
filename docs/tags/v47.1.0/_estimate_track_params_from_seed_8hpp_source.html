87template <std::ranges::range space_point_range_t>
88[[deprecated(
89 "The broadly templated versions of estimateTrackParamsFromSeed will be "
90 "removed in the future.")]]
91FreeVector estimateTrackParamsFromSeed(space_point_range_t spRange,
92 const Vector3& bField) {
93 // Check the number of provided space points
94 if (spRange.size() != 3) {
95 throw std::invalid_argument(
96 "There should be exactly three space points provided.");
97 }
98
99 // The global positions of the bottom, middle and space points
100 std::array<Vector3, 3> spPositions = {Vector3::Zero(), Vector3::Zero(),
101 Vector3::Zero()};
102 std::array<std::optional<double>, 3> spTimes = {std::nullopt, std::nullopt,
103 std::nullopt};
104 // The first, second and third space point are assumed to be bottom, middle
105 // and top space point, respectively
106 for (auto [sp, spPosition, spTime] :
107 Acts::zip(spRange, spPositions, spTimes)) {
108 if (sp == nullptr) {
109 throw std::invalid_argument("Empty space point found.");
110 }
111 spPosition = Vector3(sp->x(), sp->y(), sp->z());
112 spTime = sp->t();
113 }
114
115 return estimateTrackParamsFromSeed(spPositions[0], spTimes[0].value_or(0),
116 spPositions[1], spPositions[2], bField);
117}
