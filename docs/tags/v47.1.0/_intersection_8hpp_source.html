119 constexpr static bool pathLengthOrder(
120 const Intersection& aIntersection,
121 const Intersection& bIntersection) noexcept {

132 constexpr static bool closestOrder(
133 const Intersection& aIntersection,
134 const Intersection& bIntersection) noexcept {
135 using enum IntersectionStatus;
136
137 if ((aIntersection.status() == unreachable) &&
138 (bIntersection.status() != unreachable)) {
139 return false;
140 }
141 if ((aIntersection.status() != unreachable) &&
142 (bIntersection.status() == unreachable)) {
143 return true;
144 }
145 // both are reachable or onSurface now
146 auto a = aIntersection.pathLength();
147 auto b = bIntersection.pathLength();
148 return std::abs(a) < std::abs(b);
149 }
150
156 constexpr static bool closestForwardOrder(
157 const Intersection& aIntersection,
158 const Intersection& bIntersection) noexcept {
159 auto a = aIntersection.pathLength();
160 auto b = bIntersection.pathLength();
161 return std::signbit(a) == std::signbit(b) ? std::abs(a) < std::abs(b)
162 : a > b;
163 }
164
165 private:
167 std::array<double, DIM> m_position{};
169 double m_pathLength = std::numeric_limits<double>::infinity();
171 IntersectionStatus m_status = IntersectionStatus::unreachable;
172
173 constexpr Intersection() noexcept = default;
174};
