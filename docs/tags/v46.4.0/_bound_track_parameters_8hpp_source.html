58 static Result<BoundTrackParameters> create(

185 std::optional<SquareMatrix<3>> impactParameterCovariance() const {

189
190 SquareMatrix<3> ipCov;
191 ipCov.template topLeftCorner<2, 2>() =
192 m_cov.value().template topLeftCorner<2, 2>();
193 ipCov.template block<2, 1>(0, 2) =
194 m_cov.value().template block<2, 1>(0, eBoundTime);
195 ipCov.template block<1, 2>(2, 0) =
196 m_cov.value().template block<1, 2>(eBoundTime, 0);
197 ipCov(2, 2) = m_cov.value()(eBoundTime, eBoundTime);
198 return ipCov;
199 }
