45 struct Element {
47 std::shared_ptr<const NodeContext> context = nullptr;
48
49 friend bool operator==(const Element& lhs, const Element& rhs) {
50 const NodeContext* lc = lhs.context.get();
51 const NodeContext* rc = rhs.context.get();
52
53 while (lc != nullptr && rc != nullptr) {
54 if (lc->node != rc->node) {
55 return false;
56 }
57 lc = lc->parent.get();
58 rc = rc->parent.get();
59 }
60
61 return lc == nullptr && rc == nullptr;
62 }
63 };
