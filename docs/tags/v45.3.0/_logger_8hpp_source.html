177inline std::string_view levelName(Level level) {
178 switch (level) {
179 case Level::VERBOSE:
180 return "VERBOSE";
181 case Level::DEBUG:
182 return "DEBUG";
183 case Level::INFO:
184 return "INFO";
185 case Level::WARNING:

188 return "ERROR";
189 case Level::FATAL:
190 return "FATAL";
191 case Level::MAX:
192 return "MAX";
193 default:
194 throw std::invalid_argument{"Unknown level"};
195 }

498 public:
503 explicit TimedOutputDecorator(std::unique_ptr<OutputPrintPolicy> wrappee,
504 const std::string& format = "%X")
505 : OutputDecorator(std::move(wrappee)), m_format(format) {}

514 void flush(const Level& lvl, const std::string& input) override {
515 std::ostringstream os;
516 os << std::left << std::setw(12) << now() << input;
517 OutputDecorator::flush(lvl, os.str());

519
523 std::unique_ptr<OutputPrintPolicy> clone(
524 const std::string& name) const override {
525 return std::make_unique<TimedOutputDecorator>(m_wrappee->clone(name),
526 m_format);

529 private:
533 std::string now() const {
534 char buffer[20];
535 time_t t{};
536 std::time(&t);
537 struct tm tbuf {};
538 std::strftime(buffer, sizeof(buffer), m_format.c_str(),
539 localtime_r(&t, &tbuf));
540 return buffer;
541 }

544 std::string m_format;

629class DefaultPrintPolicy final : public OutputPrintPolicy {
630 public:
636 explicit DefaultPrintPolicy(std::ostream* out = &std::cout) : m_out(out) {}
637
642 void flush(const Level& lvl, const std::string& input) final {
643 // Mutex to serialize access to std::cout
644 static std::mutex s_stdoutMutex;
645 std::unique_lock lock{s_stdoutMutex,
646 std::defer_lock}; // prep empty, we might not need it
647
648 if (m_out == &std::cout) {
649 lock.lock(); // lock only if we are printing to std::cout
650 }
651
652 (*m_out) << input << std::endl;
653 if (lvl >= getFailureThreshold()) {
654 throw ThresholdFailure(
655 "Previous debug message exceeds the "
656 "ACTS_LOG_FAILURE_THRESHOLD=" +
657 std::string{levelName(getFailureThreshold())} +
658 " configuration, bailing out. See "
659 "https://acts.readthedocs.io/en/latest/core/misc/"
660 "logging.html#logging-thresholds");
661 }
662 }
663
670 const std::string& name() const override {
671 throw std::runtime_error{
672 "Default print policy doesn't have a name. Is there no named output in "
673 "the decorator chain?"};
674 };
675
678 std::unique_ptr<OutputPrintPolicy> clone(
679 const std::string& /*name*/) const override {
680 return std::make_unique<DefaultPrintPolicy>(m_out);
681 };
682
683 private:
685 std::ostream* m_out;
686};
