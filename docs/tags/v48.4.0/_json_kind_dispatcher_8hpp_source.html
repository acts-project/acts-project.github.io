26template <typename return_t, typename... args_t>
27class JsonKindDispatcher {
28 public:
30 using self_type = JsonKindDispatcher<return_t, args_t...>;
32 using decoder_signature = return_t(const nlohmann::json&, args_t...);
34 using decoder_type = std::function<decoder_signature>;
36 using decoder_pointer_type = return_t (*)(args_t...);
37
42 explicit JsonKindDispatcher(std::string kindKey = "kind",
43 std::string context = "JSON payload")
44 : m_kindKey(std::move(kindKey)), m_context(std::move(context)) {
45 if (m_kindKey.empty()) {
46 throw std::invalid_argument(
47 "JsonKindDispatcher kind key must be non-empty");
48 }
49 if (m_context.empty()) {
50 m_context = "JSON payload";
51 }
52 }
53
60 self_type& registerKind(std::string kind, decoder_type decoder) {
61 if (kind.empty()) {
62 throw std::invalid_argument("JsonKindDispatcher kind must be non-empty");
63 }
64 if (!decoder) {
65 throw std::invalid_argument("JsonKindDispatcher decoder must be valid");
66 }
67 auto [_, inserted] =
68 m_decoders.emplace(std::move(kind), std::move(decoder));
69 if (!inserted) {
70 throw std::invalid_argument(
71 "JsonKindDispatcher duplicate kind registration");
72 }
73 return *this;
74 }
75
82 template <typename... func_args_t>
83 return_t operator()(const nlohmann::json& encoded,
84 func_args_t&&... args) const
85 requires std::invocable<decoder_pointer_type, func_args_t...>
86 {
87 if (!encoded.contains(m_kindKey)) {
88 throw std::invalid_argument("Missing '" + m_kindKey + "' in " +
89 m_context);
90 }
91
92 const auto& kindValue = encoded.at(m_kindKey);
93 if (!kindValue.is_string()) {
94 throw std::invalid_argument("Invalid '" + m_kindKey + "' type in " +
95 m_context);
96 }
97
98 const auto kind = kindValue.template get<std::string>();
99 auto decoder = m_decoders.find(kind);
100 if (decoder == m_decoders.end()) {
101 throw std::invalid_argument("Unsupported " + m_context +
102 " kind: " + kind);
103 }
104
105 if constexpr (std::is_void_v<return_t>) {
106 decoder->second(encoded, std::forward<func_args_t>(args)...);
107 return;
108 } else {
109 return decoder->second(encoded, std::forward<func_args_t>(args)...);
110 }
111 }
112
118 bool hasKind(std::string_view kind) const {
119 return m_decoders.contains(std::string{kind});
120 }
121
123 void clear() { m_decoders.clear(); }
124
128 std::size_t size() const { return m_decoders.size(); }
129
130 private:
131 std::string m_kindKey;
132 std::string m_context;
133 std::unordered_map<std::string, decoder_type> m_decoders;
134};
