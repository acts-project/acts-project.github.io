119 void assignSourceLinks(std::span<const SourceLink> sourceLinks) const
120 requires(!ReadOnly)
121 {
122 if (m_index >= m_container->size()) {
123 throw std::out_of_range("Index out of range in SpacePointContainer");
124 }
125 if (!m_container->m_sourceLinkOffsetColumn.has_value() ||
126 !m_container->m_sourceLinkCountColumn.has_value()) {
127 throw std::logic_error("No source links column available");
128 }
129 if (accessImpl(m_container->m_sourceLinkCountColumn) != 0) {
130 throw std::logic_error(
131 "Source links already assigned to the space point");
132 }
133
134 accessImpl(m_container->m_sourceLinkOffsetColumn) =
135 static_cast<std::uint32_t>(m_container->m_sourceLinks.size());
136 accessImpl(m_container->m_sourceLinkCountColumn) =
137 static_cast<std::uint8_t>(sourceLinks.size());
138 m_container->m_sourceLinks.insert(m_container->m_sourceLinks.end(),
139 sourceLinks.begin(), sourceLinks.end());
140 }
141
144 std::span<SourceLink> sourceLinks() const noexcept
145 requires(!ReadOnly)
146 {
147 const std::size_t offset =
148 accessImpl(m_container->m_sourceLinkOffsetColumn);
149 const std::size_t count = accessImpl(m_container->m_sourceLinkCountColumn);
150 return std::span<SourceLink>(m_container->m_sourceLinks.data() + offset,
151 count);
152 }

380 template <typename OptColumn>
381 auto &accessImpl(OptColumn &&column) const {
382 assert(column.has_value() && "Column does not exist");
383 assert(m_index < column->size() && "Index out of bounds");
384 return column->proxy(*m_container)[m_index];
