40template <typename index_type, typename iterator_t>
41struct EnumerateIterator {
42 using iterator_category = std::input_iterator_tag;
43 using value_type = std::tuple<index_type, std::iter_value_t<iterator_t>>;
44 using reference = std::tuple<index_type, std::iter_reference_t<iterator_t>>;
45 using pointer = void;
46 using difference_type = std::iter_difference_t<iterator_t>;
47
48 index_type i;
49 iterator_t iter;
50
51 bool operator==(const EnumerateIterator &rhs) const {
52 return iter == rhs.iter;
53 }
54
56 EnumerateIterator &operator++() {
57 ++i;
58 ++iter;
59 return *this;
60 }
61
63 EnumerateIterator operator++(int) {
64 EnumerateIterator tmp = *this;
65 ++*this;
66 return tmp;
67 }
68
69 reference operator*() const { return reference{i, *iter}; }
70};

74template <std::ranges::range container_type, typename index_type,
75 typename iterator_t>
76struct EnumerateWrapper {
77 container_type iterable;
78 auto begin() {
79 return EnumerateIterator<index_type, iterator_t>{0, std::begin(iterable)};
80 }
81 auto end() {
82 return EnumerateIterator<index_type, iterator_t>{0, std::end(iterable)};
83 }
84};
