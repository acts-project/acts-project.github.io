229 [[nodiscard]] Result<void> initialize(
230 State& state, const NavigatorInitializeArguments& args) const {
231 ACTS_VERBOSE("Initialize. Surface sequence for navigation:");
232 for (const Surface* surface : state.options.surfaceSequence) {
233 ACTS_VERBOSE(surface->geometryId()
234 << " - "
235 << surface->center(state.options.geoContext).transpose());
236 }
237
238 state.direction = args.propagationDirection;
239 ACTS_VERBOSE("Navigation direction is " << args.propagationDirection);
240
241 state.startSurface = args.startSurface;
242 state.targetSurface = args.targetSurface;
243
244 // We set the current surface to the start surface
245 state.currentSurface = state.startSurface;
246 if (state.currentSurface != nullptr) {
247 ACTS_VERBOSE("Current surface set to start surface "
248 << state.currentSurface->geometryId());
249 } else {
250 ACTS_VERBOSE("Current surface set to nullptr");
251 }
252
253 // Find initial index.
254 auto found =
255 std::ranges::find(state.options.surfaceSequence, state.startSurface);
256
257 if (found != state.options.surfaceSequence.end()) {
258 // The index should be the index before the start surface, depending on
259 // the direction
260 state.surfaceIndex =
261 std::distance(state.options.surfaceSequence.begin(), found);
262 state.surfaceIndex += state.direction == Direction::Backward() ? 1 : -1;
263 } else {
264 ACTS_DEBUG(
265 "Did not find the start surface in the sequence. Assuming it is not "
266 "part of the sequence. Trusting the correctness of the input "
267 "sequence. Resetting the surface index.");
268 state.resetSurfaceIndex();
269 }
270
271 state.navigationBreak = false;
272
273 return Result<void>::success();
274 }

286 NavigationTarget nextTarget(State& state, const Vector3& position,
287 const Vector3& direction) const {
288 // Navigator target always resets the current surface
289 state.currentSurface = nullptr;
290
291 if (state.navigationBreak) {
292 return NavigationTarget::None();
293 }
294
295 ACTS_VERBOSE("DirectNavigator::nextTarget");
296
297 // Move the sequence to the next surface
298 state.nextSurface();
299
300 while (!state.endOfSurfaces()) {
301 ACTS_VERBOSE("Next surface candidate is "
302 << state.navSurface().geometryId() << ". "
303 << state.remainingSurfaces() << " out of "
304 << state.options.surfaceSequence.size()
305 << " surfaces remain to try.");
306
307 // Establish & update the surface status
308 // TODO we do not know the intersection index - passing the closer one
309 const Surface& surface = state.navSurface();
310 const double farLimit = std::numeric_limits<double>::max();
311 const NavigationTarget target = chooseIntersection(
312 state.options.geoContext, surface, position, direction,
313 BoundaryTolerance::Infinite(), state.options.nearLimit, farLimit,
314 state.options.surfaceTolerance);
315 if (target.isValid()) {
316 return target;
317 }
318
319 ACTS_VERBOSE("No valid intersection found with surface "
320 << surface.geometryId() << ", trying next surface.");
321 state.nextSurface();
322 }
323
324 ACTS_VERBOSE("End of surfaces reached, navigation break.");
325 state.navigationBreak = true;
326 return NavigationTarget::None();
327 }

376 private:
377 NavigationTarget chooseIntersection(
378 const GeometryContext& gctx, const Surface& surface,
379 const Vector3& position, const Vector3& direction,
380 const BoundaryTolerance& boundaryTolerance, double nearLimit,
381 double farLimit, double tolerance) const {
382 auto intersections = surface.intersect(gctx, position, direction,
383 boundaryTolerance, tolerance);
384
385 for (auto [intersectionIndex, intersection] :
386 Acts::enumerate(intersections)) {
387 if (intersection.isValid() &&
388 detail::checkPathLength(intersection.pathLength(), nearLimit,
389 farLimit, logger())) {
390 return NavigationTarget(intersection, intersectionIndex, surface,
391 boundaryTolerance);
392 }
393 }
394
395 return NavigationTarget::None();
396 }
