111 template <bool EnsureConst, typename T>
112 static std::any component_impl(T& instance, Acts::HashedString key,
113 IndexType itrack) {
114 if constexpr (EnsureConst) {
115 static_assert(std::is_const_v<std::remove_reference_t<T>>,
116 "Is not const");
117 }
118
119 using namespace Acts::HashedStringLiteral;
120 auto track = instance.m_collection->at(itrack);
121 std::conditional_t<EnsureConst, const ActsPodioEdm::TrackInfo*,
122 ActsPodioEdm::TrackInfo*>
123 dataPtr;
124 if constexpr (EnsureConst) {
125 dataPtr = &track.getData();
126 } else {
127 dataPtr = &PodioUtil::getDataMutable(track);
128 }
129 auto& data = *dataPtr;
130 switch (key) {
131 case "tipIndex"_hash:
132 return &data.tipIndex;
133 case "stemIndex"_hash:
134 return &data.stemIndex;
135 case "params"_hash:
136 return data.parameters.data();
137 case "cov"_hash:
138 return data.covariance.data();
139 case "nMeasurements"_hash:
140 return &data.nMeasurements;
141 case "nHoles"_hash:
142 return &data.nHoles;
143 case "chi2"_hash:
144 return &data.chi2;
145 case "ndf"_hash:
146 return &data.ndf;
147 case "nOutliers"_hash:
148 return &data.nOutliers;
149 case "nSharedHits"_hash:
150 return &data.nSharedHits;
151 default:
152 auto it = instance.m_dynamic.find(key);
153 if (it == instance.m_dynamic.end()) {
154 throw std::runtime_error("Unable to handle this component");
155 }
156
157 std::conditional_t<EnsureConst,
158 const podio_detail::ConstDynamicColumnBase*,
159 podio_detail::DynamicColumnBase*>
160 col = it->second.get();
161 assert(col && "Dynamic column is null");
162 return col->get(itrack);
163 }
164 }
165
169 template <typename T>
170 static auto dynamicKeys_impl(T& instance) {
171 using column_type =
172 typename decltype(instance.m_dynamic)::mapped_type::element_type;
173 return Acts::detail::DynamicKeyRange<column_type>{
174 instance.m_dynamic.begin(), instance.m_dynamic.end()};
175 }

181 template <typename T>
182 static Acts::ParticleHypothesis particleHypothesis_impl(T& instance,
183 IndexType itrack) {
184 auto track = instance.m_collection->at(itrack);
185 const auto& src = track.getParticleHypothesis();
186 return Acts::ParticleHypothesis{static_cast<Acts::PdgParticle>(src.absPdg),
187 src.mass, Acts::ChargeHypothesis{src.absQ}};
188 }
189
194 static void populateSurfaceBuffer(
195 const PodioUtil::ConversionHelper& helper,
196 const ActsPodioEdm::TrackCollection& collection,
197 std::vector<std::shared_ptr<const Acts::Surface>>& surfaces) noexcept {
198 surfaces.reserve(collection.size());
199 for (ActsPodioEdm::Track track : collection) {
200 surfaces.push_back(PodioUtil::convertSurfaceFromPodio(
201 helper, track.getReferenceSurface()));
202 }
203 }
204
206 std::reference_wrapper<const PodioUtil::ConversionHelper> m_helper;
208 std::vector<std::shared_ptr<const Acts::Surface>> m_surfaces;
209};

418 void releaseInto(podio::Frame& frame, const std::string& suffix = "")
419 requires(std::is_same_v<holder_t<ActsPodioEdm::TrackCollection>,
420 Acts::ValueHolder<ActsPodioEdm::TrackCollection>> ||
421 std::is_same_v<holder_t<ActsPodioEdm::TrackCollection>,
422 std::unique_ptr<ActsPodioEdm::TrackCollection>>)
423 {
424 std::string s = suffix;
425 if (!s.empty()) {
426 s = "_" + s;
427 }
428
429 if constexpr (std::is_same_v<
430 holder_t<ActsPodioEdm::TrackCollection>,
431 std::unique_ptr<ActsPodioEdm::TrackCollection>>) {
432 frame.put(std::move(m_collection), "tracks" + s);
433 } else {
434 frame.put(std::make_unique<ActsPodioEdm::TrackCollection>(
435 std::move(*m_collection)),
436 "tracks" + s);
437 }
438 m_surfaces.clear();
439
440 for (const auto& [key, col] : m_dynamic) {
441 col->releaseInto(frame, "tracks" + s + "_extra__");
442 }
443 }

505 ConstPodioTrackContainer(const PodioUtil::ConversionHelper& helper,
506 const podio::Frame& frame,
507 const std::string& suffix = "")

509 holder_t<const ActsPodioEdm::TrackCollection>,
510 Acts::ConstRefHolder<const ActsPodioEdm::TrackCollection>>

517 podio_detail::recoverDynamicColumns(frame, tracksKey, m_dynamic);
