148 NeighborHoodIndices neighborHoodIndices(std::size_t idx,
149 std::pair<int, int> sizes = {-1,
150 1}) const
151 requires(bdt == AxisBoundaryType::Closed)
152 {
153 // Handle invalid indices
154 if (idx <= 0 || idx >= (getNBins() + 1)) {
155 return NeighborHoodIndices();
156 }
157
158 // Handle corner case where user requests more neighbours than the number
159 // of bins on the axis. All bins are returned in this case.
160
161 const int max = getNBins();
162 sizes.first = std::clamp(sizes.first, -max, max);
163 sizes.second = std::clamp(sizes.second, -max, max);
164 if (std::abs(sizes.first - sizes.second) >= max) {
165 sizes.first = 1 - idx;
166 sizes.second = max - idx;
167 }
168
169 // If the entire index range is not covered, we must wrap the range of
170 // targeted neighbor indices into the range of valid bin indices. This may
171 // split the range of neighbor indices in two parts:
172 //
173 // Before wraparound - [ XXXXX]XXX
174 // After wraparound - [ XXXX XXXX ]
175 //
176 const int itmin = idx + sizes.first;
177 const int itmax = idx + sizes.second;
178 const std::size_t itfirst = wrapBin(itmin);
179 const std::size_t itlast = wrapBin(itmax);
180 if (itfirst <= itlast) {
181 return NeighborHoodIndices(itfirst, itlast + 1);
182 } else {
183 return NeighborHoodIndices(itfirst, max + 1, 1, itlast + 1);
184 }
185 }

292 std::vector<double> binEdges;
293 for (std::size_t i = 1; i <= m_bins; i++) {
294 binEdges.push_back(getBinLowerBound(i));

296 binEdges.push_back(getBinUpperBound(m_bins));
