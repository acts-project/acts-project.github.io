66 static MultiComponentBoundTrackParameters createCurvilinear(
67 const GeometryContext& geoCtx,
68 const std::vector<ConstructionTuple>& curvi,
69 ParticleHypothesis particleHypothesis) {
70 // Construct and average surface
71 Vector3 avgPos = Vector3::Zero();
72 Vector3 avgDir = Vector3::Zero();
73 for (const auto& [w, pos4, dir, qop, cov] : curvi) {
74 avgPos += w * pos4.template segment<3>(0);
75 avgDir += w * dir;
76 }
77
78 std::shared_ptr<PlaneSurface> s =
79 CurvilinearSurface(avgPos, avgDir).planeSurface();
80
81 std::vector<Component> bound;
82 bound.reserve(curvi.size());
83
84 // Project the position onto the surface, keep everything else as is
85 for (const auto& [w, pos4, dir, qop, cov] : curvi) {
86 const Intersection3D closestIntersection =
87 s->intersect(geoCtx, pos4.template segment<3>(eFreePos0), dir,
88 BoundaryTolerance::Infinite())
89 .closest();
90 const Vector3& newPos = closestIntersection.position();
91
92 BoundVector bv =
93 transformFreeToCurvilinearParameters(pos4[eTime], dir, qop);
94
95 // Because of the projection this should never fail
96 bv.template segment<2>(eBoundLoc0) =
97 *(s->globalToLocal(geoCtx, newPos, dir));
98
99 bound.emplace_back(w, bv, cov);
100 }
101
102 return MultiComponentBoundTrackParameters(
103 s, bound,
104 [](const Component& cmp) {
105 return std::tie(std::get<0>(cmp), std::get<1>(cmp),
106 *std::get<2>(cmp));
107 },
108 particleHypothesis);
109 }

146 template <std::ranges::range component_range_t, typename projector_t>
147 MultiComponentBoundTrackParameters(std::shared_ptr<const Surface> surface,
148 const component_range_t& cmps,
149 const projector_t& proj,
150 ParticleHypothesis particleHypothesis)
151 requires detail::ComponentRangeAndProjectorWithoutCovarianceConcept<
152 component_range_t, projector_t>
153 : m_surface(std::move(surface)),
154 m_particleHypothesis(particleHypothesis) {
155 if (cmps.begin() == cmps.end()) {
156 throw std::invalid_argument(
157 "Cannot construct MultiComponentBoundTrackParameters without "
158 "components");
159 }
160
161 for (const auto& cmp : cmps) {
162 const auto& [weight, parameters] = proj(cmp);
163 m_weights.push_back(weight);
164 m_parameters.push_back(parameters);
165 }
166 }

173 template <std::ranges::range component_range_t, typename projector_t>
174 MultiComponentBoundTrackParameters(std::shared_ptr<const Surface> surface,
175 const component_range_t& cmps,
176 const projector_t& proj,
177 ParticleHypothesis particleHypothesis)
178 requires detail::ComponentRangeAndProjectorWithCovarianceConcept<
179 component_range_t, projector_t>
180 : m_surface(std::move(surface)),
181 m_particleHypothesis(particleHypothesis),
182 m_hasCovariance(true) {
183 if (cmps.begin() == cmps.end()) {
184 throw std::invalid_argument(
185 "Cannot construct MultiComponentBoundTrackParameters without "
186 "components");
187 }
188
189 for (const auto& cmp : cmps) {
190 const auto& [weight, parameters, covariance] = proj(cmp);
191 m_weights.push_back(weight);
192 m_parameters.push_back(parameters);
193 m_covariances.push_back(covariance);
194 }
195 }

262 std::vector<Component> toComponents() const {
263 std::vector<Component> cmps;
264 cmps.reserve(size());
265 for (std::size_t i = 0; i < size(); ++i) {
266 cmps.emplace_back(
267 m_weights[i], m_parameters[i],
268 hasCovariance() ? std::optional(m_covariances[i]) : std::nullopt);
