151template <std::size_t SbSize, bool Copyable = true, typename Base = void>
152class AnyBase : public AnyBaseAll {
153 static_assert(sizeof(void*) <= SbSize, "Size is too small for a pointer");

160 template <typename U>
161 static constexpr bool isStorable() {
162 if constexpr (std::is_base_of_v<AnyBaseAll, U>) {
163 return false;
164 } else if constexpr (!std::is_void_v<Base> &&
165 !std::is_convertible_v<U*, Base*>) {
166 return false;
167 } else if constexpr (Copyable) {
168 return std::is_copy_assignable_v<U> && std::is_copy_constructible_v<U> &&
169 (sizeof(U) > SbSize || (std::is_move_assignable_v<U> &&
170 std::is_move_constructible_v<U>));
171 } else {
172 return std::is_move_constructible_v<U> &&
173 (sizeof(U) > SbSize || std::is_move_assignable_v<U>);
174 }
175 }

182 template <typename T, typename... Args>

184 explicit AnyBase(std::in_place_type_t<T> /*unused*/, Args&&... args) {
185 using U = std::decay_t<T>;
186 m_handler = makeHandler<U>();
187 constructValue<U>(std::forward<Args>(args)...);

202 template <typename T>
203 explicit AnyBase(T&& value)
204 requires(!std::is_base_of_v<AnyBaseAll, std::decay_t<T>> &&
205 isStorable<std::decay_t<T>>())
206 : AnyBase{std::in_place_type<T>, std::forward<T>(value)} {}

213 template <typename T, typename... Args>
214 requires(isStorable<std::decay_t<T>>())
215 T& emplace(Args&&... args) {
216 using U = std::decay_t<T>;
217 destroy();
218 // Construct the new value before installing the handler. constructValue
219 // does not depend on m_handler, so if the constructor throws this object is
220 // left empty (m_handler == nullptr) rather than claiming to hold a value
221 // whose storage was never set up, which would make the next destroy()
222 // operate on stale/freed memory.
223 U* ptr = constructValue<U>(std::forward<Args>(args)...);
224 m_handler = makeHandler<U>();
225 return *ptr;
226 }

232 template <typename T>
233 T& as() {
234 static_assert(std::is_same_v<T, std::decay_t<T>>,
235 "Please pass the raw type, no const or ref");
236 if (m_handler == nullptr || m_handler->typeHash != typeHash<T>()) {
237 throw std::bad_any_cast{};
238 }
239
240 _ACTS_ANY_VERBOSE("Get as "
241 << (m_handler->heapAllocated ? "heap" : "local"));
242
243 return *std::bit_cast<T*>(dataPtr());
244 }

315 AnyBase(const AnyBase& other)
316 requires Copyable
317 {
318 if (m_handler == nullptr && other.m_handler == nullptr) {
319 // both are empty, noop
320 return;
321 }
322
323 _ACTS_ANY_VERBOSE("Copy construct (this="
324 << this << ") at: " << static_cast<void*>(m_data.data()));
325
326 m_handler = other.m_handler;
327 copyConstruct(other);
328 }

340 AnyBase& operator=(const AnyBase& other)
341 requires Copyable
342 {
343 _ACTS_ANY_VERBOSE("Copy assign (this="
344 << this << ") at: " << static_cast<void*>(m_data.data()));
345
346 if (this == &other) {
347 // self-assignment, noop
348 return *this;
349 }
350
351 if (m_handler == nullptr && other.m_handler == nullptr) {
352 // both are empty, noop
353 return *this;
354 }
355
356 if (m_handler == other.m_handler) {
357 // same type, but checked before they're not both nullptr
358 copy(other);
359 } else {
360 if (m_handler != nullptr) {
361 // this object is not empty, but have different types => destroy
362 destroy();
363 }
364 assert(m_handler == nullptr);
365 m_handler = other.m_handler;
366 try {
367 copyConstruct(other);
368 } catch (...) {
369 // copyConstruct allocates and/or runs the stored type's copy
370 // constructor, either of which may throw. If it does, no value was
371 // established, so leave *this empty rather than with a handler that
372 // claims a value that does not exist (which would make destruction
373 // operate on uninitialized storage).
374 m_handler = nullptr;
375 throw;
376 }
377 }
378 return *this;
379 }

388 AnyBase(AnyBase&& other) noexcept(detail::kAnyNoexcept) {

390 << this << ") at: " << static_cast<void*>(m_data.data()));
391 if (m_handler == nullptr && other.m_handler == nullptr) {

403 AnyBase& operator=(AnyBase&& other) noexcept(detail::kAnyNoexcept) {
404 _ACTS_ANY_VERBOSE("Move assign (this="
405 << this << ") at: " << static_cast<void*>(m_data.data()));
406 if (this == &other) {
407 // self-assignment, noop (avoids destroying our own value)
408 return *this;
409 }
410
411 if (m_handler == nullptr && other.m_handler == nullptr) {
412 // both are empty, noop
413 return *this;
414 }
415
416 // At this point they can't be equal and nullptr, so it's safe to
417 // dereference
418 if (m_handler == other.m_handler &&
419 m_handler->typeHash == other.m_handler->typeHash) {
420 // same type, but checked before they're not both nullptr
421 move(std::move(other));
422 } else {
423 if (m_handler != nullptr) {
424 // this object is not empty, but have different types => destroy
425 destroy();
426 }
427 assert(m_handler == nullptr);
428 m_handler = other.m_handler;
429 moveConstruct(std::move(other));
430 }
431
432 return *this;
433 }

445 // The base accessors below are member templates on a dummy @c B defaulting to
446 // @c Base. This keeps their return types (@c B& / @c B*) from being formed
447 // when @c Base is @c void: the member template is only instantiated on use,
448 // and the @c requires clause removes it entirely for the untyped variant. A
449 // plain non-template member would try to form @c void& at class instantiation
450 // and fail to compile.
451
455 template <typename B = Base>
456 requires(!std::is_void_v<B>)
457 B* asBase() {
458 if (m_handler == nullptr) {
459 return nullptr;
460 }
461 return m_handler->upcast(dataPtr());
462 }

535 struct Handler {
536 // Maps the stored payload to Base*. Collapses to void*(*)(void*) and stays
537 // nullptr (never read) when Base == void.
538 Base* (*upcast)(void*) = nullptr;
539 // Const-qualified counterpart used by the const accessors so they don't
540 // have to cast away const on the data pointer. Same as above when
541 // Base == void.
542 const Base* (*upcastConst)(const void*) = nullptr;
543 void (*destroy)(void* ptr) = nullptr;
544 void (*moveConstruct)(void* from, void* to) = nullptr;
545 void (*move)(void* from, void* to) = nullptr;
546 void* (*copyConstruct)(const void* from, void* to) = nullptr;
547 void (*copy)(const void* from, void* to) = nullptr;
548 bool heapAllocated{false};
549 std::uint64_t typeHash{0};
550 const std::type_info* typeInfo{nullptr};
551 };
552
553 template <typename T>
554 static const Handler* makeHandler() {
555 static_assert(!std::is_base_of_v<AnyBaseAll, std::decay_t<T>>,
556 "Cannot wrap Any in Any");
557 static const Handler static_handler = []() {
558 Handler h;
559 h.heapAllocated = heapAllocated<T>();
560 if constexpr (!std::is_trivially_destructible_v<T> ||
561 heapAllocated<T>()) {
562 h.destroy = &destroyImpl<T>;
563 }
564 if constexpr (!heapAllocated<T>() &&
565 !std::is_trivially_move_constructible_v<T>) {
566 h.moveConstruct = &moveConstructImpl<T>;
567 }
568 if constexpr (!heapAllocated<T>() &&
569 !std::is_trivially_move_assignable_v<T>) {
570 h.move = &moveImpl<T>;
571 }
572 if constexpr (std::is_copy_constructible_v<T> &&
573 (!std::is_trivially_copy_constructible_v<T> ||
574 heapAllocated<T>())) {
575 h.copyConstruct = &copyConstructImpl<T>;
576 }
577
578 if constexpr (std::is_copy_assignable_v<T> &&
579 (!std::is_trivially_copy_assignable_v<T> ||
580 heapAllocated<T>())) {
581 h.copy = &copyImpl<T>;
582 }
583
584 if constexpr (!std::is_void_v<Base>) {
585 h.upcast = [](void* p) -> Base* {
586 return static_cast<Base*>(static_cast<T*>(p));
587 };
588 h.upcastConst = [](const void* p) -> const Base* {
589 return static_cast<const Base*>(static_cast<const T*>(p));
590 };
591 }
592
593 h.typeHash = typeHash<T>();
594 h.typeInfo = &typeid(T);
595
596 _ACTS_ANY_DEBUG("Type: " << typeid(T).name());
597 _ACTS_ANY_DEBUG(" -> destroy: " << h.destroy);
598 _ACTS_ANY_DEBUG(" -> moveConstruct: " << h.moveConstruct);
599 _ACTS_ANY_DEBUG(" -> move: " << h.move);
600 _ACTS_ANY_DEBUG(" -> copyConstruct: " << h.copyConstruct);
601 _ACTS_ANY_DEBUG(" -> copy: " << h.copy);
602 _ACTS_ANY_DEBUG(
603 " -> heapAllocated: " << (h.heapAllocated ? "yes" : "no"));
604
605 return h;
606 }();
607 return &static_handler;
608 }

610 template <typename T>
611 static constexpr bool heapAllocated() {
612 return sizeof(T) > SbSize;
613 }

615 template <typename T, typename... Args>
616 T* constructValue(Args&&... args) {
617 if constexpr (!heapAllocated<T>()) {
618 // construct into local buffer
619 auto* ptr = new (m_data.data()) T(std::forward<Args>(args)...);
620 _ACTS_ANY_VERBOSE("Construct local (this="
621 << this
622 << ") at: " << static_cast<void*>(m_data.data()));
623 return ptr;
624 } else {
625 // too large, heap allocate
626 auto* heap = new T(std::forward<Args>(args)...);
627 _ACTS_ANY_DEBUG("Allocate type: " << typeid(T).name() << " at " << heap);
628 _ACTS_ANY_TRACK_ALLOCATION(T, heap);
629 setDataPtr(heap);
630 return heap;
631 }
632 }

634 void destroy() {
635 _ACTS_ANY_VERBOSE("Destructor this=" << this << " handler: " << m_handler);
636 if (m_handler != nullptr && m_handler->destroy != nullptr) {
637 _ACTS_ANY_VERBOSE("Non-trivial destruction");
638 m_handler->destroy(dataPtr());
639 }
640 m_handler = nullptr;
641 }
642
643 void moveConstruct(AnyBase&& fromAny) {
644 if (m_handler == nullptr) {
645 return;
646 }
647
648 void* to = dataPtr();
649 void* from = fromAny.dataPtr();
650 if (m_handler->heapAllocated) {
651 // stored on heap: just copy the pointer
652 setDataPtr(fromAny.dataPtr());
653 // do not delete in moved-from any
654 fromAny.m_handler = nullptr;
655 return;
656 }
657
658 if (m_handler->moveConstruct == nullptr) {
659 _ACTS_ANY_VERBOSE("Trivially move construct");
660 // trivially move constructible
661 m_data = std::move(fromAny.m_data);
662 } else {
663 m_handler->moveConstruct(from, to);
664 }
665 }
666
667 void move(AnyBase&& fromAny) {
668 if (m_handler == nullptr) {
669 return;
670 }
671
672 void* to = dataPtr();
673 void* from = fromAny.dataPtr();
674 if (m_handler->heapAllocated) {
675 // stored on heap: just copy the pointer
676 // need to delete existing pointer
677 m_handler->destroy(dataPtr());
678 setDataPtr(fromAny.dataPtr());
679 // do not delete in moved-from any
680 fromAny.m_handler = nullptr;
681 return;
682 }
683
684 if (m_handler->move == nullptr) {
685 _ACTS_ANY_VERBOSE("Trivially move");
686 // trivially movable
687 m_data = std::move(fromAny.m_data);
688 } else {
689 m_handler->move(from, to);
690 }
691 }
692
693 void copyConstruct(const AnyBase& fromAny) {
694 if (m_handler == nullptr) {
695 return;
696 }
697
698 const void* from = fromAny.dataPtr();
699
700 if (m_handler->copyConstruct == nullptr) {
701 _ACTS_ANY_VERBOSE("Trivially copy construct");
702 // trivially copy constructible
703 m_data = fromAny.m_data;
704 } else if (m_handler->heapAllocated) {
705 // heap-allocated: always allocate fresh storage. We must not read the
706 // current contents of m_data here: when copyConstruct runs as part of a
707 // copy assignment with a type change, the previous value has already been
708 // destroyed but m_data still holds its stale bytes (a freed pointer, or
709 // the previous local value's bytes). Passing those as the destination
710 // would placement-new into garbage. Allocate and store the new pointer.
711 void* copyAt = m_handler->copyConstruct(from, nullptr);
712 assert(copyAt != nullptr);
713 setDataPtr(copyAt);
714 } else {
715 // local storage: construct into the internal buffer
716 m_handler->copyConstruct(from, dataPtr());
717 }
718 }
719
720 void copy(const AnyBase& fromAny) {
721 if (m_handler == nullptr) {
722 return;
723 }
724
725 void* to = dataPtr();
726 const void* from = fromAny.dataPtr();
727
728 if (m_handler->copy == nullptr) {
729 _ACTS_ANY_VERBOSE("Trivially copy");
730 // trivially copyable
731 m_data = fromAny.m_data;
732 } else {
733 m_handler->copy(from, to);
734 }
735 }

737 template <typename T>
738 static void destroyImpl(void* ptr) {
739 assert(ptr != nullptr && "Address to destroy is nullptr");
740 auto* obj = static_cast<T*>(ptr);
741 if constexpr (!heapAllocated<T>()) {
742 // stored in place: just call the destructor
743 _ACTS_ANY_VERBOSE("Destroy local at: " << ptr);
744 obj->~T();
745 } else {
746 // stored on heap: delete
747 _ACTS_ANY_DEBUG("Delete type: " << typeid(T).name()
748 << " heap at: " << obj);
749 _ACTS_ANY_TRACK_DEALLOCATION(T, obj);
750 delete obj;
751 }
752 }

754 template <typename T>
755 static void moveConstructImpl(void* from, void* to) {
756 _ACTS_ANY_VERBOSE("move const: " << from << " -> " << to);
757 assert(from != nullptr && "Source is null");
758 assert(to != nullptr && "Target is null");
759 auto* _from = static_cast<T*>(from);
760 /*T* ptr =*/new (to) T(std::move(*_from));
761 }
762
763 template <typename T>
764 static void moveImpl(void* from, void* to) {
765 _ACTS_ANY_VERBOSE("move: " << from << " -> " << to);
766 assert(from != nullptr && "Source is null");
767 assert(to != nullptr && "Target is null");
768
769 auto* _from = static_cast<T*>(from);
770 auto* _to = static_cast<T*>(to);
771
772 (*_to) = std::move(*_from);
773 }
774
775 template <typename T>
776 static void* copyConstructImpl(const void* from, void* to) {
777 _ACTS_ANY_VERBOSE("copy const: " << from << " -> " << to);
778 assert(from != nullptr && "Source is null");
779 const auto* _from = static_cast<const T*>(from);
780 if (to == nullptr) {
781 assert(heapAllocated<T>() && "Received nullptr in local buffer case");
782 to = new T(*_from);
783 _ACTS_ANY_TRACK_ALLOCATION(T, to);
784
785 } else {
786 assert(!heapAllocated<T>() && "Received non-nullptr in heap case");
787 /*T* ptr =*/new (to) T(*_from);
788 }
789 return to;
790 }
791
792 template <typename T>
793 static void copyImpl(const void* from, void* to) {
794 _ACTS_ANY_VERBOSE("copy: " << from << " -> " << to);
795 assert(from != nullptr && "Source is null");
796 assert(to != nullptr && "Target is null");
797
798 const auto* _from = static_cast<const T*>(from);
799 auto* _to = static_cast<T*>(to);
800
801 (*_to) = *_from;
802 }
