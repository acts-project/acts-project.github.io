41struct BoundParameterVariationGenerator {
42 virtual ~BoundParameterVariationGenerator() = default;
48 virtual BoundParameterVariation variationMap(
49 const BoundVector& input, const BoundMatrix& covariance) const = 0;
50};
51
53struct DeltaBoundParameterVariationGenerator
54 : public BoundParameterVariationGenerator {
56 std::vector<BoundVector> deltas;
57
60 explicit DeltaBoundParameterVariationGenerator(const double delta)
61 : deltas(1, BoundVector::Constant(delta)) {}
62
65 explicit DeltaBoundParameterVariationGenerator(
66 const std::vector<double>& deltas_) {
67 this->deltas.reserve(deltas_.size());
68 for (const auto& delta : deltas_) {
69 this->deltas.emplace_back(BoundVector::Constant(delta));
70 }
71 }
72
75 explicit DeltaBoundParameterVariationGenerator(const BoundVector& delta)
76 : deltas(1, delta) {}
77
80 explicit DeltaBoundParameterVariationGenerator(
81 std::vector<BoundVector> deltas_)
82 : deltas(std::move(deltas_)) {}
83
86 BoundParameterVariation variationMap(
87 const BoundVector& /*input*/,
88 const BoundMatrix& /*covariance*/) const override {
89 BoundParameterVariation variationMap;
90 variationMap.reserve(deltas.size() * eBoundSize);
91 for (std::size_t i = 0; i < deltas.size(); ++i) {
92 for (std::size_t j = 0; j < eBoundSize; ++j) {
93 variationMap.emplace_back(j, deltas[i][j]);
94 }
95 }
96 return variationMap;
97 }
98};

101struct CovarianceBoundParameterVariationGenerator
102 : public BoundParameterVariationGenerator {
104 std::vector<BoundVector> sigmaFactors;
105
108 explicit CovarianceBoundParameterVariationGenerator(const double sigmaFactor)
109 : sigmaFactors(1, BoundVector::Constant(sigmaFactor)) {}
110
114 explicit CovarianceBoundParameterVariationGenerator(
115 const std::vector<double>& sigmaFactors_) {
116 sigmaFactors.reserve(sigmaFactors_.size());
117 for (const auto& factor : sigmaFactors_) {
118 sigmaFactors.emplace_back(BoundVector::Constant(factor));
119 }
120 }
121
124 explicit CovarianceBoundParameterVariationGenerator(
125 const BoundVector& sigmaFactor)
126 : sigmaFactors(1, sigmaFactor) {}
127
130 explicit CovarianceBoundParameterVariationGenerator(
131 std::vector<BoundVector> sigmaFactors_)
132 : sigmaFactors(std::move(sigmaFactors_)) {}
133
137 BoundParameterVariation variationMap(
138 const BoundVector& /*input*/,
139 const BoundMatrix& covariance) const override {
140 BoundParameterVariation variationMap;
141 variationMap.reserve(sigmaFactors.size() * eBoundSize);
142 for (std::size_t i = 0; i < sigmaFactors.size(); ++i) {
143 for (std::size_t j = 0; j < eBoundSize; ++j) {
144 variationMap.emplace_back(
145 j, sigmaFactors[i][j] * std::sqrt(covariance(j, j)));
146 }
147 }
148 return variationMap;
149 }
150};

360 IntersectionStatus updateSurfaceStatus(
361 State& state, const Surface& surface, std::uint8_t index,
362 Direction navDir, const BoundaryTolerance& boundaryTolerance,
363 double surfaceTolerance, ConstrainedStep::Type stype,
364 const Logger& logger = getDummyLogger()) const {
365 state.lastSurfaceTolerance = surfaceTolerance;
366 state.lastStepConstraintType = stype;
367
368 IntersectionStatus overallStatus = m_stepperImpl.updateSurfaceStatus(
369 state.primaryStepperState, surface, index, navDir, boundaryTolerance,
370 surfaceTolerance, stype, logger);
371 if (overallStatus == IntersectionStatus::unreachable) {
372 return overallStatus;
373 }
374
375 const auto combine = [](const IntersectionStatus a,
376 const IntersectionStatus b) {
377 using enum IntersectionStatus;
378 if ((a == unreachable) || (b == unreachable)) {
379 return unreachable;
380 }
381 if ((a == reachable) || (b == reachable)) {
382 return reachable;
383 }
384 return onSurface;
385 };
386
387 for (auto& secondaryState : state.secondaryStepperStates) {
388 const IntersectionStatus status = m_stepperImpl.updateSurfaceStatus(
389 secondaryState, surface, index, navDir, BoundaryTolerance::Infinite(),
390 surfaceTolerance, stype, logger);
391 overallStatus = combine(overallStatus, status);
392 }
393
394 return overallStatus;
395 }

564 Result<Jacobian> transportToBound(
565 State& state, const Surface& surface,
566 [[maybe_unused]] const FreeToBoundCorrection& freeToBoundCorrection =
567 FreeToBoundCorrection(false)) const {
568 if (!state.cov.has_value()) {
569 return Result<Jacobian>::success(Jacobian::Identity());
570 }
571
572 Result<BoundParameters> primaryResult =
573 m_stepperImpl.boundParameters(state.primaryStepperState, surface);
574 if (!primaryResult.ok()) {
575 return Result<Jacobian>::failure(primaryResult.error());
576 }
577
578 std::vector<BoundVector> boundVectors;
579 boundVectors.reserve(state.secondaryStepperStates.size());
580 for (const auto& secondaryState : state.secondaryStepperStates) {
581 const Result<BoundParameters> result =
582 m_stepperImpl.boundParameters(secondaryState, surface);
583 if (!result.ok()) {
584 return Result<Jacobian>::failure(result.error());
585 }
586 boundVectors.push_back(result->parameters());
587 }
588 const BoundVector& referenceVector = primaryResult->parameters();
589
590 Jacobian jacobian = Jacobian::Zero();
591 std::array<std::size_t, eBoundSize> numberOfVariationsPerParameter{};
592
593 for (std::size_t i = 0; i < state.variationMap.size(); ++i) {
594 const auto& [index, delta] = state.variationMap[i];
595 const auto& nudgedVector = boundVectors[i];
596
597 const auto diff = [&]() {
598 BoundVector d = nudgedVector - referenceVector;
599 d[eBoundPhi] = detail::difference_periodic(nudgedVector[eBoundPhi],
600 referenceVector[eBoundPhi],
601 2 * std::numbers::pi);
602 return d;
603 };
604
605 jacobian.col(index) += diff() / delta;
606 ++numberOfVariationsPerParameter[index];
607 }
608 for (std::size_t i = 0; i < eBoundSize; ++i) {
609 jacobian.col(i) /= numberOfVariationsPerParameter[i];
610 }
611
612 *state.cov = jacobian * *state.cov * jacobian.transpose();
613
614 // The secondary states restart on the surface, so the next jacobian starts
615 // here as well
616 resetSecondaryStates(state, referenceVector, surface);
617 for (auto& secondaryState : state.secondaryStepperStates) {
618 secondaryState.stepSize = state.primaryStepperState.stepSize;
619 }
620
621 return Result<Jacobian>::success(jacobian);
622 }
623
631 Jacobian transportToCurvilinear(State& state) const {
632 if (!state.cov.has_value()) {
633 return Jacobian::Identity();
634 }
635
636 const auto curvilinearSurface =
637 CurvilinearSurface(position(state), direction(state)).surface();
638
639 for (std::size_t i = 0; i < m_config.maxStepsToCurvilinearSurface; ++i) {
640 const auto surfaceStatus = updateSurfaceStatus(
641 state, *curvilinearSurface, 0, state.lastStepPropagationDirection,
642 BoundaryTolerance::Infinite(), state.lastSurfaceTolerance,
643 state.lastStepConstraintType);
644 if (surfaceStatus == IntersectionStatus::onSurface) {
645 break;
646 }
647 if (surfaceStatus == IntersectionStatus::unreachable) {
648 throw std::runtime_error(
649 "RiddersStepper: Curvilinear surface is unreachable");
650 }
651 const auto stepResult = step(state, state.lastStepPropagationDirection,
652 state.lastStepMaterial);
653 if (!stepResult.ok()) {
654 throw std::runtime_error(
655 "RiddersStepper: Failed to step towards curvilinear surface: " +
656 stepResult.error().message());
657 }
658 }
659
660 // the surface is built at the current position, so this cannot fail
661 return transportToBound(state, *curvilinearSurface).value();
662 }

700 Result<double> step(State& state, Direction propagationDirection,
701 const IVolumeMaterial* material) const {
702 state.lastStepPropagationDirection = propagationDirection;
703 state.lastStepMaterial = material;
704
705 const Result<double> stepResult = m_stepperImpl.step(
706 state.primaryStepperState, propagationDirection, material);
707 if (!stepResult.ok()) {
708 return stepResult.error();
709 }
710 for (auto& secondaryState : state.secondaryStepperStates) {
711 const Result<double> secondaryStepResult =
712 m_stepperImpl.step(secondaryState, propagationDirection, material);
713 if (!secondaryStepResult.ok()) {
714 return secondaryStepResult.error();
715 }
716 }
717 state.pathAccumulated += *stepResult;
718 return *stepResult;
719 }
720
721 private:
726 void resetSecondaryStates(State& state, const BoundVector& boundVector,
727 const Surface& surface) const {
728 state.variationMap =
729 m_config.parameterVariation->variationMap(boundVector, *state.cov);
730
731 state.secondaryStepperStates.clear();
732 for (const auto& [index, delta] : state.variationMap) {
733 BoundVector nudgedParams = boundVector;
734 nudgedParams[index] += delta;
735
736 state.secondaryStepperStates.push_back(
737 m_stepperImpl.makeState(state.primaryStepperState.options));
738 m_stepperImpl.initialize(state.secondaryStepperStates.back(),
739 nudgedParams, std::nullopt,
740 particleHypothesis(state), surface);
741 }
742 }
