73template <typename Derived>
74double perp(const Eigen::MatrixBase<Derived>& v) noexcept {
75 constexpr int rows = Eigen::MatrixBase<Derived>::RowsAtCompileTime;
76 if constexpr (rows != -1) {
77 // static size, do compile time check
78 static_assert(rows >= 2,
79 "Perp function not valid for vectors not at least 2D");
80 } else {
81 // dynamic size
82 assert(v.rows() >= 2 &&
83 "Perp function not valid for vectors not at least 2D");
84 }
85 return v.template head<2>().norm();
86}

118template <typename Derived>
119double eta(const Eigen::MatrixBase<Derived>& v) noexcept
120 requires(Eigen::MatrixBase<Derived>::RowsAtCompileTime == 3)
121{
122 if (v[0] == 0. && v[1] == 0.) {
123 return std::copysign(std::numeric_limits<double>::infinity(), v[2]);
124 } else {
125 return std::asinh(v[2] / perp(v));
126 }
127}

165inline std::array<double, 4> evaluateTrigonomics(const Vector3& direction) {
166 const double x = direction(0); // == cos(phi) * sin(theta)
167 const double y = direction(1); // == sin(phi) * sin(theta)
168 const double z = direction(2); // == cos(theta)
169 // can be turned into cosine/sine
170 const double cosTheta = z;
171 const double sinTheta = fastCathetus(1, z);
172 assert(sinTheta != 0 &&
173 "VectorHelpers: Vector is parallel to the z-axis "
174 "which leads to division by zero");
175 const double invSinTheta = 1. / sinTheta;
176 const double cosPhi = x * invSinTheta;
177 const double sinPhi = y * invSinTheta;
178
179 return {cosPhi, sinPhi, cosTheta, sinTheta};
180}

191inline double cast(const Vector3& position, AxisDirection aDir) {
192 using enum AxisDirection;
193 switch (aDir) {
194 case AxisX:
195 return position[0];
196 case AxisY:
197 return position[1];
198 case AxisZ:
199 return position[2];
200 case AxisR:
201 return perp(position);
202 case AxisPhi:
203 return phi(position);
204 case AxisRPhi:
205 return perp(position) * phi(position);
206 case AxisTheta:
207 return theta(position);
208 case AxisEta:
209 return eta(position);
210 case AxisMag:
211 return position.norm();
212 default:
213 assert(false && "Invalid AxisDirection enum value");
214 return std::numeric_limits<double>::quiet_NaN();
215 }
216}

263inline std::pair<double, double> incidentAngles(
264 const Acts::Vector3& direction,
265 const Acts::RotationMatrix3& globalToLocal) {
266 Acts::Vector3 trfDir = globalToLocal * direction;
267 // The angles are defined with respect to the measurement axis
268 // i.e. "head-on" == pi/2, parallel = 0
269 double phi = std::atan2(trfDir[2], trfDir[0]);
270 double theta = std::atan2(trfDir[2], trfDir[1]);
271 return {phi, theta};
272}

280template <typename Derived>
281double deltaR(const Eigen::MatrixBase<Derived>& v1,
282 const Eigen::MatrixBase<Derived>& v2)
283 requires(Eigen::MatrixBase<Derived>::RowsAtCompileTime == 3)
284{
285 const double dphi =
286 detail::difference_periodic(phi(v1), phi(v2), 2 * std::numbers::pi);
287 const double deta = eta(v1) - eta(v2);
288 return fastHypot(dphi, deta);
289}
