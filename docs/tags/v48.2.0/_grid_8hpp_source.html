468 template <class Point>
469 T interpolate(const Point& point) const {
470 // get local indices for current bin
471 // value of bin is interpreted as being the field value at its lower left
472 // corner
473 const auto llIndices = m_axes.getLocalBinsFromPoint(point);
474
475 // get global indices for all surrounding corner points
476 const auto closestIndices =
477 m_axes.getNeighborHoodIndices(llIndices, {0, 1});
478
479 // there are 2^DIM corner points used during the interpolation
480 constexpr std::size_t nCorners = 1 << DIM;
481
482 // construct vector of pairs of adjacent bin centers and values
483 std::array<value_type, nCorners> neighbors{};
484
485 // get values on grid points
486 std::size_t i = 0;
487 for (const auto index : closestIndices) {
488 neighbors.at(i) = at(index);
489 ++i;
490 }
491
492 return Acts::interpolate(point, m_axes.getLowerLeftBinEdge(llIndices),
493 m_axes.getUpperRightBinEdge(llIndices), neighbors);
494 }
