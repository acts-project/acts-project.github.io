83 template <NavigationPolicyConcept P, typename... Args>
84 requires(std::is_constructible_v<P, const GeometryContext&,
85 const TrackingVolume&, const Logger&,
86 Args...> &&
87 (std::is_copy_constructible_v<Args> && ...))
88 constexpr NavigationPolicyFactory add(Args&&... args) && {
89 auto factory = [=](const GeometryContext& gctx,
90 const TrackingVolume& volume, const Logger& logger) {
91 return std::make_unique<P>(gctx, volume, logger, args...);
92 };
93
94 m_factories.push_back(std::move(factory));
95 return std::move(*this);
96 }

109 template <typename Fn, typename... Args>
110 requires(detail::NavigationPolicyPointerFactoryConcept<Fn, Args...> &&
111 (std::is_copy_constructible_v<Args> && ...))
112 constexpr NavigationPolicyFactory add(Fn&& fn, Args&&... args) && {
113 auto factory = [=](const GeometryContext& gctx,
114 const TrackingVolume& volume,
115 const Logger& logger) -> factory_type::result_type {
116 return fn(gctx, volume, logger, args...);
117 };
118
119 m_factories.push_back(std::move(factory));
120 return std::move(*this);
121 }
122
125 std::unique_ptr<NavigationPolicyFactory> asUniquePtr() && {
126 return std::make_unique<NavigationPolicyFactory>(std::move(*this));
127 }
128
135 std::unique_ptr<MultiNavigationPolicy> operator()(
136 const GeometryContext& gctx, const TrackingVolume& volume,
137 const Logger& logger) const {
138 if (m_factories.empty()) {
139 throw std::runtime_error(
140 "No factories registered in the navigation policy factory");
141 }
142
143 std::vector<std::unique_ptr<INavigationPolicy>> policies;
144 policies.reserve(m_factories.size());
145 for (auto& factory : m_factories) {
146 policies.push_back(factory(gctx, volume, logger));
147 }
148
149 return std::make_unique<MultiNavigationPolicy>(std::move(policies));
150 }
151
157 std::unique_ptr<INavigationPolicy> build(const GeometryContext& gctx,
158 const TrackingVolume& volume,
159 const Logger& logger) const {
160 return operator()(gctx, volume, logger);
161 }
162
163 private:
165 std::vector<factory_type> m_factories;
166};
