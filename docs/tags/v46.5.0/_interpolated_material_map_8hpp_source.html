30template <typename G>
31struct MaterialMapLookup {
32 public:
34 using Grid_t = G;
36 static constexpr std::size_t DIM_POS = Grid_t::DIM;
37
42 struct MaterialCell {
44 static constexpr unsigned int N = 1 << DIM_POS;
45
58 MaterialCell(std::function<Vector<DIM_POS>(const Vector3&)> transformPos,
59 std::array<double, DIM_POS> lowerLeft,
60 std::array<double, DIM_POS> upperRight,
61 std::array<Material::ParametersVector, N> materialValues)
62 : m_transformPos(std::move(transformPos)),
63 m_lowerLeft(std::move(lowerLeft)),
64 m_upperRight(std::move(upperRight)),
65 m_materialValues(std::move(materialValues)) {}
66
73 Material getMaterial(const Vector3& position) const {
74 // defined in Interpolation.hpp
75 return Material(interpolate(m_transformPos(position), m_lowerLeft,
76 m_upperRight, m_materialValues));
77 }
78
84 bool isInside(const Vector3& position) const {
85 const auto& gridCoordinates = m_transformPos(position);
86 for (unsigned int i = 0; i < DIM_POS; ++i) {
87 if (gridCoordinates[i] < m_lowerLeft.at(i) ||
88 gridCoordinates[i] >= m_upperRight.at(i)) {
89 return false;
90 }
91 }
92 return true;
93 }
94
95 private:
97 std::function<Vector<DIM_POS>(const Vector3&)> m_transformPos;
98
100 std::array<double, DIM_POS> m_lowerLeft;
101
103 std::array<double, DIM_POS> m_upperRight;
104
109 std::array<Material::ParametersVector, N> m_materialValues;
110 };
111
117 MaterialMapLookup(std::function<Vector<DIM_POS>(const Vector3&)> transformPos,
118 Grid_t grid)
119 : m_transformPos(std::move(transformPos)), m_grid(std::move(grid)) {}
120
128 Material material(const Vector3& position) const {
129 return Material(m_grid.atLocalBins(
130 m_grid.localBinsFromLowerLeftEdge(m_transformPos(position))));
131 }
132
140 Material getMaterial(const Vector3& position) const {
141 return Material(m_grid.interpolate(m_transformPos(position)));
142 }
143
151 MaterialCell getMaterialCell(const Vector3& position) const {
152 const auto& gridPosition = m_transformPos(position);
153 std::size_t bin = m_grid.globalBinFromPosition(gridPosition);
154 const auto& indices = m_grid.localBinsFromPosition(bin);
155 const auto& lowerLeft = m_grid.lowerLeftBinEdge(indices);
156 const auto& upperRight = m_grid.upperRightBinEdge(indices);
157
158 // Loop through all corner points
159 constexpr std::size_t nCorners = 1 << DIM_POS;
160 std::array<Material::ParametersVector, nCorners> neighbors{};
161 const auto& cornerIndices = m_grid.closestPointsIndices(gridPosition);
162
163 std::size_t i = 0;
164 for (std::size_t index : cornerIndices) {
165 neighbors.at(i++) = m_grid.at(index);
166 }
167
168 return MaterialCell(m_transformPos, lowerLeft, upperRight,
169 std::move(neighbors));
170 }
171
175 std::vector<std::size_t> getNBins() const {
176 auto nBinsArray = m_grid.numLocalBins();
177 return std::vector<std::size_t>(nBinsArray.begin(), nBinsArray.end());
178 }
179
183 std::vector<double> getMin() const {
184 auto minArray = m_grid.minPosition();
185 return std::vector<double>(minArray.begin(), minArray.end());
186 }
187
191 std::vector<double> getMax() const {
192 auto maxArray = m_grid.maxPosition();
193 return std::vector<double>(maxArray.begin(), maxArray.end());
194 }
195
201 bool isInside(const Vector3& position) const {
202 return m_grid.isInside(m_transformPos(position));
203 }
204
208 const Grid_t& getGrid() const { return m_grid; }
209
210 private:
212 std::function<Vector<DIM_POS>(const Vector3&)> m_transformPos;
214 Grid_t m_grid;
215};

238template <typename Mapper_t>
239class InterpolatedMaterialMap : public IVolumeMaterial {
240 public:
242 struct Cache {
244 std::optional<typename Mapper_t::MaterialCell> matCell;

252 explicit InterpolatedMaterialMap(Mapper_t&& mapper)
253 : m_mapper(std::move(mapper)) {}
254
259 InterpolatedMaterialMap(Mapper_t&& mapper, BinUtility bu)
260 : m_mapper(std::move(mapper)), m_binUtility(std::move(bu)) {}
261
267 const Material material(const Vector3& position) const override {
268 return m_mapper.material(position);
269 }
270
276 Material getMaterial(const Vector3& position) const {
277 return m_mapper.getMaterial(position);
278 }
279
287 Material getMaterial(const Vector3& position, Cache& cache) const {
288 if (!cache.initialized || !(*cache.matCell).isInside(position)) {
289 cache.matCell = getMaterialCell(position);
290 cache.initialized = true;
291 }
292 return (*cache.matCell).getMaterial(position);
293 }

347 return sl;
348 }

350 private:

359 const Vector3& position) const {
360 return m_mapper.getMaterialCell(position);
361 }

368 Mapper_t m_mapper;
369
370 BinUtility m_binUtility{};
371};
