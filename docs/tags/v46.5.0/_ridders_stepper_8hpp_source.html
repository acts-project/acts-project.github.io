40struct BoundParameterVariationGenerator {
41 virtual ~BoundParameterVariationGenerator() = default;
47 virtual BoundParameterVariation variationMap(
48 const BoundVector& input, const BoundMatrix& covariance) const = 0;
49};

52struct DeltaBoundParameterVariationGenerator
53 : public BoundParameterVariationGenerator {
55 std::vector<BoundVector> deltas;
56
59 explicit DeltaBoundParameterVariationGenerator(const double delta)
60 : deltas(1, BoundVector::Constant(delta)) {}
61
64 explicit DeltaBoundParameterVariationGenerator(
65 const std::vector<double>& deltas_) {
66 this->deltas.reserve(deltas_.size());
67 for (const auto& delta : deltas_) {
68 this->deltas.emplace_back(BoundVector::Constant(delta));
69 }
70 }
71
74 explicit DeltaBoundParameterVariationGenerator(const BoundVector& delta)
75 : deltas(1, delta) {}
76
79 explicit DeltaBoundParameterVariationGenerator(
80 std::vector<BoundVector> deltas_)
81 : deltas(std::move(deltas_)) {}
82
85 BoundParameterVariation variationMap(
86 const BoundVector& /*input*/,
87 const BoundMatrix& /*covariance*/) const override {
88 BoundParameterVariation variationMap;
89 variationMap.reserve(deltas.size() * eBoundSize);
90 for (std::size_t i = 0; i < deltas.size(); ++i) {
91 for (std::size_t j = 0; j < eBoundSize; ++j) {
92 variationMap.emplace_back(j, deltas[i][j]);
93 }
94 }
95 return variationMap;
96 }
97};

100struct CovarianceBoundParameterVariationGenerator
101 : public BoundParameterVariationGenerator {
103 std::vector<BoundVector> sigmaFactors;
104
107 explicit CovarianceBoundParameterVariationGenerator(const double sigmaFactor)
108 : sigmaFactors(1, BoundVector::Constant(sigmaFactor)) {}
109
113 explicit CovarianceBoundParameterVariationGenerator(
114 const std::vector<double>& sigmaFactors_) {
115 sigmaFactors.reserve(sigmaFactors_.size());
116 for (const auto& factor : sigmaFactors_) {
117 sigmaFactors.emplace_back(BoundVector::Constant(factor));
118 }
119 }
120
123 explicit CovarianceBoundParameterVariationGenerator(
124 const BoundVector& sigmaFactor)
125 : sigmaFactors(1, sigmaFactor) {}
126
129 explicit CovarianceBoundParameterVariationGenerator(
130 std::vector<BoundVector> sigmaFactors_)
131 : sigmaFactors(std::move(sigmaFactors_)) {}
132
136 BoundParameterVariation variationMap(
137 const BoundVector& /*input*/,
138 const BoundMatrix& covariance) const override {
139 BoundParameterVariation variationMap;
140 variationMap.reserve(sigmaFactors.size() * eBoundSize);
141 for (std::size_t i = 0; i < sigmaFactors.size(); ++i) {
142 for (std::size_t j = 0; j < eBoundSize; ++j) {
143 variationMap.emplace_back(
144 j, sigmaFactors[i][j] * std::sqrt(covariance(j, j)));
145 }
146 }
147 return variationMap;
148 }
149};

274 void initialize(State& state, const BoundVector& boundVector,
275 const std::optional<BoundMatrix>& covariance,
276 ParticleHypothesis particleHypothesis,
277 const Surface& surface) const {
278 state.variationMap.clear();
279 state.secondaryStepperStates.clear();
280
281 m_stepperImpl.initialize(state.primaryStepperState, boundVector,
282 std::nullopt, particleHypothesis, surface);
283
284 if (!covariance.has_value()) {
285 state.covTransport = false;
286 state.cov = BoundMatrix::Zero();
287 // intentionally not resetting `jacobian` here as otherwise `update`
288 // break. this should be revisitted at some point - might be best to
289 // rework the stepper interface and how the jacobian is accessed
290
291 return;
292 }
293
294 state.variationMap =
295 m_config.parameterVariation->variationMap(boundVector, *covariance);
296
297 for (const auto& [index, delta] : state.variationMap) {
298 BoundVector nudgedParams = boundVector;
299 nudgedParams[index] += delta;
300
301 state.secondaryStepperStates.push_back(
302 m_stepperImpl.makeState(state.primaryStepperState.options));
303 m_stepperImpl.initialize(state.secondaryStepperStates.back(),
304 nudgedParams, std::nullopt, particleHypothesis,
305 surface);
306 }
307
308 state.covTransport = true;
309 state.cov = *covariance;
310 // same as above:
311 // intentionally not resetting `jacobian` here as otherwise `update`
312 // break. this should be revisitted at some point - might be best to
313 // rework the stepper interface and how the jacobian is accessed
314 }

392 IntersectionStatus updateSurfaceStatus(
393 State& state, const Surface& surface, std::uint8_t index,
394 Direction navDir, const BoundaryTolerance& boundaryTolerance,
395 double surfaceTolerance, ConstrainedStep::Type stype,
396 const Logger& logger = getDummyLogger()) const {
397 state.lastSurfaceTolerance = surfaceTolerance;
398 state.lastStepConstraintType = stype;
399
400 IntersectionStatus overallStatus = m_stepperImpl.updateSurfaceStatus(
401 state.primaryStepperState, surface, index, navDir, boundaryTolerance,
402 surfaceTolerance, stype, logger);
403 if (overallStatus == IntersectionStatus::unreachable) {
404 return overallStatus;
405 }
406
407 const auto combine = [](const IntersectionStatus a,
408 const IntersectionStatus b) {
409 using enum IntersectionStatus;
410 if ((a == unreachable) || (b == unreachable)) {
411 return unreachable;
412 }
413 if ((a == reachable) || (b == reachable)) {
414 return reachable;
415 }
416 return onSurface;
417 };
418
419 for (auto& secondaryState : state.secondaryStepperStates) {
420 const IntersectionStatus status = m_stepperImpl.updateSurfaceStatus(
421 secondaryState, surface, index, navDir, BoundaryTolerance::Infinite(),
422 surfaceTolerance, stype, logger);
423 overallStatus = combine(overallStatus, status);
424 }
425
426 return overallStatus;
427 }

497 Result<BoundState> boundState(
498 State& state, const Surface& surface, bool transportCovariance = true,
499 const FreeToBoundCorrection& freeToBoundCorrection =
500 FreeToBoundCorrection(false)) const {
501 Result<BoundState> primaryResult = m_stepperImpl.boundState(
502 state.primaryStepperState, surface, false, freeToBoundCorrection);
503
504 if (!transportCovariance) {
505 std::get<1>(*primaryResult) = state.jacobian;
506 return primaryResult;
507 }
508
509 std::vector<BoundVector> boundVectors;
510 std::vector<double> pathLenghts;
511 boundVectors.reserve(state.secondaryStepperStates.size());
512 pathLenghts.reserve(state.secondaryStepperStates.size());
513 for (auto& secondaryState : state.secondaryStepperStates) {
514 const Result<BoundState> result = m_stepperImpl.boundState(
515 secondaryState, surface, false, freeToBoundCorrection);
516 if (!result.ok()) {
517 return result.error();
518 }
519 boundVectors.push_back(std::get<0>(*result).parameters());
520 pathLenghts.push_back(std::get<2>(*result));
521 }
522 const auto& referenceVector = std::get<0>(*primaryResult).parameters();
523
524 state.jacobian = BoundMatrix::Zero();
525 std::array<std::size_t, eBoundSize> numberOfVariationsPerParameter{};
526
527 for (std::size_t i = 0; i < state.variationMap.size(); ++i) {
528 const auto& [index, delta] = state.variationMap[i];
529 const auto& nudgedVector = boundVectors[i];
530
531 const auto diff = [&]() {
532 BoundVector d = nudgedVector - referenceVector;
533 d[eBoundPhi] = detail::difference_periodic(nudgedVector[eBoundPhi],
534 referenceVector[eBoundPhi],
535 2 * std::numbers::pi);
536 return d;
537 };
538
539 state.jacobian.col(index) += diff() / delta;
540 ++numberOfVariationsPerParameter[index];
541 }
542 for (std::size_t i = 0; i < eBoundSize; ++i) {
543 state.jacobian.col(i) /= numberOfVariationsPerParameter[i];
544 }
545
546 state.cov = state.jacobian * state.cov * state.jacobian.transpose();
547
548 BoundTrackParameters newParams(surface.getSharedPtr(), referenceVector,
549 state.cov, particleHypothesis(state));
550
551 state.variationMap =
552 m_config.parameterVariation->variationMap(referenceVector, state.cov);
553
554 return Result<BoundState>::success(
555 BoundState{std::move(newParams), state.jacobian, pathLenghts.front()});
556 }

574 BoundState curvilinearState(State& state,
575 bool transportCovariance = true) const {
576 if (!transportCovariance) {
577 return m_stepperImpl.curvilinearState(state.primaryStepperState, false);
578 }
579
580 const auto curvilinearSurface =
581 CurvilinearSurface(position(state), direction(state)).surface();
582
583 for (std::size_t i = 0; i < m_config.maxStepsToCurvilinearSurface; ++i) {
584 const auto surfaceStatus = updateSurfaceStatus(
585 state, *curvilinearSurface, 0, state.lastStepPropagationDirection,
586 BoundaryTolerance::Infinite(), state.lastSurfaceTolerance,
587 state.lastStepConstraintType);
588 if (surfaceStatus == IntersectionStatus::onSurface) {
589 break;
590 }
591 if (surfaceStatus == IntersectionStatus::unreachable) {
592 throw std::runtime_error(
593 "RiddersStepper: Curvilinear surface is unreachable");
594 }
595 const auto stepResult = step(state, state.lastStepPropagationDirection,
596 state.lastStepMaterial);
597 if (!stepResult.ok()) {
598 throw std::runtime_error(
599 "RiddersStepper: Failed to step towards curvilinear surface: " +
600 stepResult.error().message());
601 }
602 }
603
604 return boundState(state, *curvilinearSurface, true).value();
605 }

629 const std::optional<BoundMatrix> cov =

632 initialize(state,
633 transformFreeToBoundParameters(
634 position, time, direction, qOverP, *curvilinearSurface,
635 state.primaryStepperState.options.geoContext)
636 .value(),
637 cov, particleHypothesis(state), *curvilinearSurface);
638 }

662 Result<double> step(State& state, Direction propagationDirection,
663 const IVolumeMaterial* material) const {
664 state.lastStepPropagationDirection = propagationDirection;
665 state.lastStepMaterial = material;
666
667 const Result<double> stepResult = m_stepperImpl.step(
668 state.primaryStepperState, propagationDirection, material);
669 if (!stepResult.ok()) {
670 return stepResult.error();
671 }
672 for (auto& secondaryState : state.secondaryStepperStates) {
673 const Result<double> secondaryStepResult =
674 m_stepperImpl.step(secondaryState, propagationDirection, material);
675 if (!secondaryStepResult.ok()) {
676 return secondaryStepResult.error();
677 }
678 }
679 state.pathAccumulated += *stepResult;
680 return *stepResult;
681 }
