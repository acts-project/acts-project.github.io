34inline Transform3 makeTransform3(const RotationMatrix3& rotation,
35 const Vector3& translation = Vector3::Zero()) {
36 assert(isOrthogonal(rotation) &&
37 "Transform3 requires an orthogonal rotation part");
38 Transform3 transform = Transform3::Identity();
39 transform.linear() = rotation;
40 transform.translation() = translation;

48inline Transform3 makeTransform3(const AffineTransform3& transform) {
