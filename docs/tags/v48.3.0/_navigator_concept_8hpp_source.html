57template <typename Navigator, typename Options = typename Navigator::Options,
58 typename State = typename Navigator::State>
59concept NavigatorConcept = requires {
60 typename Navigator::Config;
61 typename Navigator::Options;
62 typename Navigator::State;
63
64 requires requires(const Navigator& n, const Options& o, State& s,
65 const Surface& sf, const Vector3& position,
66 const Vector3& direction,
67 const NavigatorInitializeArguments& args) {
68 { n.makeState(o) } -> std::same_as<State>;
69 { n.currentSurface(s) } -> std::same_as<const Surface*>;
70 { n.currentVolume(s) } -> std::same_as<const TrackingVolume*>;
71 { n.currentVolumeMaterial(s) } -> std::same_as<const IVolumeMaterial*>;
72 { n.startSurface(s) } -> std::same_as<const Surface*>;
73 { n.targetSurface(s) } -> std::same_as<const Surface*>;
74 { n.endOfWorldReached(s) } -> std::same_as<bool>;
75 { n.navigationBreak(s) } -> std::same_as<bool>;
76 { n.initialize(s, args) } -> std::same_as<Result<void>>;
77 // Keep pre-curvature custom navigators compatible until the next breaking
78 // release. Prefer the curvature overload whenever it is available.
79 requires Concepts::NavigatorNextTargetWithCurvature<Navigator, State> ||
80 requires {
81 {
82 n.nextTarget(s, position, direction)
83 } -> std::same_as<NavigationTarget>;
84 };
85 requires Concepts::NavigatorCheckTargetValidWithCurvature<Navigator,
86 State> ||
87 requires {
88 {
89 n.checkTargetValid(s, position, direction)
90 } -> std::same_as<bool>;
91 };
92 requires Concepts::NavigatorHandleSurfaceReachedWithCurvature<Navigator,
93 State> ||
94 requires {
95 {
96 n.handleSurfaceReached(s, position, direction, sf)
97 } -> std::same_as<void>;
98 };
99 };
100};
