54struct DeltaBoundParameterVariationGenerator
55 : public BoundParameterVariationGenerator {
57 std::vector<BoundVector> deltas;
58
61 explicit DeltaBoundParameterVariationGenerator(const double delta)
62 : deltas(1, BoundVector::Constant(delta)) {}
63
66 explicit DeltaBoundParameterVariationGenerator(
67 const std::vector<double>& deltas_) {
68 this->deltas.reserve(deltas_.size());
69 for (const auto& delta : deltas_) {
70 this->deltas.emplace_back(BoundVector::Constant(delta));
71 }
72 }
73
76 explicit DeltaBoundParameterVariationGenerator(const BoundVector& delta)
77 : deltas(1, delta) {}
78
81 explicit DeltaBoundParameterVariationGenerator(
82 std::vector<BoundVector> deltas_)
83 : deltas(std::move(deltas_)) {}
84
87 BoundParameterVariation variationMap(
88 const BoundVector& /*input*/,
89 const BoundMatrix& /*covariance*/) const override {
90 BoundParameterVariation variationMap;
91 variationMap.reserve(deltas.size() * eBoundSize);
92 for (std::size_t i = 0; i < deltas.size(); ++i) {
93 for (std::size_t j = 0; j < eBoundSize; ++j) {
94 variationMap.emplace_back(j, deltas[i][j]);
95 }
96 }
97 return variationMap;
98 }
99};

102struct CovarianceBoundParameterVariationGenerator
103 : public BoundParameterVariationGenerator {
105 std::vector<BoundVector> sigmaFactors;
106
109 explicit CovarianceBoundParameterVariationGenerator(const double sigmaFactor)
110 : sigmaFactors(1, BoundVector::Constant(sigmaFactor)) {}
111
115 explicit CovarianceBoundParameterVariationGenerator(
116 const std::vector<double>& sigmaFactors_) {
117 sigmaFactors.reserve(sigmaFactors_.size());
118 for (const auto& factor : sigmaFactors_) {
119 sigmaFactors.emplace_back(BoundVector::Constant(factor));
120 }
121 }
122
125 explicit CovarianceBoundParameterVariationGenerator(
126 const BoundVector& sigmaFactor)
127 : sigmaFactors(1, sigmaFactor) {}
128
131 explicit CovarianceBoundParameterVariationGenerator(
132 std::vector<BoundVector> sigmaFactors_)
133 : sigmaFactors(std::move(sigmaFactors_)) {}
134
138 BoundParameterVariation variationMap(
139 const BoundVector& /*input*/,
140 const BoundMatrix& covariance) const override {
141 BoundParameterVariation variationMap;
142 variationMap.reserve(sigmaFactors.size() * eBoundSize);
143 for (std::size_t i = 0; i < sigmaFactors.size(); ++i) {
144 for (std::size_t j = 0; j < eBoundSize; ++j) {
145 variationMap.emplace_back(
146 j, sigmaFactors[i][j] * std::sqrt(covariance(j, j)));
147 }
148 }
149 return variationMap;
150 }

370 IntersectionStatus updateSurfaceStatus(
371 State& state, const Surface& surface, std::uint8_t index,
372 Direction navDir, const BoundaryTolerance& boundaryTolerance,
373 double surfaceTolerance, ConstrainedStep::Type stype,
374 const Logger& logger = getDummyLogger()) const {
375 state.lastSurfaceTolerance = surfaceTolerance;
376 state.lastStepConstraintType = stype;
377
378 IntersectionStatus overallStatus = m_stepperImpl.updateSurfaceStatus(
379 state.primaryStepperState, surface, index, navDir, boundaryTolerance,
380 surfaceTolerance, stype, logger);
381 if (overallStatus == IntersectionStatus::unreachable) {
382 return overallStatus;
383 }
384
385 const auto combine = [](const IntersectionStatus a,
386 const IntersectionStatus b) {
387 using enum IntersectionStatus;
388 if ((a == unreachable) || (b == unreachable)) {
389 return unreachable;
390 }
391 if ((a == reachable) || (b == reachable)) {
392 return reachable;
393 }
394 return onSurface;
395 };
396
397 for (auto& secondaryState : state.secondaryStepperStates) {
398 const IntersectionStatus status = m_stepperImpl.updateSurfaceStatus(
399 secondaryState, surface, index, navDir, BoundaryTolerance::Infinite(),
400 surfaceTolerance, stype, logger);
401 overallStatus = combine(overallStatus, status);
402 }
403
404 return overallStatus;
405 }

574 Result<Jacobian> transportToBound(
575 State& state, const Surface& surface,
576 [[maybe_unused]] const FreeToBoundCorrection& freeToBoundCorrection =
577 FreeToBoundCorrection(false)) const {
578 if (!state.cov.has_value()) {
579 return Result<Jacobian>::success(Jacobian::Identity());
580 }
581
582 Result<BoundParameters> primaryResult =
583 m_stepperImpl.boundParameters(state.primaryStepperState, surface);
584 if (!primaryResult.ok()) {
585 return Result<Jacobian>::failure(primaryResult.error());
586 }
587
588 std::vector<BoundVector> boundVectors;
589 boundVectors.reserve(state.secondaryStepperStates.size());
590 for (const auto& secondaryState : state.secondaryStepperStates) {
591 const Result<BoundParameters> result =
592 m_stepperImpl.boundParameters(secondaryState, surface);
593 if (!result.ok()) {
594 return Result<Jacobian>::failure(result.error());
595 }
596 boundVectors.push_back(result->parameters());
597 }
598 const BoundVector& referenceVector = primaryResult->parameters();
599
600 Jacobian jacobian = Jacobian::Zero();
601 std::array<std::size_t, eBoundSize> numberOfVariationsPerParameter{};
602
603 for (std::size_t i = 0; i < state.variationMap.size(); ++i) {
604 const auto& [index, delta] = state.variationMap[i];
605 const auto& nudgedVector = boundVectors[i];
606
607 const auto diff = [&]() {
608 BoundVector d = nudgedVector - referenceVector;
609 d[eBoundPhi] = detail::difference_periodic(nudgedVector[eBoundPhi],
610 referenceVector[eBoundPhi],
611 2 * std::numbers::pi);
612 return d;
613 };
614
615 jacobian.col(index) += diff() / delta;
616 ++numberOfVariationsPerParameter[index];
617 }
618 for (std::size_t i = 0; i < eBoundSize; ++i) {
619 jacobian.col(i) /= numberOfVariationsPerParameter[i];
620 }
621
622 *state.cov = jacobian * *state.cov * jacobian.transpose();
623
624 // The secondary states restart on the surface, so the next jacobian starts
625 // here as well
626 resetSecondaryStates(state, referenceVector, surface);
627 for (auto& secondaryState : state.secondaryStepperStates) {
628 secondaryState.stepSize = state.primaryStepperState.stepSize;
629 }
630
631 return Result<Jacobian>::success(jacobian);
632 }
633
641 Jacobian transportToCurvilinear(State& state) const {
642 if (!state.cov.has_value()) {
643 return Jacobian::Identity();
644 }
645
646 const auto curvilinearSurface =
647 CurvilinearSurface(position(state), direction(state)).surface();
648
649 for (std::size_t i = 0; i < m_config.maxStepsToCurvilinearSurface; ++i) {
650 const auto surfaceStatus = updateSurfaceStatus(
651 state, *curvilinearSurface, 0, state.lastStepPropagationDirection,
652 BoundaryTolerance::Infinite(), state.lastSurfaceTolerance,
653 state.lastStepConstraintType);
654 if (surfaceStatus == IntersectionStatus::onSurface) {
655 break;
656 }
657 if (surfaceStatus == IntersectionStatus::unreachable) {
658 throw std::runtime_error(
659 "RiddersStepper: Curvilinear surface is unreachable");
660 }
661 const auto stepResult = step(state, state.lastStepPropagationDirection,
662 state.lastStepMaterial);
663 if (!stepResult.ok()) {
664 throw std::runtime_error(
665 "RiddersStepper: Failed to step towards curvilinear surface: " +
666 stepResult.error().message());
667 }
668 }
669
670 // the surface is built at the current position, so this cannot fail
671 return transportToBound(state, *curvilinearSurface).value();
672 }

704
710 Result<double> step(State& state, Direction propagationDirection,
711 const IVolumeMaterial* material) const {
712 state.lastStepPropagationDirection = propagationDirection;
713 state.lastStepMaterial = material;
714
715 const Result<double> stepResult = m_stepperImpl.step(
716 state.primaryStepperState, propagationDirection, material);
717 if (!stepResult.ok()) {
718 return stepResult.error();
719 }
720 for (auto& secondaryState : state.secondaryStepperStates) {
721 const Result<double> secondaryStepResult =
722 m_stepperImpl.step(secondaryState, propagationDirection, material);
723 if (!secondaryStepResult.ok()) {
724 return secondaryStepResult.error();
725 }
726 }
727 state.pathAccumulated += *stepResult;
728 return *stepResult;
729 }

731 private:
736 void resetSecondaryStates(State& state, const BoundVector& boundVector,
737 const Surface& surface) const {
738 state.variationMap =
739 m_config.parameterVariation->variationMap(boundVector, *state.cov);
740
741 state.secondaryStepperStates.clear();
742 for (const auto& [index, delta] : state.variationMap) {
743 BoundVector nudgedParams = boundVector;
744 nudgedParams[index] += delta;
745
746 state.secondaryStepperStates.push_back(
747 m_stepperImpl.makeState(state.primaryStepperState.options));
748 m_stepperImpl.initialize(state.secondaryStepperStates.back(),
749 nudgedParams, std::nullopt,
750 particleHypothesis(state), surface);
751 }
752 }
