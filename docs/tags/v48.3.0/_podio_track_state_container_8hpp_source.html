170 template <typename T>
171 static constexpr bool has_impl(T& instance, Acts::HashedString key,
172 Acts::TrackIndexType istate) {
173 constexpr auto kInvalid = Acts::kTrackIndexInvalid;
174 using namespace Acts::HashedStringLiteral;
175 auto trackState = instance.m_collection->at(istate);
176 const auto& data = trackState.getData();
177 switch (key) {
178 case "predicted"_hash:
179 return data.ipredicted != kInvalid;
180 case "filtered"_hash:
181 return data.ifiltered != kInvalid;
182 case "smoothed"_hash:
183 return data.ismoothed != kInvalid;
184 case "calibrated"_hash:
185 return data.measdim != kInvalid;
186 case "calibratedCov"_hash:
187 return data.measdim != kInvalid;
188 case "jacobian"_hash:
189 return data.ijacobian != kInvalid;
190 case "projector"_hash:
191 return data.hasProjector;
192 case "uncalibratedSourceLink"_hash:
193 return hasUncalibratedSourceLink_impl(instance, trackState);
194 case "previous"_hash:
195 case "next"_hash:
196 case "measdim"_hash:
197 case "referenceSurface"_hash:
198 case "chi2"_hash:
199 case "pathLength"_hash:
200 case "typeFlags"_hash:
201 return true;
202 default:
203 return instance.m_dynamic.contains(key);
204 }
205
206 return false;
207 }

213 template <typename T>
214 static bool hasUncalibratedSourceLink_impl(
215 T& instance, const ActsPodioEdm::TrackState& trackState) {
216 if (instance.storeUncalibrated()) {
217 for (const auto& link : *instance.m_links) {
218 if (link.getFrom() == trackState) {
219 return true;
220 }
221 }
222 return false;
223 } else {
224 return trackState.getData().uncalibratedIdentifier !=
225 PodioUtil::kNoIdentifier;
226 }
227 }

233 template <typename T>
234 static Acts::SourceLink getUncalibratedSourceLink_impl(
235 T& instance, Acts::TrackIndexType istate) {
236 auto trackState = instance.m_collection->at(istate);
237 if (instance.storeUncalibrated()) {
238 std::optional<ActsPodioEdm::TrackerHitLocal> hit;
239 for (const auto& link : *instance.m_links) {
240 if (link.getFrom() == trackState) {
241 if (hit.has_value()) {
242 throw std::invalid_argument(
243 "Track state has multiple links to hits");
244 }
245 hit = link.getTo();
246 }
247 }
248 if (!hit.has_value()) {
249 throw std::invalid_argument("Track state has no link to a hit");
250 }
251 return Acts::SourceLink(hit.value());
252 } else {
253 auto sl = instance.m_helper.get().identifierToSourceLink(
254 trackState.getData().uncalibratedIdentifier);
255 if (!sl.has_value()) {
256 throw std::invalid_argument(
257 "Helper could not convert identifier to source link");
258 }
259 return sl.value();
260 }
261 }

268 template <bool EnsureConst, typename T>
269 static std::any component_impl(T& instance, Acts::HashedString key,
270 Acts::TrackIndexType istate) {
271 if constexpr (EnsureConst) {
272 static_assert(std::is_const_v<std::remove_reference_t<T>>,
273 "Is not const");
274 }
275 using namespace Acts::HashedStringLiteral;
276 auto trackState = instance.m_collection->at(istate);
277 std::conditional_t<EnsureConst, const ActsPodioEdm::TrackStateInfo*,
278 ActsPodioEdm::TrackStateInfo*>
279 dataPtr;
280 if constexpr (EnsureConst) {
281 dataPtr = &trackState.getData();
282 } else {
283 dataPtr = &PodioUtil::getDataMutable(trackState);
284 }
285 auto& data = *dataPtr;
286 switch (key) {
287 case "previous"_hash:
288 return &data.previous;
289 case "next"_hash:
290 return &data.next;
291 case "predicted"_hash:
292 return &data.ipredicted;
293 case "filtered"_hash:
294 return &data.ifiltered;
295 case "smoothed"_hash:
296 return &data.ismoothed;
297 case "projector"_hash:
298 return &data.projector;
299 case "measdim"_hash:
300 return &data.measdim;
301 case "chi2"_hash:
302 return &data.chi2;
303 case "pathLength"_hash:
304 return &data.pathLength;
305 case "typeFlags"_hash:
306 return &data.typeFlags;
307 default:
308 auto it = instance.m_dynamic.find(key);
309 if (it == instance.m_dynamic.end()) {
310 throw std::runtime_error("Unable to handle this component");
311 }
312 std::conditional_t<EnsureConst,
313 const podio_detail::ConstDynamicColumnBase*,
314 podio_detail::DynamicColumnBase*>
315 col = it->second.get();
316 assert(col && "Dynamic column is null");
317 return col->get(istate);
318 }
319 }

345 }

352 static void populateSurfaceBuffer(
353 const PodioUtil::ConversionHelper& helper,
354 const ActsPodioEdm::TrackStateCollection& collection,
355 std::vector<std::shared_ptr<const Acts::Surface>>& surfaces) noexcept {
356 surfaces.reserve(collection.size());

784 IndexType addTrackState_impl(
785 Acts::TrackStatePropMask mask = Acts::TrackStatePropMask::All,
786 Acts::TrackIndexType iprevious = Acts::kTrackIndexInvalid) {
787 auto trackState = m_collection->create();
788 auto& data = PodioUtil::getDataMutable(trackState);
789 data.previous = iprevious;
790 data.ipredicted = kInvalid;
791 data.ifiltered = kInvalid;
792 data.ismoothed = kInvalid;
793 data.ijacobian = kInvalid;
794
795 PodioUtil::getReferenceSurfaceMutable(trackState).surfaceType =
796 PodioUtil::kNoSurface;
797
798 using enum Acts::TrackStatePropMask;
799
800 if (ACTS_CHECK_BIT(mask, Predicted)) {
801 m_params->create();
802 data.ipredicted = m_params->size() - 1;
803 }
804 if (ACTS_CHECK_BIT(mask, Filtered)) {
805 m_params->create();
806 data.ifiltered = m_params->size() - 1;
807 }
808 if (ACTS_CHECK_BIT(mask, Smoothed)) {
809 m_params->create();
810 data.ismoothed = m_params->size() - 1;
811 }
812 if (ACTS_CHECK_BIT(mask, Jacobian)) {
813 m_jacs->create();
814 data.ijacobian = m_jacs->size() - 1;
815 }
816 data.measdim = kInvalid;
817 data.hasProjector = false;
818 if (ACTS_CHECK_BIT(mask, Calibrated)) {
819 data.hasProjector = true;
820 }
821 m_surfaces.emplace_back();
822
823 data.uncalibratedIdentifier = PodioUtil::kNoIdentifier;
824 assert(m_collection->size() == m_surfaces.size() &&
825 "Inconsistent surface buffer");
826
827 for (const auto& [key, vec] : m_dynamic) {
828 vec->add();
829 }
830
831 return m_collection->size() - 1;
832 }

837 void addTrackStateComponents_impl(IndexType istate,
838 Acts::TrackStatePropMask mask) {
839 auto& data = PodioUtil::getDataMutable(m_collection->at(istate));
840
841 using enum Acts::TrackStatePropMask;
842
843 if (ACTS_CHECK_BIT(mask, Predicted) && data.ipredicted == kInvalid) {
844 m_params->create();
845 data.ipredicted = m_params->size() - 1;
846 }
847
848 if (ACTS_CHECK_BIT(mask, Filtered) && data.ifiltered == kInvalid) {
849 m_params->create();
850 data.ifiltered = m_params->size() - 1;
851 }
852
853 if (ACTS_CHECK_BIT(mask, Smoothed) && data.ismoothed == kInvalid) {
854 m_params->create();
855 data.ismoothed = m_params->size() - 1;
856 }
857
858 if (ACTS_CHECK_BIT(mask, Jacobian) && data.ijacobian == kInvalid) {
859 m_jacs->create();
860 data.ijacobian = m_jacs->size() - 1;
861 }
862
863 if (ACTS_CHECK_BIT(mask, Calibrated) && !data.hasProjector) {
864 data.hasProjector = true;
865 }

875 Acts::TrackStatePropMask shareTarget) {
876 auto& self = PodioUtil::getDataMutable(m_collection->at(iself));
877 auto& other = PodioUtil::getDataMutable(m_collection->at(iother));
878
879 assert(ACTS_CHECK_BIT(this->getTrackState(iother).getMask(), shareSource) &&
880 "Source has incompatible allocation");
881
882 using enum Acts::TrackStatePropMask;
883
884 IndexType sourceIndex{kInvalid};
885 switch (shareSource) {
886 case Predicted:
887 sourceIndex = other.ipredicted;
888 break;
889 case Filtered:
890 sourceIndex = other.ifiltered;
891 break;
892 case Smoothed:
893 sourceIndex = other.ismoothed;
894 break;
895 case Jacobian:
896 sourceIndex = other.ijacobian;
897 break;
898 default:
899 throw std::domain_error{"Unable to share this component"};
900 }
901
902 assert(sourceIndex != kInvalid);
903
904 switch (shareTarget) {
905 case Predicted:
906 assert(shareSource != Jacobian);
907 self.ipredicted = sourceIndex;
908 break;
909 case Filtered:
910 assert(shareSource != Jacobian);
911 self.ifiltered = sourceIndex;
912 break;
913 case Smoothed:
914 assert(shareSource != Jacobian);
915 self.ismoothed = sourceIndex;
916 break;
917 case Jacobian:
918 assert(shareSource == Jacobian);
919 self.ijacobian = sourceIndex;
920 break;
921 default:
922 throw std::domain_error{"Unable to share this component"};
923 }

1014 void setUncalibratedSourceLink_impl(IndexType istate,
1015 const Acts::SourceLink& sourceLink) {
1016 auto trackState = m_collection->at(istate);
1017 if (storeUncalibrated()) {
1018 // In this case, we assume the source link is a TrackerHitLocal
1019 std::optional<ActsPodioEdm::TrackerHitLocal> hit;
1020
1021 if (auto* p = sourceLink.getPtr<ActsPodioEdm::TrackerHitLocal>();
1022 p != nullptr) {
1023 // The given source link is a TrackerHitLocal, so we can use it directly
1024 hit = *p;
1025 } else if (auto* mp =
1026 sourceLink.getPtr<ActsPodioEdm::MutableTrackerHitLocal>();
1027 mp != nullptr) {
1028 // The given source link is a MutableTrackerHitLocal; convert to the
1029 // immutable view via TrackerHitLocal's converting constructor
1030 hit = *mp;
1031 } else {
1032 // The given source link is not a TrackerHitLocal, so we need to ask the
1033 // helper to convert it to one
1034 hit = m_helper.get().sourceLinkToTrackerHitLocal(sourceLink);
1035 }
1036
1037 if (!hit.has_value()) {
1038 throw std::invalid_argument(
1039 "Source link could not be converted to a TrackerHitLocal");
1040 }
1041
1042 // Check if this track state already has a link to this hit
1043 std::optional<ActsPodioEdm::MutableTrackStateHitLink> existingLink;
1044 for (const auto& link : *m_links) {
1045 if (link.getFrom() == trackState) {
1046 if (existingLink.has_value()) {
1047 throw std::invalid_argument(
1048 "Track state has multiple links to the same hit");
1049 }
1050 existingLink = link;
1051 }
1052 }
1053
1054 if (!existingLink.has_value()) {
1055 // No link a new one
1056 auto link = m_links->create();
1057 link.setFrom(trackState);
1058 link.setTo(hit.value());
1059 } else {
1060 // Update the existing link
1061 existingLink->setTo(hit.value());
1062 }
1063 } else {
1064 // Call out to helper to convert source link to identifier
1065 auto id = m_helper.get().sourceLinkToIdentifier(sourceLink);
1066 if (!id.has_value()) {
1067 throw std::invalid_argument(
1068 "Helper could not convert source link to identifier");
1069 }
1070 auto& data = PodioUtil::getDataMutable(trackState);
1071 data.uncalibratedIdentifier = id.value();
1072 }
1073 }
