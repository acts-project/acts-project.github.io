208 template <detray::ranges::range sf_container_t>
209 requires std::is_same_v<typename sf_container_t::value_type, value_t>
210 DETRAY_HOST auto push_back(const sf_container_t& surfaces) noexcept(false)
211 -> void {
212 m_surfaces.reserve(m_surfaces.size() + surfaces.size());
213 m_surfaces.insert(m_surfaces.end(), surfaces.begin(), surfaces.end());
214 // End of this range is the start of the next range
215 m_offsets.push_back(static_cast<dindex>(m_surfaces.size()));
216 }
217
221 DETRAY_HOST auto erase(typename vector_type<value_t>::iterator pos) noexcept(
222 false) {
223 // Remove one element
224 auto next = m_surfaces.erase(pos);
225
226 // Update the upper bound of the range and all following ranges
227 const auto idx{static_cast<std::size_t>(pos - m_surfaces.begin())};
228 auto offset = detray::upper_bound(m_offsets.begin(), m_offsets.end(), idx);
229 for (auto itr = offset; itr != m_offsets.end(); ++itr) {
230 --(*itr);
231 }
232
233 return next;
234 }

246 detray::get_data(m_surfaces)};
247 }

249 private:
251 vector_type<size_type> m_offsets{};
253 vector_type<value_t> m_surfaces{};
254};
