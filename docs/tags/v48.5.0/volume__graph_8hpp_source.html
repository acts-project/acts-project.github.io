433 inline std::string to_string() const {
434 std::stringstream stream;
435 dindex dim = n_nodes() + 1u;
436 for (const auto &n : _nodes) {
437 stream << "[>>] Node with index " << n.index() << std::endl;
438 stream << " -> edges: " << std::endl;
439 for (unsigned int i = 0u; i < dim; ++i) {
440 const dindex degr = _adj_matrix[dim * n.index() + i];
441 if (degr == 0) {
442 continue;
443 }
444 std::string n_occur =
445 degr > 1 ? "\t\t\t\t(" + std::to_string(degr) + "x)" : "";
446
447 // Edge that leads out of the detector world
448 if (i == dim - 1u && degr != 0u) {
449 stream << " -> leaving world " + n_occur << std::endl;
450 } else {
451 stream << " -> " << std::to_string(i) + "\t" + n_occur
452 << std::endl;
453 }
454 }
455 }
456 return stream.str();
457 }

460 inline std::string to_dot_string() const {
461 std::stringstream stream;
462 dindex dim = n_nodes() + 1u;
463
464 stream << "strict graph {" << std::endl;
465 stream << " layout=neato;" << std::endl;
466 stream << " node [margin=0,shape=circle,style=filled];" << std::endl;
467 stream << " overlap=scale;" << std::endl;
468 stream << " splines=true;" << std::endl;
469 stream << " mode=KK;" << std::endl;
470 stream << std::endl;
471 stream << R"( exit [label="OOB",fillcolor="firebrick1"];)" << std::endl;
472
473 for (const auto &n : _nodes) {
474 stream << " v" << n.index() << ";" << std::endl;
475 }
476
477 stream << std::endl;
478
479 for (const auto &n : _nodes) {
480 for (unsigned int i = 0u; i < dim; ++i) {
481 const dindex degr = _adj_matrix[dim * n.index() + i];
482 if (degr == 0) {
483 continue;
484 }
485
486 bool to_oob = i == dim - 1u;
487 bool to_self = n.index() == i;
488
489 std::string label = degr > 1 ? std::to_string(degr) : "";
490 std::string dest = to_oob ? "exit" : ("v" + std::to_string(i));
491 std::string dir = (to_oob || to_self) ? "forward" : "both";
492
493 stream << " v" << n.index() << " -- " << dest << " [label=\""
494 << label << "\", dir=" << dir << "];" << std::endl;
495 }
496 }
497
498 stream << "}" << std::endl;
499
500 return stream.str();
501 }

503 private:
507 void build() {
508 // Leave space for the world volume (links to dindex_invalid)
509 const dindex dim{n_nodes() + 1u};
510 _adj_matrix.resize(dim * dim);
511
512 for (const auto &n : _nodes) {
513 // Only works for non batched geometries
514 for (const auto &edg_link : n.half_edges()) {
515 // Build an edge
516 for (const auto edg : _edges(n.index(), edg_link)) {
517 const dindex elem =
518 edg.to() < detail::invalid_value<
519 typename edge_generator::mask_edge_t>()
520 ? dim * n.index() + edg.to()
521 : dim * n.index() + dim - 1u;
522 _adj_matrix[elem]++;
523 }
524 }
525 }
526 }
