130template <std::size_t DIM>
131class MultiAxisSpecXD : public MultiAxisSpec {
132 public:
133 static_assert(DIM >= 1 && DIM <= 3,
134 "MultiAxisSpecXD supports 1 to 3 dimensions");
135
138 explicit MultiAxisSpecXD(std::array<AxisSpec, DIM> axisSpecs)
139 : MultiAxisSpec(
140 std::vector<AxisSpec>(std::make_move_iterator(axisSpecs.begin()),
141 std::make_move_iterator(axisSpecs.end()))) {}
142
145 using Options = std::array<AxisSpec::Options, DIM>;
146
152 std::unique_ptr<IMultiAxisXD<DIM>> buildMultiAxis(
153 const Options& options = {}) const {
154 return downcast(buildMultiAxisImpl(options));
155 }
156
157 private:
161 static std::unique_ptr<IMultiAxisXD<DIM>> downcast(
162 std::unique_ptr<IMultiAxis> multiAxis) {
163 auto* xd = dynamic_cast<IMultiAxisXD<DIM>*>(multiAxis.get());
164 if (xd == nullptr) {
165 throw std::logic_error(
166 "MultiAxisSpecXD: unexpected multi-axis dimension");
167 }
168 multiAxis.release();
169 return std::unique_ptr<IMultiAxisXD<DIM>>(xd);
170 }
171};
