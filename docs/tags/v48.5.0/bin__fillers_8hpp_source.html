69struct fill_by_pos {
76 template <concepts::surface_grid grid_t, typename volume_t,
77 typename surface_container_t, typename mask_container,
78 typename transform_container, typename context_t, typename... Args>
79 DETRAY_HOST auto operator()(grid_t &grid, const volume_t &vol,
80 const surface_container_t &surfaces,
81 const transform_container &transforms,
82 const mask_container & /*masks*/,
83 const context_t ctx, Args &&.../*unused*/) const
84 -> void {
85 // Fill the volumes surfaces into the grid
86 for (const auto &sf : surfaces) {
87 // no portals in grids allowed
88 if (sf.volume() == vol.index() && sf.is_sensitive()) {
89 const auto &sf_trf = transforms.at(sf.transform(), ctx);
90 const auto &t = sf_trf.translation();
91
92 // transform to axis coordinate system
93 const auto loc_pos = grid.project(vol.transform(), t, t);
94
95 // Populate
96 grid.template populate<attach<>>(loc_pos, sf);
97 }
98 }
99 }
100};

108struct bin_associator {
114 template <concepts::detector detector_t, typename volume_type,
115 concepts::surface_grid grid_t, typename... Args>
116 DETRAY_HOST auto operator()(grid_t &grid, detector_t &det,
117 const volume_type &vol,

130 template <concepts::surface_grid grid_t, typename volume_t,
131 typename surface_container_t, typename mask_container,
132 typename transform_container, typename context_t, typename... Args>
133 DETRAY_HOST auto operator()(grid_t &grid, const volume_t & /*unused*/,
134 const surface_container_t &surfaces,
135 const transform_container &transforms,
136 const mask_container &masks, const context_t ctx,
137 Args &&.../*unused*/) const -> void {
138 using scalar_t = dscalar<typename grid_t::algebra_type>;
139
140 // Fill the surfaces into the grid by matching their contour onto the
141 // grid bins
142 bin_association(ctx, surfaces, transforms, masks, grid,
143 darray<scalar_t, 2>{0.1f, 0.1f}, false);
144 }
145};
