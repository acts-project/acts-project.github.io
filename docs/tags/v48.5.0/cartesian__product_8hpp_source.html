52 template <detray::ranges::range... deduced_range_ts>

82 template <int I = static_cast<int>(sizeof...(range_ts)) - 1>
83 DETRAY_HOST_DEVICE constexpr auto size(std::size_t s = 1u) const noexcept
84 -> std::size_t {
85 decltype(auto) begin = detray::get<I>(m_begins);
86 decltype(auto) end = detray::get<I>(m_ends);
87
88 s *= static_cast<std::size_t>(detray::ranges::distance(begin, end));
89
90 if constexpr (I > 0) {
91 return size<I - 1>(s);
92 } else {
93 return s;
94 }
95 }

100 iterator_coll_t m_ends{};
101};

134template <std::input_iterator... iterator_ts>
135struct cartesian_product_iterator {
137 using difference_type = std::ptrdiff_t;
139 using value_type = std::tuple<std::iter_reference_t<iterator_ts>...>;
141 using pointer = value_type *;
143 using reference = value_type;
144 // TODO: Adapt to the weakest iterator category in pack
146 using iterator_category = detray::ranges::bidirectional_iterator_tag;
147
149 constexpr cartesian_product_iterator() = default;
150
154 DETRAY_HOST_DEVICE
155 constexpr cartesian_product_iterator(dtuple<iterator_ts...> begins,
156 dtuple<iterator_ts...> ends)
157 : m_begins(begins), m_ends(ends), m_itrs(begins) {}
158
163 DETRAY_HOST_DEVICE constexpr bool operator==(
164 const cartesian_product_iterator &rhs) const {
165 return (detray::get<0>(m_itrs) == detray::get<0>(rhs.m_itrs));
166 }
167
171 DETRAY_HOST_DEVICE constexpr auto operator++()
172 -> cartesian_product_iterator & {
173 unroll_increment();
174 return *this;
175 }
176
179 DETRAY_HOST_DEVICE constexpr auto operator++(int)
180 -> cartesian_product_iterator {
181 auto tmp(*this);
182 ++(*this);
183 return tmp;
184 }
185
186
190 DETRAY_HOST_DEVICE constexpr auto operator--()
191 -> cartesian_product_iterator & {
192 unroll_decrement();
193 return *this;
194 }
195
198 DETRAY_HOST_DEVICE constexpr auto operator--(int)
199 -> cartesian_product_iterator {
200 auto tmp(*this);
201 ++(*this);
202 return tmp;
203 }
204
205
207 DETRAY_HOST_DEVICE
208 constexpr decltype(auto) operator*() const {
209 return unroll_values(
210 std::make_integer_sequence<std::size_t, sizeof...(iterator_ts)>{});
211 }
212
213 private:
218 template <int I = static_cast<int>(sizeof...(iterator_ts)) - 1>
219 DETRAY_HOST_DEVICE constexpr void unroll_increment() {
220 auto &itr = detray::get<I>(m_itrs);
221 ++itr;
222 if constexpr (I > 0) {
223 if (itr == detray::get<I>(m_ends)) {
224 itr = detray::get<I>(m_begins);
225 unroll_increment<I - 1>();
226 }
227 }
228 }
229
234 template <int I = static_cast<int>(sizeof...(iterator_ts)) - 1>
235 DETRAY_HOST_DEVICE constexpr void unroll_decrement() {
236 auto &itr = detray::get<I>(m_itrs);
237
238 if constexpr (I > 0) {
239 if (itr == detray::get<I>(m_begins)) {
240 itr = detray::ranges::prev(detray::get<I>(m_ends));
241 unroll_decrement<I - 1>();
242 }
243 }
244 --itr;
245 }
246
249 template <std::size_t... I>
250 DETRAY_HOST_DEVICE constexpr auto unroll_values(
251 std::index_sequence<I...> /*seq*/) const {
252 return std::tuple<std::iter_reference_t<iterator_ts>...>(
253 *detray::get<I>(m_itrs)...);
254 }
255
257 dtuple<iterator_ts...> m_begins;
258 dtuple<iterator_ts...> m_ends;
260 dtuple<iterator_ts...> m_itrs;
261};
