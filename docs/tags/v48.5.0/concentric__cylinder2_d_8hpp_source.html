69 template <concepts::scalar scalar_t, concepts::point point_t>
70 DETRAY_HOST_DEVICE inline scalar_t min_dist_to_boundary(
71 const bounds_type<scalar_t> &bounds, const point_t &loc_p) const {
72 return math::min(math::fabs(loc_p[1] - bounds[e_lower_z]),

83 template <concepts::algebra algebra_t>
84 DETRAY_HOST_DEVICE constexpr result_type<dbool<algebra_t>> check_boundaries(

111 template <concepts::scalar scalar_t, concepts::point point_t>
112 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
113 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
114 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
115 const scalar_t /*edge_tol*/ = 0.f) const {
116 return (bounds[e_lower_z] - tol <= loc_p[1] &&
117 loc_p[1] <= bounds[e_upper_z] + tol);
118 }

137 template <concepts::scalar scalar_t>
138 DETRAY_HOST_DEVICE constexpr scalar_t area(
139 const bounds_type<scalar_t> &bounds) const {
140 return 2.f * constant<scalar_t>::pi * bounds[e_r] *
141 (bounds[e_upper_z] - bounds[e_lower_z]);
142 }

150 template <concepts::scalar scalar_t>
151 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
152 const bounds_type<scalar_t> &bounds,
153 const bounds_type<scalar_t> &o_bounds) const {
154 bounds_type<scalar_t> new_bounds{};
155
156 new_bounds[e_r] = math::max(bounds[e_r], o_bounds[e_r]);
157 new_bounds[e_lower_z] = math::min(bounds[e_lower_z], o_bounds[e_lower_z]);
158 new_bounds[e_upper_z] = math::max(bounds[e_upper_z], o_bounds[e_upper_z]);
159
160 return new_bounds;
161 }

172 template <concepts::algebra algebra_t>
173 DETRAY_HOST_DEVICE inline darray<dscalar<algebra_t>, 6> local_min_bounds(
174 const bounds_type<dscalar<algebra_t>> &bounds,
175 const dscalar<algebra_t> env =
176 std::numeric_limits<dscalar<algebra_t>>::epsilon()) const {
177 assert(env > 0.f);
178 const dscalar<algebra_t> xy_bound{bounds[e_r] + env};
179 return {-xy_bound, -xy_bound, bounds[e_lower_z] - env,
180 xy_bound, xy_bound, bounds[e_upper_z] + env};
181 }

185 template <concepts::algebra algebra_t>
186 DETRAY_HOST_DEVICE dpoint3D<algebra_t> centroid(
187 const bounds_type<dscalar<algebra_t>> &bounds) const {
188 return {static_cast<dscalar<algebra_t>>(0.f),
189 static_cast<dscalar<algebra_t>>(0.f),
190 0.5f * (bounds[e_lower_z] + bounds[e_upper_z])};
191 }

215 template <concepts::scalar scalar_t>
216 DETRAY_HOST constexpr bool check_consistency(
217 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
218 constexpr auto tol{10.f * std::numeric_limits<scalar_t>::epsilon()};
219
220 if (bounds[e_r] < tol) {
221 os << "DETRAY ERROR (HOST): Radius must be in the range (0, "
222 "numeric_max)"
223 << std::endl;
224 return false;
225 }
226 if (bounds[e_lower_z] >= bounds[e_upper_z] ||
227 math::fabs(bounds[e_lower_z] - bounds[e_upper_z]) < tol) {
228 os << "DETRAY ERROR (HOST): Neg. half length must be smaller than "
229 "pos. half "
230 "length.";
231 return false;
232 }
233
234 return true;
235 }
