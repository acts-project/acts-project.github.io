47struct aggregate_inspector {
49 using view_type = dmulti_view<detail::get_view_t<Inspectors>...>;
51 using const_view_type = dmulti_view<detail::get_view_t<const Inspectors>...>;

54 using inspector_tuple_t = std::tuple<Inspectors...>;
56 inspector_tuple_t _inspectors{};

59 aggregate_inspector() = default;

62 template <concepts::device_view view_t>
63 DETRAY_HOST_DEVICE explicit aggregate_inspector(view_t &view)
64 : _inspectors(unroll_views(
65 view, std::make_index_sequence<sizeof...(Inspectors)>{})) {}

75 template <unsigned int current_id = 0, typename state_type,
76 concepts::point3D point3_t, concepts::vector3D vector3_t,
77 typename... Args>
78 DETRAY_HOST_DEVICE auto operator()(state_type &state,
79 const navigation::config &cfg,
80 const point3_t &pos, const vector3_t &dir,
81 const char *message, Args &&...args) {
82 // Call inspector
83 std::get<current_id>(_inspectors)(state, cfg, pos, dir, message,
84 std::forward<Args>(args)...);
85
86 // Next inspector
87 if constexpr (current_id < std::tuple_size<inspector_tuple_t>::value - 1) {
88 return operator()<current_id + 1>(state, cfg, pos, dir, message,
89 std::forward<Args>(args)...);
90 }
91 }

98 template <unsigned int current_id = 0, typename state_type, typename... Args>
99 DETRAY_HOST_DEVICE auto operator()(state_type &state, const char *message,
100 Args &&...args) {

103
104 // Next inspector
105 if constexpr (current_id < std::tuple_size<inspector_tuple_t>::value - 1) {
106 return operator()<current_id + 1>(state, message,
107 std::forward<Args>(args)...);
108 }
109 }

150template <concepts::detector detector_t,
151 template <typename...> class vector_t = dvector,
152 status... navigation_status>
153struct object_tracer {
155 using candidate_record_t = intersection_record<detector_t>;
157 using scalar_type = typename candidate_record_t::scalar_type;
159 using point3_type = typename candidate_record_t::point3_type;
161 using vector3_type = typename candidate_record_t::vector3_type;
162
164 using view_type = dvector_view<candidate_record_t>;
166 using const_view_type = dvector_view<const candidate_record_t>;
167
169 object_tracer() = default;
170
173 DETRAY_HOST_DEVICE explicit object_tracer(
174 dvector_view<candidate_record_t> &view)
175 : object_trace(view) {}
176
178 vector_t<candidate_record_t> object_trace;
180 dindex current_vol{dindex_invalid};
182 const scalar_type inv_pos{detray::detail::invalid_value<scalar_type>()};
184 point3_type last_pos = {inv_pos, inv_pos, inv_pos};
186 vector3_type last_dir = {0., 0., 0.};
187
193 template <typename state_type, concepts::point3D point3_t,
194 concepts::vector3D vector3_t, typename... Args>
195 DETRAY_HOST_DEVICE auto operator()(const state_type &state,
196 const navigation::config & /*unused*/,
197 const point3_t &pos, const vector3_t &dir,
198 const char * /*message*/,
199 Args &&.../*unused*/) {
200 DETRAY_VERBOSE_HOST_DEVICE("Actor: Check navigation status...");
201
202 // Record the candidate of an encountered object
203 if ((is_status(state.status(), navigation_status) || ...)) {
204 DETRAY_VERBOSE_HOST_DEVICE("Actor: ...Status matched:");
205 // Reached a new position: log it
206 // Also log volume switches that happen without position update
207 const bool first_obj{
208 (last_pos == point3_type{inv_pos, inv_pos, inv_pos}) &&
209 (last_dir == point3_type{0., 0., 0.})};
210
211 if ((state.is_on_portal() && current_vol != state.volume()) ||
212 object_trace.empty() || first_obj ||
213 state.current().surface().identifier() !=
214 object_trace.back().intersection().surface().identifier()) {
215 DETRAY_VERBOSE_HOST_DEVICE("Actor: -> Record surface %d",
216 state.current_surface().index());
217
218 last_pos = algebra::cast_to<double>(pos);
219 last_dir = algebra::cast_to<double>(dir);
220 current_vol = state.volume();
221 object_trace.push_back({pos, dir, state.current()});
222 }
223 }
224 }
225
228 template <typename state_type>
229 DETRAY_HOST_DEVICE auto operator()(const state_type & /*state*/,
230 const char * /*message*/) { /* Do nothing*/
231 }
232
235 DETRAY_HOST_DEVICE
236 constexpr const candidate_record_t &operator[](std::size_t i) const {
237 return object_trace[i];
238 }
239
241 DETRAY_HOST_DEVICE
242 constexpr const auto &trace() const { return object_trace; }
243
248 DETRAY_HOST_DEVICE
249 constexpr bool is_status(const status &nav_stat, const status &ref_stat) {
250 return (nav_stat == ref_stat);
251 }
252};

305 template <typename state_type, concepts::point3D point3_t,
306 concepts::vector3D vector3_t,
307 typename message_generator_t = void_generator>
308 auto operator()(const state_type &state, const navigation::config &cfg,
309 const point3_t &track_pos, const vector3_t &track_dir,
310 const char *message,
311 const message_generator_t &msg_gen = {}) {
312 std::string msg(message);
313 debug_stream << msg;
314 if constexpr (!std::same_as<message_generator_t, void_generator>) {
315 debug_stream << msg_gen();
316
317 if (state.status() == navigation::status::e_abort) {
318 fata_error_msg = msg_gen();
319 }
320 }
321 debug_stream << std::endl;
322 debug_stream << "----------------------------------------" << std::endl;
323
324 debug_stream << navigation::print_state(state);
325 debug_stream << navigation::print_candidates(state, cfg, track_pos,
326 track_dir);
327
328 debug_stream << std::endl << std::endl;
329 }
