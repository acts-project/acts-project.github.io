76 // Calculates the (optimal) link starting point for rings.
77 template <concepts::transform3D transform3_t, concepts::algebra algebra_t>
78 dpoint3D<algebra_t> inline link_start(
79 const detray::mask<detray::ring2D, algebra_t>& mask,
80 const transform3_t& transform) const {
81 using shape_t = detray::ring2D;
82 using mask_t = detray::mask<shape_t, algebra_t>;
83 using point3_t = typename mask_t::point3_type;
84 using scalar_t = typename mask_t::scalar_type;
85
86 const scalar_t r{0.5f *
87 (mask[shape_t::e_inner_r] + mask[shape_t::e_outer_r])};
88 const scalar_t phi{detray::constant<scalar_t>::pi_2};
89
90 return mask_t::to_global_frame(transform, point3_t{r, phi, 0.f});
91 }

181 template <typename mask_group_t, concepts::index index_t,
182 concepts::detector detector_t, concepts::point3D point3_t,
183 concepts::vector3D vector3_t, concepts::scalar scalar_t>
184 DETRAY_HOST inline vector3_t operator()(
185 const mask_group_t& mask_group, const index_t& index,
186 const detector_t& detector,
187 const detray::tracking_volume<detector_t>& volume,
188 const point3_t& surface_point, const vector3_t& surface_normal,
189 const scalar_t& link_length, const std::size_t /*unused*/ = 0) const {
190 return link_dir(mask_group.at(index), detector, volume, surface_point,
191 surface_normal) *
192 link_length +
193 surface_point;

212 template <typename mask_group_t, concepts::interval idx_range_t,
213 concepts::detector detector_t, concepts::point3D point3_t,
214 concepts::vector3D vector3_t, concepts::scalar scalar_t>
215 DETRAY_HOST inline vector3_t operator()(
216 const mask_group_t& mask_group, const idx_range_t& idx_range,
217 const detector_t& detector,
218 const detray::tracking_volume<detector_t>& volume,
219 const point3_t& surface_point, const vector3_t& surface_normal,
220 const scalar_t& link_length, const std::size_t mask_idx) const {
221 assert(mask_idx < idx_range.size());
222 return link_dir(mask_group.at(idx_range.lower() + mask_idx), detector,
223 volume, surface_point, surface_normal) *
224 link_length +
225 surface_point;
226 }
227
228 private:
230 template <concepts::detector detector_t, typename mask_t,
231 concepts::point3D point3_t, concepts::vector3D vector3_t>
232 inline vector3_t link_dir(const mask_t& /*mask*/,
233 const detector_t& /*detector*/,
234 const detray::tracking_volume<detector_t>& volume,
235 const point3_t& surface_point,
236 const vector3_t& surface_normal) const {
237 const auto dir = volume.center() - surface_point;
238 const auto dot_prod = vector::dot(dir, surface_normal);
239
240 // Should geometrically not happen with a local point 'surface_point'
241 assert(dot_prod != 0.f);
242
243 return math::copysign(1.f, dot_prod) * surface_normal;
244 }

247 template <concepts::detector detector_t, concepts::point3D point3_t,
248 concepts::vector3D vector3_t, typename shape_t>
249 requires std::is_same_v<shape_t, cylinder2D> ||
250 std::is_same_v<shape_t, concentric_cylinder2D>
251 inline vector3_t link_dir(
252 const detray::mask<shape_t, typename detector_t::algebra_type>& mask,
253 const detector_t& detector,
254 const detray::tracking_volume<detector_t>& volume,
255 const point3_t& /*surface_point*/,
256 const vector3_t& surface_normal) const {
257 for (const auto& desc : volume.portals()) {
258 const detray::geometry::surface surface{detector, desc};
259
260 if (auto r =
261 surface
262 .template visit_mask<detray::detail::outer_radius_getter>()) {
263 if (*r > mask[shape_t::e_r]) {
264 return surface_normal;
265 }
266 }
267 }
268 return vector3_t{-1.f * surface_normal};
269 }
