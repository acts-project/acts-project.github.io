72 template <concepts::scalar scalar_t, concepts::point point_t>
73 DETRAY_HOST_DEVICE inline scalar_t min_dist_to_boundary(
74 const bounds_type<scalar_t> &bounds, const point_t &loc_p) const {
75 const scalar_t min_x_dist =
76 math::min(math::fabs(loc_p[0] - bounds[e_min_x]),
77 math::fabs(bounds[e_max_x] - loc_p[0]));
78 const scalar_t min_y_dist =
79 math::min(math::fabs(loc_p[1] - bounds[e_min_y]),
80 math::fabs(bounds[e_max_y] - loc_p[1]));
81 const scalar_t min_z_dist =
82 math::min(math::fabs(loc_p[2] - bounds[e_min_z]),
83 math::fabs(bounds[e_max_z] - loc_p[2]));
84
85 // Use the chord for the phi distance
86 return math::min(math::min(min_x_dist, min_y_dist), min_z_dist);
87 }

97 template <concepts::algebra algebra_t>
98 DETRAY_HOST_DEVICE constexpr result_type<dbool<algebra_t>> check_boundaries(
99 const bounds_type<dscalar<algebra_t>> &bounds,
100 const dtransform3D<algebra_t> &trf, const dpoint3D<algebra_t> &glob_p,
101 const dscalar<algebra_t> tol =
102 std::numeric_limits<dscalar<algebra_t>>::epsilon(),
103 const dscalar<algebra_t> /*edge_tol*/ = 0.f) const {
104 // Get the full local position
105 const dpoint3D<algebra_t> loc_p =
106 local_frame_type<algebra_t>::global_to_local(trf, glob_p, {});
107
108 return check_boundaries(bounds, loc_p, tol);

121 template <concepts::scalar scalar_t, concepts::point3D point_t>
122 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
123 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
124 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
125 const scalar_t /*edge_tol*/ = 0.f) const {
126 return ((bounds[e_min_x] - tol) <= loc_p[0] &&
127 (bounds[e_min_y] - tol) <= loc_p[1] &&
128 (bounds[e_min_x] - tol) <= loc_p[2] &&
129 loc_p[0] <= (bounds[e_max_x] + tol) &&
130 loc_p[1] <= (bounds[e_max_y] + tol) &&
131 loc_p[2] <= (bounds[e_max_z] + tol));
132 }

151 template <concepts::scalar scalar_t>
152 DETRAY_HOST_DEVICE constexpr scalar_t volume(
153 const bounds_type<scalar_t> &bounds) const {
154 return (bounds[e_max_x] - bounds[e_min_x]) *
155 (bounds[e_max_y] - bounds[e_min_y]) *
156 (bounds[e_max_z] - bounds[e_min_z]);
157 }

165 template <concepts::scalar scalar_t>
166 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
167 const bounds_type<scalar_t> &bounds,
168 const bounds_type<scalar_t> &o_bounds) const {
169 bounds_type<scalar_t> new_bounds{};
170
171 new_bounds[e_min_x] = math::min(bounds[e_min_x], o_bounds[e_min_x]);
172 new_bounds[e_min_y] = math::min(bounds[e_min_y], o_bounds[e_min_y]);
173 new_bounds[e_min_z] = math::min(bounds[e_min_z], o_bounds[e_min_z]);
174 new_bounds[e_max_x] = math::max(bounds[e_max_x], o_bounds[e_max_x]);
175 new_bounds[e_max_y] = math::max(bounds[e_max_y], o_bounds[e_max_y]);
176 new_bounds[e_max_z] = math::max(bounds[e_max_z], o_bounds[e_max_z]);
177
178 return new_bounds;
179 }

190 template <concepts::algebra algebra_t>
191 DETRAY_HOST_DEVICE inline darray<dscalar<algebra_t>, 6> local_min_bounds(
192 const bounds_type<dscalar<algebra_t>> &bounds,
193 const dscalar<algebra_t> env =
194 std::numeric_limits<dscalar<algebra_t>>::epsilon()) const {
195 assert(env > 0.f);
196 bounds_type<dscalar<algebra_t>> o_bounds{bounds};
197 for (unsigned int i{0}; i < 3u; ++i) {
198 o_bounds[i] -= env;
199 o_bounds[i + 3u] += env;
200 }
201 return o_bounds;
202 }

206 template <concepts::algebra algebra_t>
207 DETRAY_HOST_DEVICE dpoint3D<algebra_t> centroid(
208 const bounds_type<dscalar<algebra_t>> &bounds) const {
209 return 0.5f * dpoint3D<algebra_t>{bounds[e_min_x] + bounds[e_max_x],

235 template <concepts::scalar scalar_t>
236 DETRAY_HOST constexpr bool check_consistency(
237 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
238 constexpr auto tol{10.f * std::numeric_limits<scalar_t>::epsilon()};
239
240 if (bounds[e_min_x] >= bounds[e_max_x] ||
241 math::fabs(bounds[e_min_x] - bounds[e_max_x]) < tol) {
242 os << "DETRAY ERROR (HOST): Min x must be smaller than max x.";
243 return false;
244 }
245 if (bounds[e_min_y] >= bounds[e_max_y] ||
246 math::fabs(bounds[e_min_y] - bounds[e_max_y]) < tol) {
247 os << "DETRAY ERROR (HOST): Min y must be smaller than max y.";
248 return false;
249 }
250 if (bounds[e_min_z] >= bounds[e_max_z] ||
251 math::fabs(bounds[e_min_z] - bounds[e_max_z]) < tol) {
252 os << "DETRAY ERROR (HOST): Min z must be smaller than max z.";
253 return false;
254 }
255
256 return true;
257 }
