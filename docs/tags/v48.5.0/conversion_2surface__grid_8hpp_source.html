40struct bin_association_getter {
50 template <typename group_t, typename index_t, typename volume_t>
51 DETRAY_HOST_DEVICE std::vector<std::vector<std::size_t>> operator()(
52 [[maybe_unused]] const group_t& group,
53 [[maybe_unused]] const index_t index,
54 [[maybe_unused]] const volume_t& vol_desc,
55 [[maybe_unused]] const std::array<dindex, 2>& search_window) const {
56 using accel_t = typename group_t::value_type;
57
58 if constexpr (concepts::surface_grid<accel_t>) {
59 using transform3_t = typename accel_t::local_frame_type::transform3_type;
60 using algebra_t = typename accel_t::local_frame_type::algebra_type;
61 using scalar_t = typename transform3_t::scalar_type;
62 using point2_t = typename transform3_t::point2;
63
64 // The actsvg display only works for 2-dimensional grids
65 if constexpr (accel_t::dim != 2u) {
66 DETRAY_ERROR_HOST("Only 2D grids can be displayed in actvg");
67 return {};
68 }
69
70 const accel_t grid = group.at(index);
71 const std::size_t n_bins{grid.nbins()};
72
73 std::vector<std::vector<std::size_t>> bin_assoc;
74 bin_assoc.reserve(n_bins);
75
76 // Create the bin associations
77 auto edges0 = grid.template get_axis<0>().bin_edges();
78 auto edges1 = grid.template get_axis<1>().bin_edges();
79
80 // In the svg convention the phi axis has to be the second axis to
81 // loop over
82 constexpr bool is_cyl{
83 std::is_same_v<typename accel_t::local_frame_type,
84 detray::cylindrical2D<algebra_t>> ||
85 std::is_same_v<typename accel_t::local_frame_type,
86 detray::concentric_cylindrical2D<algebra_t>>};
87 if constexpr (is_cyl) {
88 edges0.swap(edges1);
89 }
90
91 for (std::size_t i = 1u; i < edges0.size(); ++i) {
92 scalar_t p0 = 0.5f * (edges0.at(i) + edges0.at(i - 1));
93
94 for (std::size_t j = 1u; j < edges1.size(); ++j) {
95 scalar_t p1 = 0.5f * (edges1.at(j) + edges1.at(j - 1));
96
97 // Create the bin center position estimates for detray
98 // (swap cylinder coordinates back)
99 point2_t bin_center{p0, p1};
100 if constexpr (is_cyl) {
101 bin_center = {p1, p0};
102 }
103
104 // Get all the bin entries and calculate the loc index
105 std::vector<std::size_t> entries;
106 for (const auto& sf_desc : grid.search(bin_center, search_window)) {
107 // actsvg expects the sensitive surfaces to be numbered
108 // starting from zero (per volume)
109 entries.push_back(vol_desc.to_local_sf_index(sf_desc.index()));
110 }
111
112 // Remove duplicates
113 std::ranges::sort(entries);
114 auto last = std::unique(entries.begin(), entries.end());
115 entries.erase(last, entries.end());
116
117 bin_assoc.push_back(std::move(entries));
118 }
119 }
120
121 return bin_assoc;
122 }
123
124 return {};
125 }
126};
127
128} // namespace detail

138template <concepts::detector detector_t, typename view_t>
139auto surface_grid(const detector_t& detector, const dindex index,
140 const view_t& view,
141 const styling::grid_style& style =
142 styling::tableau_colorblind::grid_style) {
143 using scalar_t = dscalar<typename detector_t::algebra_type>;
144 using geo_object_ids = typename detector_t::geo_obj_ids;
145
146 const auto vol_desc = detector.volume(index);
147 const auto link = vol_desc.template accel_link<geo_object_ids::e_sensitive>();
148
149 // Proactively calculate the reference radius for a cylinder grid
150 // (will only be used if the volume actually holds a barrel grid)
151
152 // Get the the radii of the volume portal surfaces
153 std::vector<scalar_t> radii{};
154 const auto vol = tracking_volume{detector, vol_desc};
155
156 // Passive surfaces could be in the brute force finder, but no
157 // sensitive surfaces, since the volume has a grid. Their radii are,
158 // however, always within the interval of the portal radii
159 for (const auto& pt_desc : vol.portals()) {
160 auto r = detector.mask_store()
161 .template visit<detray::detail::outer_radius_getter>(
162 pt_desc.mask());
163 if (r.has_value()) {
164 radii.push_back(*r);
165 }
166 }
167
168 scalar_t cyl_ref_radius{0.f};
169 if (!radii.empty()) {
170 scalar_t inner_r = *std::ranges::min_element(radii);
171 scalar_t outer_r = *std::ranges::max_element(radii);
172
173 cyl_ref_radius = 0.5f * static_cast<actsvg::scalar>(inner_r + outer_r);
174 }
175
176 return svgtools::conversion::grid(detector.accelerator_store(), link, view,
177 cyl_ref_radius, style);
178}

187template <concepts::detector detector_t>
188std::vector<std::vector<std::size_t>> get_bin_association(
189 const detector_t& det, const detray::tracking_volume<detector_t>& vol,
190 const std::array<dindex, 2>& search_window = {2u, 2u}) {
191 using geo_object_ids = typename detector_t::geo_obj_ids;
192
193 const auto& vol_desc = det.volume(vol.index());
194
195 if (const auto& link =
196 vol_desc.template accel_link<geo_object_ids::e_sensitive>();
197 !link.is_invalid()) {
198 return det.accelerator_store()
199 .template visit<detail::bin_association_getter>(link, vol_desc,
200 search_window);
201 }
202
203 return {};
204}
