59 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
60 const trajectory_type<algebra_t> &ray, const dtransform3D<algebra_t> &trf,
61 const dscalar<algebra_t> /*overstep_tol*/ = 0.f) const {
62 using scalar_t = dscalar<algebra_t>;
63 using point3_t = dpoint3D<algebra_t>;
64 using vector3_t = dvector3D<algebra_t>;
65
66 const vector3_t &rd = ray.dir();
67 const point3_t &ro = ray.pos();
68
69 // line direction
70 const vector3_t &_z = trf.z();
71 // line center
72 const point3_t &_t = trf.translation();
73
74 // Projection of line to track direction
75 const scalar_t zd{vector::dot(_z, rd)};
76
77 const scalar_t denom{1.f - (zd * zd)};
78
79 // Case for wire is parallel to track
80 if (denom < 1e-5f) [[unlikely]] {
81 return {};
82 }
83
84 // vector from track position to line center
85 const point3_t t2l = _t - ro;
86
87 // t2l projection on line direction
88 const scalar_t t2l_on_line{vector::dot(t2l, _z)};
89
90 // t2l projection on track direction
91 const scalar_t t2l_on_track{vector::dot(t2l, rd)};
92
93 // path length to the point of closest approach on the track
94 const scalar_t s{1.f / denom * (t2l_on_track - t2l_on_line * zd)};
95 const point3_t glob_pos = ro + s * rd;
96
97 return {s, glob_pos};
98 }
99};
