49 template <value_t mask>
50 requires(mask != static_cast<value_t>(0))
51 DETRAY_HOST_DEVICE static constexpr value_t get_bits(
52 const value_t v) noexcept {
53 // Use integral constant to enforce compile time evaluation of shift
54 return (v & mask) >> extract_shift(mask);
55 }

61 template <value_t mask>
62 requires(mask != static_cast<value_t>(0))
63 DETRAY_HOST_DEVICE static constexpr void set_bits(value_t& v,
64 const value_t id) noexcept {
65 // Use integral constant to enforce compile time evaluation of shift
66 v = (v & ~mask) | ((id << extract_shift(mask)) & mask);
67
68 // Make sure, the value 'id' can be safely encoded with the bits
69 // specified by 'mask' (unless it is an invalid value which has all bits
70 // set to 1)
71 assert(((id == ~static_cast<value_t>(0)) ||
72 (id == static_cast<value_t>(~static_cast<unsigned int>(0))) ||
73 (get_bits<mask>(v) == id)) &&
74 "Not enough bits in mask to encode value");
75 }
76
77 private:
81 DETRAY_HOST_DEVICE
82 static constexpr int extract_shift(value_t mask) noexcept {
83 return __builtin_ctzll(mask);
84 }
85};
