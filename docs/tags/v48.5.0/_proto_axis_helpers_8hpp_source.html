39[[deprecated("Unused and will be removed - use the IAxis interface instead")]]
40inline std::size_t totalBinsFromProtoAxes(
41 std::span<const DirectedProtoAxis> axes) {
42 if (axes.empty() || axes.size() > 3) {
43 throw std::runtime_error(
44 "Unsupported number of axes, must be 1-3, instead got " +
45 std::to_string(axes.size()) + ")");
46 }
47 return axes[0].getAxis().getNBins() *
48 (axes.size() > 1 ? axes[1].getAxis().getNBins() : 1) *
49 (axes.size() > 2 ? axes[2].getAxis().getNBins() : 1);
50}

98[[deprecated("Unused and will be removed - use the IAxis interface instead")]]
99inline std::array<std::size_t, 3> binTripleFromProtoAxes(
100 std::span<const DirectedProtoAxis> axes, const Vector3& gp) {
101 const Vector3& bPosition = gp;
102 std::array<std::size_t, 3> bTriple = {0, 0, 0};
103 if (axes.empty() || axes.size() > 3) {
104 throw std::runtime_error(
105 "Unsupported number of axes, must be 1-3, instead got " +
106 std::to_string(axes.size()) + ")");
107 }
108 if (axes.size() == 1) {
109 BinningData bd0(axes[0]);
110 bTriple[0] = bd0.searchGlobal(bPosition);
111 }
112 if (axes.size() == 2) {
113 BinningData bd1(axes[1]);
114 bTriple[1] = bd1.searchGlobal(bPosition);
115 }
116 if (axes.size() == 3) {
117 BinningData bd2(axes[2]);
118 bTriple[2] = bd2.searchGlobal(bPosition);
119 }
120 return bTriple;
121}

128[[deprecated("Unused and will be removed - use the IAxis interface instead")]]
129inline std::size_t maxBin(std::span<const DirectedProtoAxis> axes,
130 std::size_t ba = 0) {
131 if (axes.empty() || axes.size() > 3) {
132 throw std::runtime_error(
133 "Unsupported number of axes, must be 1-3, instead got " +
134 std::to_string(axes.size()) + ")");
135 }
136 std::vector<BinningData> binningDataVec;
137 binningDataVec.reserve(axes.size());
138 for (const auto& axis : axes) {
139 binningDataVec.emplace_back(axis.getAxisDirection(), axis.getAxis());
140 }
141 if (ba >= binningDataVec.size()) {
142 return 0;
143 }
144 return (binningDataVec.at(ba).bins() - 1);
145}
