254template <concepts::algebra algebra_t, typename mask_shape_t = rectangle2D,
255 template <typename> class trajectory_t = detail::ray>
256inline auto build_telescope_detector(
257 vecmem::memory_resource &resource,
258 const tel_det_config<algebra_t, mask_shape_t, trajectory_t> &cfg =
259 tel_det_config<algebra_t, mask_shape_t, trajectory_t>{
260 20.f * unit<dscalar<algebra_t>>::mm,
261 20.f * unit<dscalar<algebra_t>>::mm}) {
262 using scalar_t = dscalar<algebra_t>;
263 using metadata_t = telescope_metadata<algebra_t, mask_shape_t>;
264 using builder_t = detector_builder<metadata_t, volume_builder>;
265 using detector_t = typename builder_t::detector_type;
266
267 builder_t det_builder;
268 det_builder.set_name("telescope_detector");
269
270 // Create an empty cuboid volume
271 auto v_builder = det_builder.new_volume(volume_id::e_cuboid);
272 v_builder->set_name("telescope_world_0");
273
274 // Identity transform (volume is centered at origin)
275 v_builder->add_volume_placement();
276
277 // Add module surfaces to volume
278 using telescope_factory =
279 telescope_generator<detector_t, mask_shape_t, trajectory_t<algebra_t>>;
280 std::unique_ptr<surface_factory_interface<detector_t>> tel_generator;
281
282 if (cfg.positions().empty()) {
283 // Automatically position the modules along pilot track
284 tel_generator = std::make_unique<telescope_factory>(
285 cfg.length(), cfg.n_surfaces(), cfg.module_mask().values(),
286 cfg.pilot_track());
287 } else {
288 // Put the modules in the requested portions along pilot track
289 tel_generator = std::make_unique<telescope_factory>(
290 cfg.positions(), cfg.module_mask().values(), cfg.pilot_track());
291 }
292
293 // Add homogeneous material description if a valid material was configured
294 volume_builder_interface<detector_t> *vm_builder{v_builder};
295 std::shared_ptr<surface_factory_interface<detector_t>> module_generator;
296
297 if (cfg.module_material() != detray::vacuum<scalar_t>{}) {
298 // Decorate the builders with homogeneous material
299 vm_builder =
300 det_builder.template decorate<homogeneous_material_builder<detector_t>>(
301 v_builder);
302
303 if (!vm_builder) {
304 throw std::runtime_error("Surface material decoration failed");
305 }
306
307 auto tel_mat_generator =
308 std::make_shared<homogeneous_material_generator<detector_t>>(
309 std::move(tel_generator), cfg.material_config());
310
311 module_generator = std::move(tel_mat_generator);
312
313 } else {
314 module_generator = std::move(tel_generator);
315 }
316
317 // Add a bounding box of portals to the cuboid volume
318 auto portal_generator =
319 std::make_shared<cuboid_portal_generator<detector_t>>(cfg.envelope());
320
321 vm_builder->add_surfaces(module_generator);
322 // (!) The portals must be added after the modules to fit them correctly
323 vm_builder->add_surfaces(portal_generator);
324
325 // TODO: Add brute force volume searcher
326
327 // If requested, add homogeneous volume material
328 if (cfg.volume_material() != detray::vacuum<scalar_t>{}) {
329 auto full_v_builder =
330 det_builder
331 .template decorate<homogeneous_volume_material_builder<detector_t>>(
332 vm_builder);
333
334 if (full_v_builder) {
335 full_v_builder->set_material(cfg.volume_material());
336 } else {
337 throw std::runtime_error("Volume material decoration failed");
338 }
339 }
340
341 // Build and return the detector and fill the name map
342 typename detector_t::name_map name_map{};
343 auto det = det_builder.build(resource, cfg.builder_options(), name_map);
344
345 if (cfg.do_check()) {
346 const bool verbose_check{false};
347 detray::detail::check_consistency(det, verbose_check, name_map);
348 }
349
350 DETRAY_DEBUG_HOST("\n" << detray::utils::print_detector(det, name_map));
351
352 return std::make_pair(std::move(det), std::move(name_map));
353}
