40template <class detector_t, class backend_t>
41 requires concepts::reader_backend<detector_t, backend_t>
42class json_converter<detector_t, backend_t> final
43 : public reader_interface<detector_t> {
44 using io_backend = backend_t;
45
46 public:
48 json_converter() : reader_interface<detector_t>(".json") {}
49
53 void read(detector_builder<typename detector_t::metadata, volume_builder>&
54 det_builder,
55 const std::string& file_name) override {
56 // Read json from file
57 io::file_handle file{file_name, std::ios_base::in | std::ios_base::binary};
58
59 // Reads the data from file and returns the corresponding io payloads
60 nlohmann::json in_json;
61 *file >> in_json;
62
63 // Add the data from the payload to the detray detector builder
64 io_backend::template from_payload<detector_t>(det_builder, in_json["data"]);
65 }
66};

76template <class detector_t, class backend_t>
77 requires concepts::writer_backend<detector_t, backend_t>
78class json_converter<detector_t, backend_t> final
79 : public writer_interface<detector_t> {
80 using io_backend = backend_t;
81
82 public:
84 json_converter() : writer_interface<detector_t>(".json") {}
85
92 std::string write(const detector_t& det,
93 const typename detector_t::name_map& names,
94 const std::ios_base::openmode mode = std::ios::out |
95 std::ios::binary,
96 const std::filesystem::path& file_path = {"./"}) override {
97 // Assert output stream
98 assert(((mode == std::ios_base::out) ||
99 (mode == (std::ios_base::out | std::ios_base::binary)) ||
100 (mode == (std::ios_base::out | std::ios_base::trunc)) ||
101 (mode == (std::ios_base::out | std::ios_base::trunc |
102 std::ios_base::binary))) &&
103 "Illegal file mode for json writer");
104
105 std::string det_name = det.name(names);
106
107 // Create a new file
108 std::string file_stem{det_name + "_" + std::string(io_backend::tag)};
109 io::file_handle file{file_path / file_stem, this->file_extension(), mode};
110
111 // Write some general information
112 nlohmann::ordered_json out_json;
113 out_json["header"] = io_backend::header_to_payload(det, det_name);
114
115 // Write the detector data into the json stream by using the
116 // conversion functions defined in "detray/io/json/json_io.hpp"
117 out_json["data"] = io_backend::to_payload(det, names);
118
119 // Write to file
120 *file << std::setw(4) << out_json << std::endl;
121
122 return file_stem + this->file_extension();
123 }
124};
