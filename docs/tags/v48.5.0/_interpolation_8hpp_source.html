29template <typename T, std::size_t N, typename P1, typename P2, typename P3>
30concept Interpolatable = requires {
31 std::has_single_bit(N);
32 {
33 std::declval<double>() * std::declval<T>() +
34 std::declval<double>() * std::declval<T>()
35 } -> std::convertible_to<T>;
36 { std::declval<P1>()[0] } -> std::convertible_to<double>;
37 { std::declval<P2>()[0] } -> std::convertible_to<double>;
38 { std::declval<P3>()[0] } -> std::convertible_to<double>;
39};

96template <typename T, std::size_t N, class Point1, class Point2 = Point1,
97 class Point3 = Point2>
98T interpolate(const Point1& point, const Point2& lowerCorner,
99 const Point3& upperCorner, const std::array<T, N>& values)
100 requires Concepts::Interpolatable<T, N, Point1, Point2, Point3>
101{
102 static constexpr std::size_t D = std::bit_width(N) - 1;
103 static constexpr std::size_t I = D - 1;
104
105 // get distance to lower boundary relative to total bin width
106 const double f =
107 (point[I] - lowerCorner[I]) / (upperCorner[I] - lowerCorner[I]);
108 assert(f >= 0 && f <= 1 && "point must be inside the given hyper box");
109
110 std::array<T, N / 2> newValues{};
111 for (std::size_t i = 0; i < N / 2; ++i) {
112 newValues[i] = (1 - f) * values[2 * i] + f * values[2 * i + 1];
113 }
114
115 if constexpr (D == 1) {
116 return newValues[0];
117 } else {
118 return interpolate(point, lowerCorner, upperCorner, newValues);
119 }
120}
