111 template <bool EnsureConst, typename T>
112 static std::any component_impl(T& instance, Acts::HashedString key,
113 IndexType itrack) {
114 if constexpr (EnsureConst) {
115 static_assert(std::is_const_v<std::remove_reference_t<T>>,
116 "Is not const");
117 }
118
119 using namespace Acts::HashedStringLiteral;
120 auto track = instance.m_collection->at(itrack);
121 std::conditional_t<EnsureConst, const ActsPodioEdm::TrackInfo*,
122 ActsPodioEdm::TrackInfo*>
123 dataPtr;
124 if constexpr (EnsureConst) {
125 dataPtr = &track.getData();
126 } else {
127 dataPtr = &PodioUtil::getDataMutable(track);
128 }
129 auto& data = *dataPtr;
130 switch (key) {
131 case "tipIndex"_hash:
132 return &data.tipIndex;
133 case "stemIndex"_hash:
134 return &data.stemIndex;
135 case "params"_hash:
136 return data.parameters.data();
137 case "cov"_hash:
138 return data.covariance.data();
139 case "nMeasurements"_hash:
140 return &data.nMeasurements;
141 case "nHoles"_hash:
142 return &data.nHoles;
143 case "chi2"_hash:
144 return &data.chi2;
145 case "ndf"_hash:
146 return &data.ndf;
147 case "nOutliers"_hash:
148 return &data.nOutliers;
149 case "nSharedHits"_hash:
150 return &data.nSharedHits;
151 case "nSplitHits"_hash:
152 return &data.nSplitHits;
153 default:
154 auto it = instance.m_dynamic.find(key);
155 if (it == instance.m_dynamic.end()) {
156 throw std::runtime_error("Unable to handle this component");
157 }
158
159 std::conditional_t<EnsureConst,
160 const podio_detail::ConstDynamicColumnBase*,
161 podio_detail::DynamicColumnBase*>
162 col = it->second.get();
163 assert(col && "Dynamic column is null");
164 return col->get(itrack);
165 }
166 }

183 template <typename T>
184 static Acts::ParticleHypothesis particleHypothesis_impl(T& instance,
185 IndexType itrack) {
186 auto track = instance.m_collection->at(itrack);
187 const auto& src = track.getParticleHypothesis();
188 return Acts::ParticleHypothesis{static_cast<Acts::PdgParticle>(src.absPdg),
189 src.mass, Acts::ChargeHypothesis{src.absQ}};
190 }
191
196 static void populateSurfaceBuffer(
197 const PodioUtil::ConversionHelper& helper,
198 const ActsPodioEdm::TrackCollection& collection,
199 std::vector<std::shared_ptr<const Acts::Surface>>& surfaces) noexcept {
200 surfaces.reserve(collection.size());
201 for (ActsPodioEdm::Track track : collection) {
202 surfaces.push_back(PodioUtil::convertSurfaceFromPodio(
203 helper, track.getReferenceSurface()));
204 }
205 }
206
208 std::reference_wrapper<const PodioUtil::ConversionHelper> m_helper;
210 std::vector<std::shared_ptr<const Acts::Surface>> m_surfaces;
211};

420 void releaseInto(podio::Frame& frame, const std::string& suffix = "")
421 requires(std::is_same_v<holder_t<ActsPodioEdm::TrackCollection>,
422 Acts::ValueHolder<ActsPodioEdm::TrackCollection>> ||
423 std::is_same_v<holder_t<ActsPodioEdm::TrackCollection>,
424 std::unique_ptr<ActsPodioEdm::TrackCollection>>)
425 {
426 std::string s = suffix;
427 if (!s.empty()) {
428 s = "_" + s;
429 }
430
431 if constexpr (std::is_same_v<
432 holder_t<ActsPodioEdm::TrackCollection>,
433 std::unique_ptr<ActsPodioEdm::TrackCollection>>) {
434 frame.put(std::move(m_collection), "tracks" + s);
435 } else {
436 frame.put(std::make_unique<ActsPodioEdm::TrackCollection>(
437 std::move(*m_collection)),
438 "tracks" + s);
439 }
440 m_surfaces.clear();
441
442 for (const auto& [key, col] : m_dynamic) {
443 col->releaseInto(frame, "tracks" + s + "_extra__");
444 }
445 }

507 ConstPodioTrackContainer(const PodioUtil::ConversionHelper& helper,
508 const podio::Frame& frame,

512 Acts::ConstRefHolder<const ActsPodioEdm::TrackCollection>>
