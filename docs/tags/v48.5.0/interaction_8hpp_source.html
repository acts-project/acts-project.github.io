54 DETRAY_HOST_DEVICE scalar_type
55 compute_bethe_bloch(const detray::material<scalar_type>& mat,
56 const pdg_particle<scalar_type>& /*ptc*/,
57 const relativistic_quantities& rq) const {
58 const scalar_type eps_per_length{rq.compute_epsilon_per_length(mat)};
59 if (eps_per_length <= 0.f) {
60 return 0.f;
61 }
62
63 const scalar_type dhalf{rq.compute_delta_half(mat)};
64 const scalar_type A = rq.compute_bethe_bloch_log_term(mat);
65 const scalar_type running{A - rq.m_beta2 - dhalf};
66 return 2.f * eps_per_length * running;
67 }

69 // Function to calculate the Bremsstrahlung energy loss of electron based on
70 // Rossi's approximation.
71 // For more rigorous calculation, check "G4eBremsstrahlungRelModel.cc" of
72 // Geant4 library. The Fig 34.13 of PDG 2024 shows the difference between
73 // the approximated and precise calculation.
80 DETRAY_HOST_DEVICE scalar_type
81 compute_bremsstrahlung(const detray::material<scalar_type>& mat,
82 const pdg_particle<scalar_type>& ptc,
83 const relativistic_quantities& rq) const {
84 scalar_type stopping_power{0.f};
85
86 // Only consider electrons and positrons at the moment
87 // For middle-heavy particles muons, the bremss is negligibe
88 //
89 // Also, over 10 MeV, positron and electron bremss might be similar.
90 // In ICRU 37, it is written that "In our tabulations, the radiative
91 // stopping power for positrons has been assumed to be the same as that
92 // for electrons, which is a good approximation at energies above, say,
93 // 10 MeV. However, it should be mentioned that exploratory calculations
94 // by Feng et at. (1981), employing the same method as that previously
95 // used by them for electrons, indicate significant differences between
96 // positrons and electrons in regard to the differential bremsstrahlung
97 // cross sections in oxygen and uranium at 500, 50, and 10 keV"
98 if (ptc.pdg_num() == electron<scalar_type>().pdg_num() ||
99 ptc.pdg_num() == positron<scalar_type>().pdg_num()) {
100 // Stopping power ~ E/X (B. B. Rossi, High-energy particles,1952)
101 // This approximation gets poor in low energy below 10 MeV
102 stopping_power = rq.m_gamma * ptc.mass() / mat.X0();
103 }
104
105 return stopping_power;
106 }

137 DETRAY_HOST_DEVICE scalar_type
138 derive_bethe_bloch(const detray::material<scalar_type>& mat,
139 const pdg_particle<scalar_type>& ptc,
140 const relativistic_quantities& rq) const {
141 const scalar_type bethe_stopping_power = compute_bethe_bloch(mat, ptc, rq);
142
143 // (K/2) * (Z/A) * z^2 / beta^2 * density
144 const scalar_type eps_per_length{rq.compute_epsilon_per_length(mat)};
145 if (eps_per_length <= 0.f) {
146 return 0.f;
147 }
148
149 /*-----------------------------------------------------------------------
150 * Calculation of d(-dE/dx)/dqop
151 *
152 * d(-dE/dx)/dqop = 2/(qop * gamma^2) * (-dE/dx)
153 * + 2 * (eps/x) * [dA/dqop - dB/dqop - dC/dqop]
154 *
155 * where
156 * A = 1/2 ln ( 2m_e c^2 beta^2 gamma^2 W_max/ I^2)
157 * B = beta^2
158 * C = delta/2
159 *
160 * dA/dqop = -1 / (2 * qop) * [4 - W_max/ (gamma M c^2) ]
161 * dB/dqop = -2 * beta^2/(qop gamma^2)
162 *
163 * dC/dqop = 1/2*(-2/qop) if (x>x_1)
164 * = 1/2*(-2/qop + ak/(qop ln10)*(x_1 - x)^(k-1)) if (x_0<x<x_1)
165 * = 0 if (x<x_0) (for nonconductors)
166 ------------------------------------------------------------------------*/
167
168 const scalar_type first_term =
169 2.f / (rq.m_qOverP * rq.m_gamma2) * bethe_stopping_power;
170
171 const scalar_type dAdqop = rq.derive_bethe_bloch_log_term();
172 const scalar_type dBdqop = rq.derive_beta2();
173 const scalar_type dCdqop = rq.derive_delta_half(mat);
174
175 const scalar_type second_term =
176 2.f * eps_per_length * (dAdqop - dBdqop - dCdqop);
177
178 return first_term + second_term;
179 }

186 DETRAY_HOST_DEVICE scalar_type
187 derive_bremsstrahlung(const detray::material<scalar_type>& mat,
188 const pdg_particle<scalar_type>& ptc,
189 const relativistic_quantities& rq) const {
190 scalar_type derivative{0.f};
191
192 if (ptc.pdg_num() == electron<scalar_type>().pdg_num() ||
193 ptc.pdg_num() == positron<scalar_type>().pdg_num()) {
194 // Stopping power ~ E/X = gamma * m/X
195 // Derivative = dgamma/dqop * m/X = -beta^2 gamma/qop * m/X
196 // (Eq (D.5) of arXiv:2403.16720)
197 derivative =
198 -rq.m_beta2 * rq.m_gamma / rq.m_qOverP * ptc.mass() / mat.X0();
199 }
200
201 return derivative;
202 }

222 DETRAY_HOST_DEVICE scalar_type compute_energy_loss_landau(
223 const scalar_type path_segment, const material<scalar_type>& mat,
224 const pdg_particle<scalar_type>& /*ptc*/,
225 const relativistic_quantities& rq) const {
226 const scalar_type I{mat.mean_excitation_energy()};
227 const scalar_type eps{rq.compute_epsilon(mat, path_segment)};
228
229 if (eps <= 0.f) {
230 return 0.f;
231 }
232
233 const scalar_type dhalf{rq.compute_delta_half(mat)};
234 const scalar_type t{rq.compute_mass_term(constant<scalar_type>::m_e)};
235 // uses RPP2018 eq. 33.11
236 const scalar_type running{math::log(t / I) + math::log(eps / I) + 0.2f -
237 rq.m_beta2 - 2.f * dhalf};
238 return eps * running;
239 }

246 DETRAY_HOST_DEVICE scalar_type compute_energy_loss_landau_fwhm(
247 const scalar_type path_segment, const material<scalar_type>& mat,
248 const pdg_particle<scalar_type>& /*ptc*/,
249 const relativistic_quantities& rq) const {
250 // the Landau-Vavilov fwhm is 4*eps (see RPP2018 fig. 33.7)

276 template <typename material_t>
277 DETRAY_HOST_DEVICE scalar_type compute_energy_loss_landau_sigma_QOverP(
278 const scalar_type path_segment, const material_t& mat,
279 const pdg_particle<scalar_type>& ptc,
280 const relativistic_quantities& rq) const {
281 const scalar_type sigmaE{
282 compute_energy_loss_landau_sigma(path_segment, mat, ptc, rq)};
283
284 // var(q/p) = (d(q/p)/dE)² * var(E)
285 // d(q/p)/dE = d/dE (q/sqrt(E²-m²))
286 // = q * -(1/2) * 1/p³ * 2E
287 // var(q/p) = (q/p)^4 * (q/beta)² * (1/q)^4 * var(E)
288 // = -q/p² E/p = -(q/p)² * 1/(q*beta) = -(q/p)² * (q/beta)
289 // / q² = (1/p)^4 * (q/beta)² * var(E)
290 // do not need to care about the sign since it is only used squared
291 const scalar_type pInv{rq.m_qOverP / ptc.charge()};
292 return math::sqrt(rq.m_q2OverBeta2) * pInv * pInv * sigmaE;
293 }

300 DETRAY_HOST_DEVICE scalar_type compute_multiple_scattering_theta0(
301 const scalar_type xOverX0, const pdg_particle<scalar_type>& ptc,
302 const relativistic_quantities& rq) const {
303 // 1/p = q/(pq) = (q/p)/q
304 const scalar_type momentumInv{math::fabs(rq.m_qOverP / ptc.charge())};
305 // q²/beta²; a smart compiler should be able to remove the unused
306 // computations
307
308 // if electron or positron
309 if ((ptc.pdg_num() == 11) || (ptc.pdg_num() == -11)) {
310 //@todo (Beomki): Not sure if we need this function. At least we
311 // need to find the reference for this equation
312 return theta0RossiGreisen(xOverX0, momentumInv, rq.m_q2OverBeta2);
313 } else {
314 return theta0Highland(xOverX0, momentumInv, rq.m_q2OverBeta2);
315 }
316 }

318 private:
322 DETRAY_HOST_DEVICE scalar_type
323 theta0Highland(const scalar_type xOverX0, const scalar_type momentumInv,
324 const scalar_type q2OverBeta2) const {
325 if (xOverX0 <= 0.f) {
326 return 0.f;
327 }
328
329 // RPP2018 eq. 33.15 (treats beta and q² consistently)
330 const scalar_type t{math::sqrt(xOverX0 * q2OverBeta2)};
331 // log((x/X0) * (q²/beta²)) = log((sqrt(x/X0) * (q/beta))²)
332 // = 2 * log(sqrt(x/X0) * (q/beta))
333 return 13.6f * unit<scalar_type>::MeV * momentumInv * t *
334 (1.0f + 0.038f * 2.f * math::log(t));
335 }

338 DETRAY_HOST_DEVICE scalar_type
339 theta0RossiGreisen(const scalar_type xOverX0, const scalar_type momentumInv,
340 const scalar_type q2OverBeta2) const {
341 if (xOverX0 <= 0.f) {
342 return 0.f;
343 }
344
345 // TODO add source paper/ resource
346 const scalar_type t{math::sqrt(xOverX0 * q2OverBeta2)};
347 return 17.5f * unit<scalar_type>::MeV * momentumInv * t *
348 (1.0f + 0.125f * math::log10(10.0f * xOverX0));
349 }

360 DETRAY_HOST_DEVICE scalar_type
361 convert_landau_fwhm_to_gaussian_sigma(const scalar_type fwhm) const {
362 return 0.5f * constant<scalar_type>::inv_sqrt2 * fwhm /
363 math::sqrt(constant<scalar_type>::ln2);
