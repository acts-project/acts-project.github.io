34template <concepts::scalar scalar_t, concepts::value value_t>
35 requires std::convertible_to<scalar_t, value_t>
36DETRAY_HOST_DEVICE constexpr auto approx_equal(
37 const scalar_t a, const scalar_t b,
38 const value_t rel_error = 16.f * std::numeric_limits<value_t>::epsilon(),
39 const value_t max_error = std::numeric_limits<value_t>::epsilon()) {
40 if constexpr (std::integral<scalar_t>) {
41 return a == b;
42 } else {
43 // Calculate the difference.
44 const scalar_t diff{math::fabs(a - b)};
45
46 // If the numbers are close to zero
47 if (diff <= max_error) {
48 return true;
49 }
50
51 // Find the largest value to scale the epsilon
52 const scalar_t largest = math::max(math::fabs(a), math::fabs(b));
53
54 return (diff <= largest * rel_error);
55 }
56}

68template <typename vector1_t, typename vector2_t>
69 requires((concepts::vector<vector1_t> || concepts::point<vector1_t>) &&
70 (concepts::vector<vector2_t> || concepts::point<vector2_t>) &&
71 !concepts::simd_scalar<vector1_t> &&
72 !concepts::simd_scalar<vector2_t>)
73DETRAY_HOST_DEVICE constexpr auto approx_equal(
74 const vector1_t& v1, const vector2_t& v2,
75 const detray::traits::value_t<vector1_t> rel_error =
76 16.f *
77 std::numeric_limits<detray::traits::value_t<vector1_t>>::epsilon(),
78 const detray::traits::value_t<vector1_t> max_error =
79 std::numeric_limits<detray::traits::value_t<vector1_t>>::epsilon()) {
80 static_assert(std::same_as<detray::traits::value_t<vector1_t>,
81 detray::traits::value_t<vector2_t>>);
82
83 using index_t = detray::traits::index_t<vector1_t>;
84
85 constexpr index_t size{detray::traits::size<vector1_t>};
86
87 bool ret{true};
88 for (index_t i = 0; i < size; ++i) {
89 ret = ret &&
90 ::detray::algebra::approx_equal(v1[i], v2[i], rel_error, max_error);
91 }
92
93 return ret;
94}

183template <concepts::transform3D transform_t>
184DETRAY_HOST_DEVICE constexpr auto approx_equal(
185 const transform_t& trf1, const transform_t& trf2,
186 const detray::traits::value_t<typename transform_t::scalar_type> rel_error =
187 16.f * std::numeric_limits<detray::traits::value_t<
188 typename transform_t::scalar_type>>::epsilon(),
189 const detray::traits::value_t<typename transform_t::scalar_type> max_error =
190 std::numeric_limits<detray::traits::value_t<
191 typename transform_t::scalar_type>>::epsilon()) {
192 return (::detray::algebra::approx_equal(trf1.matrix(), trf2.matrix(),
193 rel_error, max_error) &&
194 ::detray::algebra::approx_equal(trf1.matrix_inverse(),
195 trf2.matrix_inverse(), rel_error,
196 max_error));
197}
