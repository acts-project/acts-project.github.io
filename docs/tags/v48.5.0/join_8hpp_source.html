41template <detray::ranges::range range_t>
42struct join_view : public detray::ranges::view_interface<join_view<range_t>> {
44 using outer_iterator_t = detray::ranges::iterator_t<range_t>;
46 using outer_value_t =
47 std::conditional_t<std::is_const_v<range_t>,
48 const detray::ranges::range_value_t<range_t>,
49 detray::ranges::range_value_t<range_t>>;
51 using inner_iterator_t = detray::ranges::iterator_t<outer_value_t>;
52
54 using iterator_t = detray::ranges::detail::join_iterator<range_t>;
56 using value_t = std::iter_value_t<iterator_t>;

67 m_end{detray::ranges::end(ranges)} {}

74 DETRAY_HOST_DEVICE
75 constexpr auto end() const -> iterator_t { return {m_end, m_end}; }
76
79 DETRAY_HOST_DEVICE
80 constexpr auto data() const -> const value_t * { return &(*(*m_begin)); }
81
84 DETRAY_HOST_DEVICE
85 constexpr auto data() -> value_t * { return &(*(*m_begin)); }
86
88 DETRAY_HOST_DEVICE
89 constexpr auto size() const noexcept -> std::size_t {
90 std::size_t size{0u};
91 for (outer_iterator_t itr = m_begin; itr != m_end; ++itr) {
92 // subrange
93 const auto begin = detray::ranges::begin(*itr);
94 const auto end = detray::ranges::end(*itr);
95 size += static_cast<std::size_t>(detray::ranges::distance(begin, end));
96 }
97 return size;
98 }

101 outer_iterator_t m_begin{};
102 outer_iterator_t m_end{};
103};

159template <detray::ranges::range range_t>
160struct join_iterator {
162 using outer_iterator_t =
163 std::conditional_t<std::is_const_v<range_t>,
164 detray::ranges::const_iterator_t<range_t>,
165 detray::ranges::iterator_t<range_t>>;
167 using outer_value_t = decltype(*std::declval<outer_iterator_t>());
169 using inner_iterator_t = std::conditional_t<
170 std::is_same_v<outer_iterator_t,
171 detray::ranges::const_iterator_t<range_t>>,
172 detray::ranges::const_iterator_t<outer_value_t>,
173 detray::ranges::iterator_t<outer_value_t>>;
174
176 using iterator_t = inner_iterator_t;
178 using difference_type = std::iter_difference_t<iterator_t>;
180 using value_type = std::iter_value_t<iterator_t>;
182 using pointer = typename std::iterator_traits<iterator_t>::pointer;
184 using reference = std::iter_reference_t<iterator_t>;
186 using iterator_category =
187 typename std::iterator_traits<iterator_t>::iterator_category;
188
190 constexpr join_iterator() = default;
191
193 DETRAY_HOST_DEVICE
194 constexpr join_iterator(outer_iterator_t begin, outer_iterator_t end)
195 : m_outer_begin(begin), m_outer_end(end), m_outer_itr(begin) {
196 if (m_outer_itr != m_outer_end) {
197 m_inner_itr = (*m_outer_itr).begin();
198 } else {
199 m_inner_itr = {};
200 }
201 next_inner();
202 }
203
207 DETRAY_HOST_DEVICE auto operator++() -> join_iterator & {
208 ++m_inner_itr;
209 next_inner();
210
211 return *this;
212 }
213
216 DETRAY_HOST_DEVICE constexpr auto operator++(int) -> join_iterator {
217 auto tmp(*this);
218 ++(*this);
219 return tmp;
220 }
221
222
226 DETRAY_HOST_DEVICE constexpr auto operator--() -> join_iterator &
227 requires std::bidirectional_iterator<inner_iterator_t> &&
228 std::bidirectional_iterator<outer_iterator_t>
229 {
230 // If we are calling this at the end of the join iteration, go back into
231 // the valid range
232 if (m_outer_itr == m_outer_end) {
233 --m_outer_itr;
234 }
235
236 previous_inner();
237 --m_inner_itr;
238
239 return *this;
240 }
241
244 DETRAY_HOST_DEVICE constexpr auto operator--(int) -> join_iterator
245 requires std::bidirectional_iterator<inner_iterator_t> &&
246 std::bidirectional_iterator<outer_iterator_t>
247 {
248 auto tmp(*this);
249 --(*this);
250 return tmp;
251 }
252
253
255 DETRAY_HOST_DEVICE
256 constexpr decltype(auto) operator*() const { return *m_inner_itr; }
257
261 DETRAY_HOST_DEVICE constexpr auto operator+=(const difference_type j)
262 -> join_iterator &
263 requires std::random_access_iterator<inner_iterator_t> &&
264 std::random_access_iterator<outer_iterator_t>
265 {
266 // walk through join to catch the switch between intermediate ranges
267 if (difference_type i{j}; i >= difference_type{0}) {
268 while (i--) {
269 ++(*this);
270 }
271 } else {
272 while (i++) {
273 --(*this);
274 }
275 }
276 return *this;
277 }
278
282 DETRAY_HOST_DEVICE constexpr auto operator-=(const difference_type j)
283 -> join_iterator &
284 requires std::random_access_iterator<inner_iterator_t> &&
285 std::random_access_iterator<outer_iterator_t>
286 {
287 m_inner_itr += (-j);
288 return *this;
289 }
290
294 DETRAY_HOST_DEVICE constexpr decltype(auto) operator[](
295 const difference_type i) const
296 requires std::random_access_iterator<inner_iterator_t> &&
297 std::random_access_iterator<outer_iterator_t>
298 {
299 difference_type offset{i -
300 (m_inner_itr - detray::ranges::begin(*m_outer_itr))};
301 return *(*this + offset);
302 }
303
304 private:
306 DETRAY_HOST_DEVICE friend constexpr bool operator==(
307 const join_iterator &lhs, const join_iterator &rhs) {
308 return (lhs.m_inner_itr == rhs.m_inner_itr);
309 }
310
313 DETRAY_HOST_DEVICE friend constexpr bool operator!=(
314 const join_iterator &lhs, const join_iterator &rhs) {
315 return (lhs.m_outer_itr != rhs.m_outer_itr);
316 }
317
319 DETRAY_HOST_DEVICE friend constexpr auto operator<=>(const join_iterator &lhs,
320 const join_iterator &rhs)
321 requires detray::ranges::random_access_iterator<inner_iterator_t> &&
322 std::random_access_iterator<outer_iterator_t>
323 {
324#if defined(__apple_build_version__)
325 const auto l_o_itr{lhs.m_outer_itr};
326 const auto l_i_itr{lhs.m_inner_itr};
327 const auto r_o_itr{rhs.m_outer_itr};
328 const auto r_i_itr{rhs.m_inner_itr};
329 if (l_o_itr == r_o_itr) {
330 if (l_i_itr < r_i_itr || (l_i_itr == r_i_itr && l_i_itr < r_i_itr)) {
331 return std::strong_ordering::less;
332 }
333 if (l_i_itr > r_i_itr || (l_i_itr == r_i_itr && l_i_itr > r_i_itr)) {
334 return std::strong_ordering::greater;
335 }
336 return std::strong_ordering::equivalent;
337 } else {
338 if (l_o_itr < r_o_itr || (l_o_itr == r_o_itr && l_o_itr < r_o_itr)) {
339 return std::strong_ordering::less;
340 }
341 if (l_o_itr > r_o_itr || (l_o_itr == r_o_itr && l_o_itr > r_o_itr)) {
342 return std::strong_ordering::greater;
343 }
344 return std::strong_ordering::equivalent;
345 }
346#else
347 if (lhs.m_outer_itr == rhs.m_outer_itr) {
348 return lhs.m_inner_itr <=> rhs.m_inner_itr;
349 } else {
350 return lhs.m_outer_itr <=> rhs.m_outer_itr;
351 }
352#endif
353 }

359 DETRAY_HOST_DEVICE friend constexpr auto operator+(const join_iterator &itr,
360 const difference_type j)
361 -> join_iterator
362 requires std::random_access_iterator<inner_iterator_t> &&
363 std::random_access_iterator<outer_iterator_t>
364 {
365 join_iterator<range_t> tmp(itr);
366 // walk through join to catch the switch between intermediate ranges
367 if (difference_type i{j}; i >= difference_type{0}) {

405 DETRAY_HOST_DEVICE friend constexpr auto operator-(const join_iterator &left,
406 const join_iterator &right)
407 -> difference_type
408 requires std::random_access_iterator<inner_iterator_t> &&
409 std::random_access_iterator<outer_iterator_t>
410 {
411 const join_iterator lhs{left};
412 const join_iterator rhs{right};
413 outer_iterator_t tmp_outer_itr;
414
415 if (tmp_outer_itr == rhs.m_outer_itr) {
416 return lhs.m_inner_itr - rhs.m_inner_itr;
417 }
418 if ((tmp_outer_itr - rhs.m_outer_itr) < 0) {
419 // Negative distance
420 difference_type diff{lhs.m_inner_itr -
421 detray::ranges::end(*lhs.m_outer_itr)};
422 for (tmp_outer_itr + 1; tmp_outer_itr != rhs.m_outer_itr;
423 ++tmp_outer_itr) {
424 diff += detray::ranges::end(*tmp_outer_itr) -
425 detray::ranges::begin(*tmp_outer_itr);
426 }
427 diff += rhs.m_inner_itr - detray::ranges::end(*rhs.m_outer_itr);
428 return diff;
429 } else {
430 // Positive distance
431 difference_type diff{lhs.m_inner_itr -
432 detray::ranges::begin(*tmp_outer_itr)};
433 for (tmp_outer_itr - 1; tmp_outer_itr != rhs.m_outer_itr;
434 --tmp_outer_itr) {
435 diff += detray::ranges::end(*tmp_outer_itr) -
436 detray::ranges::begin(*tmp_outer_itr);
437 }
438 diff += rhs.m_inner_itr - detray::ranges::begin(*rhs.m_outer_itr);
439 return diff;
440 }
441 }
442
444 constexpr void next_inner() {
445 if (m_outer_itr == m_outer_end) {
446 return;
447 }
448 while (m_inner_itr == detray::ranges::end(*m_outer_itr)) {
449 ++m_outer_itr;
450 if (m_outer_itr != m_outer_end) {
451 m_inner_itr = (*m_outer_itr).begin();
452 } else {
453 break;
454 }
455 }
456 }
457
459 constexpr void previous_inner() {
460 // Get the start of the current inner range
461 inner_iterator_t inner_begin = detray::ranges::begin(*m_outer_itr);
462
463 // Iterator has reached last valid position in this range
464 // during the previous decrement. Now go to the end of the
465 // previous range
466 while (m_inner_itr == inner_begin) {
467 // No more inner ranges to try
468 if (m_outer_itr == m_outer_begin) {
469 m_inner_itr = detray::ranges::end(*m_outer_begin);
470 return;
471 }
472
473 --m_outer_itr;
474
475 inner_begin = detray::ranges::begin(*m_outer_itr);
476 m_inner_itr = detray::ranges::end(*m_outer_itr);
477 }
478 }
479
481 outer_iterator_t m_outer_begin{};
482 outer_iterator_t m_outer_end{};
484 outer_iterator_t m_outer_itr{};
486 inner_iterator_t m_inner_itr{};
487};
