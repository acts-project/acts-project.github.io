46template <std::random_access_iterator rand_iter_t,
47 std::sentinel_for<rand_iter_t> sentinel_t, class Predicate>
48DETRAY_HOST_DEVICE inline auto find_if(rand_iter_t first, sentinel_t last,
49 Predicate&& comp) {
50 for (rand_iter_t i = first; i != last; ++i) {
51 if (std::forward<Predicate>(comp)(*i)) {
52 return i;
53 }
54 }
55
56 return last;
57}
