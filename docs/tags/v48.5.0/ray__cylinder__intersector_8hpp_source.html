34 using algebra_type = algebra_t;
36 using frame_type = cylindrical2D<algebra_t>;
38 using point_type = dpoint3D<algebra_t>;
39
41 template <typename surface_descr_t>
42 using intersection_type =

44
46 template <typename other_algebra_t>
47 using trajectory_type = detail::ray<other_algebra_t>;
48
49 // Maximum number of solutions this intersector can produce
51 static constexpr std::uint8_t n_solutions{2u};
52
54 using result_type = darray<
55 intersection_point<algebra_t, point_type, intersection::contains_pos>,
56 n_solutions>;
57
64 template <typename mask_t>
65 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
66 const trajectory_type<algebra_t> &ray, const dtransform3D<algebra_t> &trf,
67 const mask_t &mask,
68 const dscalar<algebra_t> /*overstep_tol*/ = 0.f) const {
69 // One or both of these solutions might be invalid
70 const auto qe = solve_intersection(ray, mask, trf);
71
72 result_type results;
73
74 switch (qe.solutions()) {
75 [[likely]] case 2: {
76 results[1].path = qe.larger();
77 results[1].point = ray.pos() + qe.larger() * ray.dir();
78 // If there are two solutions, reuse the case for a single
79 // solution to setup the intersection point with the smaller
80 // path in results[0]
81 [[fallthrough]];
82 }
83 case 1: {
84 results[0].path = qe.smaller();
85 results[0].point = ray.pos() + qe.smaller() * ray.dir();
86 break;
87 }
88 [[unlikely]] default: { /* Do nothing */
89 }
90 }
91
92 return results;
93 }

103 template <typename mask_t>
104 DETRAY_HOST_DEVICE inline detail::quadratic_equation<dscalar<algebra_t>>
105 solve_intersection(const trajectory_type<algebra_t> &ray, const mask_t &mask,
106 const dtransform3D<algebra_t> &trf) const {
107 using scalar_t = dscalar<algebra_t>;
108 using point3_t = dpoint3D<algebra_t>;
109 using vector3_t = dvector3D<algebra_t>;
110
111 const scalar_t r{mask[mask_t::shape::e_r]};
112 const vector3_t &sz = trf.z();
113 const vector3_t &sc = trf.translation();
114
115 const point3_t &ro = ray.pos();
116 const vector3_t &rd = ray.dir();
117
118 const auto pc_cross_sz = vector::cross(ro - sc, sz);
119 const auto rd_cross_sz = vector::cross(rd, sz);
120 const scalar_t a{vector::dot(rd_cross_sz, rd_cross_sz)};
121 const scalar_t b{2.f * vector::dot(rd_cross_sz, pc_cross_sz)};
122 const scalar_t c{vector::dot(pc_cross_sz, pc_cross_sz) - (r * r)};
123
124 return detail::quadratic_equation<scalar_t>{a, b, c};
125 }
126};
