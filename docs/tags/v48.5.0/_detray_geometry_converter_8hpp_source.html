116 template <detray::concepts::metadata metadata_t>
117 DetrayGeometry<metadata_t> convert(
118 vecmem::memory_resource& mr, const Acts::GeometryContext& gctx,
119 const std::shared_ptr<const Acts::TrackingGeometry>& trackingGeometry,
120 const std::string& detectorName = "") const {
121 using detector_t = detray::host::detector<metadata_t>;
122
123 if (trackingGeometry == nullptr) {
124 throw std::invalid_argument(
125 "DetrayGeometryConverter: trackingGeometry must not be null");
126 }
127
128 // ── Convert TrackingGeometry → detray payloads ────────────────────────
129 auto payloads = m_cfg.payloadConverter->convertTrackingGeometry(
130 gctx, *trackingGeometry);
131
132 // ── Build detray detector from payloads ───────────────────────────────
133 detray::detector_builder<metadata_t> detectorBuilder{};
134
135 detray::io::geometry_reader::from_payload<detector_t>(detectorBuilder,
136 *payloads.detector);
137
138 if (m_cfg.convertMaterial) {
139 if (!payloads.homogeneousMaterial) {
140 ACTS_DEBUG("No homogeneous material payload found, skipping");
141 } else {
142 const auto& materialVolumes = payloads.homogeneousMaterial->volumes;
143 ACTS_DEBUG("Found homogeneous material payload with "
144 << materialVolumes.size() << " volumes");
145
146 auto hasMaterialType =
147 [&materialVolumes](detray::io::material_id type) {
148 return std::ranges::any_of(
149 materialVolumes, [type](const auto& volume) {
150 return std::ranges::any_of(
151 volume.surface_mat,
152 [type](const auto& sm) { return sm.type == type; });
153 });
154 };
155
156 if (hasMaterialType(detray::io::material_id::slab) &&
157 !detray::concepts::has_material_slabs<detector_t>) {
158 throw std::invalid_argument(
159 "DetrayGeometryConverter: the tracking geometry contains "
160 "homogeneous material slabs, but the target Detray metadata "
161 "type does not support material slabs");
162 }
163 if (hasMaterialType(detray::io::material_id::rod) &&
164 !detray::concepts::has_material_rods<detector_t>) {
165 throw std::invalid_argument(
166 "DetrayGeometryConverter: the tracking geometry contains "
167 "homogeneous material rods, but the target Detray metadata "
168 "type does not support material rods");
169 }
170
171 if constexpr (detray::concepts::has_material_slabs<detector_t> ||
172 detray::concepts::has_material_rods<detector_t>) {
173 detray::io::homogeneous_material_reader::from_payload<detector_t>(
174 detectorBuilder, *payloads.homogeneousMaterial);
175 }
176 }
177
178 detray::io::material_map_reader<std::integral_constant<std::size_t, 2>>::
179 from_payload<detector_t>(detectorBuilder,
180 std::move(*payloads.materialGrids));
181 }
182
183 if (m_cfg.convertSurfaceGrids) {
184 detray::io::surface_grid_reader<typename detector_t::surface_type,
185 std::integral_constant<std::size_t, 0>,
186 std::integral_constant<std::size_t, 2>>::
187 template from_payload<detector_t>(detectorBuilder,
188 *payloads.surfaceGrids);
189 }
190
191 if (!detectorName.empty()) {
192 detectorBuilder.set_name(detectorName);
193 } else if (payloads.names.contains(0)) {
194 detectorBuilder.set_name(payloads.names.at(0));
195 }
196
197 DetrayGeometry<metadata_t> result{};
198 detray::volume_builder_options build_opts{};
199 build_opts.deduplicate(m_cfg.deduplicateMaterial);
200 result.detector = std::make_shared<detector_t>(
201 detectorBuilder.build(mr, build_opts, result.names));
202
203 // detray assigns the surface identifiers during the build
204 result.surfaceIds = DetraySurfaceIdMap(*result.detector, *trackingGeometry);
205 ACTS_DEBUG("Mapped " << result.surfaceIds.size()
206 << " sensitive surfaces between ACTS and detray");
207
208 return result;
209 }
210
211 private:
212 Config m_cfg;
213
214 const Acts::Logger& logger() const { return *m_logger; }
215 std::unique_ptr<const Acts::Logger> m_logger;
216};
