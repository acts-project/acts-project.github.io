26template <concepts::detector detector_t, typename intersection_t>
27inline auto intersection(const detector_t& detector,
28 const std::vector<intersection_t>& intersections,
29 const typename detector_t::vector3_type& dir = {},
30 const typename detector_t::geometry_context& gctx = {},
31 const dindex_range highlight_idx =
32 {detray::detail::invalid_value<dindex>(),
33 detray::detail::invalid_value<dindex>()},
34 const styling::landmark_style& style =
35 styling::svg_default::intersection_style) {
36 using point3_t = typename detector_t::point3_type;
37 using point2_t = typename detector_t::point2_type;
38 using p_intersection_t = svgtools::meta::proto::intersection<point3_t>;
39 p_intersection_t p_ir;
40
41 for (const auto& intr : intersections) {
42 const detray::geometry::surface sf{detector, intr.surface()};
43 if (sf.identifier().is_invalid()) {
44 continue;
45 }
46
47 const point2_t bound{intr.local()[0], intr.local()[1]};
48 const auto position = sf.local_to_global(gctx, bound, dir);
49 const auto p_lm = svgtools::conversion::landmark(position, style);
50
51 p_ir._landmarks.push_back(p_lm);
52 }
53
54 svgtools::styling::apply_style(p_ir, style);
55
56 // Place highlights
57 assert(highlight_idx[0] <= highlight_idx[1]);
58 if (highlight_idx[1] - highlight_idx[0] != 0u) {
59 styling::landmark_style highlight_style{style};
60
61 constexpr actsvg::scalar opacity{1.f};
62 highlight_style._fill_color = {styling::colors::red, opacity};
63
64 for (std::size_t i = highlight_idx[0]; i <= highlight_idx[1]; ++i) {
65 styling::apply_style(p_ir._landmarks.at(i), highlight_style);
66 }
67 }
68
69 return p_ir;
70}
