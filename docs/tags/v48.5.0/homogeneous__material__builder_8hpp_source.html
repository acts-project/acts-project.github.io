32class homogeneous_material_builder final : public volume_decorator<detector_t> {
33 public:
35 using material_id = typename detector_t::material::id;
37 using scalar_type = dscalar<typename detector_t::algebra_type>;

42 DETRAY_HOST
43 explicit homogeneous_material_builder(
44 std::unique_ptr<volume_builder_interface<detector_t>> vol_builder)
45 : volume_decorator<detector_t>(std::move(vol_builder)) {
46 DETRAY_VERBOSE_HOST(
47 "Add hom. material builder to volume: " << this->name());
48
49 static_assert(concepts::has_material_slabs<detector_t> ||
50 concepts::has_material_rods<detector_t>,
51 "No homogeneous surface material in detector type");

59 DETRAY_HOST
60 void add_surfaces(
61 std::shared_ptr<surface_factory_interface<detector_t>> sf_factory,
62 typename detector_t::geometry_context ctx = {}) override {
63 DETRAY_VERBOSE_HOST("Add [material] surface factory:");
64
65 // If the factory also holds surface data, call base volume builder
66 volume_decorator<detector_t>::add_surfaces(sf_factory, ctx);
67
68 // Add material
69 auto mat_factory =
70 std::dynamic_pointer_cast<homogeneous_material_factory<detector_t>>(
71 sf_factory);
72 if (mat_factory) {
73 DETRAY_VERBOSE_HOST("-> found decoration: " << DETRAY_TYPENAME(
74 homogeneous_material_factory<detector_t>));
75 (*mat_factory)(this->surfaces(), m_materials);
76 return;
77 }
78 auto mat_generator =
79 std::dynamic_pointer_cast<homogeneous_material_generator<detector_t>>(
80 sf_factory);
81 if (mat_generator) {
82 DETRAY_VERBOSE_HOST("-> found decoration: " << DETRAY_TYPENAME(
83 homogeneous_material_generator<detector_t>));
84 (*mat_generator)(this->surfaces(), m_materials);
85 return;
86 }
87
88 if (!mat_factory && !mat_generator) {
89 DETRAY_VERBOSE_HOST("No material found in this surface factory");
90 DETRAY_VERBOSE_HOST("-> Built non-material surfaces");
91 }
92 }

100 DETRAY_HOST
101 auto build(detector_t &det, const volume_builder_options &opt,
102 typename detector_t::geometry_context ctx = {}) ->
103 typename detector_t::volume_type * override {
104 DETRAY_VERBOSE_HOST("Build homogeneous material...");
105
106 DETRAY_DEBUG_HOST("-> n_surfaces=" << this->surfaces().size());
107
108 if (opt.deduplicate()) {
109 add_deduplicated_material(det);
110 } else {
111 add_material(det);
112 }
113
114 DETRAY_VERBOSE_HOST(
115 "Successfully built homogeneous material for volume: " << this->name());
116
117 // Call the underlying volume builder(s) and give the volume to the
118 // next decorator
119 return volume_decorator<detector_t>::build(det, opt, ctx);
120 }

125 DETRAY_HOST
126 void add_material(detector_t &det) {
127 const auto &material = det.material_store();
128
129 // Update the surface material links and shift them according to the
130 // number of material slabs/rods that were in the detector previously
131 for (auto &sf : this->surfaces()) {
132 DETRAY_DEBUG_HOST("-> sf=" << sf);
133 DETRAY_DEBUG_HOST(" -> material_id=" << sf.material().id());
134 if constexpr (concepts::has_material_slabs<detector_t>) {
135 if (sf.material().id() == material_id::e_material_slab) {
136 dindex offset =
137 material.template size<material_id::e_material_slab>();
138 DETRAY_DEBUG_HOST("-> update material slab offset: " << offset);
139 sf.update_material(offset);
140 DETRAY_DEBUG_HOST("-> material now: " << sf.material());
141 }
142
143 DETRAY_DEBUG_HOST(
144 "-> Appending "
145 << m_materials.template size<material_id::e_material_slab>()
146 << " slabs into detector materials");
147 }
148 if constexpr (concepts::has_material_rods<detector_t>) {
149 if (sf.material().id() == material_id::e_material_rod) {
150 DETRAY_DEBUG_HOST(
151 "-> update material rod offset: "
152 << material.template size<material_id::e_material_rod>());
153 sf.update_material(
154 material.template size<material_id::e_material_rod>());
155 }
156
157 DETRAY_DEBUG_HOST(
158 "-> Appending "
159 << m_materials.template size<material_id::e_material_rod>()
160 << " rods into detector materials");
161 }
162 }
163
164 // Add material to the detector
165 det._materials.append(std::move(m_materials));
166 m_materials.clear_all();
167 }
168
172 DETRAY_HOST
173 void add_deduplicated_material(detector_t &det) {
174 DETRAY_VERBOSE_HOST("-> Deduplicate homogeneous material");
175
176 if constexpr (concepts::has_material_slabs<detector_t>) {
177 deduplicate<material_id::e_material_slab>(det);
178 }
179 if constexpr (concepts::has_material_rods<detector_t>) {
180 deduplicate<material_id::e_material_rod>(det);
181 }
182
183 // Add remaining material types, if any
184 det._materials.append(std::move(m_materials));
185 m_materials.clear_all();
186 }
187
190 template <material_id mat_id>
191 DETRAY_HOST void deduplicate(detector_t &det) {
192 auto &local_coll = m_materials.template get<mat_id>();
193 if (local_coll.empty()) {
194 return;
195 }
196
197 auto &det_coll = det._materials.template get<mat_id>();
198 [[maybe_unused]] const std::size_t n_before{det_coll.size()};
199
200 // Global index for every volume local material entry
201 detail::homogeneous_material_lookup lookup{det_coll};
202 std::vector<dindex> global_idx;
203 global_idx.reserve(local_coll.size());
204 for (const auto &mat : local_coll) {
205 global_idx.push_back(lookup.insert(mat));
206 }
207
208 for (auto &sf : this->surfaces()) {
209 if (sf.material().id() == mat_id) {
210 sf.material().set_index(global_idx.at(sf.material().index()));
211 DETRAY_DEBUG_HOST("-> sf=" << sf
212 << ": material now: " << sf.material());
213 }
214 }
215
216 DETRAY_DEBUG_HOST("-> Appended " << det_coll.size() - n_before << " of "
217 << local_coll.size() << " entries of type "
218 << mat_id << " to detector materials");
219 local_coll.clear();
220 }
221
222 // Material container for this volume
223 typename detector_t::material_container m_materials{};
224};
