66 DETRAY_HOST_DEVICE static inline point3_type local_to_global(
67 const transform3_type & /*trf*/, const point3_type &p) {
68 const scalar_type x{p[2] * math::cos(p[0])};
69 const scalar_type y{p[2] * math::sin(p[0])};
70 const scalar_type z{p[1]};
71
72 return point3_type{x, y, z};
73 }

80 template <typename mask_t>
81 DETRAY_HOST_DEVICE static inline point3_type local_to_global(
82 const transform3_type & /*trf*/, const mask_t &mask, const loc_point &p,
83 const vector3_type & /*dir*/) {
84 const scalar_type r{mask[mask_t::shape::e_r]};
85 const scalar_type x{r * math::cos(p[0])};
86 const scalar_type y{r * math::sin(p[0])};
87 const scalar_type z{p[1]};

95 template <typename mask_t>
96 DETRAY_HOST_DEVICE static inline vector3_type normal(
97 const transform3_type & /*trf*/, const point2_type &p,
98 const mask_t & /*mask*/) {
99 // normal vector in global coordinates (concentric cylinders have no
100 // rotation)
101 return {math::cos(p[0]), math::sin(p[0]), static_cast<scalar_type>(0)};
102 }
103
107 DETRAY_HOST_DEVICE static inline vector3_type normal(
108 const transform3_type & /*trf*/, const point3_type &p) {
109 // normal vector in global coordinates (concentric cylinders have no
