26template <concepts::detector detector_t>
27inline auto is_not_world_portal(
28 const detray::geometry::surface<detector_t>& d_portal) {
29 const auto d_link_idx = d_portal.volume_links();
30 bool is_world_pt{false};
31 for (const auto vol_link : d_link_idx) {
32 is_world_pt = is_world_pt || detray::detail::is_invalid_value(vol_link);
33 }
34 return !is_world_pt;
35}

39template <concepts::detector detector_t>
40inline auto get_linked_volume(
41 const detector_t& detector,
42 const detray::geometry::surface<detector_t>& d_portal,
43 [[maybe_unused]] const std::size_t mask_idx) {
44 assert(is_not_world_portal(d_portal));
45 const auto d_link_idx = d_portal.volume_links();
46
47 return tracking_volume{detector, d_link_idx.at(mask_idx)};
48}

53template <concepts::detector detector_t>
54inline auto link_points(const typename detector_t::geometry_context& context,
55 const detector_t& detector,
56 const detray::geometry::surface<detector_t>& d_portal,
57 typename detector_t::vector3_type dir,
58 const double link_length,
59 const std::size_t mask_idx = 0u) {
60 assert(is_not_world_portal(d_portal));
61
62 // Calculating the start position:
63 const auto start = d_portal.template visit_mask<link_start_getter>(
64 d_portal.transform(context), mask_idx);
65
66 // Calculating the end position:
67 const auto n =
68 d_portal.normal(context, d_portal.global_to_local(context, start, dir));
69 const auto volume = get_linked_volume(detector, d_portal, mask_idx);
70 const auto end = d_portal.template visit_mask<link_end_getter>(
71 detector, volume, start, n, link_length, mask_idx);
72
73 return std::make_tuple(start, end);
74}
