20template <concepts::algebra algebra_t>
21struct axis_rotation {
22 public:
24 using algebra_type = algebra_t;
26 using scalar_type = dscalar<algebra_t>;
28 using vector3_type = dvector3D<algebra_t>;
30 template <std::size_t ROWS, std::size_t COLS>
31 using matrix_type = dmatrix<algebra_t, ROWS, COLS>;
32
37 DETRAY_HOST_DEVICE
38 axis_rotation(const vector3_type& axis, const scalar_type theta) {
39 // normalize the axis
40 const vector3_type U = vector::normalize(axis);
41
42 scalar_type cos_theta{math::cos(theta)};
43
44 auto I = matrix::identity<matrix_type<3, 3>>();
45 matrix_type<3, 3> axis_cross = matrix::cross_matrix(U);
46 matrix_type<3, 3> axis_outer = matrix::outer_product(U, U);
47
48 R = cos_theta * I + math::sin(theta) * axis_cross +
49 (1.f - cos_theta) * axis_outer;
50 }
51
54 template <typename vector3_t>
55 requires(concepts::vector3D<vector3_t> ||
56 concepts::column_matrix3D<vector3_t>)
57 DETRAY_HOST_DEVICE vector3_t operator()(const vector3_t& v) const {
58 return R * v;
59 }
60
61 private:
63 matrix_type<3, 3> R{matrix::identity<matrix_type<3, 3>>()};
64};

68template <concepts::algebra algebra_t>
69struct euler_rotation {
70 public:
72 using algebra_type = algebra_t;
74 using scalar_type = dscalar<algebra_t>;
76 using vector3_type = dvector3D<algebra_t>;
77
79 vector3_type x{1.0f, 0.f, 0.f};
81 vector3_type z{0.f, 0.f, 1.f};
83 scalar_type alpha{0.f};
85 scalar_type beta{0.f};
87 scalar_type gamma{0.f};
88
90 DETRAY_HOST_DEVICE std::pair<vector3_type, vector3_type> operator()() const {
91 // alpha around z axis
92 axis_rotation<algebra_t> axis_rot_alpha(z, alpha);
93 auto new_x = axis_rot_alpha(x);
94
95 // beta around x' axis
96 axis_rotation<algebra_t> axis_rot_beta(new_x, beta);
97 const auto new_z = axis_rot_beta(z);
98
99 axis_rotation<algebra_t> axis_rot_gamma(new_z, gamma);
100 new_x = axis_rot_gamma(new_x);
101
102 return std::make_pair(new_x, new_z);
103 }
104
108 DETRAY_HOST_DEVICE vector3_type operator()(const vector3_type& v0) const {
109 // alpha around z axis
110 axis_rotation<algebra_t> axis_rot_alpha(z, alpha);
111 const auto new_x = axis_rot_alpha(x);
112 const auto v1 = axis_rot_alpha(v0);
113
114 // beta around x' axis
115 axis_rotation<algebra_t> axis_rot_beta(new_x, beta);
116 const auto new_z = axis_rot_beta(z);
117 const auto v2 = axis_rot_beta(v1);
118
119 // gamma around z' axis
120 axis_rotation<algebra_t> axis_rot_gamma(new_z, gamma);
121 const auto v3 = axis_rot_gamma(v2);
122
123 return v3;
124 }
125};
