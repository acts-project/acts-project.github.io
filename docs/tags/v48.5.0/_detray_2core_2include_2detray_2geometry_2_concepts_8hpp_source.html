24template <class F>
25concept coordinate_frame = requires() {
26 requires concepts::point<typename F::loc_point>;
27 requires concepts::algebra<typename F::algebra_type>;
28
29 {
30 F::global_to_local_3D(dtransform3D<typename F::algebra_type>(),
31 dpoint3D<typename F::algebra_type>(),
32 dvector3D<typename F::algebra_type>())
33 } -> std::same_as<dpoint3D<typename F::algebra_type>>;
34
35 {
36 F::global_to_local(dtransform3D<typename F::algebra_type>(),
37 dpoint3D<typename F::algebra_type>(),
38 dvector3D<typename F::algebra_type>())
39 } -> std::same_as<typename F::loc_point>;
40
41 {
42 F::local_to_global(dtransform3D<typename F::algebra_type>(),
43 dpoint3D<typename F::algebra_type>())
44 } -> std::same_as<dpoint3D<typename F::algebra_type>>;
45
46 {
47 F::normal(dtransform3D<typename F::algebra_type>(),
48 dpoint3D<typename F::algebra_type>())
49 } -> std::same_as<dvector3D<typename F::algebra_type>>;
50};

53template <typename S, typename A>
54concept shape = concepts::algebra<A> && requires(const S sh, std::ostream os) {
55 requires std::is_enum_v<typename S::boundaries>;
56
57 requires concepts::coordinate_frame<typename S::template local_frame_type<A>>;
58
59 typename S::template result_type<dbool<A>>;
60 typename S::template bounds_type<dscalar<A>>;
61
62 S::dim;
63
64 {
65 sh.min_dist_to_boundary(typename S::template bounds_type<dscalar<A>>(),
66 dpoint2D<A>())
67 } -> std::same_as<dscalar<A>>;
68
69 {
70 sh.template check_boundaries<A>(
71 typename S::template bounds_type<dscalar<A>>(), dtransform3D<A>(),
72 dpoint3D<A>(), dscalar<A>())
73 } -> std::same_as<typename S::template result_type<dbool<A>>>;
74
75 {
76 sh.measure(typename S::template bounds_type<dscalar<A>>())
77 } -> std::same_as<dscalar<A>>;
78
79 {
80 sh.merge(typename S::template bounds_type<dscalar<A>>(),
81 typename S::template bounds_type<dscalar<A>>())
82 } -> std::same_as<typename S::template bounds_type<dscalar<A>>>;
83
84 {
85 sh.template local_min_bounds<A>(
86 typename S::template bounds_type<dscalar<A>>(), dscalar<A>())
87 } -> std::same_as<darray<dscalar<A>, 6>>;
88
89 {
90 sh.template centroid<A>(typename S::template bounds_type<dscalar<A>>())
91 } -> std::same_as<dpoint3D<A>>;
92
93 {

95 dindex())
96 } -> std::same_as<dvector<dpoint3D<A>>>;
97
98 {
99 sh.check_consistency(typename S::template bounds_type<dscalar<A>>(), os)
100 } -> std::same_as<bool>;
101};

121template <class F>
122concept planar_frame =
123 concepts::coordinate_frame<F> &&
