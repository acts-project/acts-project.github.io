61 constexpr static int hexToInt(std::string_view hex) {
62 constexpr auto hexCharToInt = [](char c) {
63 if (c >= '0' && c <= '9') {
64 return c - '0';
65 } else if (c >= 'a' && c <= 'f') {
66 return c - 'a' + 10;
67 } else if (c >= 'A' && c <= 'F') {
68 return c - 'A' + 10;
69 } else {
70 throw std::invalid_argument("Invalid hex character");
71 }
72 };
73
74 int value = 0;
75 for (char c : hex) {
76 value = (value << 4) + hexCharToInt(c);
77 }
78 return value;
79 };
80
81 public:
84 constexpr explicit Color(std::string_view hex) {
85 if (hex[0] == '#' && hex.size() == 7) {
86 rgb[0] = hexToInt(hex.substr(1, 2)); // Extract R component
87 rgb[1] = hexToInt(hex.substr(3, 2)); // Extract G component
88 rgb[2] = hexToInt(hex.substr(5, 2)); // Extract B component
89 } else {
90 throw std::invalid_argument{
91 "Invalid hex color format. Expected format: #RRGGBB"};
92 }
93 }
