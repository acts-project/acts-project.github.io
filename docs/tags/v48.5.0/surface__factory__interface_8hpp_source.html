129 DETRAY_HOST

136 virtual void push_back(surface_data<detector_t> &&data) = 0;
137
139 DETRAY_HOST
140 virtual void push_back(std::vector<surface_data<detector_t>> &&data) = 0;
142
144 DETRAY_HOST
145 virtual void clear() = 0;
146
155 DETRAY_HOST
156 virtual auto operator()(
157 typename detector_t::volume_type &volume,
158 typename detector_t::surface_lookup_container &surfaces,
159 typename detector_t::transform_container &transforms,
160 typename detector_t::mask_container &masks,
161 typename detector_t::geometry_context ctx = {}) -> dindex_range = 0;
162
163 protected:
172 template <typename container_t, typename... Args>
173 DETRAY_HOST dindex insert_in_container(
174 container_t &cont, const typename container_t::value_type value,
175 const dindex idx, Args &&...args) const {
176 // If no valid position is given, perform push back
177 if (detail::is_invalid_value(idx)) {
178 cont.push_back(value, std::forward<Args>(args)...);
179
180 return static_cast<dindex>(cont.size() - 1u);
181 } else {
182 // Make sure the container size encompasses the new value
183 if (cont.size() < idx + 1) {
184 cont.resize(idx + 1, std::forward<Args>(args)...);
185 }
186
187 cont.at(idx, std::forward<Args>(args)...) = value;
188
189 return idx;
190 }
191 }
192};
