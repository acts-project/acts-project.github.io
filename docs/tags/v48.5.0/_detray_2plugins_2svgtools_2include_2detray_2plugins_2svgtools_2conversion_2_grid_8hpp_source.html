43template <concepts::grid grid_t, typename view_t>
44 requires std::is_same_v<
45 typename grid_t::local_frame_type,
46 detray::concentric_cylindrical2D<
47 typename grid_t::local_frame_type::algebra_type>> ||
48 std::is_same_v<typename grid_t::local_frame_type,
49 detray::cylindrical2D<
50 typename grid_t::local_frame_type::algebra_type>>
51inline auto grid_type_and_edges(const grid_t& grid, const view_t& /*unused*/) {
52 using scalar_t = typename grid_t::local_frame_type::scalar_type;
53 using axis_label = detray::axis::label;
54
55 auto edges_phi = grid.template get_axis<axis_label::e_rphi>().bin_edges();
56 auto edges_z = grid.template get_axis<axis_label::e_cyl_z>().bin_edges();
57 // Unknown for 2D cylinder
58 dvector<scalar_t> edges_r{};
59
60 if constexpr (std::is_same_v<view_t, actsvg::views::x_y>) {
61 return std::tuple(grid_type::e_barrel, actsvg::proto::grid::e_r_phi,
62 edges_r, edges_phi);
63 }
64 if constexpr (std::is_same_v<view_t, actsvg::views::z_r>) {
65 return std::tuple(grid_type::e_barrel, actsvg::proto::grid::e_x_y, edges_z,
66 edges_r);
67 }
68 if constexpr (std::is_same_v<view_t, actsvg::views::z_phi> ||
69 std::is_same_v<view_t, typename actsvg::views::z_rphi>) {
70 return std::tuple(grid_type::e_barrel, actsvg::proto::grid::e_z_phi,
71 edges_z, edges_phi);
72 }
73
74 return std::tuple(grid_type::e_barrel, actsvg::proto::grid::e_x_y,
75 dvector<scalar_t>{}, dvector<scalar_t>{});
76}

121template <concepts::scalar scalar_t>
122struct type_and_edge_getter {
131 template <typename group_t, typename index_t, typename view_t>
132 DETRAY_HOST_DEVICE inline auto operator()(
133 [[maybe_unused]] const group_t& group,
134 [[maybe_unused]] const index_t index,
135 [[maybe_unused]] const view_t& view) const {
136 using value_t = typename group_t::value_type;
137
138 if constexpr (concepts::grid<value_t>) {
139 // Only two dimensional grids for actsvg
140 if constexpr (value_t::dim == 2) {
141 return grid_type_and_edges(group.at(index), view);
142 }
143 }
144
145 return std::tuple(grid_type::e_unknown, actsvg::proto::grid::e_x_y,
146 dvector<scalar_t>{}, dvector<scalar_t>{});
147 }
148};

161template <typename store_t, typename link_t, typename view_t,
162 concepts::scalar scalar_t>
163auto grid(const store_t& store, const link_t& link, const view_t& view,
164 const scalar_t ref_radius,
165 const styling::grid_style& style =
166 styling::tableau_colorblind::grid_style) {
167 if (link.is_invalid()) {
168 return std::tuple(std::optional<actsvg::proto::grid>{},
169 detail::grid_type::e_unknown);
170 }
171
172 auto [gr_type, view_type, edges0, edges1] =
173 store.template visit<detail::type_and_edge_getter<scalar_t>>(link, view);
174 actsvg::proto::grid p_grid;
175 p_grid._type = view_type;
176
177 // Find the correct grid radius
178 if (gr_type == detail::grid_type::e_barrel) {
179 p_grid._reference_r = static_cast<actsvg::scalar>(ref_radius);
180
181 // Add the cylinder radius to the axis binning
182 if constexpr (std::is_same_v<view_t, actsvg::views::x_y>) {
183 if (edges0.empty()) {
184 edges0 = {p_grid._reference_r, p_grid._reference_r};
185 }
186 }
187 if constexpr (std::is_same_v<view_t, actsvg::views::z_r>) {
188 if (edges1.empty()) {
189 edges1 = {p_grid._reference_r, p_grid._reference_r};
190 }
191 }
192
193 } else if (gr_type == detail::grid_type::e_endcap) {
194 // An axis is always sorted
195 p_grid._reference_r = static_cast<actsvg::scalar>(edges0.back());
196 }
197
198 std::ranges::transform(
199 edges0, std::back_inserter(p_grid._edges_0),
200 [](scalar_t v) { return static_cast<actsvg::scalar>(v); });
201 std::ranges::transform(
202 edges1, std::back_inserter(p_grid._edges_1),
203 [](scalar_t v) { return static_cast<actsvg::scalar>(v); });
204
205 svgtools::styling::apply_style(p_grid, style);
206
207 return std::tuple(std::optional<actsvg::proto::grid>{p_grid}, gr_type);
208}
