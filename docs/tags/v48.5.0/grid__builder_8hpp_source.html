107 template <typename grid_shape_t>
108 DETRAY_HOST void init_grid(
109 const mask<grid_shape_t, algebra_type> &m,
110 const darray<std::size_t, grid_t::dim> &n_bins,
111 const std::vector<std::pair<typename grid_t::loc_bin_index, dindex>>

128 DETRAY_VERBOSE_HOST("Created empty grid:\n"
129 << DETRAY_TYPENAME(grid_t) << "\n"

198 DETRAY_HOST
199 auto build(detector_t &det, const volume_builder_options &opt,
200 typename detector_t::geometry_context ctx = {}) ->
201 typename detector_t::volume_type * override {
202 DETRAY_VERBOSE_HOST("Build surface grid...");
203
204 if (opt.deduplicate()) {
205 DETRAY_WARN_HOST(this->name()
206 << ": Surface grid deduplication not yet implemented");
207 }
208
209 DETRAY_VERBOSE_HOST(
210 " -> Defer to other builders to get complete surface descriptors "
211 "first:");
212
213 using surface_desc_t = typename detector_t::surface_type;
214
215 // Add the surfaces (portals and/or passives) that are owned by the vol
216 typename detector_t::volume_type *vol_ptr =
217 volume_decorator<detector_t>::build(det, opt, ctx);
218
219 DETRAY_VERBOSE_HOST("Resume building with updated surface descriptors");
220
221 // Find the surfaces that should be filled into the grid
222 const auto vol = tracking_volume{det, vol_ptr->index()};
223
224 // Grid has not been filled previously, fill it automatically
225 if (m_grid.size() == 0u) {
226 DETRAY_DEBUG_HOST("-> Filling grid automatically...");
227
228 std::vector<surface_desc_t> surfaces{};
229 for (auto &sf_desc : vol.surfaces()) {
230 if (sf_desc.is_sensitive() ||
231 (m_add_passives && sf_desc.is_passive())) {
232 surfaces.push_back(sf_desc);
233 }
234 }
235
236 this->fill_grid(
237 tracking_volume{det, volume_decorator<detector_t>::operator()()},
238 surfaces, det.transform_store(), det.mask_store(), ctx);
239 } else {
240 DETRAY_DEBUG_HOST("-> Grid is prefilled...");
241
242 // The grid is prefilled with surface descriptors that contain the
243 // correct LOCAL surface indices per bin (e.g. from file IO).
244 // Now add the rest of the linking information, which is only
245 // available after the volume builder ran
246 for (surface_desc_t &sf_desc : m_grid.all()) {
247 assert(!detail::is_invalid_value(sf_desc.index()));
248
249 dindex glob_idx{vol_ptr->to_global_sf_index(sf_desc.index())};
250 const auto &new_sf_desc = det.surface(glob_idx);
251
252 assert(new_sf_desc.index() == glob_idx);
253 assert(!new_sf_desc.identifier().is_invalid());
254
255 sf_desc = new_sf_desc;
256 }
257 }
258
259 // For cylinder grids, additional mask information is needed
260 if constexpr (concepts::cylindrical<typename grid_t::local_frame_type>) {
261 assert(vol_ptr->id() == volume_id::e_cylinder);
262
263 constexpr auto inv_vol_link{
264 detray::detail::invalid_value<std::uint8_t>()};
265 constexpr auto inf{std::numeric_limits<scalar_type>::max()};
266
267 DETRAY_DEBUG_HOST("Find radius for cylinder grid...");
268
269 // Passive surfaces could be in the brute force finder, but no
270 // sensitive surfaces, since the volume has a grid. Their radii are,
271 // however, always within the interval of the portal radii
272 std::vector<scalar_type> radii{};
273 for (auto pt_desc : vol.portals()) {
274 auto r = det.mask_store().template visit<detail::outer_radius_getter>(
275 pt_desc.mask());
276
277 if (r.has_value()) {
278 DETRAY_DEBUG_HOST("Found radius " << *r
279 << " mm of portal: " << pt_desc);
280 radii.push_back(*r);
281 }
282 }
283
284 scalar_type grid_r{0.f};
285 if (!radii.empty()) {
286 const scalar_type inner_r{*std::ranges::min_element(radii)};
287 const scalar_type outer_r{*std::ranges::max_element(radii)};
288
289 grid_r = 0.5f * (inner_r + outer_r);
290 }
291
292 typename spatial_grid_t::mask_type cyl_mask{inv_vol_link, grid_r, -inf,
293 inf};
294
295 DETRAY_DEBUG_HOST("Setting mask for cylinder grid: " << cyl_mask);
296
297 m_grid.mask(cyl_mask);
298 }
299
300 constexpr auto gid{types::id<typename detector_t::accel, spatial_grid_t>};
301 const dindex grid_idx{det.accelerator_store().template size<gid>()};
302
303 DETRAY_DEBUG_HOST("Adding grid to volume in detector. Surface type: "
304 << m_id << ", grid: " << gid
305 << ", grid idx: " << grid_idx);
306
307 // Set: contained surface type, grid type, grid instance index
308 vol_ptr->set_accel_link(m_id, gid, grid_idx);
309
310 // Add to detector
311 det._accelerators.template push_back<gid>(m_grid);
312
313 DETRAY_DEBUG_HOST("Accelerator link: " << vol_ptr->accel_link());
314 DETRAY_VERBOSE_HOST("Successfully built "
315 << gid << " for volume: " << this->name());
316
317 DETRAY_DEBUG_HOST("Finished grid: " << m_grid);
318
319 return vol_ptr;
320 }
