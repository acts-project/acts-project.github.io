106 template <concepts::scalar scalar_t, concepts::point point_t>
107 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
108 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
109 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
110 const scalar_t edge_tol = 0.f) const {
111 const scalar_t loc_0{math::fabs(loc_p[0])};
112 const scalar_t loc_1{math::fabs(loc_p[1])};
113
114 auto inside_mask{(loc_0 <= (bounds[e_half_x] + tol) &&
115 loc_1 <= (bounds[e_half_y] + tol))};
116
117 decltype(inside_mask) inside_edge{false};
118 if (detail::any_of(edge_tol > 0.f)) {
119 const scalar_t full_tol{tol + edge_tol};
120
121 inside_edge = (loc_0 <= (bounds[e_half_x] + full_tol) &&
122 loc_1 <= (bounds[e_half_y] + full_tol));
123 }
124
125 return result_type<decltype(inside_mask)>{inside_mask, inside_edge};
126 }

157 template <concepts::scalar scalar_t>
158 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
159 const bounds_type<scalar_t> &bounds,
160 const bounds_type<scalar_t> &o_bounds) const {
161 bounds_type<scalar_t> new_bounds{};
162
163 new_bounds[e_half_x] = math::max(bounds[e_half_x], o_bounds[e_half_x]);
164 new_bounds[e_half_y] = math::max(bounds[e_half_y], o_bounds[e_half_y]);
165
166 return new_bounds;
167 }

178 template <concepts::algebra algebra_t>
179 DETRAY_HOST_DEVICE inline darray<dscalar<algebra_t>, 6> local_min_bounds(
180 const bounds_type<dscalar<algebra_t>> &bounds,
181 const dscalar<algebra_t> env =
182 std::numeric_limits<dscalar<algebra_t>>::epsilon()) const {
183 using scalar_t = dscalar<algebra_t>;
184
185 assert(env > 0.f);
186 const scalar_t x_bound{bounds[e_half_x] + env};
187 const scalar_t y_bound{bounds[e_half_y] + env};
188 return {-x_bound, -y_bound, -env, x_bound, y_bound, env};
189 }

205 template <concepts::algebra algebra_t>
206 DETRAY_HOST dvector<dpoint3D<algebra_t>> vertices(
207 const bounds_type<dscalar<algebra_t>> &bounds, dindex /*ignored*/) const {
208 using point3_t = dpoint3D<algebra_t>;
209
210 // left hand lower corner
211 point3_t lh_lc{-bounds[e_half_x], -bounds[e_half_y],
212 static_cast<dscalar<algebra_t>>(0.f)};
213 // right hand lower corner
214 point3_t rh_lc{bounds[e_half_x], -bounds[e_half_y],
215 static_cast<dscalar<algebra_t>>(0.f)};
216 // right hand upper corner
217 point3_t rh_uc{bounds[e_half_x], bounds[e_half_y],
218 static_cast<dscalar<algebra_t>>(0.f)};
219 // left hand upper corner
220 point3_t lh_uc{-bounds[e_half_x], bounds[e_half_y],
221 static_cast<dscalar<algebra_t>>(0.f)};
222
223 // Return the confining vertices
224 return {lh_lc, rh_lc, rh_uc, lh_uc};
225 }

233 template <concepts::scalar scalar_t>
234 DETRAY_HOST constexpr bool check_consistency(
235 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
236 if (constexpr auto tol{10.f * std::numeric_limits<scalar_t>::epsilon()};
237 bounds[e_half_x] < tol || bounds[e_half_y] < tol) {
238 os << "DETRAY ERROR (HOST): Half lengths must be in the range (0, "
239 "numeric_max)"
240 << std::endl;
241 return false;
242 }
243
244 return true;
245 }
