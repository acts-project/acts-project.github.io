40template <typename track_t,
41 typename generator_t =
42 detail::random_numbers<dscalar<typename track_t::algebra_type>>>
43class uniform_track_generator
44 : public detray::ranges::view_interface<
45 uniform_track_generator<track_t, generator_t>> {

48 using vector3_t = dvector3D<algebra_t>;

73 DETRAY_HOST_DEVICE
74 constexpr iterator(std::shared_ptr<generator_t> rand_gen, configuration cfg,
75 std::size_t iph = 1u, std::size_t ith = 0u)
76 : m_rnd_numbers{std::move(rand_gen)},
77 m_cfg{cfg},
78 m_phi_step_size{(cfg.phi_range()[1] - cfg.phi_range()[0]) /
79 static_cast<scalar_t>(cfg.phi_steps())},
80 m_theta_step_size{(cfg.theta_range()[1] - cfg.theta_range()[0]) /
81 static_cast<scalar_t>(cfg.theta_steps() - 1u)},
82 m_eta_step_size{(cfg.eta_range()[1] - cfg.eta_range()[0]) /
83 static_cast<scalar_t>(cfg.eta_steps() - 1u)},
84 m_phi{cfg.phi_range()[0]},
85 m_theta{cfg.uniform_eta() ? get_theta(cfg.eta_range()[0])
86 : cfg.theta_range()[0]},
87 i_phi{iph},
88 i_theta{ith} {}

92 DETRAY_HOST_DEVICE
93 constexpr bool operator==(const iterator& rhs) const {
94 return rhs.i_phi == i_phi && rhs.i_theta == i_theta;
95 }

101 DETRAY_HOST_DEVICE
102 constexpr auto operator++() -> iterator& {
103 // Check theta range according to step size
104 if (i_theta < m_cfg.theta_steps()) {
105 // Check phi sub-range
106 if (i_phi < m_cfg.phi_steps()) {
107 // Calculate new phi in the given range
108 m_phi = m_cfg.phi_range()[0] +
109 static_cast<scalar_t>(i_phi) * m_phi_step_size;
110 ++i_phi;
111 return *this;
112 }
113 // Reset phi range
114 i_phi = 1;
115 m_phi = m_cfg.phi_range()[0];
116
117 // Calculate new theta in the given range
118 ++i_theta;
119
120 if (m_cfg.uniform_eta()) {
121 const scalar_t eta = m_cfg.eta_range()[0] +
122 static_cast<scalar_t>(i_theta) * m_eta_step_size;
123 m_theta = get_theta(eta);
124 } else {
125 m_theta = m_cfg.theta_range()[0] +
126 static_cast<scalar_t>(i_theta) * m_theta_step_size;
127 }
128 }
129 return *this;
130 }

137 ++(*this);
138 return tmp;
139 }

143 DETRAY_HOST_DEVICE
144 track_t operator*() const {
145 if (!m_rnd_numbers) {
146 std::string err_str{"Invalid random number generator"};
147 DETRAY_FATAL_HOST(err_str);
148 throw std::invalid_argument(err_str);
149 }
150
151 scalar_t sin_theta{math::sin(m_theta)};
152
153 // Momentum direction from angles
154 vector3_t p{math::cos(m_phi) * sin_theta, math::sin(m_phi) * sin_theta,
155 math::cos(m_theta)};
156
157 // Magnitude of momentum
158 if constexpr (std::is_same_v<track_t, detail::ray<algebra_t>>) {
159 p = vector::normalize(p);
160 } else {
161 sin_theta = (sin_theta == scalar_t{0.f})
162 ? std::numeric_limits<scalar_t>::epsilon()
163 : sin_theta;
164 p = (m_cfg.is_pT() ? 1.f / sin_theta : 1.f) * m_cfg.m_p_mag *
165 vector::normalize(p);
166 }
167
168 const auto& ori = m_cfg.origin();
169
170 // Randomly flip the charge sign
171 darray<double, 2> signs{1., -1.};
172 const auto sign{static_cast<scalar_t>(
173 signs[m_cfg.randomize_charge() ? m_rnd_numbers->coin_toss() : 0u])};
174
175 return track_t{
176 {ori[0], ori[1], ori[2]}, m_cfg.time(), p, sign * m_cfg.charge()};
177 }
178
180 std::shared_ptr<generator_t> m_rnd_numbers;
181
183 configuration m_cfg{};
184
186 scalar_t m_phi_step_size{0.f};
187 scalar_t m_theta_step_size{0.f};
188 scalar_t m_eta_step_size{0.f};
189
191 scalar_t m_phi{-constant<scalar_t>::pi};
192 scalar_t m_theta{0.f};
193
195 std::size_t i_phi{0u};
196 std::size_t i_theta{0u};
197
198 private:
202 DETRAY_HOST_DEVICE
203 scalar_t get_theta(const scalar_t eta) {
204 return 2.f * math::atan(math::exp(-eta));
205 }
206 };
