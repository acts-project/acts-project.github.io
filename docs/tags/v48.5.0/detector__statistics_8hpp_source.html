63template <std::size_t I = 0u, typename store_t>
64DETRAY_HOST_DEVICE inline std::size_t n_grids(const store_t& store,
65 std::size_t n = 0u) {
66 constexpr auto coll_id{types::id_cast<typename store_t::value_types, I>};
67 using value_type = types::get<typename store_t::value_types, coll_id>;
68
69 if constexpr (detray::concepts::grid<value_type>) {
70 n += store.template size<coll_id>();
71 }
72
73 if constexpr (I < store_t::n_collections() - 1u) {
74 return n_grids<I + 1>(store, n);
75 }
76 return n;
77}
