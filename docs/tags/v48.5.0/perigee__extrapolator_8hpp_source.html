122 DETRAY_HOST_DEVICE auto extrapolate(
123 state &propagation,
124 typename actor_chain_type::state_ref_tuple actor_state_refs =
125 typename actor_chain_type::state_ref_tuple{}) const {
126 propagation.navigation().set_direction(navigation::direction::e_backward);
127 m_propagator.propagate(propagation, actor_state_refs);
128
129 assert(finished(propagation));
130 assert(!propagation.stepping()().is_invalid());
131
132 return detail::free_to_bound_vector<line2D<algebra_type>>(
133 m_perigee_surface.transform(), propagation.stepping()());
134 }

139 DETRAY_HOST_DEVICE auto extrapolate(state &propagation) {
140 // Run propagation
141 propagation.navigation().set_direction(navigation::direction::e_backward);
142 m_propagator.propagate(propagation);
143
144 assert(finished(propagation));
145 assert(!propagation.stepping()().is_invalid());
146
147 return detail::free_to_bound_vector<line2D<algebra_type>>(
148 m_perigee_surface.transform(), propagation.stepping()());
149 }
150
151 private:
153 static constexpr typename perigee_surface::material_type perigee_material{
154 detray::vacuum<scalar_type>{}, std::numeric_limits<scalar_type>::max()};
156 static constexpr typename perigee_surface::mask_type perigee_mask{
157 0u, std::numeric_limits<scalar_type>::max(),
158 std::numeric_limits<scalar_type>::max()};
160 perigee_surface m_perigee_surface{dtransform3D<algebra_type>{}, perigee_mask,
161 perigee_material, 0u,
162 surface_id::e_passive};
164 propagator_type m_propagator{};
165};
