354 template <typename propagator_state_t>
355 DETRAY_HOST_DEVICE result operator()(state& updater_state,
356 propagator_state_t& propagation) const {
357 const auto& navigation = propagation.navigation();
358
359 // Do covariance transport only when the track is on surface
360 if (!(navigation.is_on_sensitive() ||
361 navigation.encountered_sf_material())) {
362 return {};
363 }
364 assert(navigation.is_on_surface());
365
366 // Current track position and transport jacobian
367 auto& stepping = propagation.stepping();
368
369 // Destination surface (current)
370 const auto dest_sf = navigation.current_surface();
371
372 // Bound track params of departure surface (not yet updated)
373 auto& departure_params = updater_state.bound_params();
374
375 // Result to be passed on to observing actors
376 result res{actor::status::e_notify, departure_params};
377
378 // Covariance is transported only when the departure surface is an
379 // actual tracking surface (i.e. this disables the covariance
380 // transport from curvilinear frames).
381 if (!departure_params.surface_link().is_invalid()) {
382 DETRAY_DEBUG_HOST(
383 "Actor: Departure surface: " << departure_params.surface_link());
384
385 // There is no need to transport the identical/initial parameters
386 assert(!dest_sf.identifier().is_invalid());
387 if (departure_params.surface_link() == dest_sf.identifier()) {
388 DETRAY_VERBOSE_HOST_DEVICE(
389 "Actor: On initial surface (%d), parameter transport not required",
390 dest_sf.index());
391
392 // Notify observers?
393 if (!updater_state.m_notify_on_initial) {
394 res.status = actor::status::e_unknown;
395 }
396
397 return res;
398 }
399
400 DETRAY_DEBUG_HOST("Actor: Destination surface: " << dest_sf.identifier());
401 DETRAY_VERBOSE_HOST_DEVICE(
402 "Actor: Transport track covariance to surface %d", dest_sf.index());
403
404 DETRAY_DEBUG_HOST("Actor: Departure bound param.: " << departure_params);
405
406 // Transport the covariance
407 const bound_matrix<algebra_t> propagation_step_jacobian =
408 get_full_jacobian(propagation, departure_params);
409
410 // Update the full Jacobian, if required
411 if (math::fabs(stepping.path_length()) > 0.f) {
412 if (updater_state.has_full_jacobian()) {
413 updater_state.set_full_jacobian(propagation_step_jacobian *
414 updater_state.full_jacobian());
415 }
416
417 // Reset transport Jacobian to identity matrix
418 stepping.reset_transport_jacobian();
419 }
420
421 // Copy old covariance in order to update new covariance in place
422 const bound_matrix_type old_cov = departure_params.covariance();
423 bound_matrix_type& new_cov = departure_params.covariance();
424
425 detray::detail::transport_covariance_to_bound_impl(
426 old_cov, propagation_step_jacobian, new_cov);
427 } else {
428 DETRAY_VERBOSE_HOST_DEVICE("Actor: Departure surface link invalid");
429 }
430
431 DETRAY_VERBOSE_HOST_DEVICE(
432 "Actor: Convert track parameter vector to local for surface %d",
433 dest_sf.index());
434
435 // Get the bound parameters at the destination surface
436 res.destination_params().set_parameter_vector(
437 dest_sf.free_to_bound_vector(propagation.context(), stepping()));
438
439 // Set new surface link
440 res.destination_params().set_surface_link(dest_sf.identifier());
441
442 assert(!res.destination_params().is_invalid());
443
444 DETRAY_DEBUG_HOST("Actor: Transported bound param.:\n"
445 << res.destination_params());
446
447 return res;
448 }

455 template <typename propagator_state_t>
456 DETRAY_HOST_DEVICE constexpr bound_matrix_type get_full_jacobian(
457 propagator_state_t& propagation,
458 const bound_track_parameters_type& departure_params) const {
459 // Map the surface shapes of the detector down to the common frames
460 using detector_t = typename propagator_state_t::detector_type;
461 using frame_registry_t =
462 types::mapped_registry<typename detector_t::masks, select_frame>;
463
464 const auto& gctx = propagation.context();
465 const auto& stepping = propagation.stepping();
466 const auto& navigation = propagation.navigation();
467
468 // Our goal here is to compute the full Jacobian, which is given as:
469 //
470 // J_full = J_F2B * (D + I) * J_transport * J_B2F
471 //
472 // The transport Jacobian is given, but the free-to-bound and
473 // bound-to-free matrices as well as the derivative matrix $D$ need
474 // to be computed still. In order to avoid full-rank matrix
475 // multiplications, we use the known substructure of the
476 // aforementioned matrices in order to perform fewer operations. We
477 // describe in detail the mathematics performed throughout the
478 // remainder of this function.
479
480 // Departure surface
481 const tracking_surface dep_sf{navigation.detector(),
482 departure_params.surface_link()};
483
484 // Destination Surface
485 const tracking_surface dest_sf = navigation.current_surface();
486
487 // Free track params of departure surface
488 const free_track_parameters<algebra_t> dep_free_params =
489 dep_sf.bound_to_free_vector(gctx, departure_params);
490
491 // Free track params of destination surface
492 const free_track_parameters<algebra_t>& dest_free_params = stepping();
493
494 // First, we will compute the bound-to-free Jacobian, which is given
495 // by three sub-Jacobians. In particular, the matrix has an 8x6
496 // shape and looks like this:
497 //
498 // l0 l1 phi th q/p t
499 // px [[ A, A, B, B, 0, 0],
500 // py [ A, A, B, B, 0, 0],
501 // pz [ A, A, B, B, 0, 0],
502 // t [ 0, 0, 0, 0, 0, 1],
503 // dx [ 0, 0, C, C, 0, 0],
504 // dy [ 0, 0, C, C, 0, 0],
505 // dz [ 0, 0, 0, C, 0, 0],
506 // q/p [ 0, 0, 0, 0, 1, 0]]
507 //
508 // Note that we thus only have 17 out of 48 non-trivial matrix
509 // elements.
510 //
511 // In this matrix, A represents the d(pos)/d(loc) submatrix, B is
512 // the d(pos)/d(angle) submatrix, and C is the d(dir)/d(angle)
513 // submatrix, all of which are 3x2 in size. Also, A and B depend on
514 // the frame type while C is computed in the same way for all
515 // frames. Finally, note that submatrix B is the zero matrix for
516 // most frame types.
517 const auto& dep_trf3 = dep_sf.transform(gctx);
518 const dmatrix<algebra_t, 3, 2> b2f_dpos_dloc =
519 types::visit<frame_registry_t, get_bound_to_free_dpos_dloc_visitor>(
520 dep_sf.shape_id(), dep_trf3, dep_free_params);
521
522 const dmatrix<algebra_t, 3, 2> b2f_ddir_dangle =
523 detail::jacobian_engine<algebra_t>::
524 bound_to_free_jacobian_submatrix_ddir_dangle(departure_params);
525
526 const dmatrix<algebra_t, 3, 2> b2f_dpos_dangle =
527 types::visit<frame_registry_t, get_bound_to_free_dpos_dangle_visitor>(
528 dep_sf.shape_id(), dep_trf3, dep_free_params, b2f_ddir_dangle);
529
530 // Next, we compute the derivative which is defined as the outer
531 // product of the two 8x1 vectors representing the path to free
532 // derivative and the free to path derivative. Thus, it is the
533 // product of the following two vectors:
534 //
535 // px [[ A], [[ B],^T
536 // py [ A], [ B],
537 // pz [ A], [ B],
538 // t [ 0]] [ 0],
539 // dx [ A], [ C],
540 // dy [ A], [ C],
541 // dz [ A], [ C],
542 // q/p [ A], [ 0]]
543 //
544 // Where A is frame independent and non-zero, B is frame-dependent
545 // and non-zero, while C is frame-dependent and non-zero only for
546 // line frames.
547 const dpoint3D<algebra_t> dest_glob_pos{dest_free_params.pos()};
548 const dvector3D<algebra_t> dest_glob_dir{dest_free_params.dir()};
549
550 auto vol = navigation.current_volume();
551 const auto vol_mat_ptr =
552 vol.has_material() ? vol.material_parameters(dest_glob_pos) : nullptr;
553
554 const auto path_to_free_derivative =
555 detail::jacobian_engine<algebra_t>::path_to_free_derivative(
556 dest_glob_dir, stepping.dtds(), stepping.dqopds(vol_mat_ptr));
557
558 const auto free_to_path_derivative = dest_sf.free_to_path_derivative(
559 gctx, dest_glob_pos, dest_glob_dir, stepping.dtds());
560
561 // Now, we compute the free-to-bound Jacobian which is of size 6x8
562 // and has the following structure:
563 //
564 // px py pz t dx dy dz q/p
565 // l0 [[ A, A, A, 0, 0, 0, 0, 0],
566 // l1 [ A, A, A, 0, 0, 0, 0, 0],
567 // phi [ 0, 0, 0, 0, B, B, 0, 0],
568 // th [ 0, 0, 0, 0, B, B, B, 0],
569 // q/p [ 0, 0, 0, 0, 0, 0, 0, 1],
570 // t [ 0, 0, 0, 1, 0, 0, 0, 0]]
571 //
572 // Thus, the number of non-trivial elements is only 11 out of the 48
573 // matrix elements. Also, submatrix A depends on the frame type
574 // while submatrix B is the same for all frame types.
575 const dmatrix<algebra_t, 2, 3> f2b_dloc_dpos =
576 types::visit<frame_registry_t, get_free_to_bound_dloc_dpos_visitor>(
577 dest_sf.shape_id(), dest_sf.transform(gctx), stepping);
578
579 const dmatrix<algebra_t, 2, 3> f2b_dangle_ddir = detail::jacobian_engine<
580 algebra_t>::free_to_bound_jacobian_submatrix_dangle_ddir(dest_glob_dir);
581
582 // Finally, we can use our Sympy-generated full Jacobian computation
583 // and return its result.
584 bound_matrix_type full_jacobian;
585
586 if constexpr (std::decay_t<propagator_state_t>::stepper_uses_gradient) {
587 detail::update_full_jacobian_with_gradient_impl(
588 stepping.internal_transport_jacobian(), b2f_dpos_dloc,
589 b2f_ddir_dangle, b2f_dpos_dangle, path_to_free_derivative,
590 free_to_path_derivative, f2b_dloc_dpos, f2b_dangle_ddir,
591 full_jacobian);
592 } else {
593 detail::update_full_jacobian_without_gradient_impl(
594 stepping.internal_transport_jacobian(), b2f_dpos_dloc,
595 b2f_ddir_dangle, b2f_dpos_dangle, path_to_free_derivative,
596 free_to_path_derivative, f2b_dloc_dpos, f2b_dangle_ddir,
597 full_jacobian);
598 }
599
600 return full_jacobian;
601 }

606template <concepts::algebra algebra_t>
607struct parameter_setter : base_actor {
609 using state = parameter_updater_state<algebra_t>;
610
616 template <typename propagator_state_t>
617 DETRAY_HOST_DEVICE void operator()(
618 state& updater_state, propagator_state_t& propagation,
619 const parameter_transporter_result<algebra_t>& res) const {
620 // Update the track parameters only if necessary
621 if (res.status != actor::status::e_success &&
622 !updater_state.m_always_update) {
623 return;
624 }
625
626 const auto& navigation = propagation.navigation();
627 auto& stepping = propagation.stepping();
628
629 // One of the transporter observers might have exited the navigation
630 assert(navigation.is_on_surface() || !navigation.is_alive());
631
632 DETRAY_VERBOSE_HOST_DEVICE("Actor: Update the track parameters");
633
634 // Updated bound track parameters
635 const bound_track_parameters<algebra_t>& bound_params =
636 res.destination_params();
637
638 // Update free params after bound params were changed by actors
639 const tracking_surface dest_sf{navigation.detector(),
640 bound_params.surface_link()};
641 stepping() =
642 dest_sf.bound_to_free_vector(propagation.context(), bound_params);
643
644 assert(!stepping().is_invalid());
645
646 // Track pos/dir is not always known precisely: adjust navigation
647 // tolerances according to bound covariance
648 if (const noise_cfg& cfg = updater_state.noise_estimation_cfg();
649 cfg.estimate_scattering_noise) {
650 detail::estimate_external_mask_tolerance(
651 bound_params, propagation,
652 static_cast<dscalar<algebra_t>>(cfg.n_stddev), cfg.accumulated_error);
653 }
654
655 DETRAY_DEBUG_HOST("Actor: Updated bound param.:\n" << bound_params);
656 }
657};
