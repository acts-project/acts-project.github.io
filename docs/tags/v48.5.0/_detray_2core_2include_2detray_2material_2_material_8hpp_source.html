153 DETRAY_HOST_DEVICE
154 scalar_type mean_excitation_energy() const {
155 if (!m_has_density_effect_data) {
156 // use approximative computation as defined in ATL-SOFT-PUB-2008-003
157 return 16.f * unit<scalar_type>::eV *
158 math::pow(m_z, static_cast<scalar_type>(0.9));
159 } else {
160 return m_density.get_mean_excitation_energy();
161 }
162 }

166 DETRAY_HOST_DEVICE
167 constexpr scalar_type fraction() const {
168 if constexpr (ratio::num == 0) {
169 return 0.f;
170 } else if constexpr (ratio::den == 0) {
171 return detail::invalid_value<scalar_type>();
172 } else {
173 return static_cast<scalar_type>(ratio::num) /
174 static_cast<scalar_type>(ratio::den);
175 }
176 }

179 DETRAY_HOST
180 std::string to_string() const {
181 std::stringstream strm;
182 if (m_ar <= 0.f) {
183 strm << "vacuum";
184 return strm.str();
185 }
186 strm << "material: ";
187 strm << " X0 = " << m_x0;
188 strm << " | L0 = " << m_l0;
189 strm << " | Z = " << m_z;
190
191 strm << " | state = ";
192 switch (m_state) {
193 using enum material_state;
194 case e_solid: {
195 strm << "solid";
196 break;
197 }
198 case e_liquid: {
199 strm << "liquid";
200 break;
201 }
202 case e_gas: {
203 strm << "gaseous";
204 break;
205 }
206 default: {
207 strm << "unknown";
208 break;
209 }
210 }
211
212 return strm.str();
213 }

259 DETRAY_HOST_DEVICE
260 constexpr scalar_type mass_to_molar_density(double ar, double mass_rho) {
261 if (mass_rho == 0.) {
262 return 0.f;
263 }
264
265 const double molar_mass{ar * unit<double>::g / unit<double>::mol};
266
267 return static_cast<scalar_type>(mass_rho / molar_mass);
268 }

280 bool m_has_density_effect_data = false;
