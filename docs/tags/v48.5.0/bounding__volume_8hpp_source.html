89 template <typename other_shape_t>
90 requires std::is_same_v<
91 typename shape::template local_frame_type<algebra_t>,
92 typename other_shape_t::template local_frame_type<algebra_t>>
93 DETRAY_HOST constexpr axis_aligned_bounding_volume(
94 const std::vector<axis_aligned_bounding_volume<other_shape_t, algebra_t>>&
95 aabbs,
96 std::size_t box_id, const scalar_t env) {
97 using loc_point_t = darray<scalar_t, other_shape_t::dim>;
98
99 // Find min/max extent of the local aabb in local coordinates
100 constexpr scalar_t inv{detail::invalid_value<scalar_t>()};
101 scalar_t min_x{inv};
102 scalar_t min_y{inv};
103 scalar_t min_z{inv};
104 scalar_t max_x{-inv};
105 scalar_t max_y{-inv};
106 scalar_t max_z{-inv};
107 for (const auto& vol : aabbs) {
108 const auto min_point = vol.template loc_min<loc_point_t>();
109 const auto max_point = vol.template loc_max<loc_point_t>();
110
111 // Check every coordinate of the points
112 min_x = min_point[0] < min_x ? min_point[0] : min_x;
113 min_y = min_point[1] < min_y ? min_point[1] : min_y;
114
115 max_x = max_point[0] > max_x ? max_point[0] : max_x;
116 max_y = max_point[1] > max_y ? max_point[1] : max_y;
117
118 if constexpr (min_point.size() > 2) {
119 min_z = min_point[2] < min_z ? min_point[2] : min_z;
120 max_z = max_point[2] > max_z ? max_point[2] : max_z;
121 }
122 }
123 m_mask = mask<shape, algebra_t, std::size_t>{
124 box_id, min_x - env, min_y - env, min_z - env,
125 max_x + env, max_y + env, max_z + env};
126 }

149 template <typename point_t>
150 DETRAY_HOST_DEVICE constexpr auto loc_min() const -> point_t {
151 if constexpr (std::is_same_v<shape, cuboid3D>) {
152 return {m_mask[cuboid3D::e_min_x], m_mask[cuboid3D::e_min_y],
153 m_mask[cuboid3D::e_min_z]};
154 } else if constexpr (std::is_same_v<shape, cylinder3D>) {
155 return {-m_mask[cylinder3D::e_max_r], m_mask[cylinder3D::e_min_phi],
156 m_mask[cylinder3D::e_min_z]};
157 }
158
159 // If the volume shape is not supported, return universal minimum
160 assert(false);
161 constexpr scalar_t inv{detail::invalid_value<scalar_t>()};
162 return point_t{-inv, -inv, -inv};
163 }
164
166 template <typename point_t>
167 DETRAY_HOST_DEVICE constexpr auto loc_max() const -> point_t {
168 if constexpr (std::is_same_v<shape, cuboid3D>) {
169 return {m_mask[cuboid3D::e_max_x], m_mask[cuboid3D::e_max_y],
170 m_mask[cuboid3D::e_max_z]};
171 } else if constexpr (std::is_same_v<shape, cylinder3D>) {
172 return {m_mask[cylinder3D::e_max_r], m_mask[cylinder3D::e_max_phi],
173 m_mask[cylinder3D::e_max_z]};
174 }
175
176 // If the volume shape is not supported, return universal minimum
177 // (or compilation error for 2D point)
178 assert(false);
179 constexpr scalar_t inv{detail::invalid_value<scalar_t>()};
180 return point_t{inv, inv, inv};
181 }

186 template <concepts::transform3D transform3_t>
187 DETRAY_HOST_DEVICE constexpr auto glob_min(const transform3_t& trf) const ->
188 typename transform3_t::point3 {
189 using point3_t = typename transform3_t::point3;
190
191 if constexpr (std::is_same_v<shape, cuboid3D>) {
192 return trf.point_to_global(loc_min<point3_t>());
193 } else if constexpr (std::is_same_v<shape, cylinder3D>) {
194 return cylindrical3D<transform3_t>{}.local_to_global(trf, m_mask,
195 loc_min<point3_t>());
196 }
197
198 // If the volume shape is not supported, return universal minimum
199 // (or compilation error for 2D point)
200 assert(false);
201 constexpr scalar_t inv{detail::invalid_value<scalar_t>()};
202 return point3_t{-inv, -inv, -inv};
203 }

228 template <concepts::point3D point3_t>
229 DETRAY_HOST_DEVICE constexpr auto center() const -> point3_t {
230 const scalar_t center_x{
231 0.5f * (m_mask[cuboid3D::e_max_x] + m_mask[cuboid3D::e_min_x])};
232 const scalar_t center_y{
233 0.5f * (m_mask[cuboid3D::e_max_y] + m_mask[cuboid3D::e_min_y])};
234 const scalar_t center_z{
235 0.5f * (m_mask[cuboid3D::e_max_z] + m_mask[cuboid3D::e_min_z])};
236
237 return {detail::is_invalid_value(center_x) ? 0.f : center_x,
238 detail::is_invalid_value(center_y) ? 0.f : center_y,
239 detail::is_invalid_value(center_z) ? 0.f : center_z};
240 }

251 template <concepts::transform3D transform3_t, typename S = shape_t>
252 requires std::is_same_v<S, cuboid3D>
253 DETRAY_HOST_DEVICE auto transform(const transform3_t& trf) const
254 -> axis_aligned_bounding_volume {
255 using point3_t = typename transform3_t::point3;
256
257 const scalar_t scalor_x{
258 (m_mask[cuboid3D::e_max_x] - m_mask[cuboid3D::e_min_x])};
259 const scalar_t scalor_y{
260 (m_mask[cuboid3D::e_max_y] - m_mask[cuboid3D::e_min_y])};
261 const scalar_t scalor_z{
262 (m_mask[cuboid3D::e_max_z] - m_mask[cuboid3D::e_min_z])};
263
264 // Cannot handle 'inv' propagation through the calculation for now
265 if (detail::is_invalid_value(scalor_x) ||
266 detail::is_invalid_value(scalor_y) ||
267 detail::is_invalid_value(scalor_z)) {
268 // If the box was infinite to begin with, it stays that way
269 assert(detail::is_invalid_value(scalor_x) &&
270 detail::is_invalid_value(scalor_y) &&
271 detail::is_invalid_value(scalor_z));
272
273 return *this;
274 }
275
276 // The new axis vectors, scaled to the aabb dimensions
277 // (e.g. max_x - min_x)
278 const point3_t new_box_x = scalor_x * trf.x();
279 const point3_t new_box_y = scalor_y * trf.y();
280 const point3_t new_box_z = scalor_z * trf.z();
281
282 // Transform the old min and max points to the global frame and
283 // construct all corner points of the local aabb in global coordinates
284 darray<point3_t, 8> glob_c_points;
285 glob_c_points[0] = glob_min(trf);
286 glob_c_points[1] = glob_max(trf);
287 glob_c_points[2] = glob_c_points[0] + new_box_x;
288 glob_c_points[3] = glob_c_points[0] + new_box_y;
289 glob_c_points[4] = glob_c_points[0] + new_box_z;
290 glob_c_points[5] = glob_c_points[1] - new_box_x;
291 glob_c_points[6] = glob_c_points[1] - new_box_y;
292 glob_c_points[7] = glob_c_points[1] - new_box_z;
293
294 // Find min/max extent of the local aabb in global coordinates
295 constexpr scalar_t inv{detail::invalid_value<scalar_t>()};
296 scalar_t min_x{inv};
297 scalar_t min_y{inv};
298 scalar_t min_z{inv};
299 scalar_t max_x{-inv};
300 scalar_t max_y{-inv};
301 scalar_t max_z{-inv};
302 for (const point3_t& p : glob_c_points) {
303 // Check every coordinate of the point
304 min_x = p[0] < min_x ? p[0] : min_x;
305 max_x = p[0] > max_x ? p[0] : max_x;
306 min_y = p[1] < min_y ? p[1] : min_y;
307 max_y = p[1] > max_y ? p[1] : max_y;
308 min_z = p[2] < min_z ? p[2] : min_z;
309 max_z = p[2] > max_z ? p[2] : max_z;
310 }
311
312 // Construct the transformed aabb
313 return axis_aligned_bounding_volume{
314 m_mask.volume_link(), min_x, min_y, min_z, max_x, max_y, max_z};
315 }
