59template <concepts::detector detector_t>
60DETRAY_HOST inline std::string print_detector(
61 const detector_t &det, const typename detector_t::name_map &names = {}) {
62 // Gathers navigation information across navigator update calls
63 std::stringstream debug_stream{};
64
65 debug_stream << std::left << "[>] Detector " << det.name(names) << " has "
66 << det.volumes().size() << " volumes." << std::endl;
67
68 for (const auto [i, v_desc] : detray::views::enumerate(det.volumes())) {
69 tracking_volume v{det, v_desc};
70
71 debug_stream << "[>>] Volume " << v.name(names) << std::endl;
72 debug_stream << v << std::endl;
73
74 debug_stream << "[>>>] Acceleration Structures:" << std::endl;
75 const auto acc_link = v_desc.accel_link();
76 for (std::size_t j = 0u; j < acc_link.size(); ++j) {
77 // An acceleration data structure link was set, but is invalid
78 if (!acc_link[j].is_invalid_id() && !acc_link[j].is_invalid_index()) {
79 debug_stream << "Surfaces registered in '" << acc_link[j].id()
80 << "':" << std::endl;
81 det.accelerator_store().template visit<detail::accelerator_printer>(
82 acc_link[j], debug_stream);
83 }
84 }
85
86 debug_stream << "[>>>] Surfaces:" << std::endl;
87 for (const auto sf_desc : v.template surfaces<>()) {
88 geometry::surface sf{det, sf_desc};
89 debug_stream << sf << std::endl;
90
91 // Check the surface material, if present
92 if (sf.has_material()) {
93 debug_stream << "[>>>>] Surface material: ";
94 sf.template visit_material<detail::material_printer>(debug_stream);
95 }
96 }
97
98 debug_stream << std::endl;
99 }
100
101 return debug_stream.str();
102}
