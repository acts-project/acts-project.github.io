211// Specialize std::hash so surface descriptors can be used in STL containers
212template <typename mask_link_t, typename material_link_t,
213 typename transform_link_t, typename navigation_link_t>
214struct hash<detray::surface_descriptor<mask_link_t, material_link_t,
215 transform_link_t, navigation_link_t>> {
216 using descr_t =
217 detray::surface_descriptor<mask_link_t, material_link_t, transform_link_t,
218 navigation_link_t>;
219
220 auto operator()(const descr_t sf_desc) const noexcept {
221 return std::hash<detray::geometry::identifier::value_t>()(
222 sf_desc.identifier().value());
223 }
224};
