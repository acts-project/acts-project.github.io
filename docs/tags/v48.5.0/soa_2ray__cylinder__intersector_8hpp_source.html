68 template <typename mask_t, concepts::algebra other_algebra_t>
69 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
70 const trajectory_type<other_algebra_t> &ray, const transform3_type &trf,
71 const mask_t &mask, const scalar_type /*overstep_tol*/ = 0.f) const {
72 // One or both of these solutions might be invalid
73 const auto qe = solve_intersection(ray, mask, trf);
74
75 const auto &pos = ray.pos();
76 const auto &dir = ray.dir();
77 const point3_type ro{pos[0], pos[1], pos[2]};
78 const vector3_type rd{dir[0], dir[1], dir[2]};
79
80 result_type results;
81
82 results[0].path = qe.smaller();
83 results[0].point = ro + qe.smaller() * rd;
84
85 results[1].path = qe.larger();
86 results[1].point = ro + qe.larger() * rd;
87
88 return results;
89 }
90
91 protected:
99 template <typename mask_t, typename other_algebra_t>
100 DETRAY_HOST_DEVICE inline auto solve_intersection(
101 const trajectory_type<other_algebra_t> &ray, const mask_t &mask,
102 const transform3_type &trf) const {
103 const vector3_type &sz = trf.z();
104 const point3_type &sc = trf.translation();
105
106 const scalar_type r = mask[mask_t::shape::e_r];
107
108 const auto &pos = ray.pos();
109 const auto &dir = ray.dir();
110 const point3_type ro{pos[0], pos[1], pos[2]};
111 const vector3_type rd{dir[0], dir[1], dir[2]};
112
113 const vector3_type tmp = ro - sc;
114 const auto pc_cross_sz = vector::cross(tmp, sz);
115 const auto rd_cross_sz = vector::cross(rd, sz);
116 const scalar_type a = vector::dot(rd_cross_sz, rd_cross_sz);
117 const scalar_type b = 2.f * vector::dot(rd_cross_sz, pc_cross_sz);
118 const scalar_type c = vector::dot(pc_cross_sz, pc_cross_sz) - (r * r);
119
120 return detail::quadratic_equation<scalar_type>{a, b, c};
121 }
122};
