24template <typename A>

37template <typename A>
38concept actor_chain = requires(const A c, typename A::state_tuple s) {
39 typename A::actor_tuple;
40 typename A::state_tuple;
41 typename A::state_ref_tuple;
42
43 { c.actors() } -> std::same_as<const typename A::actor_tuple &>;
44
45 {
46 c.make_default_actor_states()
47 } -> detray::concepts::any_of<typename A::state_tuple, std::nullopt_t>;
48
49 { c.setup_actor_states(s) } -> std::same_as<typename A::state_ref_tuple>;
50};

53template <typename S, typename T>
54concept is_state_of =
55 (actor_chain<T> && (detail::is_permutation_v<std::remove_cvref_t<S>,
56 typename T::state_ref_tuple> ||
57 detail::is_permutation_v<std::remove_cvref_t<S>,
58 typename T::state_tuple>)) ||
59
60 (actor<T> && std::same_as<std::remove_cvref_t<S>, typename T::state>);

63template <typename S>
64concept propagator_state = requires {
65 typename S::detector_type;
66 typename S::detector_type::scalar_type;
67 typename S::stepper_state_type;
68 typename S::navigator_state_type;
69 typename S::context_type;
70} && requires(S &s) {
71 requires requires(
72 const pdg_particle<typename S::detector_type::scalar_type> &ptc) {
73 { s.set_particle(ptc) };
74 };
75
76 requires requires(bool b) {
77 { s.heartbeat(b) };
78 { s.debug(b) };
79 };
80
81 { s.stepping() } -> std::same_as<typename S::stepper_state_type &>;
82 { s.navigation() } -> std::same_as<typename S::navigator_state_type &>;
83 { s.context() } -> std::same_as<typename S::context_type &>;
84} && requires(const S &s) {
85 { s.debug() } -> std::same_as<bool>;
86 { s.is_alive() } -> std::same_as<bool>;
87 { s.heartbeat() } -> std::same_as<bool>;
88 { s.stepping() } -> std::same_as<const typename S::stepper_state_type &>;
89 { s.navigation() } -> std::same_as<const typename S::navigator_state_type &>;
90 { s.context() } -> std::same_as<const typename S::context_type &>;
91};

95template <typename S, typename P>
96concept is_propagator_state_of = propagator_state<S> && requires {
97 typename P::detector_type;
98 typename P::stepper_type;
99 typename P::navigator_type;
100
101 requires std::same_as<typename P::detector_type, typename S::detector_type>;
102 requires std::same_as<typename P::stepper_type::state,
103 typename S::stepper_state_type>;
104 requires std::same_as<typename P::navigator_type::state,
105 typename S::navigator_state_type>;
106};
