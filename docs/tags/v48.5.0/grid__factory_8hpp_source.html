48template <typename bin_t, template <std::size_t> class serializer_t,
49 concepts::algebra algebra_t>
50class grid_factory {
51 public:
52 // All grids are owning since they are used to fill the data
54 static constexpr bool is_owning = true;
55
57 using bin_type = bin_t;
59 template <typename grid_shape_t>
60 using grid_type = grid<algebra_t, axes<grid_shape_t>, bin_type, serializer_t>;
62 template <typename grid_shape_t>
63 using loc_bin_index = typename grid_type<grid_shape_t>::loc_bin_index;
64
66 using scalar_type = dscalar<algebra_t>;
68 template <typename T>
69 using vector_type = host_container_types::template vector_type<T>;
71 using algebra_type = algebra_t;
72
73 grid_factory() = default;
74
77 explicit grid_factory(vecmem::memory_resource &resource)
78 : m_resource(&resource) {}

80 //
81 // annulus 2D
82 //
90 template <
91 typename r_bounds = axis::closed<axis::label::e_r>,
92 typename phi_bounds = axis::circular<>,
93 typename r_binning = axis::regular<scalar_type, host_container_types>,
94 typename phi_binning = axis::regular<scalar_type, host_container_types>>
95 requires std::is_enum_v<decltype(r_bounds::label)>
96 auto new_grid(const mask<annulus2D, algebra_type> &grid_bounds,
97 const darray<std::size_t, 2UL> n_bins,
98 const std::vector<std::pair<loc_bin_index<annulus2D>, dindex>>
99 &bin_capacities = {},
100 const darray<std::vector<scalar_type>, 2UL> &bin_edges =
101 darray<std::vector<scalar_type>, 2UL>(),
102 const darray<std::vector<scalar_type>, 2UL> &axis_spans =
103 darray<std::vector<scalar_type>, 2UL>()) const {
104 static_assert(
105 std::is_same_v<phi_bounds, axis::circular<>>,
106 "Phi axis bounds need to be circular for stereo annulus shape");
107
108 // Axes boundaries and local indices
109 using boundary = annulus2D::boundaries;
110 using axes_t = axes<annulus2D>::template type<algebra_t>;
111 using local_frame = typename axes_t::template frame<algebra_t>;
112
113 constexpr auto e_r_axis = static_cast<dindex>(axes_t::label0);
114 constexpr auto e_phi_axis = static_cast<dindex>(axes_t::label1);
115
116 auto b_values = grid_bounds.values();
117 // Overwrite the mask values if axis spans are provided
118 if (!axis_spans[0UL].empty()) {
119 assert(axis_spans[0UL].size() == 2UL);
120 b_values[boundary::e_min_r] = axis_spans[0UL].at(0UL);
121 b_values[boundary::e_max_r] = axis_spans[0UL].at(1UL);
122 }
123 scalar_type min_phi =
124 b_values[boundary::e_average_phi] - b_values[boundary::e_min_phi_rel];
125 scalar_type max_phi =
126 b_values[boundary::e_average_phi] + b_values[boundary::e_max_phi_rel];
127 if (!axis_spans[1UL].empty()) {
128 assert(axis_spans[1UL].size() == 2UL);
129 min_phi = axis_spans[1UL].at(0UL);
130 max_phi = axis_spans[1UL].at(1UL);
131 }
132
133 return new_grid<local_frame>(
134 {b_values[boundary::e_min_r], b_values[boundary::e_max_r], min_phi,
135 max_phi},
136 {n_bins[e_r_axis], n_bins[e_phi_axis]}, bin_capacities,
137 {bin_edges[e_r_axis], bin_edges[e_phi_axis]},
138 types::list<r_bounds, phi_bounds>{},
139 types::list<r_binning, phi_binning>{});
140 }

551 template <concepts::coordinate_frame grid_frame_t, typename... bound_ts,
552 typename... binning_ts>
553 auto new_grid(
554 const std::vector<scalar_type> &spans,
555 const std::vector<std::size_t> &n_bins,
556 const std::vector<std::pair<axis::multi_bin<sizeof...(bound_ts)>, dindex>>
557 &bin_capacities = {},
558 const std::vector<std::vector<scalar_type>> &ax_bin_edges = {},
559 const types::list<bound_ts...> & /*unused*/ = {},
560 const types::list<binning_ts...> & /*unused*/ = {}) const {
561 static_assert(sizeof...(bound_ts) == sizeof...(binning_ts),
562 "number of axis bounds and binning types has to match");
563
564 // Build the coordinate axes and the grid
565 using axes_t = axis::multi_axis<is_owning, grid_frame_t,
566 axis::single_axis<bound_ts, binning_ts>...>;
567 using grid_t = grid<algebra_t, axes_t, bin_t, serializer_t>;
568
569 return new_grid<grid_t>(spans, n_bins, bin_capacities, ax_bin_edges);
570 }

650 template <concepts::grid grid_t>
651 auto new_grid(
652 const std::vector<scalar_type> &spans,
653 const std::vector<std::size_t> &n_bins,
654 [[maybe_unused]] const std::vector<
655 std::pair<typename grid_t::loc_bin_index, dindex>> &bin_capacities =
656 {},
657 const std::vector<std::vector<scalar_type>> &ax_bin_edges = {}) const {
658 using owning_grid_t = typename grid_t::template type<true>;
659 using axes_t = typename owning_grid_t::axes_type;
660
661 // Prepare data
662 vector_type<dsized_index_range> axes_data{};
663 vector_type<scalar_type> bin_edges{};
664
665 // Call init for every axis
666 unroll_axis_init<typename axes_t::binnings>(
667 spans, n_bins, ax_bin_edges, axes_data, bin_edges,
668 std::make_index_sequence<axes_t::dim>{});
669
670 // Assemble the grid and return it
671 axes_t axes(std::move(axes_data), std::move(bin_edges));
672
673 typename owning_grid_t::bin_container_type bin_data{};
674
675 // Bins with dynamic capacity and "index grids" need different treatment
676 if constexpr (std::is_same_v<bin_t, bins::dynamic_array<
677 typename grid_t::value_type>>) {
678 // Bin and bin entries vector are separate containers
679 bin_data.bins.resize(axes.nbins());
680 // Set correct bin capacities, if they were provided
681 if (!bin_capacities.empty()) {
682 // Set memory offsets correctly
683 typename grid_t::template serializer_type<grid_t::dim> serializer{};
684 dindex total_cap{0u};
685 for (const auto &capacity : bin_capacities) {
686 // Get the empty bin by its global index
687 auto &data = bin_data.bins[serializer(axes, capacity.first)];
688
689 data.offset = total_cap;
690 data.capacity = capacity.second;
691 data.size = 0u;
692
693 total_cap += data.capacity;
694 }
695 assert(total_cap > 0);
696 bin_data.entries.resize(
697 total_cap, detail::invalid_value<typename grid_t::value_type>());
698 }
699 } else {
700 // Bin entries are contained in the bins directly
701 bin_data.resize(axes.nbins(), bin_type{});
702 }
703
704 return owning_grid_t(std::move(bin_data), std::move(axes));
705 }

710 template <std::size_t I, typename binnings>
711 auto axis_init([[maybe_unused]] const std::vector<scalar_type> &spans,
712 [[maybe_unused]] const std::vector<std::size_t> &n_bins,
713 [[maybe_unused]] const std::vector<std::vector<scalar_type>>
714 &ax_bin_edges,
715 vector_type<dsized_index_range> &axes_data,
716 vector_type<scalar_type> &bin_edges) const {
717 if constexpr (std::is_same_v<
718 types::at<binnings, I>,
719 axis::regular<scalar_type, host_container_types>>) {
720 axes_data.push_back({static_cast<dindex>(bin_edges.size()),
721 static_cast<dindex>(n_bins.at(I))});
722 bin_edges.push_back(spans.at(I * 2u));
723 bin_edges.push_back(spans.at(I * 2u + 1u));
724 } else {
725 const auto &bin_edges_loc = ax_bin_edges.at(I);
726 axes_data.push_back({static_cast<dindex>(bin_edges.size()),
727 static_cast<dindex>(bin_edges_loc.size() - 1u)});
728 bin_edges.insert(bin_edges.end(), bin_edges_loc.begin(),
729 bin_edges_loc.end());
730 }
731 }
