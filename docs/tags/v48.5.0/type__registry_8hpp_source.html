101template <concepts::type_registry registry_t, class type_selector_t>
102class mapped_registry : public registry_t {
103 public:
105 using id = typename registry_t::id;
107 using orig_types = typename registry_t::type_list;
109 using type_list =
110 decltype(detray::types::filtered_list<orig_types, type_selector_t>());
111
113 enum : std::size_t {
114 n_types = detray::types::size<type_list>,
115 e_any = detray::types::size<type_list>,
116 e_unknown = detray::types::size<type_list> + 1u,
117 };
118
125 DETRAY_HOST_DEVICE static constexpr bool is_valid(
126 const std::size_t filtered_type_idx) {
127 return filtered_type_idx < n_types;
128 }
129
133 template <typename id_t = id>
134 requires(!std::convertible_to<id_t, std::size_t>)
135 DETRAY_HOST_DEVICE static constexpr bool is_valid(const id_t orig_id) {
136 if (static_cast<std::size_t>(orig_id) >= index_map().size()) {
137 return false;
138 } else {
139 return mapped_index(orig_id) < n_types;
140 }
141 }
142
143
145 DETRAY_HOST_DEVICE
146 static consteval auto index_map() {
147 return detray::types::filtered_indices<orig_types, type_selector_t>(
148 detray::types::list{});
149 }
150
155 DETRAY_HOST_DEVICE
156 static constexpr std::size_t mapped_index(const std::size_t orig_idx) {
157 return index_map_v<orig_types, type_selector_t>[orig_idx];
158 }
159
163 DETRAY_HOST_DEVICE
164 static constexpr std::size_t mapped_index(const id orig_id) {
165 return index_map_v<orig_types, type_selector_t>[static_cast<std::size_t>(
166 orig_id)];
167 }
168
171 DETRAY_HOST_DEVICE
172 static constexpr std::size_t original_index(const std::size_t pos) {
173 std::size_t j{0u};
174 for (std::size_t i : index_map_v<orig_types, type_selector_t>) {
175 if (i == pos) {
176 assert(registry_t::is_valid(j));
177 return j;
178 }
179 j++;
180 }
181 return detray::detail::invalid_value<std::size_t>();
182 }
183
186 DETRAY_HOST_DEVICE
187 static constexpr id original_id(const std::size_t pos) {
188 return static_cast<id>(original_index(pos));
189 }
190};

229template <typename T, std::size_t I, typename R, class S>
230struct contains_impl<T, I, mapped_registry<R, S>> {
231 // Figure out which type list to check
232 // (use original list if 'T' is from there, otherwise try filtered list)
234 using orig_list_t = typename mapped_registry<R, S>::orig_types;
236 using filt_list_t = typename mapped_registry<R, S>::type_list;
238 using type_list_t =
239 std::conditional_t<contains<orig_list_t, T>, orig_list_t, filt_list_t>;
240
242 using base_t = contains_impl<T, I, type_list_t>;
243
245 static constexpr bool value = base_t::value;
246
247 // Depending on which type list the type belongs to, map the index
249 static constexpr std::size_t mapped_idx =
250 std::same_as<type_list_t, orig_list_t>
251 ? mapped_registry<R, S>::mapped_index(base_t::pos)
252 : base_t::pos;
253
254 // The type is not contained in 'type_list', the position will be invalid:
255 // Do not pass that to the range-checked array of the filtered indices
257 static constexpr std::size_t pos =
258 !base_t::value ? std::numeric_limits<std::size_t>::max() : mapped_idx;
259};

448template <concepts::type_registry registry_t, typename functor_t,
449 typename... Args, std::size_t current_idx = 0u,
450 std::size_t... remaining_idcs>
451 requires(std::invocable<functor_t,
452 const types::front<typename registry_t::type_list>&,
453 Args...>)
454DETRAY_HOST_DEVICE constexpr decltype(auto) visit_helper(
455 const std::size_t idx,
456 std::index_sequence<current_idx, remaining_idcs...> /*seq*/,
457 Args&&... args) {
458 // Check if the current tuple index is matched to the target index
459 if constexpr (concepts::mapped_type_registry<registry_t>) {
460 if (registry_t::mapped_index(idx) == current_idx) {
461 return functor_t{}(types::at<registry_t, current_idx>{},
462 std::forward<Args>(args)...);
463 }
464 } else {
465 if (idx == current_idx) {
466 return functor_t{}(types::at<registry_t, current_idx>{},
467 std::forward<Args>(args)...);
468 }
469 }
470
471 // Check the next index
472 using return_t = std::invoke_result_t<
473 functor_t, const types::front<typename registry_t::type_list>&, Args...>;
474
475 if constexpr (sizeof...(remaining_idcs) >= 1u) {
476 using next_elem_t = types::at<registry_t, current_idx + 1u>;
477 using next_return_t =
478 std::invoke_result_t<functor_t, const next_elem_t&, Args...>;
479 static_assert(
480 std::same_as<return_t, next_return_t>,
481 "Functor return type must be the same for all elements of the "
482 "tuple.");
483
484 return visit_helper<registry_t, functor_t>(
485 idx, std::index_sequence<remaining_idcs...>{},
486 std::forward<Args>(args)...);
487 } else if constexpr (std::is_same_v<return_t, void>) {
488 return;
489 } else {
490 return return_t{};
491 }
492}

501template <concepts::type_registry registry_t, typename functor_t,
502 typename... Args>
503DETRAY_HOST_DEVICE constexpr decltype(auto) visit(const std::size_t idx,
504 Args&&... args) {
505 return visit_helper<registry_t, functor_t>(
506 idx, std::make_index_sequence<registry_t::n_types>{},
507 std::forward<Args>(args)...);
508}

517template <concepts::type_registry registry_t, typename functor_t,
518 typename... Args>
519DETRAY_HOST_DEVICE constexpr decltype(auto) visit(
520 const typename registry_t::id type_id, Args&&... args) {
521 assert(registry_t::is_valid(type_id));
522
523 return visit<registry_t, functor_t>(static_cast<std::size_t>(type_id),
524 std::forward<Args>(args)...);
