100struct transform_payload {
102 std::array<io::scalar, 3u> tr{};
103 // Column major rotation matrix
106 std::array<io::scalar, 9u> rot{};
107
108 private:
109 DETRAY_HOST friend inline std::ostream& operator<<(
110 std::ostream& os, const transform_payload& transform) {
112 const auto& r = transform.rot;
114 os << "rot: ";
115 os << std::fixed << std::setw(4);
116 auto print_line = [&](std::size_t i) {
117 os << r[i] << " " << r[i + 3] << " " << r[i + 6] << "\n";
118 };
119 print_line(0);
120 os << " ";
121 print_line(1);
122 os << " ";
123 print_line(2);
124
126 const auto& t = transform.tr;
127
128 os << "tr: ";
129 os << t[0] << " " << t[1] << " " << t[2];
130 return os;
131 }
132};

277struct axis_payload {
279 axis::binning binning{axis::binning::e_regular};
281 axis::bounds bounds{axis::bounds::e_closed};
283 axis::label label{axis::label::e_r};
284
286 std::size_t bins{0u};
288 std::vector<io::scalar> edges{};
289
290 private:
291 DETRAY_HOST friend inline std::ostream& operator<<(std::ostream& os,
292 const axis_payload& axis) {
293 os << "axis_payload{binning: " << axis.binning
294 << ", bounds: " << axis.bounds << ", label: " << axis.label
295 << ", bins: " << axis.bins << ", edges: [";
296 for (std::size_t i = 0; i < axis.edges.size(); ++i) {
297 if (i > 0) {
298 os << ", ";
299 }
300 os << axis.edges[i];
301 }
302 os << "]}";
303 return os;
304 }
305};
