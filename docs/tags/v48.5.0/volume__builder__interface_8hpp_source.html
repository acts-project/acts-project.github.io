222 DETRAY_HOST std::string_view name() override { return m_builder->name(); }

228 DETRAY_HOST

237 DETRAY_HOST

267 void add_surfaces(
268 std::shared_ptr<surface_factory_interface<detector_t>> sf_factory,

272
273 protected:
278 volume_builder_interface<detector_t> *get_builder() {

288 typename detector_t::transform_container &transforms() override {

293 typename detector_t::mask_container &masks() override {
294 return m_builder->masks();
295 }
296
297
298 private:
