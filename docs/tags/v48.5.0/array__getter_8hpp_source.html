155template <std::size_t I, std::size_t J, concepts::matrix M>
156DETRAY_HOST_DEVICE decltype(auto) element(M &matrix) {
157 if constexpr (concepts::has_compile_time_2d_access<M>) {
158 return matrix.template element<I, J>();
159 } else {
160 using index_t = detray::traits::index_t<std::decay_t<M>>;
161 return element(matrix, static_cast<index_t>(I), static_cast<index_t>(J));
162 }
163}

165template <std::size_t I, concepts::vector V>
166DETRAY_HOST_DEVICE decltype(auto) element(V &vector) {
167 if constexpr (concepts::has_compile_time_1d_access<V>) {
168 return vector.template element<I>();
169 } else {
170 using index_t = detray::traits::index_t<std::decay_t<V>>;
171 return element(vector, static_cast<index_t>(I));
172 }

176struct block_getter {
182 template <std::size_t ROWS, std::size_t COLS, std::size_t oROWS,
183 std::size_t oCOLS, concepts::scalar scalar_t,
184 template <typename, std::size_t> class array_t>
185 DETRAY_HOST_DEVICE constexpr auto operator()(
186 const array_t<array_t<scalar_t, oROWS>, oCOLS> &m, std::size_t row,
187 std::size_t col) const {
188 array_t<array_t<scalar_t, ROWS>, COLS> submatrix{};
189
190 for (std::size_t icol = col; icol < col + COLS; ++icol) {
191 for (std::size_t irow = row; irow < row + ROWS; ++irow) {
192 submatrix[icol - col][irow - row] = m[icol][irow];
193 }
194 }
195 return submatrix;
196 }

203 template <std::size_t SIZE, std::size_t ROWS, std::size_t COLS,
204 concepts::scalar scalar_t,
205 template <typename, std::size_t> class array_t>
206 DETRAY_HOST_DEVICE constexpr array_t<scalar_t, SIZE> vector(
207 const array_t<array_t<scalar_t, ROWS>, COLS> &m, std::size_t row,
208 std::size_t col) const {
209 assert(col < COLS);
210 assert(row + SIZE <= ROWS);
211
212 array_t<scalar_t, SIZE> subvector{};
213
214 for (std::size_t irow = row; irow < row + SIZE; ++irow) {
215 subvector[irow - row] = m[col][irow];
216 }
217
218 return subvector;
219 }

227 template <std::size_t ROWS, std::size_t COLS, class input_matrix_type,
228 concepts::scalar scalar_t,
229 template <typename, std::size_t> class array_t>
230 DETRAY_HOST_DEVICE constexpr void set(
231 input_matrix_type &m, const array_t<array_t<scalar_t, ROWS>, COLS> &b,
232 std::size_t row, std::size_t col) const {
233 for (std::size_t j = 0u; j < COLS; ++j) {
234 for (std::size_t i = 0u; i < ROWS; ++i) {
235 m[j + col][i + row] = b[j][i];
236 }
237 }
238 }
