75 DETRAY_HOST_DEVICE static inline point3_type local_to_global(
76 const transform3_type &trf, const point3_type &p) {
77 const scalar_type r{p[2]};
78 const scalar_type phi{p[0] / r};
79 const scalar_type x{r * math::cos(phi)};
80 const scalar_type y{r * math::sin(phi)};
81 const scalar_type z{p[1]};
82
83 return trf.point_to_global(point3_type{x, y, z});
84 }

92 template <typename mask_t>
93 DETRAY_HOST_DEVICE static inline point3_type local_to_global(

95 const vector3_type & /*dir*/) {
96 return cylindrical2D<algebra_t>::local_to_global(

105 template <typename mask_t>
106 DETRAY_HOST_DEVICE static inline vector3_type normal(
107 const transform3_type &trf, const point2_type &p, const mask_t &mask) {
108 const scalar_type phi{p[0] / mask[mask_t::shape::e_r]};
109
110 // normal vector in global coordinate
111 return trf.vector_to_global(vector2_type{math::cos(phi), math::sin(phi)});
112 }

118 DETRAY_HOST_DEVICE static inline vector3_type normal(
119 const transform3_type &trf, const point3_type &p) {
120 const scalar_type phi{p[0] / p[2]};
121
122 // normal vector in global coordinate
123 return trf.vector_to_global(vector2_type{math::cos(phi), math::sin(phi)});
