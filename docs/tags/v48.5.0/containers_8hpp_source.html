93template <typename T>
94struct has_reserve {
95 private:
98 template <typename C>
99 static constexpr auto check(C*) ->
100 typename std::is_void<decltype(std::declval<C>().reserve(
101 std::declval<typename C::size_type>()))>::type;
102
105 template <typename>
106 static constexpr std::false_type check(...);
107
109 using type = decltype(check<T>(nullptr));
110
111 public:
113 static constexpr bool value = type::value;
114};

119template <typename T>
