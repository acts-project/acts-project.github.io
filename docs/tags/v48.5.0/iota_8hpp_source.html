34template <std::incrementable incr_t>
35class iota_view : public detray::ranges::view_interface<iota_view<incr_t>> {
36 private:
38 struct iterator {
39 using difference_type = std::ptrdiff_t;
40 using value_type = incr_t;
41 using pointer = incr_t *;
42 using reference = incr_t;
43 using iterator_category =
44 std::conditional_t<concepts::random_access_incrementable<incr_t> &&
45 std::totally_ordered<incr_t>,
46 detray::ranges::random_access_iterator_tag,
47 detray::ranges::bidirectional_iterator_tag>;
48
49 constexpr iterator()
50 requires std::default_initializable<incr_t>
51 = default;
52
54 DETRAY_HOST_DEVICE constexpr explicit iterator(incr_t i) : m_i{i} {}
55
58 DETRAY_HOST_DEVICE
59 constexpr auto operator++() -> iterator & {
60 ++m_i;
61 return *this;
62 }
63
64 DETRAY_HOST_DEVICE constexpr auto operator++(int) -> iterator {
65 auto tmp(*this);
66 ++(*this);
67 return tmp;
68 }
69 DETRAY_HOST_DEVICE
70 constexpr auto operator--() -> iterator & {
71 --m_i;
72 return *this;
73 }
74
75 DETRAY_HOST_DEVICE constexpr auto operator--(int) -> iterator {
76 auto tmp(*this);
77 --(*this);
78 return tmp;
79 }
80
81 DETRAY_HOST_DEVICE constexpr iterator &operator+=(const difference_type n)
82 requires concepts::random_access_incrementable<incr_t>
83 {
84 difference_type nv = static_cast<difference_type>(m_i) + n;
85 // Check if the round trip between difference type and incrementor
86 // type is valid.
87 assert(static_cast<difference_type>(static_cast<incr_t>(nv)) == nv);
88 m_i = static_cast<incr_t>(nv);
89 return *this;
90 }
91
92 DETRAY_HOST_DEVICE friend constexpr iterator operator+(
93 const iterator &i, const difference_type n)
94 requires concepts::random_access_incrementable<incr_t>
95 {
96 return iterator(i.m_i + n);
97 }
98
99 DETRAY_HOST_DEVICE friend constexpr iterator operator+(
100 const difference_type n, const iterator &i)
101 requires concepts::random_access_incrementable<incr_t>
102 {
103 return iterator(i.m_i + n);
104 }
105
106 DETRAY_HOST_DEVICE constexpr iterator &operator-=(const difference_type n)
107 requires concepts::random_access_incrementable<incr_t>
108 {
109 m_i -= n;
110 return *this;
111 }
112
113 DETRAY_HOST_DEVICE friend constexpr iterator operator-(
114 const iterator &i, const difference_type n)
115 requires concepts::random_access_incrementable<incr_t>
116 {
117 return iterator(i.m_i - n);
118 }
119
120 DETRAY_HOST_DEVICE friend constexpr difference_type operator-(
121 const iterator &i, const iterator &j)
122 requires concepts::random_access_incrementable<incr_t>
123 {
124 return i.m_i - j.m_i;
125 }
127
129 DETRAY_HOST_DEVICE
130 constexpr auto operator*() const
131 noexcept(std::is_nothrow_copy_constructible_v<incr_t>) -> incr_t {
132 return m_i;
133 }
134
135 DETRAY_HOST_DEVICE
136 constexpr incr_t operator[](const difference_type n) const
137 requires concepts::random_access_incrementable<incr_t>
138 {
139 return static_cast<incr_t>(m_i + n);
140 }
141
142 private:
143 DETRAY_HOST_DEVICE
144 friend constexpr auto operator<=>(const iterator &lhs, const iterator &rhs)
145 requires std::totally_ordered<incr_t>
146 = default;
147
149 incr_t m_i{};
150 };
151
153 incr_t m_start;
154 incr_t m_end;
155
156 public:
158 using iterator_t = iterator;
159
162 constexpr iota_view()
163 requires std::default_initializable<incr_t>
164 = default;
165
168 template <concepts::interval interval_t>
169 requires(!std::same_as<interval_t, iota_view>)
170 DETRAY_HOST_DEVICE constexpr explicit iota_view(interval_t &&interval)
171 : m_start{detray::detail::get<0>(interval)},
172 m_end{detray::detail::get<1>(interval)} {
173 assert(m_start <= m_end);
174 }
175
179 DETRAY_HOST_DEVICE constexpr iota_view(incr_t start, incr_t end)
180 : m_start{start}, m_end{end} {
181 assert(m_start <= m_end);
182 }
183
186 DETRAY_HOST_DEVICE
187 constexpr explicit iota_view(incr_t start)
188 : m_start{start}, m_end{start + 1} {
189 assert(m_start <= m_end);
190 }
191
193 DETRAY_HOST_DEVICE
194 constexpr auto begin() const -> iterator_t { return iterator_t{m_start}; }
195
197 DETRAY_HOST_DEVICE
198 constexpr auto end() const -> iterator_t { return iterator_t{m_end}; }
199
201 DETRAY_HOST_DEVICE
202 constexpr auto size() const -> incr_t { return m_end - m_start; }
203};
