103 template <detray::concepts::detector detector_t>
104 static std::vector<Entry> collectEntries(
105 const detector_t& detector,
106 const Acts::TrackingGeometry& trackingGeometry) {
107 std::vector<Entry> entries;
108 for (const auto& surface : detector.surfaces()) {
109 if (!surface.is_sensitive()) {
110 continue;
111 }
112 const Acts::GeometryIdentifier id(surface.source);
113 const Acts::Surface* actsSurface = trackingGeometry.findSurface(id);
114 if (actsSurface == nullptr) {
115 throw std::invalid_argument(
116 std::format("DetraySurfaceIdMap: no ACTS surface with id {} for "
117 "detray surface {}",
118 id, surface.identifier().value()));
119 }
120 entries.emplace_back(surface.identifier(), actsSurface);
121 }
122 return entries;
123 }
