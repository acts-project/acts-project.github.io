151template <std::size_t I, std::size_t J, concepts::matrix M>
152DETRAY_HOST_DEVICE decltype(auto) element(M &matrix) {
153 if constexpr (concepts::has_compile_time_2d_access<M>) {
154 return matrix.template element<I, J>();
155 } else {
156 using index_t = detray::traits::index_t<std::decay_t<M>>;
157 return element(matrix, static_cast<index_t>(I), static_cast<index_t>(J));
158 }
159}
160
161template <std::size_t I, concepts::vector V>
162DETRAY_HOST_DEVICE decltype(auto) element(V &vector) {
163 if constexpr (concepts::has_compile_time_1d_access<V>) {
164 return vector.template element<I>();
165 } else {
166 using index_t = detray::traits::index_t<std::decay_t<V>>;
167 return element(vector, static_cast<index_t>(I));
168 }

172struct block_getter {
178 template <std::size_t ROWS, std::size_t COLS, std::size_t oROWS,
179 std::size_t oCOLS, concepts::scalar scalar_t>
180 DETRAY_HOST_DEVICE algebra::fastor::Matrix<scalar_t, ROWS, COLS> operator()(
181 const Fastor::Tensor<scalar_t, oROWS, oCOLS> &m, std::size_t row,
182 std::size_t col) const {
183 return m(Fastor::seq(static_cast<int>(row), static_cast<int>(row + ROWS)),
184 Fastor::seq(static_cast<int>(col), static_cast<int>(col + COLS)));
185 }
186
192 template <std::size_t SIZE, std::size_t oROWS, std::size_t oCOLS,
193 concepts::scalar scalar_t>
194 DETRAY_HOST_DEVICE Fastor::Tensor<scalar_t, SIZE> vector(
195 const Fastor::Tensor<scalar_t, oROWS, oCOLS> &m, std::size_t row,
196 std::size_t col) const {
197 return Fastor::Tensor<scalar_t, SIZE>(
198 m(Fastor::seq(static_cast<int>(row), static_cast<int>(row + SIZE)),
199 static_cast<int>(col)));
200 }
201
207 template <std::size_t ROWS, std::size_t COLS, concepts::scalar scalar_t,
208 concepts::matrix input_matrix_type>
209 DETRAY_HOST_DEVICE void set(input_matrix_type &m,
210 const Fastor::Tensor<scalar_t, ROWS, COLS> &b,
211 std::size_t row, std::size_t col) const {
212 m(Fastor::seq(static_cast<int>(row), static_cast<int>(row + ROWS)),
213 Fastor::seq(static_cast<int>(col), static_cast<int>(col + COLS))) = b;
214 }
215
221 template <std::size_t ROWS, concepts::scalar scalar_t,
222 concepts::matrix input_matrix_type>
223 DETRAY_HOST_DEVICE void set(input_matrix_type &m,
224 const Fastor::Tensor<scalar_t, ROWS> &b,
225 std::size_t row, std::size_t col) const {
226 m(Fastor::seq(static_cast<int>(row), static_cast<int>(row + ROWS)),
227 static_cast<int>(col)) = b;
228 }
229
230}; // struct block_getter
