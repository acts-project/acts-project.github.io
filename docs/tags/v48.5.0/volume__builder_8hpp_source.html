51 explicit volume_builder(const volume_id id, const dindex idx = 0)

114 DETRAY_HOST
115 auto build(detector_t& det, const volume_builder_options& opt,
116 typename detector_t::geometry_context ctx = {}) ->
117 typename detector_t::volume_type* override {
118 DETRAY_VERBOSE_HOST("Build surfaces...");
119
120 assert(!m_surfaces.empty());
121 assert(!m_transforms.empty());
122 assert(!m_masks.all_empty());
123
124 // Prepare volume data
125 m_volume.set_index(static_cast<dindex>(det.volumes().size()));
126
127 m_volume.set_transform(det.transform_store().size());
128 det._transforms.push_back(m_trf);
129
130 // Add all data from the builder to the detector containers
131 add_to_detector(opt, ctx, det);
132
133 // Reset after the data was added to the detector
134 m_surfaces.clear();
135 m_transforms.clear(ctx);
136 m_masks.clear_all();
137
138 DETRAY_VERBOSE_HOST("Successfully built "
139 << m_volume.n_surfaces()
140 << " surfaces for volume: " << this->name());
141
142 // Pass to decorator builders
143 return &(det._volumes.back());
144 }

208 template <geo_obj_ids surface_id = static_cast<geo_obj_ids>(0)>
209 DETRAY_HOST auto add_to_detector(
210 const volume_builder_options& opt,
211 const typename detector_t::geometry_context ctx,
212 detector_t& det) noexcept(false) -> void {
213 if (opt.deduplicate()) {
214 DETRAY_WARN_HOST(m_volume_name
215 << ": Geometry data deduplication not yet implemented");
216 }
217
218 // Append transforms
219 const auto trf_offset = det.transform_store().size(ctx);
220 det._transforms.append(std::move(m_transforms), ctx);
221
222 // Surface index offset in the global detector container
223 auto sf_offset{static_cast<dindex>(det.surfaces().size())};
224
226 auto find_range = [&](auto sf_id) {
227 // Compare given id to surface identifier
228 auto is_sf_type = [sf_id](const auto& sf) { return sf.id() == sf_id; };
229
230 auto first = static_cast<dindex>(
231 math::abs(std::ranges::find_if(m_surfaces, is_sf_type) -
232 std::begin(m_surfaces)));
233
234 auto last = static_cast<dindex>(math::abs(
235 std::ranges::rend(m_surfaces) -
236 std::ranges::find_if(std::ranges::rbegin(m_surfaces),
237 std::ranges::rend(m_surfaces), is_sf_type)));
238
239 // Set correct empty range, otherwise shift by global surface offset
240 return (first >= last)
241 ? dindex_range{}
242 : dindex_range{first + sf_offset, last + sf_offset};
243 };
244
245 m_volume.template update_sf_link<surface_id::e_portal>(
246 find_range(surface_id::e_portal));
247
248 m_volume.template update_sf_link<surface_id::e_sensitive>(
249 find_range(surface_id::e_sensitive));
250
251 m_volume.template update_sf_link<surface_id::e_passive>(
252 find_range(surface_id::e_passive));
253
254 // Update mask and transform index of surfaces and set the
255 // correct index of the surface in container
256 std::size_t n_portals{0u};
257 for (auto& sf_desc : m_surfaces) {
258 det._masks.template visit<detail::mask_index_update>(sf_desc.mask(),
259 sf_desc);
260 sf_desc.set_volume(m_volume.index());
261 sf_desc.update_transform(trf_offset);
262 sf_desc.set_index(sf_offset++);
263
264 if (sf_desc.is_portal()) {
265 ++n_portals;
266 }
267
268 det._surfaces.insert(sf_desc);
269 }
270
271 // Place the appropriate surfaces in the brute force search method.
272 constexpr auto default_acc_id{detector_t::accel::id::e_surface_default};
273
274 // Strip the source link from the lookup data structure
275 typename detector_t::surface_container descriptors;
276 descriptors.reserve(m_surfaces.size());
277 std::ranges::transform(
278 m_surfaces, std::back_inserter(descriptors),
279 [](typename detector_t::surface_lookup_container::value_type& sf) {
280 return static_cast<
281 typename detector_t::surface_container::value_type>(sf);
282 });
283
284 // Add portals to brute force navigation method
285 if (m_has_accel) {
286 DETRAY_VERBOSE_HOST("-> Volume has acceleration structure:");
287
288 typename detector_t::surface_container portals{};
289 portals.reserve(n_portals);
290
291 std::ranges::copy_if(
292 descriptors, std::back_inserter(portals),
293 [](auto& sf_desc) { return !sf_desc.is_sensitive(); });
294
295 // Add only the portals to the brute force method
296 DETRAY_VERBOSE_HOST(
297 "-> Register only portals/passives with brute force "
298 "accelerator");
299
300 det._accelerators.template push_back<default_acc_id>(std::move(portals));
301 } else {
302 DETRAY_VERBOSE_HOST(
303 "-> Register all surfaces with brute force accelerator");
304
305 // Add all surfaces to the brute force method
306 det._accelerators.template push_back<default_acc_id>(
307 std::move(descriptors));
308 }
309
310 m_volume.template set_accel_link<surface_id>(
311 default_acc_id,
312 det.accelerator_store().template size<default_acc_id>() - 1u);
313
314 // Append masks
315 det._masks.append(std::move(m_masks));
316
317 // Finally, add the volume descriptor to the detector
318 det._volumes.push_back(m_volume);
319 }
320
321 private:
323 bool m_has_accel{false};
324
326 std::string m_volume_name{};
327
329 typename detector_t::volume_type m_volume{};
331 typename detector_t::transform3_type m_trf{};
332
335 typename detector_t::surface_lookup_container m_surfaces{};
336 typename detector_t::transform_container m_transforms{};
337 typename detector_t::mask_container m_masks{};
339};
