148 template <typename view_t>
149 inline auto draw_surface(
150 const dindex index, const view_t& view,
151 const typename detector_t::geometry_context& gctx = {}) const {
152 if (index >= _detector.surfaces().size()) {
153 throw std::invalid_argument("Surface index too large: " +
154 std::to_string(index));
155 }
156
157 const auto surface = detray::geometry::surface{_detector, index};
158
159 DETRAY_VERBOSE_HOST("Draw surface: " << surface);
160
161 actsvg::svg::object ret;
162 actsvg::svg::object material;
163 const auto& style = _style._detector_style._volume_style;
164
165 if (surface.is_portal()) {
166 auto p_portals = svgtools::conversion::portal(gctx, _detector, surface,
167 view, style._portal_style,
168 false, _hide_material);
169
170 for (auto& p_portal : p_portals) {
171 std::string id = p_portal._name + "_" + svg_id(view);
172
173 DETRAY_DEBUG_HOST("-> Portal SVG ID: " << id);
174
175 ret.add_object(actsvg::display::portal(std::move(id), p_portal, view));
176 }
177
178 if (!_hide_material && surface.has_material()) {
179 std::string id_0 = p_portals.at(0)._name + "_" + svg_id(view);
180 material = actsvg::display::surface_material(
181 id_0 + "_material_map", p_portals.at(0)._surface._material);
182 }
183 } else {
184 const auto& sf_style = surface.is_sensitive()
185 ? style._sensitive_surface_style
186 : style._passive_surface_style;
187
188 auto p_surfaces = svgtools::conversion::surface(
189 gctx, _detector, surface, view, sf_style, _hide_material);
190
191 // Draw the surfaces directly
192 for (auto& p_surface : p_surfaces) {
193 std::string id = p_surface._name + "_" + svg_id(view);
194
195 DETRAY_DEBUG_HOST("-> Surface SVG ID: " << id);
196
197 ret.add_object(
198 actsvg::display::surface(std::move(id), p_surface, view));
199 }
200
201 // The surface links to only one material for all masks
202 if (!_hide_material && surface.has_material()) {
203 std::string id_0 = p_surfaces.at(0)._name + "_" + svg_id(view);
204 material = actsvg::display::surface_material(
205 id_0 + "_material_map", p_surfaces.at(0)._material);
206 }
207 }
208 // Add an optional info box
209 if (_show_info) {
210 auto p_information_section =
211 svgtools::conversion::information_section(gctx, surface);
212
213 auto info_box = svgtools::meta::display::information_section(
214 ret._id + "_info_box", p_information_section, view,
215 _info_screen_offset, ret);
216 ret.add_object(info_box);
217 }
218
219 return std::tuple{ret, material};
220 }

230 template <detray::ranges::range range_t, typename view_t>
231 inline auto draw_surfaces(
232 const range_t& indices, const view_t& view,
233 const typename detector_t::geometry_context& gctx = {}) const {
234 DETRAY_VERBOSE_HOST("Draw surfaces...");
235
236 auto ret = svgtools::utils::group(det_name() + "_surfaces_" + svg_id(view));
237
238 auto material =
239 svgtools::utils::group(det_name() + "_material_" + svg_id(view));
240
241 for (const auto [i, index] : detray::views::enumerate(indices)) {
242 auto [sf_svg, mat_svg] = draw_surface(index, view, gctx);
243
244 ret.add_object(sf_svg);
245
246 // Material is optional
247 if (mat_svg.is_defined()) {
248 // Only add one gradient box
249 if (i == 0) {
250 material.add_object(mat_svg);
251 } else {
252 material.add_object(mat_svg._sub_objects.at(0));
253 }
254 }
255 }
256
257 return std::tuple{ret, material};
258 }

265 template <typename view_t>
266 inline auto draw_surface_material(const dindex index,
267 const view_t& view) const {
268 if (index >= _detector.surfaces().size()) {
269 throw std::invalid_argument("Surface index too large: " +
270 std::to_string(index));
271 }
272
273 const auto surface = detray::geometry::surface{_detector, index};
274
275 if (_hide_material) {
276 return actsvg::svg::object{};
277 }
278
279 DETRAY_VERBOSE_HOST("Draw material for surface: " << surface);
280
281 const styling::surface_material_style* mat_style{nullptr};
282 const auto& vol_style = _style._detector_style._volume_style;
283 switch (surface.id()) {
284 using enum surface_id;
285 case e_portal: {
286 mat_style = &vol_style._portal_style._surface_style._material_style;
287 break;
288 }
289 case e_sensitive: {
290 mat_style = &vol_style._sensitive_surface_style._material_style;
291 break;
292 }
293 case e_passive: {
294 mat_style = &vol_style._passive_surface_style._material_style;
295 break;
296 }
297 case e_unknown: {
298 throw std::runtime_error("Encountered surface of unknown type.");
299 break;
300 }
301 }
302
303 auto p_material = svgtools::conversion::surface_material(_detector, surface,
304 view, *mat_style);
305
306 std::string id = det_name() + "_material_map_" +
307 std::to_string(surface.index()) + svg_id(view);
308
309 DETRAY_DEBUG_HOST("-> Surface Material SVG ID: " << id);
310
311 return actsvg::display::surface_material(id, p_material);
312 }

321 template <detray::ranges::range range_t, typename view_t>
322 inline auto draw_surface_materials(const range_t& indices,

329 svg_id(view));

336 ret.add_object(mat_svg);
337 } else {
338 ret.add_object(mat_svg._sub_objects.at(0));
339 }
340 }
341
342 return ret;
343 }

352 template <typename view_t>
353 inline auto draw_volume(
354 const dindex index, const view_t& view,
355 const typename detector_t::geometry_context& gctx = {}) const {
356 if (index >= _detector.volumes().size()) {
357 std::string err_str{"Volume index too large: " + std::to_string(index)};
358 DETRAY_FATAL_HOST(err_str);
359 throw std::invalid_argument(err_str);
360 }
361
362 const auto d_volume = tracking_volume{_detector, index};
363
364 DETRAY_VERBOSE_HOST("Draw volume " << "(" << svg_id(view) << "-view): "
365 << d_volume.name(_name_map));
366 DETRAY_DEBUG_HOST("-> " << d_volume);
367
368 auto [p_volume, gr_type] = svgtools::conversion::volume(
369 gctx, _detector, d_volume, view, _style._detector_style._volume_style,
370 _hide_portals, _hide_passives, _hide_grids, _hide_material,
371 _search_window);
372
373 // Draw the basic volume
374 p_volume._name = d_volume.name(_name_map);
375 p_volume._name = std::regex_replace(p_volume._name, std::regex("/"), "_");
376 std::string id = p_volume._name + "_" + svg_id(view);
377 auto vol_svg = svgtools::utils::group(id);
378
379 actsvg::svg::object sf_grid_svg;
380
381 // Add the grid to the volume svg
382 auto grid_svg = actsvg::display::grid(id + "_grid", p_volume._surface_grid);
383
384 // zr and xy - views of volume including the portals
385 if constexpr (!std::is_same_v<view_t, actsvg::views::z_phi> &&
386 !std::is_same_v<view_t, actsvg::views::z_rphi>) {
387 vol_svg.add_object(actsvg::display::volume(id, p_volume, view));
388
389 if (!_hide_grids) {
390 DETRAY_VERBOSE_HOST("-> Added grid to volume SVG");
391 vol_svg.add_object(grid_svg);
392 }
393 }
394
395 if (!_hide_grids) {
396 // Draw surface grid svg, if volume has grid and the view fits
397 // the grid
398 sf_grid_svg = detray::svgtools::meta::display::surface_grid(
399 id, p_volume, gr_type, grid_svg, view);
400 }
401
402 DETRAY_DEBUG_HOST("-> Volume SVG ID: " << id);
403
404 return std::tuple{vol_svg, sf_grid_svg};
405 }

414 template <typename view_t>
415 inline auto draw_volume(
416 const std::string_view name, const view_t& view,
417 const typename detector_t::geometry_context& gctx = {}) const {
418 if (_name_map.empty() || !_name_map.contains(name)) {
419 std::string err_str{"Could not identify volume by name '" +
420 std::string{name} + "'"};
421 if (_name_map.empty()) {
422 err_str += ": name map empty!";
423 } else if (!_name_map.contains(name)) {
424 err_str += ": name map does not contain voume name!";
425 }
426
427 DETRAY_FATAL_HOST(err_str);
428 throw std::invalid_argument(err_str);
429 }
430
431 dindex index = _detector.volume(name, _name_map).index();
432
433 return draw_volume(index, view, gctx);
434 }

444 template <detray::ranges::range range_t, typename view_t>
445 inline auto draw_volumes(
446 const range_t& volume_ids, const view_t& view,
447 const typename detector_t::geometry_context& gctx = {}) const {
448 DETRAY_VERBOSE_HOST("Draw volumes...");
449
450 // Overlay the volume svgs
451 auto vol_group =
452 svgtools::utils::group(det_name() + "_volumes_" + svg_id(view));
453
454 // The surface[grid] sheets per volume
455 std::vector<actsvg::svg::object> sf_grids;
456
457 for (const auto& vol_identifier : volume_ids) {
458 auto [vol_svg, sf_grid] = draw_volume(vol_identifier, view, gctx);
459
460 // The general volume display
461 if constexpr (!std::is_same_v<view_t, actsvg::views::z_phi> &&
462 !std::is_same_v<view_t, actsvg::views::z_rphi>) {
463 vol_group.add_object(vol_svg);
464 }
465 sf_grids.push_back(std::move(sf_grid));
466 }
467
468 return std::tuple(vol_group, std::move(sf_grids));
469 }

479 template <typename view_t>
480 inline auto draw_detector(
481 const view_t& view,
482 const typename detector_t::geometry_context& gctx = {},
483 const float r_length = 1100.f, const float z_length = 3100.f) const {
484 DETRAY_VERBOSE_HOST("Draw detector...");
485
486 auto p_detector = svgtools::conversion::detector(
487 gctx, _detector, view, _style._detector_style, _hide_portals,
488 _hide_passives, _hide_grids);
489
490 std::string id = det_name() + "_" + svg_id(view);
491
492 auto det_svg = actsvg::display::detector(std::move(id), p_detector, view);
493
494 if constexpr (std::is_same_v<view_t, actsvg::views::z_r>) {
495 if (!_hide_eta_lines) {
496 DETRAY_VERBOSE_HOST("Drawing eta lines");
497
498 auto p_eta_lines = svgtools::meta::proto::eta_lines{};
499
500 svgtools::styling::apply_style(p_eta_lines, _style._eta_lines_style);
501
502 // Hardcoded until we find a way to scale axes automatically
503 p_eta_lines._r = r_length;
504 p_eta_lines._z = z_length;
505
506 det_svg.add_object(
507 svgtools::meta::display::eta_lines("eta_lines_", p_eta_lines));
508 }
509 }
510
511 DETRAY_VERBOSE_HOST("Detector SVG ID: " << id);
512 DETRAY_INFO_HOST(
513 "Finished drawing detector: " << _detector.name(_name_map));
514
515 return det_svg;
516 }

610 template <typename view_t, class trajectory_t, typename intersection_t>
611 inline auto draw_intersections_and_trajectory(
612 const std::string& prefix,
613 const std::vector<intersection_t>& intersections,
614 const trajectory_t& trajectory, const view_t& view,
615 typename detector_t::scalar_type max_path = 500.f,
616 const dindex_range highlight_idx = {detail::invalid_value<dindex>(),
617 detail::invalid_value<dindex>()},
618 const typename detector_t::geometry_context& gctx = {}) const {
619 DETRAY_VERBOSE_HOST("Draw intersections and trajectory...");
620
621 actsvg::svg::object ret;
622 ret._tag = "g";
623 ret._id = prefix;
624
625 if (!intersections.empty()) {
626 // Draw intersections with landmark style
627 auto p_ir = svgtools::conversion::intersection(
628 _detector, intersections, trajectory.dir(0.f), gctx, highlight_idx,
629 _style._landmark_style);
630
631 ret.add_object(svgtools::meta::display::intersection(prefix + "_record",
632 p_ir, view));
633
634 // The intersection record is always sorted by path length
635 const auto sf_back{
636 detray::geometry::surface{_detector, intersections.back().surface()}};
637 const auto sf_front{detray::geometry::surface{
638 _detector, intersections.front().surface()}};
639
640 const auto pos_back = sf_back.local_to_global(
641 gctx, intersections.back().local(), trajectory.dir(0.f));
642 const auto pos_front = sf_front.local_to_global(
643 gctx, intersections.front().local(), trajectory.dir(0.f));
644
645 max_path = vector::norm(pos_back - trajectory.pos(0.f));
646 max_path =
647 math::max(max_path, vector::norm(pos_front - trajectory.pos(0.f)));
648 }
649
650 ret.add_object(
651 draw_trajectory(prefix + "_trajectory", trajectory, max_path, view));
652 return ret;
653 }
