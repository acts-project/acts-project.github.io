324 DETRAY_HOST bool self_check(std::ostream &os) const {
325 if (identifier().is_invalid()) {
326 os << "DETRAY ERROR (HOST): Invalid identifier for surface:\n"
327 << *this << std::endl;
328 return false;
329 }
330 if (index() >= m_detector.surfaces().size()) {
331 os << "DETRAY ERROR (HOST): Surface index out of bounds for "
332 "surface:\n"
333 << *this << std::endl;
334 return false;
335 }
336 if (volume() >= m_detector.volumes().size()) {
337 os << "DETRAY ERROR (HOST): Surface volume index out of bounds for "
338 "surface:\n"
339 << *this << std::endl;
340 return false;
341 }
342 if (detail::is_invalid_value(m_desc.transform())) {
343 os << "DETRAY ERROR (HOST): Surface transform undefined for "
344 "surface:\n"
345 << *this << std::endl;
346 return false;
347 }
348 if (m_desc.transform() >= m_detector.transform_store().size()) {
349 os << "DETRAY ERROR (HOST): Surface transform index out of bounds "
350 "for surface:\n"
351 << *this << std::endl;
352 return false;
353 }
354 if (detail::is_invalid_value(m_desc.mask())) {
355 os << "DETRAY ERROR (HOST): Surface does not have a valid mask "
356 "link:\n"
357 << *this << std::endl;
358 return false;
359 }
360 // Only check, if there is material in the detector
361 if (!m_detector.material_store().all_empty() && has_material() &&
362 m_desc.material().is_invalid_index()) {
363 os << "DETRAY ERROR (HOST): Surface does not have valid material "
364 "link:\n"
365 << *this << std::endl;
366 return false;
367 }
368 // Check the mask boundaries
369 if (!visit_mask<typename kernels::mask_self_check>(os)) {
370 os << "\nSurface: " << *this << std::endl;
371 return false;
372 }
373
374 // Check the mask volume link
375 const auto vol_links = visit_mask<typename kernels::get_volume_links>();
376 for (const auto vol_link : vol_links) {
377 if (is_portal()) {
378 if (vol_link == volume()) {
379 os << "DETRAY ERROR (HOST): Portal surface links to mother "
380 "volume:\n"
381 << *this << std::endl;
382 return false;
383 }
384 } else if (vol_link != volume()) {
385 os << "DETRAY ERROR (HOST): Passive/sensitive surface does not "
386 "link to mother volume: Mask volume link : "
387 << vol_link << "\n"
388 << *this << std::endl;
389 return false;
390 }
391 }
392
393 return true;
394 }
