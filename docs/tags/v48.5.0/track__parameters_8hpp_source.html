76template <detray::concepts::detector detector_t>
77inline auto read_free_track_params(const std::string &file_name) {
78 using algebra_t = typename detector_t::algebra_type;
79 using scalar_t = dscalar<algebra_t>;
80 using point3_t = dpoint3D<algebra_t>;
81 using vector3_t = dvector3D<algebra_t>;
82 using track_t = detray::free_track_parameters<algebra_t>;
83
84 dfe::NamedTupleCsvReader<io::csv::free_track_parameters> track_param_reader(
85 file_name);
86
87 io::csv::free_track_parameters track_param_data{};
88 std::vector<std::vector<std::pair<scalar_t, track_t>>> track_params_per_track;
89
90 while (track_param_reader.read(track_param_data)) {
91 // Add new track state to correct track
92 dindex trk_index{track_param_data.track_id};
93 while (track_params_per_track.size() <= trk_index) {
94 track_params_per_track.emplace_back();
95 }
96
97 point3_t pos{static_cast<scalar_t>(track_param_data.x),
98 static_cast<scalar_t>(track_param_data.y),
99 static_cast<scalar_t>(track_param_data.z)};
100 vector3_t p{static_cast<scalar_t>(track_param_data.px),
101 static_cast<scalar_t>(track_param_data.py),
102 static_cast<scalar_t>(track_param_data.pz)};
103
104 track_t track_param{pos, static_cast<scalar_t>(track_param_data.t), p,
105 static_cast<scalar_t>(track_param_data.q)};
106
107 // Add to collection
108 track_params_per_track[trk_index].emplace_back(track_param_data.q,
109 track_param);
110 }
111
112 // Check the result
113 if (track_params_per_track.empty()) {
114 throw std::invalid_argument(
115 "ERROR: csv reader: Failed to read free track parameters data");
116 }
117
118 return track_params_per_track;
119}

122template <detray::concepts::scalar scalar_t, typename track_t>
123inline void write_free_track_params(
124 const std::string &file_name,
125 const std::vector<std::vector<std::pair<scalar_t, track_t>>>
126 &track_params_per_track,
127 const bool replace = true) {
128 // Don't write over existing data
129 std::string trk_file_name{file_name};
130 if (!replace && io::file_exists(file_name)) {
131 trk_file_name = io::alt_file_name(file_name);
132 } else {
133 // Make sure the output directories exit
134 io::create_path(std::filesystem::path{trk_file_name}.parent_path());
135 }
136
137 dfe::NamedTupleCsvWriter<io::csv::free_track_parameters> track_param_writer(
138 trk_file_name);
139
140 for (const auto &[track_idx, track_params] :
141 detray::views::enumerate(track_params_per_track)) {
142 // Skip empty traces
143 if (track_params.empty()) {
144 continue;
145 }
146
147 for (const auto &[charge, track_param] : track_params) {
148 const auto &glob_pos = track_param.pos();
149 // Momentum may not be retrievable for straight-line tracks
150 const auto &p{charge != 0.f ? track_param.mom(charge)
151 : track_param.dir()};
152
153 io::csv::free_track_parameters track_param_data{};
154 track_param_data.track_id = track_idx;
155 track_param_data.x = glob_pos[0];
156 track_param_data.y = glob_pos[1];
157 track_param_data.z = glob_pos[2];
158 track_param_data.t = track_param.time();
159 track_param_data.px = p[0];
160 track_param_data.py = p[1];
161 track_param_data.pz = p[2];
162 track_param_data.q = charge;
163
164 track_param_writer.append(track_param_data);
165 }
166 }
167}
