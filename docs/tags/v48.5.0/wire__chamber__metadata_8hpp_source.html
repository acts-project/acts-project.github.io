137 template <typename container_t = host_container_types>
139 using material_store =
140 multi_store<material_id, empty_context, dtuple,
141 grid_collection<concentric_cylinder2D_map_t<container_t>>,
142 grid_collection<ring2D_map_t<container_t>>,
143 typename container_t::template vector_type<raw_material_t>,
144 typename container_t::template vector_type<material_rod_t>>;
145
147 using transform_link = typename transform_store<>::single_link;
149 using mask_link = typename mask_store<>::range_link;

153 using surface_type =
154 surface_descriptor<mask_link, material_link, transform_link, nav_link>;
155
156 template <typename axes_t, typename bin_entry_t, typename container_t>
158 using dynamic_simple_grid_t =
159 spatial_grid<algebra_type, axes_t,
160 detray::bins::dynamic_array<bin_entry_t>,
161 detray::simple_serializer, container_t, false>;
162
163 template <typename container_t>
165 using surface_concentric_cylinder2D_grid_t =
166 dynamic_simple_grid_t<axes<detray::concentric_cylinder2D>, surface_type,
167 container_t>;
168 template <typename container_t>
170 using volume_cylinder3D_grid_t =
171 dynamic_simple_grid_t<axes<detray::cylinder3D>, dindex, container_t>;
