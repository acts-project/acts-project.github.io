322template <typename point3_container_t>
323inline void apply_style(actsvg::proto::surface<point3_container_t>& p_surface,
324 const surface_style& styling) {
325 // Fill color
326 p_surface._fill = actsvg::style::fill(styling._fill_color);
327 p_surface._fill._fc._hl_rgb = styling._highlight_rgb;
328 p_surface._fill._fc._highlight = styling._highlights;
329
330 // Stroke
331 p_surface._stroke =
332 actsvg::style::stroke(styling._stroke_color, styling._stroke_width);
333 p_surface._stroke._sc._hl_rgb = styling._highlight_stroke_rgb;
334 p_surface._stroke._hl_width = styling._highlight_stroke_width;
335}

338template <typename point3_container_t>
339inline void apply_style(
340 typename actsvg::proto::portal<point3_container_t>::link& p_link,
341 const link_style& styling) {
342 p_link._end_marker._size = styling._marker_size;
343}
344
346template <typename point3_container_t>
347inline void apply_style(actsvg::proto::portal<point3_container_t>& p_portal,
348 const portal_style& styling) {
349 apply_style(p_portal._surface, styling._surface_style);
350
351 for (auto& volume_link : p_portal._volume_links) {
352 apply_style<point3_container_t>(volume_link, styling._link_style);
353 }
354}

357inline void apply_style(actsvg::proto::grid& p_grid,
358 const grid_style& styling) {
359 p_grid._stroke._sc = styling._stroke_color;
360 p_grid._stroke._width = styling._stroke_width;
361 // Use dashed lines for the grid
362 // p_grid._stroke._dasharray = {4};
363}

366inline void apply_style(actsvg::proto::surface_material& p_material,
367 const surface_material_style& styling) {
368 p_material._info_pos = styling._info_pos;
369
370 p_material._gradient_pos = styling._gradient_pos;
371 p_material._gradient_box = styling._gradient_box;
372 p_material._gradient_font = styling._gradient_font;
373 p_material._gradient_label = styling._gradient_label;
374 p_material._gradient_stroke._sc = styling._gradient_stroke_color;
375 p_material._gradient_stroke._width = styling._gradient_stroke_width;
376 p_material._gradient._stops = colors::gradient::generate_stops(
377 styling._gradient_color_scale, styling._gradient_n_stops);
378
379 apply_style(p_material._grid, styling._grid_style);
380}

383template <typename point3_container_t>
384inline void apply_style(actsvg::proto::volume<point3_container_t>& p_volume,
385 const volume_style& styling) {
386 using proto_sf_t = actsvg::proto::surface<point3_container_t>;
387
388 for (auto& p_surface : p_volume._v_surfaces) {
389 const auto& sf_style =
390 (p_surface._sf_type == proto_sf_t::sf_type::e_sensitive)
391 ? styling._sensitive_surface_style
392 : styling._passive_surface_style;
393 apply_style(p_surface, sf_style);
394 }
395 for (auto& p_portals : p_volume._portals) {
396 apply_style(p_portals, styling._portal_style);
397 }
398 apply_style(p_volume._surface_grid, styling._grid_style);
399}

402template <typename point3_container_t>
403inline void apply_style(actsvg::proto::detector<point3_container_t>& p_detector,
404 const detector_style& styling) {
405 for (auto& p_volume : p_detector._volumes) {
406 apply_style(p_volume, styling._volume_style);

411inline void apply_style(meta::proto::eta_lines& p_eta_lines,
412 const eta_lines_style& styling) {
413 p_eta_lines._stroke_main = actsvg::style::stroke(styling._fill_color_main,

423template <concepts::point3D point3_t>
424inline void apply_style(meta::proto::landmark<point3_t>& p_landmark,
425 const landmark_style& styling) {
426 const auto fill = actsvg::style::fill(styling._fill_color);
427 const auto stroke =
428 actsvg::style::stroke(styling._fill_color, styling._stroke_width);
429
430 const auto marker = actsvg::style::marker{styling._marker_type,
431 styling._marker_size, fill, stroke};
432 p_landmark._marker = marker;
433}
434
436template <concepts::point3D point3_t>
437inline void apply_style(meta::proto::intersection<point3_t>& p_intersection,
438 const landmark_style& styling) {
439 const auto stroke =
440 actsvg::style::stroke(styling._fill_color, styling._stroke_width);
441
442 auto fill_color = styling._fill_color;
443 fill_color._opacity *= 0.5f;
444 const auto fill = actsvg::style::fill(fill_color);
445
446 const auto marker = actsvg::style::marker{styling._marker_type,
447 styling._marker_size, fill, stroke};
448
449 for (auto& p_landmark : p_intersection._landmarks) {
450 p_landmark._marker = marker;
451 }
452}

455template <concepts::point3D point3_t>
456inline void apply_style(meta::proto::trajectory<point3_t>& p_trajectory,
457 const trajectory_style& styling) {
458 p_trajectory._stroke =
459 actsvg::style::stroke(styling._fill_color, styling._stroke_width);
