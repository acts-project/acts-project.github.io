70 template <concepts::scalar scalar_t, concepts::point point_t>

87 template <concepts::algebra algebra_t>
88 DETRAY_HOST_DEVICE constexpr result_type<dbool<algebra_t>> check_boundaries(
89 const bounds_type<dscalar<algebra_t>> &bounds,
90 const dtransform3D<algebra_t> &trf, const dpoint3D<algebra_t> &glob_p,
91 const dscalar<algebra_t> tol =
92 std::numeric_limits<dscalar<algebra_t>>::epsilon(),
93 const dscalar<algebra_t> edge_tol = 0.f) const {
94 // Get the full local position
95 const dpoint3D<algebra_t> loc_p =
96 cartesian3D<algebra_t>::global_to_local(trf, glob_p, {});
97
98 return check_boundaries(bounds, loc_p, tol, edge_tol);
99 }

111 template <concepts::scalar scalar_t, concepts::point point_t>
112 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
113 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
114 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
115 const scalar_t edge_tol = 0.f) const {
116 auto inside_mask{(bounds[e_lower] - tol <= loc_p[kCheckIndex] &&
117 loc_p[kCheckIndex] <= bounds[e_upper] + tol)};
118
119 decltype(inside_mask) inside_edge{false};
120 if (detail::any_of(edge_tol > 0.f)) {
121 const scalar_t full_tol{tol + edge_tol};
122
123 inside_edge = (bounds[e_lower] - full_tol <= loc_p[kCheckIndex] &&
124 loc_p[kCheckIndex] <= bounds[e_upper] + full_tol);
125 }
126
127 return result_type<decltype(inside_mask)>{inside_mask, inside_edge};
128 }

159 template <concepts::scalar scalar_t>
160 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
161 const bounds_type<scalar_t> &bounds,
162 const bounds_type<scalar_t> &o_bounds) const {
163 bounds_type<scalar_t> new_bounds{};
164
165 new_bounds[e_lower] = math::min(bounds[e_lower], o_bounds[e_lower]);
166 new_bounds[e_upper] = math::max(bounds[e_upper], o_bounds[e_upper]);
167
168 return new_bounds;
169 }

180 template <concepts::algebra algebra_t>
181 DETRAY_HOST_DEVICE inline darray<dscalar<algebra_t>, 6> local_min_bounds(
182 const bounds_type<dscalar<algebra_t>> &bounds,
183 const dscalar<algebra_t> env =
184 std::numeric_limits<dscalar<algebra_t>>::epsilon()) const {
185 assert(env > 0.f);
186 darray<dscalar<algebra_t>, 6> o_bounds{-env, -env, -env, env, env, env};
187 o_bounds[kCheckIndex] += bounds[e_lower];
188 o_bounds[3u + kCheckIndex] += bounds[e_upper];
189 return o_bounds;
190 }

194 template <concepts::algebra algebra_t>
195 DETRAY_HOST_DEVICE auto centroid(
196 const bounds_type<dscalar<algebra_t>> &bounds) const {
197 using point3_t = dpoint3D<algebra_t>;
198
199 point3_t centr{0.f, 0.f, 0.f};
200 centr[kCheckIndex] = 0.5f * (bounds[e_lower] + bounds[e_upper]);
201
202 return centr;
203 }

225 template <concepts::scalar scalar_t>
226 DETRAY_HOST constexpr bool check_consistency(
227 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
228 if (bounds[e_upper] < bounds[e_lower]) {
229 os << "DETRAY ERROR (HOST): Upper bounds must be smaller than "
230 "lower bounds ";
231 return false;
232 }
233
234 return true;
235 }
