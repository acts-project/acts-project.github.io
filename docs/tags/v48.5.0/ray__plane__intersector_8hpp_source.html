61 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
62 const trajectory_type<algebra_t> &ray, const dtransform3D<algebra_t> &trf,
63 const dscalar<algebra_t> /*overstep_tol*/ = 0.f) const {
64 using scalar_t = dscalar<algebra_t>;
65 using point3_t = dpoint3D<algebra_t>;
66 using vector3_t = dvector3D<algebra_t>;
67
68 // Retrieve the surface normal & translation (context resolved)
69 const vector3_t &sn = trf.z();
70 const vector3_t &st = trf.translation();
71
72 // Intersection code
73 const point3_t &ro = ray.pos();
74 const vector3_t &rd = ray.dir();
75 const scalar_t denom = vector::dot(rd, sn);
76
77 // this is dangerous
78 if (denom == 0.f) [[unlikely]] {
79 return {};
80 }
81
82 scalar_t s{vector::dot(sn, st - ro) / denom};
83 point3_t glob_pos = ro + s * rd;
84
85 return {s, glob_pos};
86 }
87};
