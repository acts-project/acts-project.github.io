24template <concepts::algebra algebra_t>
25struct curvilinear_frame {
27 using transform3_type = dtransform3D<algebra_t>;
29 using vector3 = typename transform3_type::vector3;
31 using unit_vectors_type = unit_vectors<vector3>;
33 using bound_to_free_matrix_type = bound_to_free_matrix<algebra_t>;
35 using bound_vector_type = bound_parameters_vector<algebra_t>;
37 using jacobian_engine_type = detail::jacobian_engine<algebra_t>;
39 using free_track_parameters_type = free_track_parameters<algebra_t>;
40
43 DETRAY_HOST_DEVICE
44 explicit curvilinear_frame(const free_track_parameters_type& free_params) {
45 assert(!free_params.is_invalid());
46
47 const vector3 t = free_params.pos();
48 const vector3 z = free_params.dir();
49 const vector3 x = unit_vectors_type().make_curvilinear_unit_u(z);
50
51 m_trf = transform3_type(t, z, x);
52 m_bound_vec = detail::free_to_bound_vector<cartesian2D<algebra_t>>(
53 m_trf, free_params);
54 }
55
58 DETRAY_HOST_DEVICE
59 bound_to_free_matrix_type bound_to_free_jacobian() const {
60 return jacobian_engine_type()
61 .template bound_to_free_jacobian<cartesian2D<algebra_t>>(
62 m_trf, mask<rectangle2D, algebra_t>{}, m_bound_vec);
63 }
64
66 transform3_type m_trf{};
68 bound_vector_type m_bound_vec{};
69
70}; // curvilinear frame
