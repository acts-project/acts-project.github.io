210template <typename BackendT>
211concept BlueprintBackend =
212 HasSurfaceFactory<BackendT> && HasLayerNameMember<BackendT> &&
213 requires(const typename BackendT::Config& cfg, const Acts::Logger& logger,
214 const BackendT& backend,
215 const typename BackendT::Element& element) {
216 BackendT{cfg, logger};
217 { BackendT::kIdentifier } -> std::convertible_to<std::string_view>;
218 { backend.world() } -> std::same_as<typename BackendT::Element>;
219 { backend.nameOf(element) } -> std::same_as<std::string>;
220 {
221 backend.children(element)
222 } -> std::same_as<std::vector<typename BackendT::Element>>;
223 { backend.parent(element) } -> std::same_as<typename BackendT::Element>;
224 { backend.isSensitive(element) } -> std::same_as<bool>;
225 requires requires(const typename BackendT::Element& a,
226 const typename BackendT::Element& b) {
227 { a == b } -> std::convertible_to<bool>;
228 };
229 };
