123 DETRAY_HOST_DEVICE
124 constexpr void insert(candidate_const_itr_t /*pos*/,
125 const intersection_type &new_cadidate) {
126 // Insert the first candidate
127 if (math::fabs(new_cadidate) < math::fabs(this->candidates()[1].path)) {
128 this->candidates()[1] = new_cadidate;
129 }
130
131 assert(this->next_index() == 1u);
132 assert(detail::is_invalid_value(new_cadidate.volume_link()) ||
133 new_cadidate.volume_link() < this->detector().volumes().size());
134 }

159 template <typename track_t>
160 DETRAY_HOST_DEVICE constexpr bool update_impl(const track_t &track,
161 state &navigation,
162 const navigation::config &cfg,
163 const context_type &ctx) const {
164 const auto &det = navigation.detector();
165 constexpr bool is_init{true};
166
167 using algebra_t = typename detector_t::algebra_type;
168
169 assert(navigation.trust_level() != navigation::trust_level::e_full);
170
171 // Tangential to the track direction
172 const detray::detail::ray<algebra_t> tangential{
173 track.pos(),
174 static_cast<scalar_t>(navigation.direction()) * track.dir()};
175
176 // Update only the current candidate and the corresponding
177 // - do this only when the navigation state is still coherent
178 if (navigation.trust_level() == navigation::trust_level::e_high) {
179 DETRAY_VERBOSE_HOST_DEVICE("Called 'update()' - high trust");
180
181 // Update next candidate: If not reachable, 'high trust' is broken
182 if (!navigation::update_candidate(navigation.target(), tangential, det,
183 cfg.intersection,
184 navigation.external_tol(), ctx)) {
185 navigation.status(navigation::status::e_unknown);
186 // This will run into the fair trust case below.
187 navigation.set_fair_trust();
188 } else {
189 // Update navigation flow on the new candidate information
190 navigation::update_status(navigation, cfg);
191
192 navigation.run_inspector(cfg, track.pos(), track.dir(),
193 "Update complete: high trust: ");
194
195 // The work is done if: the track has not reached a surface yet
196 // or trust is gone (portal was reached or the cache is broken).
197 if (navigation.status() == navigation::status::e_towards_object ||
198 navigation.is_on_portal()) {
199 return !is_init;
200 }
201 // Else (if full trust): Track is on non-portal surface and
202 // cache is not exhausted. Move the current surface back in the
203 // cache and re-initialize the volume to find the next target
204 if (navigation.trust_level() == navigation::trust_level::e_full) {
205 navigation.reset_candidate();
206 }
207
208 // Find the next/different candidate
209 navigation.set_no_trust();
210 }
211 }
212 // If [next] target is not reachable or actor flagged 'no trust',
213 // re-initialize the volume.
214 assert(navigation.trust_level() == navigation::trust_level::e_no_trust);
215 DETRAY_VERBOSE_HOST_DEVICE("Called 'update()' - no trust");
216
217 constexpr bool resolve_overstepping{true};
218 navigation::local_navigation(track, navigation, cfg, ctx,
219 resolve_overstepping);
220 return is_init;
221 }
