136 DETRAY_HOST_DEVICE random_track_generator_config& eta_range(scalar_t low,
137 scalar_t high) {
138 // This value is more or less random
139 constexpr auto num_max{0.001f * std::numeric_limits<scalar_t>::max()};
140 auto min_eta{low > -num_max ? low : -num_max};
141 auto max_eta{high < num_max ? high : num_max};
142
143 assert(min_eta <= max_eta);
144
145 auto get_theta = [](const scalar_t eta) {
146 return 2.f * math::atan(math::exp(-eta));
147 };
148
149 theta_range(get_theta(max_eta), get_theta(min_eta));
150 return *this;
151 }

312 DETRAY_HOST
313 friend std::ostream& operator<<(std::ostream& out,
314 const random_track_generator_config& cfg) {
315 const auto& ori = cfg.origin();
316 const auto& mom_range = cfg.mom_range();
317 const auto& phi_range = cfg.phi_range();
318 const auto& theta_range = cfg.theta_range();
319
320 // General
321 out << "\nRandom track generator\n"
322 << "----------------------------\n"
323 << " Random seed : " << cfg.seed() << "\n"
324 << " No. tracks : " << cfg.n_tracks() << "\n"
325 << " Charge : "
326 << cfg.charge() / detray::unit<scalar_t>::e << " [e]\n"
327 << " Rand. charge : " << std::boolalpha
328 << cfg.randomize_charge() << std::noboolalpha << "\n";
329
330 // Momentum and direction
331 if (cfg.is_pT()) {
332 out << " Transverse mom. : [";
333 } else {
334 out << " Momentum : [";
335 }
336 out << mom_range[0] / detray::unit<scalar_t>::GeV << ", "
337 << mom_range[1] / detray::unit<scalar_t>::GeV << ") [GeV]\n"
338 << " Phi range : ["
339 << phi_range[0] / detray::unit<scalar_t>::rad << ", "
340 << phi_range[1] / detray::unit<scalar_t>::rad << ") [rad]\n"
341 << " Theta range : ["
342 << theta_range[0] / detray::unit<scalar_t>::rad << ", "
343 << theta_range[1] / detray::unit<scalar_t>::rad << ") [rad]\n"
344 << " Origin : [" << ori[0] / detray::unit<scalar_t>::mm
345 << ", " << ori[1] / detray::unit<scalar_t>::mm << ", "
346 << ori[2] / detray::unit<scalar_t>::mm << "] [mm]\n"
347 << " Do vertex smearing : " << std::boolalpha
348 << cfg.do_vertex_smearing() << "\n"
349 << std::noboolalpha;
350
351 if (cfg.do_vertex_smearing()) {
352 const auto& ori_stddev = cfg.origin_stddev();
353 out << " Origin stddev : ["
354 << ori_stddev[0] / detray::unit<scalar_t>::mm << ", "
355 << ori_stddev[1] / detray::unit<scalar_t>::mm << ", "
356 << ori_stddev[2] / detray::unit<scalar_t>::mm << "] [mm]\n";
357 }
358
359 return out;
360 }
