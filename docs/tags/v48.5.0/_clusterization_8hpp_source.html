26class DisjointSets {
27 public:
30 explicit DisjointSets(std::size_t initial_size = 128)
31 : m_defaultSize(initial_size),
32 m_size(initial_size),
33 m_rank(m_size),
34 m_parent(m_size),
35 m_ds(&m_rank[0], &m_parent[0]) {}
36
39 Acts::Ccl::Label makeSet() {
40 // Empirically, m_size = 128 seems to be good default. If we
41 // exceed this, take a performance hit and do the right thing.
42 while (m_globalId >= m_size) {
43 m_size *= 2;
44 m_rank.resize(m_size);
45 m_parent.resize(m_size);
46 m_ds = boost::disjoint_sets<std::size_t*, std::size_t*>(&m_rank[0],
47 &m_parent[0]);
48 }
49 m_ds.make_set(m_globalId);
50 return static_cast<Acts::Ccl::Label>(m_globalId++);
51 }
52
56 void unionSet(std::size_t x, std::size_t y) { m_ds.union_set(x, y); }
60 Acts::Ccl::Label findSet(std::size_t x) {
61 return static_cast<Acts::Ccl::Label>(m_ds.find_set(x));
62 }
63
65 void clear() {
66 m_size = m_defaultSize;
67 m_rank.clear();
68 m_parent.clear();
69 m_rank.resize(m_size);
70 m_parent.resize(m_size);
71 m_globalId = 1;
72 m_ds = boost::disjoint_sets<std::size_t*, std::size_t*>(&m_rank[0],
73 &m_parent[0]);
74 }
75
76 private:
77 std::size_t m_defaultSize{128};
78 std::size_t m_globalId = 1;
79 std::size_t m_size{m_defaultSize};
80 std::vector<std::size_t> m_rank;
81 std::vector<std::size_t> m_parent;
82 boost::disjoint_sets<std::size_t*, std::size_t*> m_ds;
83};

122// When looking for a cell connected to a reference cluster, the code
123// always loops backward, starting from the reference cell. Since
124// the cells are globally sorted column-wise, the connection function
125// can therefore tell when the search should be stopped.
126enum class ConnectResult {
127 eNoConn, // No connections, keep looking
128 eNoConnStop, // No connections, stop looking
129 eConn, // Found connection
130 eDuplicate // Found duplicate cell, throw an exception
131};
