template<concepts::detector detector_t, typename node_inspector = void_node_inspector<typename detector_t::volume_type>, template< typename... > class vector_t = dvector>
struct detray::volume_graph< detector_t, node_inspector, vector_t >::node_generator::node

A node in the graph has an index (volume index) and a collection of edges that belong to it (mask link of every surface in the volume).

One mask link is a half edge in the graph.
