45 template <class detector_t>
46 static homogeneous_material_header_payload header_to_payload(
47 const detector_t& det, const std::string_view det_name) {
48 homogeneous_material_header_payload header_data;
49
50 using algebra_t = typename detector_t::algebra_type;
51 using material_id_t = surface_material_payload::mat_type;
52
53 auto count_surface_with_material_type = [&det](material_id_t target_type) {
54 std::size_t count{0u};
55 for (const auto& [sf_idx, sf_desc] :
56 detray::views::enumerate(det.surfaces())) {
57 const auto sf = geometry::surface{det, sf_desc};
58
59 if (surface_material_payload mslp =
60 sf.has_material() ? sf.template visit_material<
61 get_material_payload<algebra_t>>(sf_idx)
62 : surface_material_payload{};
63 mslp.type == target_type) {
64 count++;
65 }
66 }
67
68 return count;
69 };
70
71 header_data.common = detail::basic_converter::to_payload(det_name, tag);
72
73 const auto& materials = det.material_store();
74
75 header_data.sub_header.emplace();
76 auto& mat_sub_header = header_data.sub_header.value();
77 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
78 mat_sub_header.n_slabs =
79 materials.template size<detector_t::material::id::e_material_slab>();
80 mat_sub_header.n_slab_surfaces =
81 count_surface_with_material_type(material_id_t::slab);
82 }
83 mat_sub_header.n_rods = 0u;
84 if constexpr (detray::concepts::has_material_rods<detector_t>) {
85 mat_sub_header.n_rods =
86 materials.template size<detector_t::material::id::e_material_rod>();
87 mat_sub_header.n_rod_surfaces =
88 count_surface_with_material_type(material_id_t::rod);
89 }
90
91 return header_data;
92 }

99 template <class detector_t>
100 static payload_type to_payload(
101 const detector_t& det, const typename detector_t::name_map& /*unused*/) {
102 payload_type dm_data;
103 dm_data.volumes.reserve(det.volumes().size());
104
105 for (const auto& vol : det.volumes()) {
106 dm_data.volumes.push_back(to_payload(vol, det));
107 }
108
109 return dm_data;
110 }

118 template <class detector_t>
119 static material_volume_payload to_payload(
120 const typename detector_t::volume_type& vol_desc, const detector_t& det) {
121 using material_type = surface_material_payload::mat_type;
122
123 material_volume_payload mv_data;
124 mv_data.volume_link = detail::basic_converter::to_payload(vol_desc.index());
125
126 // Return early if the stores for homogeneous materials are empty
127 using mat_id = typename detector_t::material::id;
128 using algebra_t = typename detector_t::algebra_type;
129
130 bool contains_rods{true};
131 bool contains_slabs{true};
132 if constexpr (detray::concepts::has_material_rods<detector_t>) {
133 if (det.material_store().template empty<mat_id::e_material_rod>()) {
134 contains_rods = false;
135 }
136 }
137 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
138 if (det.material_store().template empty<mat_id::e_material_slab>()) {
139 contains_slabs = false;
140 }
141 }
142 if (!contains_rods && !contains_slabs) {
143 return mv_data;
144 }
145
146 // Find all surfaces that belong to the volume and count them
147 std::size_t sf_idx{0u};
148 std::size_t slab_idx{0u};
149 std::size_t rod_idx{0u};
150 auto vol = tracking_volume{det, vol_desc};
151 for (const auto& sf_desc : vol.surfaces()) {
152 // Convert material slabs and rods
153 const auto sf = geometry::surface{det, sf_desc};
154
155 if (surface_material_payload mslp =
156 sf.has_material()
157 ? sf.template visit_material<get_material_payload<algebra_t>>(
158 sf_idx)
159 : surface_material_payload{};
160 mslp.type == material_type::slab) {
161 mslp.index_in_coll = slab_idx++;
162 mv_data.surface_mat.push_back(mslp);
163 } else if (mslp.type == material_type::rod) {
164 mslp.index_in_coll = rod_idx++;
165 mv_data.surface_mat.push_back(mslp);
166 } else { /* material maps are handled by another writer */
167 }
168 ++sf_idx;
169 }
170
171 return mv_data;
172 }

178 template <detray::concepts::material_params material_t>
179 static material_param_payload to_payload(const material_t& mat) {
180 material_param_payload mat_data;
181
182 mat_data.params = {mat.X0(),
183 mat.L0(),
184 mat.Ar(),
185 mat.Z(),
186 mat.mass_density(),
187 mat.molar_density(),
188 static_cast<io::scalar>(mat.state())};
189 return mat_data;
190 }

198 template <detray::concepts::algebra algebra_t,
199 detray::concepts::homogeneous_material material_t>
200 static surface_material_payload to_payload(const material_t& mat,
201 std::size_t sf_idx) {
202 surface_material_payload mat_data;
203
204 mat_data.type = io::detail::get_id<algebra_t, material_t>();
205 mat_data.surface = detail::basic_converter::to_payload(sf_idx);
206 mat_data.thickness = mat.thickness();
207 mat_data.mat = to_payload(mat.get_material());
208
209 return mat_data;
210 }

212 private:
214 template <detray::concepts::algebra algebra_t>
215 struct get_material_payload {
216 template <typename material_group_t, typename index_t>
217 inline auto operator()(const material_group_t& material_group,
218 const index_t& index,
219 [[maybe_unused]] std::size_t sf_index) const {
220 using material_t = typename material_group_t::value_type;
221
222 if constexpr (detray::concepts::material_slab<material_t> ||
223 detray::concepts::material_rod<material_t>) {
224 return homogeneous_material_writer::to_payload<algebra_t>(
225 material_group[index], sf_index);
226 } else {
227 return surface_material_payload{};
228 }
229 }
230 };
231};
