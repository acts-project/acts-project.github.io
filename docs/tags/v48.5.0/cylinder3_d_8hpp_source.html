72 template <concepts::scalar scalar_t, concepts::point point_t>
73 DETRAY_HOST_DEVICE inline scalar_t min_dist_to_boundary(
74 const bounds_type<scalar_t> &bounds, const point_t &loc_p) const {
75 const scalar_t min_r_dist =
76 math::min(math::fabs(loc_p[0] - bounds[e_min_r]),
77 math::fabs(bounds[e_max_r] - loc_p[0]));
78 const scalar_t min_phi_dist =
79 math::min(math::fabs(loc_p[1] - bounds[e_min_phi]),
80 math::fabs(bounds[e_max_phi] - loc_p[1]));
81 const scalar_t min_z_dist =
82 math::min(math::fabs(loc_p[2] - bounds[e_min_z]),
83 math::fabs(bounds[e_max_z] - loc_p[2]));
84
85 // Use the chord for the phi distance
86 return math::min(
87 math::min(min_r_dist, 2.f * loc_p[0] * math::sin(0.5f * min_phi_dist)),
88 min_z_dist);
89 }

99 template <concepts::algebra algebra_t>
100 DETRAY_HOST_DEVICE constexpr result_type<dbool<algebra_t>> check_boundaries(
101 const bounds_type<dscalar<algebra_t>> &bounds,
102 const dtransform3D<algebra_t> &trf, const dpoint3D<algebra_t> &glob_p,
103 const dscalar<algebra_t> tol =
104 std::numeric_limits<dscalar<algebra_t>>::epsilon(),
105 const dscalar<algebra_t> /*edge_tol*/ = 0.f) const {
106 // Get the full local position
107 const auto loc_p =
108 local_frame_type<algebra_t>::global_to_local(trf, glob_p, {});
109
110 return check_boundaries(bounds, loc_p, tol);

123 template <concepts::scalar scalar_t, concepts::point point_t>
124 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
125 const bounds_type<scalar_t> &bounds, const point_t &loc_p,
126 const scalar_t tol = std::numeric_limits<scalar_t>::epsilon(),
127 const scalar_t /*edge_tol*/ = 0.f) const {
128 const scalar_t phi_tol = detail::phi_tolerance(tol, loc_p[0]);
129
130 return ((bounds[e_min_r] - tol) <= loc_p[0] &&
131 (bounds[e_min_phi] - phi_tol) <= loc_p[1] &&
132 (bounds[e_min_z] - tol) <= loc_p[2] &&
133 loc_p[0] <= (bounds[e_max_r] + tol) &&
134 loc_p[1] <= (bounds[e_max_phi] + phi_tol) &&
135 loc_p[2] <= (bounds[e_max_z] + tol));
136 }

155 template <concepts::scalar scalar_t>
156 DETRAY_HOST_DEVICE constexpr scalar_t volume(
157 const bounds_type<scalar_t> &bounds) const {
158 return constant<scalar_t>::pi * (bounds[e_max_z] - bounds[e_min_z]) *
159 (bounds[e_max_r] * bounds[e_max_r] -
160 bounds[e_min_r] * bounds[e_min_r]);
161 }

169 template <concepts::scalar scalar_t>
170 DETRAY_HOST_DEVICE constexpr bounds_type<scalar_t> merge(
171 const bounds_type<scalar_t> &bounds,
172 const bounds_type<scalar_t> &o_bounds) const {
173 bounds_type<scalar_t> new_bounds{};
174
175 new_bounds[e_min_r] = math::min(bounds[e_min_r], o_bounds[e_min_r]);
176 new_bounds[e_min_phi] = math::min(bounds[e_min_phi], o_bounds[e_min_phi]);
177 new_bounds[e_min_z] = math::min(bounds[e_min_z], o_bounds[e_min_z]);
178 new_bounds[e_max_r] = math::max(bounds[e_max_r], o_bounds[e_max_r]);
179 new_bounds[e_max_phi] = math::max(bounds[e_max_phi], o_bounds[e_max_phi]);
180 new_bounds[e_max_z] = math::max(bounds[e_max_z], o_bounds[e_max_z]);
181
182 return new_bounds;
183 }

237 template <concepts::scalar scalar_t>
238 DETRAY_HOST constexpr bool check_consistency(
239 const bounds_type<scalar_t> &bounds, std::ostream &os) const {
240 constexpr auto tol{10.f * std::numeric_limits<scalar_t>::epsilon()};
241
242 if (bounds[e_min_r] < tol) {
243 os << "DETRAY ERROR (HOST): Radii must be in the range (0, "
244 "numeric_max)"
245 << std::endl;
246 return false;
247 }
248 if (bounds[e_min_r] >= bounds[e_max_r] ||
249 math::fabs(bounds[e_min_r] - bounds[e_max_r]) < tol) {
250 os << "DETRAY ERROR (HOST): Min Radius must be smaller than max "
251 "Radius.";
252 return false;
253 }
254 if (bounds[e_min_z] >= bounds[e_max_z] ||
255 math::fabs(bounds[e_min_z] - bounds[e_max_z]) < tol) {
256 os << "DETRAY ERROR (HOST): Min z must be smaller than max z.";
257 return false;
258 }
259
260 return true;
261 }
