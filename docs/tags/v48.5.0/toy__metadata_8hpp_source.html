42 template <template <typename...> class vector_t = dvector>

48 using rectangle2D_t = mask<detray::rectangle2D, algebra_type, nav_link>;
50 using trapezoid2D_t = mask<detray::trapezoid2D, algebra_type, nav_link>;
52 using concentric_cylinder2D_t =

55 using ring2D_t = mask<detray::ring2D, algebra_type, nav_link>;

58 enum class mask_id : std::uint_least8_t {
59 e_rectangle2D = 0u,
60 e_trapezoid2D = 1u,

63 };

72 os << "e_trapezoid2D";

90 concentric_cylinder2D_t, ring2D_t>;
91
92 template <typename container_t>
94 using concentric_cylinder2D_map_t =
95 material_map<algebra_type, detray::concentric_cylinder2D, container_t>;
96 template <typename container_t>
98 using ring2D_map_t = material_map<algebra_type, detray::ring2D, container_t>;
100 using material_slab_t = material_slab<scalar_t>;
101
103 enum class material_id : std::uint_least8_t {
104 e_concentric_cylinder2D_map = 0u,
105 e_ring2D_map = 1u,
106 e_material_slab = 2u,
107 e_none = 3u,
108 };
109
110 DETRAY_HOST inline friend std::ostream& operator<<(std::ostream& os,
111 material_id id) {

116 case material_id::e_ring2D_map:
117 os << "e_ring2D_map";

122 case material_id::e_none:
123 os << "e_none";

126 os << "invalid";

131 template <typename container_t = host_container_types>
133 using material_store =
134 multi_store<material_id, empty_context, dtuple,
135 grid_collection<concentric_cylinder2D_map_t<container_t>>,
136 grid_collection<ring2D_map_t<container_t>>,
137 typename container_t::template vector_type<material_slab_t>>;
138
140 using transform_link = typename transform_store<>::single_link;
142 using mask_link = typename mask_store<>::range_link;
144 using material_link = typename material_store<>::single_link;
146 using surface_type =
147 surface_descriptor<mask_link, material_link, transform_link, nav_link>;
148
149 template <typename axes_t, typename bin_entry_t, typename container_t>
151 using static_simple_grid_t =
152 spatial_grid<algebra_type, axes_t,
153 detray::bins::static_array<bin_entry_t, 1>,
154 detray::simple_serializer, container_t, false>;
155
156 template <typename container_t>
158 using surface_concentric_cylinder2D_grid_t =
159 static_simple_grid_t<axes<detray::concentric_cylinder2D>, surface_type,
160 container_t>;
161 template <typename container_t>
163 using surface_ring2D_grid_t =
164 static_simple_grid_t<axes<detray::ring2D>, surface_type, container_t>;
165
166 template <typename axes_t, typename bin_entry_t, typename container_t>
168 using single_simple_grid_t =
169 spatial_grid<algebra_type, axes_t, detray::bins::single<bin_entry_t>,
170 detray::simple_serializer, container_t, false>;
171
172 template <typename container_t>
174 using volume_cylinder3D_grid_t =
175 single_simple_grid_t<axes<detray::cylinder3D>, dindex, container_t>;

184 e_volume_default = e_volume_cylinder3D_grid,

208 template <typename container_t = host_container_types>
210 using accelerator_store = multi_store<
211 accel_id, empty_context, dtuple,

213 grid_collection<surface_concentric_cylinder2D_grid_t<container_t>>,
214 grid_collection<surface_ring2D_grid_t<container_t>>,
215 grid_collection<volume_cylinder3D_grid_t<container_t>>>;
