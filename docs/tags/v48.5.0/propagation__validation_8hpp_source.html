34template <concepts::scalar scalar_t>
35struct propagation_validation_config {
37 propagation::config propagation;
38
40 pdg_particle<scalar_t> particle = detray::muon<scalar_t>();
41
43 bool do_multiple_scattering{true};
45 bool do_energy_loss{true};
46
48 bool display_svg{true};
50 float max_percent_missed{1.f};
52 float max_percent_additional{15.f};
53};

66template <concepts::detector detector_t, typename bfield_view_t>
67bool propagation_validation(
68 const detector_t& det, const typename detector_t::name_map& names,
69 const std::optional<bfield_view_t>& bfield,
70 const propagation_validation_config<typename detector_t::scalar_type>& cfg,
71 const std::vector<free_track_parameters<typename detector_t::algebra_type>>&
72 tracks,
73 std::vector<dvector<intersection_record<detector_t>>>& truth_traces_fw) {
74 // 'false' if any failures were detected
75 bool test_successful{true};
76
77 using algebra_t = typename detector_t::algebra_type;
78 using scalar_t = dscalar<algebra_t>;
79 using stepper_t =
80 rk_stepper<bfield_view_t, algebra_t, constrained_step<scalar_t>,
81 stepper_rk_policy<scalar_t>, stepping::print_inspector>;
82 using data_record_t = intersection_record<detector_t>;
83
84 // Host memory resource
85 vecmem::host_memory_resource host_mr;
86
87 // Geometry context
88 const typename detector_t::geometry_context ctx{};
89
90 // Configure the test
91 test::navigation_validation_config<algebra_t> test_cfg{};
92 test_cfg.n_tracks(tracks.size());
93 test_cfg.ptc_hypothesis(cfg.particle);
94 test_cfg.collect_sensitives_only(true);
95 test_cfg.fail_on_diff(false);
96 test_cfg.display_svg(cfg.display_svg);
97 test_cfg.display_only_missed(true);
98 test_cfg.verbose(false);
99
100 // Create the backward truth traces by reverting the order
101 std::vector<dvector<data_record_t>> truth_traces_bw;
102 truth_traces_bw.reserve(tracks.capacity());
103
104 for (const auto& truth_trace_fw : truth_traces_fw) {
105 // Revert the forward trace for the backward propagation
106 dvector<data_record_t> truth_trace_bw(truth_trace_fw.size());
107 std::ranges::reverse_copy(truth_trace_fw, truth_trace_bw.begin());
108
109 assert(!truth_trace_bw.empty());
110
111 truth_traces_bw.push_back(std::move(truth_trace_bw));
112 }
113
114 // Check truth data
115 if (truth_traces_fw.empty() || truth_traces_bw.empty()) {
116 DETRAY_ERROR_HOST("Propagation truth data empty");
117 return false;
118 }
119
120 // Make a tuple of references from a tuple
121 auto setup_actor_states = []<typename... T>(detray::dtuple<T...>& t) {
122 return detray::tie(detray::detail::get<T>(t)...);
123 };
124
125 // Define the actors
126 using interactor = actor::pointwise_material_interactor<algebra_t>;
127 using perigee_stopper = perigee_stopper<algebra_t>;
128
129 // Reusable actor states
130 typename interactor::state interactor_state{};
131 interactor_state.do_multiple_scattering = cfg.do_multiple_scattering;
132 interactor_state.do_energy_loss = cfg.do_energy_loss;
133
134 typename perigee_stopper::state stopper_state{};
135
136 std::cout << "-----------------------------------"
137 << "\nFORWARD - Propagation" << std::endl
138 << "-----------------------------------\n";
139
140 using parameter_updater = actor::parameter_updater<algebra_t, interactor>;
141
142 using actor_chain_t = actor_chain<parameter_updater>;
143
144 typename parameter_updater::state updater_state{};
145 updater_state.noise_estimation_cfg().n_stddev =
146 cfg.propagation.navigation.n_scattering_stddev;
147 updater_state.noise_estimation_cfg().accumulated_error =
148 cfg.propagation.navigation.accumulated_error;
149 updater_state.noise_estimation_cfg().estimate_scattering_noise =
150 cfg.propagation.navigation.estimate_scattering_noise;
151
152 // Prepare the actor states for every track
153 dvector<typename actor_chain_t::state_tuple> state_tuples{};
154 dvector<typename actor_chain_t::state_ref_tuple> state_ref_tuples{};
155 state_tuples.reserve(tracks.size());
156 state_ref_tuples.reserve(tracks.size());
157
158 for (std::size_t i = 0u; i < tracks.size(); ++i) {
159 // Define the initial covariance
160 updater_state.init(tracks.at(i));
161
162 // Copy the configured updater state for this track
163 state_tuples.push_back(detray::make_tuple(updater_state, interactor_state));
164 state_ref_tuples.push_back(setup_actor_states(state_tuples.back()));
165 }
166
167 // Forward navigation
168 test_cfg.name(det.name(names) + "_GeV_fw");
169 test_cfg.navigation_direction(navigation::direction::e_forward);
170 const auto [trk_stats_fw, n_surfaces_fw, n_miss_nav_fw, n_miss_truth_fw,
171 step_traces_fw, track_mat_fw, mat_steps_fw] =
172 navigation_validator::compare_to_navigation<stepper_t, parameter_updater>(
173 test_cfg, host_mr, det, names, ctx, bfield, cfg.propagation,
174 truth_traces_fw, tracks, state_ref_tuples);
175
176 // Check stats
177 auto n_tracks{static_cast<double>(trk_stats_fw.n_tracks)};
178 if (static_cast<double>(trk_stats_fw.n_tracks_w_holes) / n_tracks >
179 cfg.max_percent_missed / 100.) {
180 DETRAY_ERROR_HOST("Too many tracks with missing surfaces");
181 test_successful = false;
182 }
183 if (static_cast<double>(trk_stats_fw.n_tracks_w_extra) / n_tracks >
184 cfg.max_percent_additional / 100.) {
185 DETRAY_ERROR_HOST("Too many tracks with additional surfaces");
186 test_successful = false;
187 }
188
189 std::cout << "\n-----------------------------------\n"
190 << "BACKWARD - Propagation" << std::endl
191 << "-----------------------------------\n";
192
193 using actor_chain_bw_t = actor_chain<parameter_updater, perigee_stopper>;
194
195 // Prepare the actor states for every track
196 dvector<typename actor_chain_bw_t::state_tuple> bw_state_tuples{};
197 dvector<typename actor_chain_bw_t::state_ref_tuple> bw_state_ref_tuples{};
198 bw_state_tuples.reserve(tracks.size());
199 bw_state_ref_tuples.reserve(tracks.size());
200
201 for (std::size_t i = 0u; i < tracks.size(); ++i) {
202 // Define the initial covariance
203 updater_state.init(tracks.at(i));
204
205 // Copy the configured updater state for this track
206 bw_state_tuples.push_back(
207 detray::make_tuple(stopper_state, interactor_state, updater_state));
208 bw_state_ref_tuples.push_back(setup_actor_states(bw_state_tuples.back()));
209 }
210
211 // Backward navigation
212 test_cfg.name(det.name(names) + "_GeV_bw");
213 test_cfg.navigation_direction(navigation::direction::e_backward);
214 const auto [trk_stats_bw, n_surfaces_bw, n_miss_nav_bw, n_miss_truth_bw,
215 step_traces_bw, track_mat_bw, mat_steps_bw] =
216 navigation_validator::compare_to_navigation<stepper_t, parameter_updater,
217 perigee_stopper>(
218 test_cfg, host_mr, det, names, ctx, bfield, cfg.propagation,
219 truth_traces_bw, tracks, bw_state_ref_tuples);
220
221 // Make sure some data was collected
222 assert(trk_stats_fw.n_tracks > 0u);
223 assert(n_surfaces_fw.n_total() > 0u);
224 assert(trk_stats_bw.n_tracks > 0u);
225 assert(n_surfaces_bw.n_total() > 0u);
226
227 assert(trk_stats_fw.n_tracks == trk_stats_bw.n_tracks);
228
229 // Check, the amount of collected material between forward and backward
230 assert(step_traces_fw.size() == trk_stats_fw.n_tracks);
231 assert(mat_steps_fw.size() == trk_stats_fw.n_tracks);
232 assert(track_mat_fw.size() == trk_stats_fw.n_tracks);
233 assert(track_mat_fw.size() == track_mat_bw.size());
234 assert(step_traces_fw.size() == step_traces_bw.size());
235 assert(mat_steps_fw.size() == mat_steps_bw.size());
236
237 // Check stats
238 n_tracks = static_cast<double>(trk_stats_bw.n_tracks);
239 if (static_cast<double>(trk_stats_bw.n_tracks_w_holes) / n_tracks >
240 cfg.max_percent_missed / 100.) {
241 DETRAY_ERROR_HOST("Too many tracks with missing surfaces");
242 test_successful = false;
243 }
244 if (static_cast<double>(trk_stats_bw.n_tracks_w_extra) / n_tracks >
245 cfg.max_percent_additional / 100.) {
246 DETRAY_ERROR_HOST("Too many tracks with additional surfaces");
247 test_successful = false;
248 }
249
250 std::cout << "\n-----------------------------------\n"
251 << "MATERIAL TRACE" << std::endl
252 << "-----------------------------------\n";
253
254 constexpr double rel_mat_error{0.01};
255
256 // Material traces contain different surfaces
257 std::size_t n_bad_comp{0u};
258 // Overall integrated material differs while surface seq. is identical
259 std::size_t n_diff_mat{0u};
260
261 // Loop over tracks
262 for (std::size_t i = 0u; i < track_mat_fw.size(); ++i) {
263 // No material on that track (e.g. detector model without material)
264 if (mat_steps_fw[i].empty() && mat_steps_bw[i].empty()) {
265 continue;
266 }
267
268 std::remove_cvref_t<decltype(mat_steps_bw[i])> inv_mat_trace_bw{};
269 if (!mat_steps_bw[i].empty()) {
270 inv_mat_trace_bw.resize(mat_steps_bw[i].size());
271
272 // Revert the backward trace to compare to the forward trace
273 std::ranges::reverse_copy(mat_steps_bw[i], inv_mat_trace_bw.begin());
274
275 assert(mat_steps_bw[i].size() == inv_mat_trace_bw.size());
276 }
277
278 // Compare the material traces and total integrated material per trk
279 const auto [is_bad_comp, is_diff_mat] = material_validator::compare_traces(
280 mat_steps_fw[i], track_mat_fw[i], inv_mat_trace_bw, track_mat_bw[i], i,
281 rel_mat_error, test_cfg.verbose());
282
283 if (is_bad_comp) {
284 n_bad_comp++;
285 }
286 if (is_diff_mat) {
287 n_diff_mat++;
288 }
289 }
290
291 std::cout << "Total no. tracks with diff. material: "
292 << (n_bad_comp + n_diff_mat) << " ("
293 << 100. * static_cast<double>(n_bad_comp + n_diff_mat) / n_tracks
294 << "%)" << std::endl;
295
296 std::cout << "No. identical tracks with diff. material: " << n_diff_mat
297 << " (" << 100. * static_cast<double>(n_diff_mat) / n_tracks << "%)"
298 << std::endl;
299 std::cout << "-----------------------------------\n" << std::endl;
300
301 // Trigger test failures
302 if (n_diff_mat != 0) {
303 DETRAY_ERROR_HOST("" << n_diff_mat << " tracks have differing material");
304 test_successful = false;
305 }
306
307 return test_successful;
308}
