107 template <typename surface_descr_t, typename mask_t,
108 concepts::algebra other_algebra_t, std::uint8_t N = n_solutions>
109 requires(N > 1u)
110 DETRAY_HOST_DEVICE constexpr darray<intersection_type<surface_descr_t>,
111 n_solutions>
112 operator()(const trajectory_type<other_algebra_t> &traj,
113 const surface_descr_t &sf, const mask_t &mask,
114 const transform3_type &trf, const intersection::config &cfg = {},
115 const scalar_type external_mask_tol = 0.f) const {
116 // One or both of these solutions might be invalid
117 result_type result =
118 call_intersector(traj, mask, trf, cfg.overstep_tolerance);
119
120 darray<intersection_type<surface_descr_t>, n_solutions> ret;
121
122 for (std::size_t i = 0u; i < n_solutions; ++i) {
123 if (detray::detail::any_of(result[i].is_valid())) {
124 resolve_mask(ret[i], traj, result[i], sf, mask, trf, cfg,
125 external_mask_tol);
126 }
127 }
128
129 // Even if there are two geometrically valid solutions, the smaller one
130 // might not be passed on if it is below the overstepping tolerance
131 return ret;
132 }

185template <typename intersection_t, typename trajectory_t,
186 concepts::algebra algebra_t, concepts::point point_t,
187 typename surface_descr_t, typename mask_t,
188 concepts::transform3D transform3_t, concepts::scalar scalar_t>
189DETRAY_HOST_DEVICE constexpr void resolve_mask(
190 intersection_t &is, const trajectory_t &traj,
191 const intersection_point<algebra_t, point_t, intersection::contains_pos>
192 &ip,
193 const surface_descr_t sf_desc, const mask_t &mask, const transform3_t &trf,
194 const intersection::config &cfg = {},
195 const scalar_t external_mask_tolerance = 0.f) {
196 // Mask out solutions that don't meet the overstepping tolerance (SoA)
197 if constexpr (concepts::soa<algebra_t>) {
198 using status_t = typename intersection_t::status_type;
199
200 is.status()(is.path() < cfg.overstep_tolerance) =
201 static_cast<status_t>(intersection::status::e_outside);
202 } else {
203 is.set_status(intersection::status::e_outside);
204 }
205
206 // Build intersection struct from test trajectory, if the distance is valid
207 if (detray::detail::none_of(ip.path >= cfg.overstep_tolerance)) {
208 // Not a valid intersection
209 return;
210 }
211
212 // Save local position for debug evaluation
213 if constexpr (intersection_t::contains_pos()) {
214 // Global position on the surface
215 dpoint3D<algebra_t> glob_pos;
216 if constexpr (concepts::soa<algebra_t>) {
217 // The trajectory is given in AoS layout. Translate...
218 const auto &origin = traj.pos();
219 const auto &dir = traj.dir();
220
221 // Broadcast
222 const dvector3D<algebra_t> ro{origin[0], origin[1], origin[2]};
223 const dvector3D<algebra_t> rd{dir[0], dir[1], dir[2]};
224
225 glob_pos = ro + ip.path * rd;
226 } else {
227 // Works for any parameterized trajectory
228 glob_pos = traj.pos(ip.path);
229 }
230 is.set_local(mask_t::to_local_frame3D(trf, glob_pos, traj.dir(ip.path)));
231 }
232
233 scalar_t base_tol = 0.f;
234 scalar_t ext_tol = 0.f;
235
236 // Tol.: scale with distance of surface to account for track bending
237 if (!sf_desc.is_portal() ||
238 cfg.min_mask_tolerance == std::numeric_limits<float>::max()) {
239 ext_tol = external_mask_tolerance;
240 base_tol =
241 math::max(static_cast<scalar_t>(cfg.min_mask_tolerance),
242 math::min(static_cast<scalar_t>(cfg.max_mask_tolerance),
243 static_cast<scalar_t>(cfg.mask_tolerance_scalor) *
244 math::fabs(ip.path)));
245 }
246
247 // Mask check results with and without external tolerance
248 typename mask_t::result_type mask_check{};
249
250 // Intersector provides specialized local point
251 if constexpr (std::same_as<point_t, dpoint2D<algebra_t>>) {
252 mask_check = mask.resolve(ip.point, base_tol, ext_tol);
253 } else {
254 // Otherwise, let the shape transform the point to local
255 mask_check = mask.resolve(trf, ip.point, base_tol, ext_tol);
256 }
257
258 // Set the less strict status first, then overwrite with more strict
259 is.set_status_if(intersection::status::e_edge,
260 detray::get<check_type::e_with_edge>(mask_check));
261 is.set_status_if(intersection::status::e_inside,
262 detray::get<check_type::e_precise>(mask_check));
263
264 is.set_path(ip.path);
265 is.set_surface(sf_desc);
266 is.set_direction(!math::signbit(ip.path));
267 is.set_volume_link(mask.volume_link());
268}
