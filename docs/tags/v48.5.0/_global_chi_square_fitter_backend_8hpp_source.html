40// A projector used for scattering. By using Jacobian * phiThetaProjector one
41// gets only the derivatives for the variables phi and theta.
42const Eigen::Matrix<double, eBoundSize, 2> phiThetaProjector = [] {
43 Eigen::Matrix<double, eBoundSize, 2> m =
44 Eigen::Matrix<double, eBoundSize, 2>::Zero();
45 m(eBoundPhi, 0) = 1.0;
46 m(eBoundTheta, 1) = 1.0;
47 return m;
