113 constexpr ParticleHypothesis(PdgParticle absPdg, float mass, float absCharge)

137 explicit ParticleHypothesis(PdgParticle absPdg)
138 : m_absPdg{absPdg},
139 m_mass{findMass(absPdg).value()},
140 m_charge{std::abs(findCharge(absPdg).value())} {
141 assert(absPdg == makeAbsolutePdgParticle(absPdg) &&
142 "pdg is expected to be absolute");
143 }

192 std::ostream& toStream(std::ostream& os) const {
193 os << "ParticleHypothesis{absPdg=";

195 os << *shortString;

199 os << ", mass=" << mass() << ", absCharge=" << absoluteCharge() << "}";
