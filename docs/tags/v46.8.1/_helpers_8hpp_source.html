33template <SmartPointerConcept T>
34std::vector<std::add_pointer_t<std::add_const_t<typename T::element_type>>>
35unpackConstSmartPointers(const std::vector<T>& items) {
36 std::vector<std::add_pointer_t<std::add_const_t<typename T::element_type>>>
37 rawPtrs{};
38 rawPtrs.reserve(items.size());
39 for (const auto& ptr : items) {
40 rawPtrs.push_back(ptr.operator->());
41 }
42 return rawPtrs;

115template <template <std::size_t> class Callable, std::size_t N,
116 std::size_t NMAX, typename... Args>
117decltype(auto) template_switch(std::size_t v, Args&&... args) {
118 if (v == N) {
119 return Callable<N>::invoke(std::forward<Args>(args)...);
120 }
121 if (v == 0) {
122 std::cerr << "template_switch<Fn, " << N << ", " << NMAX << ">(v=" << v
123 << ") is not valid (v == 0 and N != 0)" << std::endl;
124 std::abort();
125 }
126 if constexpr (N < NMAX) {
127 return template_switch<Callable, N + 1, NMAX>(v,
128 std::forward<Args>(args)...);
129 }
130 std::cerr << "template_switch<Fn, " << N << ", " << NMAX << ">(v=" << v
131 << ") is not valid (v > NMAX)" << std::endl;
132 std::abort();
133}

143template <std::size_t N, std::size_t NMAX, typename Lambda, typename... Args>
144decltype(auto) template_switch_lambda(std::size_t v, Lambda&& func,
145 Args&&... args) {
146 if (v == N) {
147 return func(std::integral_constant<std::size_t, N>{},
148 std::forward<Args>(args)...);
149 }
150 if (v == 0) {
151 std::cerr << "template_switch<Fn, " << N << ", " << NMAX << ">(v=" << v
152 << ") is not valid (v == 0 and N != 0)" << std::endl;
153 std::abort();
154 }
155 if constexpr (N < NMAX) {
156 return template_switch_lambda<N + 1, NMAX>(v, func,
157 std::forward<Args>(args)...);
158 }
159 std::cerr << "template_switch<Fn, " << N << ", " << NMAX << ">(v=" << v
160 << ") is not valid (v > NMAX)" << std::endl;
161 std::abort();
162}

169template <typename T, typename U>
170T clampValue(U value) {
171 if (std::numeric_limits<U>::has_infinity && std::isinf(value)) {
172 if (!std::numeric_limits<T>::has_infinity) {
173 throw std::logic_error(
174 "Cannot convert infinite value to type without infinity support");
175 }
176 return (value > 0) ? std::numeric_limits<T>::infinity()
177 : -std::numeric_limits<T>::infinity();
178 }
179 if (std::numeric_limits<U>::has_quiet_NaN && std::isnan(value)) {
180 if (!std::numeric_limits<T>::has_quiet_NaN) {
181 throw std::logic_error(
182 "Cannot convert NaN value to type without NaN support");
183 }
184 return std::numeric_limits<T>::quiet_NaN();
185 }
186 return static_cast<T>(
187 std::clamp(value, static_cast<U>(std::numeric_limits<T>::lowest()),
188 static_cast<U>(std::numeric_limits<T>::max())));
189}

198template <typename T>
199std::tuple<typename T::value_type, double> range_medium(const T& tseries) {
200 auto [minIt, maxIt] = std::ranges::minmax_element(tseries);
201 typename T::value_type range = (*maxIt - *minIt);
202 double medium = static_cast<double>((*maxIt + *minIt) * 0.5);
203 return {range, medium};
204}

289inline auto getMinMaxAndBinCount(std::vector<double>& xPos) {
290 // sort the values for unique()
291 std::ranges::sort(xPos);
292
293 // get the number of bins over unique values
294 // ranges::unique returns [ret, end): duplicate tail; unique elements are
295 // [begin, ret)
296 const auto uniqueTail = std::ranges::unique(xPos);
297 const std::size_t xBinCount = static_cast<std::size_t>(
298 std::ranges::distance(xPos.begin(), uniqueTail.begin()));
299
300 // get the minimum and maximum
301 auto [xMin, xMax] = std::ranges::minmax(xPos);
302
303 // calculate maxima (add one last bin, because bin value always corresponds to
304 // left boundary)
305 const double stepX = (xMax - xMin) / static_cast<double>(xBinCount - 1);
306 xMax += stepX;
307
308 // Return all values as a tuple
309 return std::make_tuple(xMin, xMax, xBinCount);
310}
