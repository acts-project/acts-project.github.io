213template <typename F>
214inline bool HoughAccumulatorSection::isLineInside(F &&function) const &
215 requires std::invocable<F, float>
216{
217 const float yB = function(m_xBegin);
218 const float yE = function(m_xBegin + m_xSize);
219 return (yE > yB) ? yB < m_yBegin + m_ySize && yE > m_yBegin
220 : yB > m_yBegin && yE < m_yBegin + m_ySize;
221}

223template <typename F>
224inline bool HoughAccumulatorSection::isCrossingInside(F &&line1,
225 F &&line2) const &
226 requires std::invocable<F, float>
227{
228 // this microalgorithm idea is illustrated below
229 // section left section right
230 // example with crossing
231 // | +2
232 // line 1 crossing left section edge +1 _|
233 // left edge mid point |_ |
234 // | +1
235 // line 2crossing left section +2 |
236 //
237 // example with no crossing
238 // | +1
239 // line 1 crossing left section edge +1 _|
240 // left edge mid point |_ |
241 // | +2
242 // line 2crossing left section +2 |
243 // The above covers most of the cases.
244 // Additional precautions are made when both lines cross
245 // left & right (x) bounds outside of vertical (y) bounds.
246
247 const float xL = xBegin();
248 const float xR = xBegin() + xSize();
249
250 // Evaluate both lines at section boundaries
251 const float y1L = line1(xL);
252 const float y1R = line1(xR);
253 const float y2L = line2(xL);
254 const float y2R = line2(xR);
255
256 // --- Step 1: Quick rejection based on ordering ---
257 // If one line is consistently above the other → no crossing
258 const float dL = y1L - y2L;
259 const float dR = y1R - y2R;
260
261 if (dL * dR > 0) {
262 return false;
263 }
264
265 // --- Step 2: Check if either line fully spans inside vertically ---
266 const auto checkVerticalOverlap = [this](float y) {
267 return yBegin() < y && y < yBegin() + ySize();
268 };
269
270 if (checkVerticalOverlap(y1L) && checkVerticalOverlap(y1R)) {
271 return true;
272 }
273
274 if (checkVerticalOverlap(y2L) && checkVerticalOverlap(y2R)) {
275 return true;
276 }
277
278 // --- Step 3: Approximate crossing position ---
279 // Linear interpolation assuming near-linear behavior
280 const float abs_dL = std::abs(dL);
281 const float abs_dR = std::abs(dR);
282
283 // Avoid division by zero (parallel & overlapping edge case)
284 if (abs_dL + abs_dR == 0.0f) {
285 return false;
286 }
287
288 const float t = abs_dL / (abs_dL + abs_dR); // fraction from left
289 const float xCross = std::lerp(xL, xR, t);
290
291 // --- Step 4: Check if crossing lies inside vertical bounds ---
292 const float yCross1 = line1(xCross);
293 const float yCross2 = line2(xCross);
294 const float yCross = 0.5f * (yCross1 + yCross2); // intersection approx
295
296 return checkVerticalOverlap(yCross);
297}

301template <typename Measurement>
302struct HoughExplorationOptions {

309 float yMinBinSize = 1.0f;

341template <typename Measurement>
342void exploreHoughParametersSpace(
343 std::vector<HoughAccumulatorSection> &sectionsStack,
344 const std::vector<Measurement> &measurements,
345 const HoughExplorationOptions<Measurement> &opt,
346 std::vector<HoughAccumulatorSection> &results) {
347 while (!sectionsStack.empty()) {
348 HoughAccumulatorSection thisSection = std::move(sectionsStack.back());
349 sectionsStack.pop_back();
350
351 std::vector<HoughAccumulatorSection> newSections;
352 const bool splitX = thisSection.xSize() > opt.xMinBinSize;
353 const bool splitY = thisSection.ySize() > opt.yMinBinSize;
354 if (splitX && splitY) {
355 // Split into 4 sections
356 newSections.push_back(thisSection.bottomLeft());
357 newSections.push_back(thisSection.topLeft());
358 newSections.push_back(thisSection.bottomRight());
359 newSections.push_back(thisSection.topRight());
360 } else if (!splitX && splitY) {
361 // Split into 2 sections horizontally
362 newSections.push_back(thisSection.bottom());
363 newSections.push_back(thisSection.top());
364 } else if (splitX && !splitY) {
365 // Split into 2 sections vertically
366 newSections.push_back(thisSection.left());
367 newSections.push_back(thisSection.right());
368 }
369
370 if (thisSection.decision() ==
371 HoughAccumulatorSection::Decision::DrillAndExpand) {
372 for (HoughAccumulatorSection &s : newSections) {
373 s.expand(opt.expandX, opt.expandY);
374 }
375 }
376
377 for (const std::uint32_t idx : thisSection.indices()) {
378 const auto &m = measurements[idx];
379 const auto line = [&](float x) { return opt.lineFunctor(m, x); };
380
381 for (HoughAccumulatorSection &s : newSections) {
382 if (s.isLineInside(line)) {
383 s.indices().push_back(idx);
384 }
385 }
386 }
387 for (HoughAccumulatorSection &s : newSections) {
388 s.updateDecision(opt.decisionFunctor(s, measurements));
389 if (s.decision() == HoughAccumulatorSection::Decision::Accept) {
390 results.push_back(std::move(s));
391 } else if (s.decision() == HoughAccumulatorSection::Decision::Drill ||
392 s.decision() ==
393 HoughAccumulatorSection::Decision::DrillAndExpand) {
394 sectionsStack.push_back(std::move(s));
395 }
396 }
397 }
398}

408template <typename Measurement, typename Functor>
409bool passIntersectionsCheck(const HoughAccumulatorSection &section,
410 const std::vector<Measurement> &measurements,
411 const Functor &lineFunctor,
412 const std::uint32_t threshold) {
413 const std::size_t count = section.count();
414 const float xLeft = section.xBegin();
415 const float xRight = xLeft + section.xSize();
416 const auto &indices = section.indices();
417
418 // Small Buffer Optimization
419 constexpr std::size_t kMaxStackLines = 64;
420
421 if (count <= kMaxStackLines) {
422 std::array<float, kMaxStackLines> yLeft{};
423 std::array<float, kMaxStackLines> yRight{};
424
425 for (std::size_t i = 0; i < count; ++i) {
426 const auto &m = measurements[indices[i]];
427 yLeft[i] = lineFunctor(m, xLeft);
428 yRight[i] = lineFunctor(m, xRight);
429 }
430
431 std::uint32_t inside = 0;
432 for (std::uint32_t i = 0; i < count; ++i) {
433 for (std::uint32_t j = i + 1; j < count; ++j) {
434 if ((yLeft[i] - yLeft[j]) * (yRight[i] - yRight[j]) < 0.0f) {
435 inside++;
436 if (inside >= threshold) {
437 return true; // Early exit
438 }
439 }
440 }
441 }
442 return false;
443 } else {
444 std::vector<float> yLeft(count);
445 std::vector<float> yRight(count);
446 for (std::uint32_t i = 0; i < count; ++i) {
447 const auto &m = measurements[indices[i]];
448 yLeft[i] = lineFunctor(m, xLeft);
449 yRight[i] = lineFunctor(m, xRight);
450 }
451 std::uint32_t inside = 0;
452 for (std::uint32_t i = 0; i < count; ++i) {
453 for (std::uint32_t j = i + 1; j < count; ++j) {
454 if ((yLeft[i] - yLeft[j]) * (yRight[i] - yRight[j]) < 0.0f) {
455 inside++;
456 if (inside >= threshold) {
457 return true; // Early exit
458 }
459 }
460 }
461 }
462 return inside >= threshold;
463 }
464}
