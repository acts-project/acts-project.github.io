186 template <SurfaceVisitor visitor_t>
187 void visitSurfaces(visitor_t&& visitor, bool restrictToSensitives) const {
188 if (restrictToSensitives) {
189 apply([&visitor](const Surface& surface) {
190 if (surface.geometryId().sensitive() == 0) {
191 return;
192 }
193 visitor(&surface);
194 });
195
196 } else {
197 apply(overloaded{
198 [&visitor](const Surface& surface) { visitor(&surface); },
199 [&visitor](const Portal& portal) { visitor(&portal.surface()); },
200 [&visitor](const BoundarySurface& bs) {
201 visitor(&bs.surfaceRepresentation());
202 }});
203 }
204 }
205
215 template <SurfaceVisitor visitor_t>
216 void visitSurfaces(visitor_t&& visitor) const {
217 visitSurfaces(std::forward<visitor_t>(visitor), true);
218 }
