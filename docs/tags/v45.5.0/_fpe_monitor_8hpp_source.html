54 : m_data{std::make_unique<std::byte[]>(bufferSize)},
55 m_size{bufferSize} {}
56
57 Buffer(const Buffer &) = delete;
60 Buffer(Buffer &&other) noexcept
61 : m_data(std::move(other.m_data)),
62 m_size(other.m_size),
63 m_offset(other.m_offset) {
64 other.m_size = 0;
65 other.m_offset = 0;
66 }
67
70 std::pair<void *, std::size_t> next() const {
71 return {m_data.get() + m_offset, m_size - m_offset};
72 }
73
76 void pushOffset(std::size_t offset) {
77 assert(m_offset + offset < m_size);
78 m_offset = offset;
79 }
80
82 void reset() { m_offset = 0; }
83
86 std::size_t size() const { return m_size; }
89 std::size_t offset() const { return m_offset; }

93 std::byte *data() { return m_data.get(); }

95 private:
96 std::unique_ptr<std::byte[]> m_data;
