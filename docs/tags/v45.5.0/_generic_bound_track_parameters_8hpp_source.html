63 static Result<GenericBoundTrackParameters> create(
64 const GeometryContext& geoCtx, std::shared_ptr<const Surface> surface,
65 const Vector4& pos4, const Vector3& dir, double qOverP,
66 std::optional<CovarianceMatrix> cov,
67 ParticleHypothesis particleHypothesis,

70 transformFreeToBoundParameters(pos4.segment<3>(ePos0), pos4[eTime], dir,

202 std::optional<SquareMatrix<3>> impactParameterCovariance() const {
203 if (!m_cov.has_value()) {
204 return std::nullopt;
205 }
206
207 SquareMatrix<3> ipCov;
208 ipCov.template topLeftCorner<2, 2>() =
209 m_cov.value().template topLeftCorner<2, 2>();
210 ipCov.template block<2, 1>(0, 2) =
211 m_cov.value().template block<2, 1>(0, eBoundTime);
212 ipCov.template block<1, 2>(2, 0) =
213 m_cov.value().template block<1, 2>(eBoundTime, 0);
214 ipCov(2, 2) = m_cov.value()(eBoundTime, eBoundTime);
215 return ipCov;
216 }
