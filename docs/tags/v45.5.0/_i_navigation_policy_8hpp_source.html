80 template <typename T, typename... Args>

90 NavigationPolicyState currentState() {

133namespace detail {
134template <typename T>
135concept HasOldInitializeCandidates = requires {
136 requires requires(T policy, const GeometryContext& gctx,
137 const NavigationArguments& args,
138 AppendOnlyNavigationStream& stream, const Logger& logger) {
139 policy.initializeCandidates(gctx, args, stream, logger);
140 };
141};
142
143template <typename T>
144concept HasNewInitializeCandidates = requires {
145 requires requires(T policy, const GeometryContext& gctx,
146 const NavigationArguments& args,
147 NavigationPolicyState& state,
148 AppendOnlyNavigationStream& stream, const Logger& logger) {
149 policy.initializeCandidates(gctx, args, state, stream, logger);
150 };
151};
152
153template <detail::HasOldInitializeCandidates T>
154void oldToNewSignatureAdapter(const void* instance, const GeometryContext& gctx,
155 const NavigationArguments& args,
156 NavigationPolicyState& /*state*/,
157 AppendOnlyNavigationStream& stream,
158 const Logger& logger) {
159 const auto* policy = static_cast<const T*>(instance);
160 policy->initializeCandidates(gctx, args, stream, logger);
161}
162} // namespace detail
163
167template <typename T>
168concept NavigationPolicyConcept = requires {
169 requires std::is_base_of_v<INavigationPolicy, T>;
170
171 // Require either of the signatures to allow backwards compatibility
172 requires(detail::HasOldInitializeCandidates<T> ||
173 detail::HasNewInitializeCandidates<T>);
174};

181class INavigationPolicy {
182 public:
185 static void noopInitializeCandidates(
186 const GeometryContext& /*unused*/, const NavigationArguments& /*unused*/,
187 NavigationPolicyState& /*unused*/,
188 const AppendOnlyNavigationStream& /*unused*/, const Logger& /*unused*/) {
189 // This is a noop
190 }
191
193 virtual ~INavigationPolicy() = default;
194
199 virtual void connect(NavigationDelegate& delegate) const = 0;
200
205 virtual void visit(
206 const std::function<void(const INavigationPolicy&)>& visitor) const {
207 visitor(*this);
208 }
209
216 virtual bool isValid([[maybe_unused]] const GeometryContext& gctx,
217 [[maybe_unused]] const NavigationArguments& args,
218 [[maybe_unused]] NavigationPolicyState& state,
219 const Logger& logger) const {
220 ACTS_VERBOSE("Default navigation policy isValid check. (always true)");
221 return true;
222 }
223
224 struct EmptyState {};
225
231 virtual void createState([[maybe_unused]] const GeometryContext& gctx,
232 [[maybe_unused]] const NavigationArguments& args,
233 NavigationPolicyStateManager& stateManager,
234 const Logger& logger) const {
235 ACTS_VERBOSE(
236 "Default navigation policy state initialization. (empty state)");
237 stateManager.pushState<EmptyState>();
238 }
239
243 virtual void popState(NavigationPolicyStateManager& stateManager,
244 const Logger& logger) const {
245 // By default, we didn't push anything, so we don't need to poop anything
246 ACTS_VERBOSE("Default navigation policy pop state. (pops empty state)");
247 stateManager.popState(); // This will be correct regardless of if a derived
248 // class pushed a concrete state type that's
249 // different from the default EmptyState.
250 }
251
252 protected:
258 template <NavigationPolicyConcept T>
259 void connectDefault(NavigationDelegate& delegate) const {
260 // This cannot be a concept because we use it in CRTP below
261 const auto* self = static_cast<const T*>(this);
262
263 if constexpr (detail::HasNewInitializeCandidates<T>) {
264 delegate.template connect<&T::initializeCandidates>(self);
265 } else {
266 // @TODO: Remove navigation policy signature compatibility eventually
267 delegate.connect(&detail::oldToNewSignatureAdapter<T>, self);
268 }
269 }
270};
