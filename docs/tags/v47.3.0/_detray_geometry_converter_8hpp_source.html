99 template <typename metadata_t>
100 DetrayGeometry<metadata_t> convert(
101 vecmem::memory_resource& mr, const Acts::GeometryContext& gctx,
102 const std::shared_ptr<const Acts::TrackingGeometry>& trackingGeometry,
103 const std::string& detectorName = "") const {
104 using detector_t = detray::detector<metadata_t>;
105
106 if (trackingGeometry == nullptr) {
107 throw std::invalid_argument(
108 "DetrayGeometryConverter: trackingGeometry must not be null");
109 }
110
111 // ── Convert TrackingGeometry → detray payloads ────────────────────────
112 auto payloads = m_cfg.payloadConverter->convertTrackingGeometry(
113 gctx, *trackingGeometry);
114
115 // ── Build detray detector from payloads ───────────────────────────────
116 detray::detector_builder<metadata_t> detectorBuilder{};
117
118 detray::io::geometry_reader::from_payload<detector_t>(detectorBuilder,
119 *payloads.detector);
120
121 if (m_cfg.convertMaterial) {
122 detray::io::homogeneous_material_reader::from_payload<detector_t>(
123 detectorBuilder, *payloads.homogeneousMaterial);
124
125 detray::io::material_map_reader<std::integral_constant<std::size_t, 2>>::
126 from_payload<detector_t>(detectorBuilder,
127 std::move(*payloads.materialGrids));
128 }
129
130 if (m_cfg.convertSurfaceGrids) {
131 detray::io::surface_grid_reader<typename detector_t::surface_type,
132 std::integral_constant<std::size_t, 0>,
133 std::integral_constant<std::size_t, 2>>::
134 template from_payload<detector_t>(detectorBuilder,
135 *payloads.surfaceGrids);
136 }
137
138 if (!detectorName.empty()) {
139 detectorBuilder.set_name(detectorName);
140 } else if (payloads.names.contains(0)) {
141 detectorBuilder.set_name(payloads.names.at(0));
142 }
143
144 DetrayGeometry<metadata_t> result{};
145 result.detector =
146 std::make_shared<detector_t>(detectorBuilder.build(mr, result.names));
147
148 return result;
149 }
150
151 private:
152 Config m_cfg;
153
154 const Acts::Logger& logger() const { return *m_logger; }
155 std::unique_ptr<const Acts::Logger> m_logger;
156};
