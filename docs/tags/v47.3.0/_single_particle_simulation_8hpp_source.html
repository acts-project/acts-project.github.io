66 template <typename generator_t>
67 Acts::Result<SingleParticleSimulationResult> simulate(
68 const Acts::GeometryContext &geoCtx,
69 const Acts::MagneticFieldContext &magCtx, generator_t &generator,
70 const Particle &particle) const {
71 // propagator-related additional types
72 using Actor = detail::SimulationActor<generator_t, decay_t, interactions_t,
73 hit_surface_selector_t>;
74 using Result = typename Actor::result_type;
75 using ActorList = Acts::ActorList<Actor>;
76 using PropagatorOptions =
77 typename propagator_t::template Options<ActorList>;
78
79 // Construct per-call options.
80 PropagatorOptions options(geoCtx, magCtx);
81 options.stepping.maxStepSize = maxStepSize;
82 options.pathLimit = pathLimit;
83 // setup the interactor as part of the propagator options
84 auto &actor = options.actorList.template get<Actor>();
85 actor.generator = &generator;
86 actor.decay = decay;
87 actor.interactions = interactions;
88 actor.selectHitSurface = selectHitSurface;
89 actor.initialParticle = particle;
90
91 if (particle.hasReferenceSurface()) {
92 auto result = propagator.propagate(
93 particle.boundParameters(geoCtx).value(), options);
94 if (!result.ok()) {
95 return result.error();
96 }
97 auto &value = result.value().template get<Result>();
98 return std::move(value);
99 }
100
101 auto result =
102 propagator.propagate(particle.curvilinearParameters(), options);
103 if (!result.ok()) {
104 return result.error();
105 }
106 auto &value = result.value().template get<Result>();
107 return std::move(value);
108 }
109};
