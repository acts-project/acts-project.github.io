129template <typename payload_t>
130[[deprecated("Use makeGrid(const IAxis&) instead")]] std::unique_ptr<IGrid>
131makeGrid(const ProtoAxis& a) {
132 if (a.isAutorange()) {
133 throw std::invalid_argument(
134 "ProtoAxis::makeGrid: Auto-range of the proto axis is not (yet) "
135 "resolved, call setRange() first.");
136 }
137
138 return makeGrid<payload_t>(a.getAxis());
139}
140
150template <typename payload_t>
151[[deprecated(
152 "Use makeGrid(const IAxis&, const IAxis&) instead")]] std::unique_ptr<IGrid>
153makeGrid(const ProtoAxis& a, const ProtoAxis& b) {
154 if (a.isAutorange() || b.isAutorange()) {
155 throw std::invalid_argument(
156 "ProtoAxis::makeGrid: Auto-range of the proto axis is not (yet) "
157 "resolved, call setRange() first.");
158 }
159
160 return makeGrid<payload_t>(a.getAxis(), b.getAxis());
161}
