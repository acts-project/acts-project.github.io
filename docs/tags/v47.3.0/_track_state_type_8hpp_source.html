277 friend std::ostream& operator<<(std::ostream& os,
278 const TrackStateTypeBase& t) {
279 bool first = true;
280 const auto append = [&](const char* name, bool condition) {
281 if (condition) {
282 if (!first) {
283 os << ",";
284 }
285 os << name;
286 first = false;
287 }
288 };
289 append("HasParams", t.hasParameters());
290 if (t.isMaterial()) {
291 append("IsMat", true);
292 } else {
293 append("HasMat", t.hasMaterial());
294 }
295 if (t.isMeasurement()) {
296 append("IsMeas", true);
297 } else if (t.isOutlier()) {
298 append("IsOutlier", true);
299 } else if (t.isHole()) {
300 append("IsHole", true);
301 } else {
302 append("HasMeas", t.hasMeasurement());
303 }
304 append("IsSharedHit", t.isSharedHit());
305 append("IsSplitHit", t.isSplitHit());
306 append("NoExpectedHit", t.hasNoExpectedHit());
307 return os;
308 }

351 void assertConsistency() const {
352 assert(!(test(HasNoExpectedHit) &&
353 (test(HasMeasurement) || test(IsOutlier) || test(IsHole) ||
354 test(IsSharedHit) || test(IsSplitHit))) &&
355 "TrackStateType - NoExpectedHit cannot be set with other "
356 "measurement flags");
357 assert(!(test(IsOutlier) && test(IsHole)) &&
358 "TrackStateType - Outlier and Hole cannot be set simultaneously");
359 assert(
360 !(test(IsHole) && test(HasMeasurement)) &&
361 "TrackStateType - Hole and Measurement cannot be set simultaneously");
362 assert(!(test(IsOutlier) && !test(HasMeasurement)) &&
363 "TrackStateType - Outlier flag requires Measurement flag to be set");
364 }
