98class Barcode {
99 public:
101 using PrimaryVertexId = std::uint16_t;
103 using SecondaryVertexId = std::uint16_t;
105 using ParticleId = std::uint32_t;
107 using GenerationId = std::uint8_t;

113 static constexpr Barcode Invalid() { return Barcode(); }
114
116 constexpr Barcode() = default;
118 constexpr Barcode(const Barcode&) = default;
120 constexpr Barcode(Barcode&&) = default;
123 Barcode& operator=(const Barcode&) = default;
126 Barcode& operator=(Barcode&&) = default;
127
130 bool operator==(const Barcode&) const = default;
131 friend constexpr auto operator<=>(Barcode lhs, Barcode rhs) {
132 return lhs.asVector() <=> rhs.asVector();
133 }
134
138 static constexpr bool isValid(const Barcode& b) { return b != Invalid(); }
141 constexpr bool isValid() const { return isValid(*this); }
142
145 constexpr PrimaryVertexId vertexPrimary() const { return vertexPrimaryID; }
146
149 constexpr SecondaryVertexId vertexSecondary() const {
150 return vertexSecondaryID;
151 }
152
155 constexpr ParticleId particle() const { return particleID; }
156
159 constexpr GenerationId generation() const { return generationID; }
160
163 constexpr SubParticleId subParticle() const { return subParticleID; }
164
167 constexpr std::vector<std::uint32_t> asVector() const {
168 return {vertexPrimary(), vertexSecondary(), particle(), generation(),
169 subParticle()};
170 }

175 [[nodiscard]]
176 constexpr Barcode withVertexPrimary(PrimaryVertexId id) const {
177 Barcode barcode = *this;

211
215 [[nodiscard]]
216 constexpr Barcode withSubParticle(SubParticleId id) const {
217 Barcode barcode = *this;
218 barcode.subParticleID = id;
219 return barcode;

221
225 [[nodiscard]]
226 constexpr Barcode withData(std::span<std::uint32_t> data) {
227 if (data.size() != 5) {
228 throw std::invalid_argument(
229 "Size of the data is " + std::to_string(data.size()) +
230 " but Barcode requires data vector to have exactly 5 elements");
231 }
232
233 Barcode barcode = *this;
234 barcode.vertexPrimaryID = data[0];
235 barcode.vertexSecondaryID = data[1];
236 barcode.particleID = data[2];
237 barcode.generationID = data[3];
238 barcode.subParticleID = data[4];
239 return barcode;
240 }
241
246 Barcode makeDescendant(SubParticleId sub = 0u) const {
247 Barcode barcode = *this;
248 barcode.generationID += 1;
249 barcode.subParticleID = sub;
250 return barcode;
251 }
252
255 constexpr Barcode vertexId() const {
256 // The vertex is identified by primary vertex, secondary vertex, and
257 // generation. The other components are set to 0 so two particle originating
258 // from the same vertex will have the same vertex ID.
259 Barcode barcode = *this;
260 barcode.particleID = 0u;
261 barcode.subParticleID = 0u;
262 return barcode;
263 }
264
267 constexpr Barcode withoutSubparticle() const {
268 // Provide a pseudo-barcode that contains all fields but not the
269 // subparticle counter. This can be used as key in a map to store the
270 // subparticle information
271 Barcode barcode = *this;
272 barcode.subParticleID = 0u;
273 return barcode;
274 }
275
277 friend inline std::ostream& operator<<(std::ostream& os, Barcode barcode) {
278 // extra "+" to ensure printing as a number and not as a character
279 os << "vp=" << +barcode.vertexPrimary()
280 << "|vs=" << +barcode.vertexSecondary() << "|p=" << +barcode.particle()
281 << "|g=" << +barcode.generation() << "|sp=" << +barcode.subParticle();
282 return os;
283 }
284
287 std::size_t hash() const {
288 std::size_t seed = 0;
289 boost::hash_combine(seed, vertexPrimary());
290 boost::hash_combine(seed, vertexSecondary());
291 boost::hash_combine(seed, particle());
292 boost::hash_combine(seed, generation());
293 boost::hash_combine(seed, subParticle());
294
295 return seed;
296 }
297
298 private:
299 PrimaryVertexId vertexPrimaryID = 0u;
300 SecondaryVertexId vertexSecondaryID = 0u;
301 ParticleId particleID = 0u;
302 GenerationId generationID = 0u;
303 SubParticleId subParticleID = 0u;
304};
