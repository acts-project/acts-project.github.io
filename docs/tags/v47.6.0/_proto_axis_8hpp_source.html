153template <typename payload_t>
154[[deprecated(
155 "Use makeGrid(const IAxis&, const IAxis&) instead")]] std::unique_ptr<IGrid>
156makeGrid(const ProtoAxis& a, const ProtoAxis& b) {
157 if (a.isAutorange() || b.isAutorange()) {

160 "resolved, call setRange() first.");
161 }
162
163 return makeGrid<payload_t>(a.getAxis(), b.getAxis());
