105template <Acts::Concepts::arithmetic T>
106class Tensor {
107 public:
109 using Shape = std::array<std::size_t, 2>;
110
115 static Tensor Create(Shape shape, const ExecutionContext &execContext) {
116 auto ptr = detail::createTensorMemory(shape[0] * shape[1] * sizeof(T),
117 execContext);
118 return Tensor(shape, std::move(ptr), execContext);
119 }
120
126 Tensor clone(const ExecutionContext &to) const {
127 auto clonedPtr = detail::cloneTensorMemory(m_ptr, nbytes(), m_device, to);
128 return Tensor(m_shape, std::move(clonedPtr), to);
129 }
130
133 T *data() { return static_cast<T *>(m_ptr.get()); }
134
137 const T *data() const { return static_cast<const T *>(m_ptr.get()); }
138
141 Shape shape() const { return m_shape; }
142
145 std::size_t size() const { return m_shape[0] * m_shape[1]; }
146
149 std::size_t nbytes() const { return size() * sizeof(T); }
150
153 Device device() const { return m_device; }
154
155 private:
156 Tensor(Shape shape, detail::TensorPtr ptr, const ExecutionContext &ctx)
157 : m_shape(shape), m_ptr(std::move(ptr)), m_device(ctx.device) {}
158
159 Shape m_shape{};
160 detail::TensorPtr m_ptr;
161 Device m_device{};
162};

175template <Acts::Concepts::arithmetic T>
176void checkMaskCompatibility(const Tensor<T> &tensor, const Tensor<bool> &mask,
177 std::size_t matchDim) {
178 if (tensor.shape()[matchDim] != mask.shape()[0]) {
179 throw std::invalid_argument(
180 "tensor shape[" + std::to_string(matchDim) +
181 "]=" + std::to_string(tensor.shape()[matchDim]) +
182 " does not match mask length " + std::to_string(mask.shape()[0]));
183 }
184 if (mask.shape()[1] != 1) {
185 throw std::invalid_argument("mask must have shape [N, 1], got [N, " +
186 std::to_string(mask.shape()[1]) + "]");
187 }
188 if (tensor.device() != mask.device()) {
189 throw std::invalid_argument(
190 "tensor and mask must reside on the same device");
191 }
192}

194} // namespace detail
