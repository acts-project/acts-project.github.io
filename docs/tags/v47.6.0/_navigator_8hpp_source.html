165 // Navigation on volume level
167 NavigationBoundaries navBoundaries = {};
169 std::optional<std::size_t> navBoundaryIndex;
170
171 // Navigation candidates (portals and surfaces together). The candidates
172 // live in `stream` (sorted by path length); the navigator works through
173 // them by index without copying them out.
175 std::optional<std::size_t> navCandidateIndex;
179 double navCandidatesFarLimit = std::numeric_limits<double>::max();
180
182 // They are stored as a pair of surface pointer
185 std::vector<std::pair<const Surface*, bool>> freeCandidates{};

189 NavigationTarget& navSurface() {

195 NavigationTarget& navLayer() { return navLayers.at(navLayerIndex.value()); }

199 NavigationTarget& navBoundary() {
200 return navBoundaries.at(navBoundaryIndex.value());
201 }
202
205 NavigationTarget& navCandidate() {
206 return stream.candidates().at(navCandidateIndex.value());
207 }

236 std::vector<const Surface*> freeSurfaces;

239 void resetAfterLayerSwitch() {
240 navSurfaces.clear();
241 navSurfaceIndex.reset();
242 }
243
245 void resetAfterVolumeSwitch() {
246 resetAfterLayerSwitch();
247
248 navLayers.clear();
249 navLayerIndex.reset();
250 navBoundaries.clear();
251 navBoundaryIndex.reset();
252 navCandidateIndex.reset();
253
254 currentLayer = nullptr;
255
256 policyStateManager.reset();
257 policyStateIsDefault = true;
258 }
259
261 void resetForRenavigation() {
262 resetAfterVolumeSwitch();
263
264 currentVolume = nullptr;
265 currentSurface = nullptr;
266
267 navigationBreak = false;
268 navigationStage = Stage::initial;
269
270 // Set the surface reached switches back to false
271 std::ranges::for_each(freeCandidates,
272 [](std::pair<const Surface*, bool>& freeSurface) {
273 freeSurface.second = false;
274 });
275
276 stream.reset();
277 }
278 };
