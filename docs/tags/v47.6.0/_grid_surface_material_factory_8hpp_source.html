30template <typename material_accessor_t>
31std::unique_ptr<
32 IGridSurfaceMaterial<typename material_accessor_t::grid_value_type>>
33create1D(
34 const IAxis& axis, material_accessor_t&& materialAccessor,
35 GridAccess::BoundToGridLocal1DimDelegate boundToGridLocal,
36 GridAccess::GlobalToGridLocal1DimDelegate globalToGridLocal,
37 const std::vector<typename material_accessor_t::grid_value_type>& payload) {
38 // Visit the axis type and create the grid surface material
39 auto ism =
40 axis.visit([&]<typename AxisType>(const AxisType& concreteAxis)
41 -> std::unique_ptr<IGridSurfaceMaterial<
42 typename material_accessor_t::grid_value_type>> {
43 using GridType =
44 Grid<typename material_accessor_t::grid_value_type, AxisType>;
45 return std::make_unique<
46 GridSurfaceMaterialT<GridType, material_accessor_t>>(
47 GridType(concreteAxis),
48 std::forward<material_accessor_t>(materialAccessor),
49 std::move(boundToGridLocal), std::move(globalToGridLocal));
50 });
51 // Fill it via the grid view
52 AnyGridView<typename material_accessor_t::grid_value_type> gv =
53 ism->gridView();
54 auto indices = gv.multiAxisAny().getNBinsAny();
55 for (std::size_t i0 = 0; i0 < indices[0]; ++i0) {
56 // Offset comes from overflow/underflow bin
57 gv.atLocalBins({i0 + 1u}) = payload[i0];
58 }
59 return ism;
60}

73template <typename material_accessor_t>
74std::unique_ptr<
75 IGridSurfaceMaterial<typename material_accessor_t::grid_value_type>>
76create2D(
77 const IAxis& axis0, const IAxis& axis1,
78 material_accessor_t&& materialAccessor,
79 GridAccess::BoundToGridLocal2DimDelegate boundToGridLocal,
80 GridAccess::GlobalToGridLocal2DimDelegate globalToGridLocal,
81 const std::vector<
82 std::vector<typename material_accessor_t::grid_value_type>>& payload) {
83 // Validate axis compatibility
84 auto ism =
85 axis0.visit([&]<typename AxisTypeA>(const AxisTypeA& axisA)
86 -> std::unique_ptr<IGridSurfaceMaterial<
87 typename material_accessor_t::grid_value_type>> {
88 return axis1.visit(
89 [&]<typename AxisTypeB>(const AxisTypeB& axisB)
90 -> std::unique_ptr<IGridSurfaceMaterial<
91 typename material_accessor_t::grid_value_type>> {
92 using GridType =
93 Grid<typename material_accessor_t::grid_value_type, AxisTypeA,
94 AxisTypeB>;
95 return std::make_unique<
96 GridSurfaceMaterialT<GridType, material_accessor_t>>(
97 GridType(axisA, axisB),
98 std::forward<material_accessor_t>(materialAccessor),
99 std::move(boundToGridLocal), std::move(globalToGridLocal));
100 });
101 });
102
103 // Fill it via the grid view
104 AnyGridView<typename material_accessor_t::grid_value_type> gv =
105 ism->gridView();
106 auto indices = gv.multiAxisAny().getNBinsAny();
107 for (std::size_t i0 = 0; i0 < indices[0]; ++i0) {
108 for (std::size_t i1 = 0; i1 < indices[1]; ++i1) {
109 // Offset comes from overflow/underflow bin
110 gv.atLocalBins({i0 + 1, i1 + 1}) = payload[i0][i1];
111 }
112 }
113
114 return ism;
115}
