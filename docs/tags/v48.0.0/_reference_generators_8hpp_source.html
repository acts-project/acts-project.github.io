45struct ShiftedCenterReferenceGenerator : public IReferenceGenerator {
47 Transform3 globalToLocal{Transform3::Identity()};
48
50 AxisDirection layerDir{AxisDirection::AxisY};
51
53 AxisDirection shiftDir{AxisDirection::AxisX};
54
57 double layerLow{0.};
58
61 double layerPitch{0.};
62
65 std::vector<double> shifts;
66
73 const std::vector<Vector3> references(const GeometryContext& gctx,
74 const Surface& surface) const override {
75 if (shifts.empty()) {
76 throw std::invalid_argument(
77 "The ShiftedCenetrReference generator called without shifts vector.");
78 }
79
80 Vector3 cLocal = globalToLocal * surface.center(gctx);
81
82 const double lc = VectorHelpers::cast(cLocal, layerDir);
83 const auto bin = static_cast<std::size_t>(
84 std::floor((lc - layerLow) / layerPitch)); // 0-based layer bin
85
86 cLocal[toUnderlying(shiftDir)] += shifts.at(bin); // shift along shiftDir
87 return {globalToLocal.inverse() * cLocal};
88 }
89};
