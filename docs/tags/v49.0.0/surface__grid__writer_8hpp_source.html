66 static payload_type to_payload_impl(
67 const detector_t& det, const typename detector_t::name_map& /*unused*/) {
68 using surface_desc_t = typename detector_t::surface_type;
69
70 payload_type grids_data;
71
72 for (const auto& vol_desc : det.volumes()) {
73 // Links to all acceleration data structures in the volume
74 const auto& multi_link = vol_desc.accel_link();
75
76 // How to convert the surface descriptors in the grid
77 auto sf_converter =
78 [&vol_desc = std::as_const(vol_desc)](const surface_desc_t& sf_desc) {
79 return vol_desc.to_local_sf_index(sf_desc.index());
80 };
81
82 // Start a 1, because the first acceleration structure is always
83 // the brute force method
84 for (dindex i = 1u; i < multi_link.size(); ++i) {
85 const auto& acc_link = multi_link[i];
86 // Don't look at empty links
87 if (acc_link.is_invalid()) {
88 continue;
89 }
90
91 // Generate the payload
92 grid_writer_t::to_payload_impl(det.accelerator_store(), acc_link,
93 vol_desc.index(), vol_desc.index(),
94 grids_data, sf_converter);
95 }
96 }
97
98 return grids_data;
99 }
100};
