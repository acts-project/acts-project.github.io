74 template <typename mask_t>
75 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
76 const trajectory_type<algebra_t> &ray,
77 const dtransform3D<algebra_t> & /*trf*/, const mask_t &mask,
78 const dscalar<algebra_t> overstep_tol = 0.f) const {
79 using scalar_t = dscalar<algebra_t>;
80 using point3_t = dpoint3D<algebra_t>;
81 using vector3_t = dvector3D<algebra_t>;
82
83 const scalar_t r{mask[mask_t::shape::e_r]};
84 constexpr scalar_t inv{detail::invalid_value<dvalue<algebra_t>>()};
85
86 const point3_t &ro = ray.pos();
87 const vector3_t &rd = ray.dir();
88 scalar_t path{inv};
89
90 const scalar_t rd_perp_2{rd[0] * rd[0] + rd[1] * rd[1]};
91
92 // The ray is parallel to the cylinder axis (z-axis)...
93 if (rd_perp_2 < std::numeric_limits<scalar_t>::epsilon()) [[unlikely]] {
94 return {};
95 }
96
97 // ...otherwise, two solutions should exist, if the descriminator is
98 // greater than zero
99 //
100 // rad_diff here is defined as:
101 //
102 // r * r - (ro[0] * ro[0] + ro[1] * ro[1])
103 //
104 // But this runs an enormous risk of catastrophic cancellation, which we
105 // would like to avoid.
106 //
107 // To compute this as accurately as possible without using any double
108 // precision operations, we'll use two core ideas:
109 //
110 // 1. The Sterbenz lemma: if x and y are floats of the same sign and
111 // y/2 <= x <= 2y, then x - y is exact and has no rounding error.
112 // 2. FMA hardware has no rounding error internally, so if we compute
113 // a = x * y we get a normal, rounded result. If we then compute
114 // e = x * y - a using fma(x, y, -a), we get the exact error of a.
115 //
116 // We can use 2) to define an exact squaring algorithm, where to square x
117 // we compute first a = x * x and then the error e = fma(x, x, -a).
118 // Clearly, the exact square of x is then just a + e. We'll call this
119 // procedure a, e = fma_square(x).
120 //
121 // To compute rad_diff, we can thus compute:
122 //
123 // rsq_a, rsq_e = fma_square(r)
124 // ro0sq_a, ro0sq_e = fma_square(ro[0])
125 // ro1sq_a, ro1sq_e = fma_square(ro[1])
126 //
127 // Such that:
128 //
129 // rad_diff = (rsq_a + rsq_e) - ((ro0sq_a + ro0sq_e) + (ro1sq_a + ro1sq_e))
130 //
131 // rad_diff = (rsq_a + rsq_e) - (ro0sq_a + ro0sq_e) - (ro1sq_a + ro1sq_e)
132 //
133 // However, this is not exact by the Sterbenz lemma 1). Full exactness is
134 // not possible, but we can still try to minimize error. We know that the
135 // magnitudes of the errors will be much smaller than the magnitudes of the
136 // approximated squares, so we can first rearrange:
137 //
138 // rad_diff = (rsq_a - ro0sq_a - ro1sq_a) + (rsq_e - ro0sq_e - ro1sq_e)
139 //
140 // What matters now is the way we associate the subtractions inside the
141 // two sets of brackets. Arguing informally, we are more likely to get
142 // exact (or at least less erroneous) results if the scale of the two
143 // numbers are as close as possible, so we will first find the bigger
144 // and the smaller value of ro:
145 //
146 // robig = max(ro[0], ro[1])
147 // rosmall = min(ro[0], ro[1])
148 //
149 // Such that we can express the most accurate result possible:
150 //
151 // rad_diff = ((rsq_a - robigsq_a) - rosmallsq_a) +
152 // ((rsq_e - robigsq_e) - rosmallsq_e)
153 const auto fma_square = [](const scalar_t x) {
154 scalar_t a = x * x;
155 scalar_t e = math::fma(x, x, -a);
156 return std::tuple<scalar_t, scalar_t>(a, e);
157 };
158
159 const auto [rsq_a, rsq_e] = fma_square(r);
160 const auto [robigsq_a, robigsq_e] =
161 fma_square(math::max(math::abs(ro[0]), math::abs(ro[1])));
162 const auto [rosmlsq_a, rosmlsq_e] =
163 fma_square(math::min(math::abs(ro[0]), math::abs(ro[1])));
164
165 const scalar_t rad_diff =
166 ((rsq_a - robigsq_a) - rosmlsq_a) + ((rsq_e - robigsq_e) - rosmlsq_e);
167
168 // Only calculate the path, when not already on surface
169 if (math::fabs(rad_diff) > 1.f * unit<scalar_t>::um) {
170 const scalar_t rd_perp_inv_2{1.f / rd_perp_2};
171 const scalar_t k{-rd_perp_inv_2 * (ro[0] * rd[0] + ro[1] * rd[1])};
172 const scalar_t c{rd_perp_inv_2 * rad_diff};
173 const scalar_t discr{c + k * k};
174
175 // No intersection found
176 if (discr < 0.f) [[unlikely]] {
177 return {};
178 }
179
180 // The options are k +- sqrt(discr). Clearly the square root of the
181 // discriminant is always positive. This means that if k is positive,
182 // then k + sqrt(discr) will have a greater absolute value than
183 // k - sqrt(discr). If k is negative, the logic is reversed.
184 const scalar_t sqrt_discr{math::sqrt(discr)};
185 const scalar_t sfar{k + math::copysign(sqrt_discr, k)};
186 const scalar_t snear{sfar != 0.f ? -c / sfar : sfar};
187
188 path = snear >= overstep_tol ? snear : sfar;
189 } else {
190 path = rad_diff;
191 }
192
193 if (path < overstep_tol) {
194 path = inv;
195 }
196
197 // Only need the global z-component for the mask check
198 return {path, point_type{inv, ro[2] + path * rd[2]}};
199 }
200};
