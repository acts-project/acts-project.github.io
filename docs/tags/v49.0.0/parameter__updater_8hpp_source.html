368 template <typename propagator_state_t>
369 DETRAY_HOST_DEVICE result operator()(state& updater_state,
370 propagator_state_t& propagation) const {
371 const auto& navigation = propagation.navigation();
372
373 // Do covariance transport only when the track is on surface
374 if (!(navigation.is_on_sensitive() ||
375 navigation.encountered_sf_material())) {
376 return {};
377 }
378 assert(navigation.is_on_surface());
379
380 // Current track position and transport jacobian
381 auto& stepping = propagation.stepping();
382
383 // Destination surface (current)
384 const auto dest_sf = navigation.current_surface();
385
386 // Bound track params of departure surface (not yet updated)
387 auto& departure_params = updater_state.bound_params();
388
389 // Result to be passed on to observing actors
390 result res{actor::status::e_notify, departure_params};
391
392 // Covariance is transported only when the departure surface is an
393 // actual tracking surface (i.e. this disables the covariance
394 // transport from curvilinear frames).
395 if (!departure_params.surface_link().is_invalid()) {
396 DETRAY_DEBUG_HOST(
397 "Actor: Departure surface: " << departure_params.surface_link());
398
399 // There is no need to transport the identical/initial parameters
400 assert(!dest_sf.identifier().is_invalid());
401 if (departure_params.surface_link() == dest_sf.identifier()) {
402 DETRAY_VERBOSE_HOST_DEVICE(
403 "Actor: On initial surface (%d), parameter transport not required",
404 dest_sf.index());
405
406 // Notify observers?
407 if (!updater_state.m_notify_on_initial) {
408 res.status = actor::status::e_unknown;
409 }
410
411 return res;
412 }
413
414 DETRAY_DEBUG_HOST("Actor: Destination surface: " << dest_sf.identifier());
415 DETRAY_VERBOSE_HOST_DEVICE(
416 "Actor: Transport track covariance to surface %d", dest_sf.index());
417
418 DETRAY_DEBUG_HOST("Actor: Departure bound param.: " << departure_params);
419
420 // Transport the covariance
421 const bound_matrix<algebra_t> propagation_step_jacobian =
422 get_full_jacobian(propagation, departure_params);
423
424 // Update the full Jacobian, if required
425 if (math::fabs(stepping.path_length()) > 0.f) {
426 if constexpr (state_t::stores_full_jacobian) {
427 if (updater_state.has_full_jacobian()) {
428 updater_state.set_full_jacobian(propagation_step_jacobian *
429 updater_state.full_jacobian());
430 }
431 }
432
433 // Reset transport Jacobian to identity matrix
434 stepping.reset_transport_jacobian();
435 }
436
437 // Copy old covariance in order to update new covariance in place
438 const bound_matrix_type old_cov = departure_params.covariance();
439 bound_matrix_type& new_cov = departure_params.covariance();
440
441 detray::detail::transport_covariance_to_bound_impl(
442 old_cov, propagation_step_jacobian, new_cov);
443 } else {
444 DETRAY_VERBOSE_HOST_DEVICE("Actor: Departure surface link invalid");
445 }
446
447 DETRAY_VERBOSE_HOST_DEVICE(
448 "Actor: Convert track parameter vector to local for surface %d",
449 dest_sf.index());
450
451 // Get the bound parameters at the destination surface
452 res.destination_params().set_parameter_vector(
453 dest_sf.free_to_bound_vector(propagation.context(), stepping()));
454
455 // Set new surface link
456 res.destination_params().set_surface_link(dest_sf.identifier());
457
458 assert(!res.destination_params().is_invalid());
459
460 DETRAY_DEBUG_HOST("Actor: Transported bound param.:\n"
461 << res.destination_params());
462
463 return res;
464 }

471 template <typename propagator_state_t>
472 DETRAY_HOST_DEVICE constexpr bound_matrix_type get_full_jacobian(
473 propagator_state_t& propagation,
474 const bound_track_parameters_type& departure_params) const {
475 // Map the surface shapes of the detector down to the common frames
476 using detector_t = typename propagator_state_t::detector_type;
477 using frame_registry_t =
478 types::mapped_registry<typename detector_t::masks, select_frame>;
479
480 const auto& gctx = propagation.context();
481 const auto& stepping = propagation.stepping();
482 const auto& navigation = propagation.navigation();
483
484 // Our goal here is to compute the full Jacobian, which is given as:
485 //
486 // J_full = J_F2B * (D + I) * J_transport * J_B2F
487 //
488 // The transport Jacobian is given, but the free-to-bound and
489 // bound-to-free matrices as well as the derivative matrix $D$ need
490 // to be computed still. In order to avoid full-rank matrix
491 // multiplications, we use the known substructure of the
492 // aforementioned matrices in order to perform fewer operations. We
493 // describe in detail the mathematics performed throughout the
494 // remainder of this function.
495
496 // Departure surface
497 const tracking_surface dep_sf{navigation.detector(),
498 departure_params.surface_link()};
499
500 // Destination Surface
501 const tracking_surface dest_sf = navigation.current_surface();
502
503 // Free track params of departure surface
504 const free_track_parameters<algebra_t> dep_free_params =
505 dep_sf.bound_to_free_vector(gctx, departure_params);
506
507 // Free track params of destination surface
508 const free_track_parameters<algebra_t>& dest_free_params = stepping();
509
510 // First, we will compute the bound-to-free Jacobian, which is given
511 // by three sub-Jacobians. In particular, the matrix has an 8x6
512 // shape and looks like this:
513 //
514 // l0 l1 phi th q/p t
515 // px [[ A, A, B, B, 0, 0],
516 // py [ A, A, B, B, 0, 0],
517 // pz [ A, A, B, B, 0, 0],
518 // t [ 0, 0, 0, 0, 0, 1],
519 // dx [ 0, 0, C, C, 0, 0],
520 // dy [ 0, 0, C, C, 0, 0],
521 // dz [ 0, 0, 0, C, 0, 0],
522 // q/p [ 0, 0, 0, 0, 1, 0]]
523 //
524 // Note that we thus only have 17 out of 48 non-trivial matrix
525 // elements.
526 //
527 // In this matrix, A represents the d(pos)/d(loc) submatrix, B is
528 // the d(pos)/d(angle) submatrix, and C is the d(dir)/d(angle)
529 // submatrix, all of which are 3x2 in size. Also, A and B depend on
530 // the frame type while C is computed in the same way for all
531 // frames. Finally, note that submatrix B is the zero matrix for
532 // most frame types.
533 const auto& dep_trf3 = dep_sf.transform(gctx);
534 const dmatrix<algebra_t, 3, 2> b2f_dpos_dloc =
535 types::visit<frame_registry_t, get_bound_to_free_dpos_dloc_visitor>(
536 dep_sf.shape_id(), dep_trf3, dep_free_params);
537
538 const dmatrix<algebra_t, 3, 2> b2f_ddir_dangle =
539 detail::jacobian_engine<algebra_t>::
540 bound_to_free_jacobian_submatrix_ddir_dangle(departure_params);
541
542 const dmatrix<algebra_t, 3, 2> b2f_dpos_dangle =
543 types::visit<frame_registry_t, get_bound_to_free_dpos_dangle_visitor>(
544 dep_sf.shape_id(), dep_trf3, dep_free_params, b2f_ddir_dangle);
545
546 // Next, we compute the derivative which is defined as the outer
547 // product of the two 8x1 vectors representing the path to free
548 // derivative and the free to path derivative. Thus, it is the
549 // product of the following two vectors:
550 //
551 // px [[ A], [[ B],^T
552 // py [ A], [ B],
553 // pz [ A], [ B],
554 // t [ 0]] [ 0],
555 // dx [ A], [ C],
556 // dy [ A], [ C],
557 // dz [ A], [ C],
558 // q/p [ A], [ 0]]
559 //
560 // Where A is frame independent and non-zero, B is frame-dependent
561 // and non-zero, while C is frame-dependent and non-zero only for
562 // line frames.
563 const dpoint3D<algebra_t> dest_glob_pos{dest_free_params.pos()};
564 const dvector3D<algebra_t> dest_glob_dir{dest_free_params.dir()};
565
566 auto vol = navigation.current_volume();
567 const auto vol_mat_ptr =
568 vol.has_material() ? vol.material_parameters(dest_glob_pos) : nullptr;
569
570 const auto path_to_free_derivative =
571 detail::jacobian_engine<algebra_t>::path_to_free_derivative(
572 dest_glob_dir, stepping.dtds(), stepping.dqopds(vol_mat_ptr));
573
574 const auto free_to_path_derivative = dest_sf.free_to_path_derivative(
575 gctx, dest_glob_pos, dest_glob_dir, stepping.dtds());
576
577 // Now, we compute the free-to-bound Jacobian which is of size 6x8
578 // and has the following structure:
579 //
580 // px py pz t dx dy dz q/p
581 // l0 [[ A, A, A, 0, 0, 0, 0, 0],
582 // l1 [ A, A, A, 0, 0, 0, 0, 0],
583 // phi [ 0, 0, 0, 0, B, B, 0, 0],
584 // th [ 0, 0, 0, 0, B, B, B, 0],
585 // q/p [ 0, 0, 0, 0, 0, 0, 0, 1],
586 // t [ 0, 0, 0, 1, 0, 0, 0, 0]]
587 //
588 // Thus, the number of non-trivial elements is only 11 out of the 48
589 // matrix elements. Also, submatrix A depends on the frame type
590 // while submatrix B is the same for all frame types.
591 const dmatrix<algebra_t, 2, 3> f2b_dloc_dpos =
592 types::visit<frame_registry_t, get_free_to_bound_dloc_dpos_visitor>(
593 dest_sf.shape_id(), dest_sf.transform(gctx), stepping);
594
595 const dmatrix<algebra_t, 2, 3> f2b_dangle_ddir = detail::jacobian_engine<
596 algebra_t>::free_to_bound_jacobian_submatrix_dangle_ddir(dest_glob_dir);
597
598 // Finally, we can use our Sympy-generated full Jacobian computation
599 // and return its result.
600 bound_matrix_type full_jacobian;
601
602 if constexpr (std::decay_t<propagator_state_t>::stepper_uses_gradient) {
603 detail::update_full_jacobian_with_gradient_impl(
604 stepping.internal_transport_jacobian(), b2f_dpos_dloc,
605 b2f_ddir_dangle, b2f_dpos_dangle, path_to_free_derivative,
606 free_to_path_derivative, f2b_dloc_dpos, f2b_dangle_ddir,
607 full_jacobian);
608 } else {
609 detail::update_full_jacobian_without_gradient_impl(
610 stepping.internal_transport_jacobian(), b2f_dpos_dloc,
611 b2f_ddir_dangle, b2f_dpos_dangle, path_to_free_derivative,
612 free_to_path_derivative, f2b_dloc_dpos, f2b_dangle_ddir,
613 full_jacobian);
614 }
615
616 return full_jacobian;
617 }

631template <concepts::algebra algebra_t, typename state_t>
632struct parameter_setter_base : base_actor {
634 using state = state_t;
635
641 template <typename propagator_state_t>
642 DETRAY_HOST_DEVICE void operator()(
643 state& updater_state, propagator_state_t& propagation,
644 const parameter_transporter_result<algebra_t>& res) const {
645 // Update the track parameters only if necessary
646 if (res.status != actor::status::e_success &&
647 !updater_state.m_always_update) {
648 return;
649 }
650
651 const auto& navigation = propagation.navigation();
652 auto& stepping = propagation.stepping();
653
654 // One of the transporter observers might have exited the navigation
655 assert(navigation.is_on_surface() || !navigation.is_alive());
656
657 DETRAY_VERBOSE_HOST_DEVICE("Actor: Update the track parameters");
658
659 // Updated bound track parameters
660 const bound_track_parameters<algebra_t>& bound_params =
661 res.destination_params();
662
663 // Update free params after bound params were changed by actors
664 const tracking_surface dest_sf{navigation.detector(),
665 bound_params.surface_link()};
666 stepping() =
667 dest_sf.bound_to_free_vector(propagation.context(), bound_params);
668
669 assert(!stepping().is_invalid());
670
671 // Track pos/dir is not always known precisely: adjust navigation
672 // tolerances according to bound covariance
673 if (const noise_cfg& cfg = updater_state.noise_estimation_cfg();
674 cfg.estimate_scattering_noise) {
675 detail::estimate_external_mask_tolerance(
676 bound_params, propagation,
677 static_cast<dscalar<algebra_t>>(cfg.n_stddev), cfg.accumulated_error);
678 }
679
680 DETRAY_DEBUG_HOST("Actor: Updated bound param.:\n" << bound_params);
681 }
