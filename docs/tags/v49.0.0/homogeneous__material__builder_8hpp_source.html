30template <typename detector_t>
31class homogeneous_material_builder final : public volume_decorator<detector_t> {
32 public:
34 using material_id = typename detector_t::material::id;
36 using scalar_type = dscalar<typename detector_t::algebra_type>;
37
38 static_assert(concepts::detector<detector_t>);
39
41 DETRAY_HOST
42 explicit homogeneous_material_builder(
43 std::unique_ptr<volume_builder_interface<detector_t>> vol_builder)
44 : volume_decorator<detector_t>(std::move(vol_builder)) {
45 DETRAY_VERBOSE_HOST(
46 "Add hom. material builder to volume: " << this->name());
47
48 static_assert(concepts::has_material_slabs<detector_t> ||
49 concepts::has_material_rods<detector_t>,
50 "No homogeneous surface material in detector type");
51 }
52
58 DETRAY_HOST
59 void add_surfaces(
60 std::shared_ptr<surface_factory_interface<detector_t>> sf_factory,
61 typename detector_t::geometry_context ctx = {}) override {
62 DETRAY_VERBOSE_HOST("Add [material] surface factory:");
63
64 // If the factory also holds surface data, call base volume builder
65 volume_decorator<detector_t>::add_surfaces(sf_factory, ctx);
66
67 // Add material
68 auto mat_factory =
69 std::dynamic_pointer_cast<homogeneous_material_factory<detector_t>>(
70 sf_factory);
71 if (mat_factory) {
72 DETRAY_VERBOSE_HOST("-> found decoration: " << DETRAY_TYPENAME(
73 homogeneous_material_factory<detector_t>));
74 (*mat_factory)(this->surfaces(), m_materials);
75 return;
76 }
77 auto mat_generator =
78 std::dynamic_pointer_cast<homogeneous_material_generator<detector_t>>(
79 sf_factory);
80 if (mat_generator) {
81 DETRAY_VERBOSE_HOST("-> found decoration: " << DETRAY_TYPENAME(
82 homogeneous_material_generator<detector_t>));
83 (*mat_generator)(this->surfaces(), m_materials);
84 return;
85 }
86
87 if (!mat_factory && !mat_generator) {
88 DETRAY_VERBOSE_HOST("No material found in this surface factory");
89 DETRAY_VERBOSE_HOST("-> Built non-material surfaces");
90 }
91 }

99 DETRAY_HOST
100 auto build(detector_t &det, const volume_builder_options &opt,
101 typename detector_t::geometry_context ctx = {}) ->
102 typename detector_t::volume_type * override {
103 DETRAY_VERBOSE_HOST("Build homogeneous material...");
104 DETRAY_DEBUG_HOST("-> n_surfaces=" << this->surfaces().size());
105
106 if constexpr (concepts::has_material_slabs<detector_t>) {
107 add_material<material_id::e_material_slab>(det, opt);
108 }
109 if constexpr (concepts::has_material_rods<detector_t>) {
110 add_material<material_id::e_material_rod>(det, opt);
111 }
112
113 m_materials.clear_all();
114
115 DETRAY_VERBOSE_HOST(
116 "Successfully built homogeneous material for volume: " << this->name());
117
118 // Call the underlying volume builder(s) and give the volume to the
119 // next decorator
120 return volume_decorator<detector_t>::build(det, opt, ctx);
121 }
122
123 private:
128 template <material_id mat_id>
129 DETRAY_HOST void add_material(detector_t &det,
130 const volume_builder_options &opt) {
131 auto &local_coll = m_materials.template get<mat_id>();
132 if (local_coll.empty()) {
133 return;
134 }
135
136 // non-const access
137 auto &det_coll = det._materials.template get<mat_id>();
138 const auto offset{static_cast<dindex>(det_coll.size())};
139
140 // Conmpute the material indices
141 std::vector<dindex> global_idx;
142 global_idx.reserve(local_coll.size());
143
144 if (opt.deduplicate()) {
145 DETRAY_VERBOSE_HOST("-> Deduplicate homogeneous material");
146
147 // Global index for every volume local material entry
148 for (const auto &mat : local_coll) {
149 DETRAY_DEBUG_HOST("Building material " << mat);
150 // Insert into map
151 const auto coll_size{static_cast<dindex>(det_coll.size())};
152 dindex new_idx{coll_size};
153
154 // Test only against the material that is already in the detector
155 // Any duplication within the new data will be resolved the same way
156 for (dindex i = 0u; i < coll_size; ++i) {
157 if (mat == det_coll.at(i)) {
158 DETRAY_DEBUG_HOST("Found identical material grid at index "
159 << i << ". Deduplicating...");
160 new_idx = i;
161 break;
162 }
163 }
164
165 // No duplicate was found, append new material
166 if (new_idx == coll_size) {
167 DETRAY_DEBUG_HOST("Adding to detector... ");
168 det_coll.push_back(mat);
169 }
170
171 // Save index
172 DETRAY_DEBUG_HOST(" -> Material index: " << new_idx);
173 global_idx.push_back(new_idx);
174 }
175 } else { // Do not deduplicate
176 // Append all material
177 det._materials.insert(local_coll);
178 }
179
180 // Update all surface links
181 for (auto &sf : this->surfaces()) {
182 DETRAY_DEBUG_HOST("-> sf = " << sf);
183
184 if (sf.material().id() == mat_id) {
185 if (opt.deduplicate()) {
186 assert(!global_idx.empty());
187 sf.material().set_index(global_idx.at(sf.material().index()));
188 } else {
189 sf.update_material(offset);
190 }
191 DETRAY_DEBUG_HOST("-> material link now: " << sf.material());
192 }
193 }
194
195 DETRAY_VERBOSE_HOST("-> Appended " << det_coll.size() - offset << " of "
196 << local_coll.size()
197 << " entries of type " << mat_id
198 << " to detector materials");
199 }
200
201 // Material container for this volume
202 typename detector_t::material_container m_materials{};
203};
