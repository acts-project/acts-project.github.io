70 output_file_converter_interface(const io::format f, const std::string& ext)
71 : m_format{f}, m_file_extension{ext} {};
72
74 io::format format() const { return m_format; }
75
77 std::string_view file_extension() const { return m_file_extension; }
78
80 std::ios_base::openmode open_mode() const { return m_mode; }
81
83 void open_mode(const std::ios_base::openmode mode) { m_mode = mode; }
84
93 virtual std::vector<std::string> write_to_file(
94 const detector_payload& payload, const std::filesystem::path& path,
95 const std::ios_base::openmode mode) = 0;
96
100 void from_payload(const detector_payload& payload,
101 const std::any& output_data) override {
102 write_to_file(payload, get_file_path(output_data), m_mode);
103 };
104
105 private:
107 std::filesystem::path get_file_path(const std::any& output_data) {
108 std::filesystem::path file_path{"./"};
109 try {
110 file_path = std::any_cast<std::filesystem::path>(output_data);
111 } catch (const std::bad_any_cast& e) {
112 DETRAY_FATAL_HOST(
113 "Unknown output data type in "
114 << m_file_extension
115 << " file converter! Should be 'std::filesystem::path'");
116 throw;
117 }
118
119 return file_path;
120 }
121
123 io::format m_format;
125 std::string m_file_extension;
127 std::ios_base::openmode m_mode{std::ios_base::out | std::ios_base::binary};
128};
