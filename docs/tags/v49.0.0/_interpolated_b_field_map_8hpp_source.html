252 explicit InterpolatedBFieldMap(Config cfg) : m_cfg{std::move(cfg)} {
253 if (!m_cfg.transformDependsOnPosition &&
254 !std::is_same_v<FieldType, Vector3>) {
255 throw std::invalid_argument(
256 "InterpolatedBFieldMap: transformDependsOnPosition = false requires "
257 "Vector3 grid values");
258 }
259 typename Grid::index_t minBin{};
260 minBin.fill(1);
261 m_lowerLeft = m_cfg.grid.multiAxis().getLowerLeftBinEdge(minBin);
262 m_upperRight = m_cfg.grid.multiAxis().getLowerLeftBinEdge(
263 m_cfg.grid.multiAxis().getNBins());
264 }

274 Result<FieldCell> getFieldCell(const Vector3& position) const {
275 const auto& gridPosition = m_cfg.transformPos(position);
276 const auto& indices =
277 m_cfg.grid.multiAxis().getLocalBinsFromPoint(gridPosition);
278 const auto& lowerLeft = m_cfg.grid.multiAxis().getLowerLeftBinEdge(indices);
279 const auto& upperRight =
280 m_cfg.grid.multiAxis().getUpperRightBinEdge(indices);
281
282 // loop through all corner points
283 constexpr std::size_t nCorners = 1 << DIM_POS;
284 std::array<FieldType, nCorners> neighbors{};
285 const auto& cornerIndices =
286 m_cfg.grid.multiAxis().getClosestPointsIndices(gridPosition);
287
288 if (!isInsideLocal(gridPosition)) {
289 return MagneticFieldError::OutOfBounds;
290 }
291
292 std::size_t i = 0;
293 if constexpr (std::is_same_v<FieldType, Vector3>) {
294 if (!m_cfg.transformDependsOnPosition) {
295 // position-independent transformation: apply it once per cell
296 for (std::size_t index : cornerIndices) {
297 neighbors.at(i++) =
298 m_cfg.transformBField(m_cfg.grid.at(index), position);
299 }
300 assert(i == nCorners);
301 return FieldCell(lowerLeft, upperRight, std::move(neighbors));
302 }
303 }
304
305 for (std::size_t index : cornerIndices) {
306 neighbors.at(i++) = m_cfg.grid.at(index);
307 }
308
309 assert(i == nCorners);
310
311 return FieldCell(lowerLeft, upperRight, std::move(neighbors));
312 }

378 Result<Vector3> getField(const Vector3& position) const {
379 const auto gridPosition = m_cfg.transformPos(position);
380 if (!isInsideLocal(gridPosition)) {
381 return Result<Vector3>::failure(MagneticFieldError::OutOfBounds);
382 }
383
384 return Result<Vector3>::success(
385 m_cfg.transformBField(m_cfg.grid.interpolate(gridPosition), position));
386 }
387
395 Vector3 getFieldUnchecked(const Vector3& position) const final {
396 const auto gridPosition = m_cfg.transformPos(position);
397 return m_cfg.transformBField(m_cfg.grid.interpolate(gridPosition),
398 position);
399 }
400
402 Result<Vector3> getField(const Vector3& position,
403 MagneticFieldProvider::Cache& cache) const final {
404 Cache& lcache = cache.as<Cache>();
405 const auto gridPosition = m_cfg.transformPos(position);
406 if (!lcache.fieldCell || !(*lcache.fieldCell).isInside(gridPosition)) {
407 auto res = getFieldCell(position);
408 if (!res.ok()) {
409 return Result<Vector3>::failure(res.error());
410 }
411 lcache.fieldCell = *res;
412 }
413 if constexpr (std::is_same_v<FieldType, Vector3>) {
414 if (!m_cfg.transformDependsOnPosition) {
415 // the cell stores already transformed values
416 return Result<Vector3>::success(
417 (*lcache.fieldCell).getField(gridPosition));
418 }
419 }
420 // the cell stores grid-frame values: transform at the lookup position
421 return Result<Vector3>::success(m_cfg.transformBField(
422 (*lcache.fieldCell).getField(gridPosition), position));
423 }
424
425 private:
426 Config m_cfg;
427
428 typename Grid::point_t m_lowerLeft;
429 typename Grid::point_t m_upperRight;
430};
