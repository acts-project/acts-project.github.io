74 DETRAY_HOST_DEVICE
75 transform3(const vector3 &t, const vector3 &x, const vector3 &y,
76 const vector3 &z, bool get_inverse = true) {
77 _data.setIdentity();
78
79 auto &matrix = _data.matrix();
80 matrix.template block<3, 1>(0, 0) = x;
81 matrix.template block<3, 1>(0, 1) = y;
82 matrix.template block<3, 1>(0, 2) = z;
83 matrix.template block<3, 1>(0, 3) = t;
84
85 if (get_inverse) {
86 _data_inv = _data.inverse();
87 } else {
88 _data_inv.setIdentity();
89 }
90 }
91
98 DETRAY_HOST_DEVICE
99 transform3(const vector3 &t, const vector3 &z, const vector3 &x,
100 bool get_inverse = true)
101 : transform3(t, x, z.cross(x), z, get_inverse) {}

106 DETRAY_HOST_DEVICE
107 explicit transform3(const vector3 &t) {
108 _data.setIdentity();
109
110 auto &matrix = _data.matrix();
111 matrix.template block<3, 1>(0, 3) = t;
112
113 _data_inv = _data.inverse();
114 }
115
119 DETRAY_HOST_DEVICE
120 explicit transform3(const matrix44 &m) {
121 _data.matrix() = m;
122
123 _data_inv = _data.inverse();
124 }
125
130 DETRAY_HOST_DEVICE
131 transform3(const matrix44 &m, const matrix44 &m_inv)
132 : _data{m}, _data_inv{m_inv} {
133 // The assertion will not hold for (casts to) int
134 if constexpr (std::floating_point<scalar_type>) {
135 [[maybe_unused]] constexpr auto epsilon{
136 std::numeric_limits<scalar_type>::epsilon()};
137 assert(algebra::approx_equal(matrix44(m * m_inv),
138 matrix44(matrix44::Identity()),
139 16.f * epsilon, 1e-6f));
140 }
141 }
142
146 DETRAY_HOST_DEVICE
147 explicit transform3(const array_type<16> &ma) {
148 _data.matrix() << ma[0], ma[1], ma[2], ma[3], ma[4], ma[5], ma[6], ma[7],
149 ma[8], ma[9], ma[10], ma[11], ma[12], ma[13], ma[14], ma[15];
150
151 _data_inv = _data.inverse();
152 }
153
155 DETRAY_HOST_DEVICE
156 transform3() {
157 _data.setIdentity();
158 _data_inv.setIdentity();
159 }

169 DETRAY_HOST_DEVICE
170 constexpr bool operator==(const transform3 &rhs) const {
