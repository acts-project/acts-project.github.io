63 DETRAY_HOST_DEVICE
64 constexpr material(const scalar_type x0, const scalar_type l0,

84 DETRAY_HOST_DEVICE
85 constexpr bool operator==(const material<scalar_type> &rhs) const {
86 // Don't check dependent parameters
87 assert(m_molar_rho == mass_to_molar_density(m_ar, m_mass_rho));
88 assert(rhs.molar_density() ==
89 mass_to_molar_density(rhs.Ar(), rhs.mass_density()));
90
91 return (m_x0 == rhs.X0() && m_l0 == rhs.L0() && m_ar == rhs.Ar() &&
92 m_z == rhs.Z() && m_mass_rho == rhs.mass_density() &&
93 m_has_density_effect_data == rhs.has_density_effect_data() &&
94 m_density == rhs.density_effect_data());
95 }
96
101 DETRAY_HOST_DEVICE
102 constexpr bool operator!=(const material<scalar_type> &rhs) const {
103 return !(*this == rhs);
104 }
105
108 DETRAY_HOST_DEVICE
109 constexpr bool is_valid() const {
110 if (m_x0 == detail::invalid_value<scalar_type>()) {
111 return false;
112 }
113 // Cannot happen independently from X0 not being invalid
114 assert(m_l0 != detail::invalid_value<scalar_type>());
115 assert(m_ar != 0.f);
116 assert(m_z != 0.f);
117 assert(m_mass_rho != 0.f);
118 assert(m_molar_rho);
119
120 return true;
121 }

157 DETRAY_HOST_DEVICE
158 scalar_type mean_excitation_energy() const {
159 if (!m_has_density_effect_data) {
160 // use approximative computation as defined in ATL-SOFT-PUB-2008-003
161 return 16.f * unit<scalar_type>::eV *
162 math::pow(m_z, static_cast<scalar_type>(0.9));
163 } else {
164 return m_density.get_mean_excitation_energy();
165 }
166 }
167
170 DETRAY_HOST_DEVICE
171 constexpr scalar_type fraction() const {
172 if constexpr (ratio::num == 0) {
173 return 0.f;
174 } else if constexpr (ratio::den == 0) {
175 return detail::invalid_value<scalar_type>();
176 } else {
177 return static_cast<scalar_type>(ratio::num) /
178 static_cast<scalar_type>(ratio::den);
179 }
180 }

183 DETRAY_HOST
184 std::string to_string() const {
185 std::stringstream strm;
186 if (m_ar <= 0.f) {
187 strm << "vacuum";
188 return strm.str();
189 }
190 strm << "material: ";
191 strm << " X0 = " << m_x0 << " mm";
192 strm << " | L0 = " << m_l0 << " mm";
193 strm << " | Z = " << m_z;
194 strm << " | rho = " << m_mass_rho << " GeV/mm^3";
195
196 return strm.str();
197 }

243 DETRAY_HOST_DEVICE
244 constexpr scalar_type mass_to_molar_density(double ar,
245 double mass_rho) const {
246 if (mass_rho == 0.) {
247 return 0.f;
248 }
249
250 const double molar_mass{ar * unit<double>::g / unit<double>::mol};
251
252 return static_cast<scalar_type>(mass_rho / molar_mass);
253 }
254
255 private:
256 // Material properties
257 scalar_type m_x0 = detail::invalid_value<scalar_type>();
258 scalar_type m_l0 = detail::invalid_value<scalar_type>();
259 scalar_type m_ar = 0.f;
260 scalar_type m_z = 0.f;
261 scalar_type m_mass_rho = 0.f;
262 scalar_type m_molar_rho = 0.f;
263 detail::density_effect_data<scalar_type> m_density = {};
264 bool m_has_density_effect_data = false;
265};
