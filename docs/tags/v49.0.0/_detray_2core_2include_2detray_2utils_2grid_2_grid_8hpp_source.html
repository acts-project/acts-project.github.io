383 DETRAY_HOST_DEVICE constexpr auto operator==(const grid_impl &rhs) const
384 -> bool {
385 // Check axes: they need to be identical
386 if (m_axes != rhs.m_axes) {
387 return false;
388 }
389 // Loop over global bin index and compare the two
390 for (glob_bin_index i = 0; i < nbins(); ++i) {
391 if (bin(i) != rhs.bin(i)) {
392 return false;
393 }
394 }
395 return true;
396 }

435template <concepts::grid grid_lhs_t, concepts::grid grid_rhs_t>
436 requires(!std::same_as<grid_lhs_t, grid_rhs_t> &&
437 (std::same_as<typename grid_lhs_t::template owning_type<false>,
438 grid_rhs_t> ||
439 std::same_as<typename grid_rhs_t::template owning_type<false>,
440 grid_lhs_t>))
441DETRAY_HOST_DEVICE constexpr bool operator==(const grid_lhs_t &lhs,
442 const grid_rhs_t &rhs) {
443 // Generate a non-owning grid from the owning grid and compare
444 if constexpr (grid_lhs_t::is_owning) {
445 typename grid_lhs_t::template owning_type<false> n_owning_lhs(lhs);
446
447 return (n_owning_lhs == rhs);
448
449 } else if constexpr (grid_rhs_t::is_owning) {
450 typename grid_rhs_t::template owning_type<false> n_owning_rhs(rhs);
451
452 return (n_owning_rhs == lhs);
453 } else {
454 // Cannot be reached as per requirements list
455 return false;
456 }
457}
