41class json_input_converter final : public input_file_converter_interface {
42 public:
44 json_input_converter() : input_file_converter_interface(".json") {}
45
49 void add_from_file(io::file_handle& file,
50 detector_payload& payload) override {
51 // Reads the data from file and returns the corresponding io payloads
52 nlohmann::json in_json;
53 *file >> in_json;
54
55 // Peek at the header to determine the kind of payload is in the file
56 const tagged_header_payload header = in_json["header"]["common"];
57 // If not already done, add common header information to detector payload
58 if (payload.header.version.empty()) {
59 payload.header = header;
60 }
61
62 if (header.version != io::detail::get_detray_version()) {
63 DETRAY_WARN_HOST(
64 " --> \'" << header.tag
65 << "\' file was generated with a different version (file: "
66 << header.version << " vs. current: "
67 << io::detail::get_detray_version() << ")");
68 }
69
70 nlohmann::json& json_payload = in_json["data"];
71 if (header.tag == "geometry") {
72 payload.geometry = json_payload;
73 } else if (header.tag == "homogeneous_material") {
74 payload.homogeneous_material = json_payload;
75 } else if (header.tag == "material_maps") {
76 payload.material_maps = json_payload;
77 } else if (header.tag == "surface_grids") {
78 payload.surface_grids = json_payload;
79 } else {
80 std::stringstream err_stream{};
81 err_stream << "Unsupported file tag '" << header.tag << "' in input file";
82 DETRAY_FATAL_HOST(err_stream.str());
83 throw std::invalid_argument(err_stream.str());
84 }
85 }
86};

96class json_output_converter final : public output_file_converter_interface {
97 public:
99 json_output_converter()
100 : output_file_converter_interface(io::format::json, ".json") {}
101
107 std::vector<std::string> write_to_file(
108 const detector_payload& payload, const std::filesystem::path& file_path,
109 const std::ios_base::openmode mode = std::ios::out |
110 std::ios::binary) override {
111 // Assert output stream
112 assert(((mode == std::ios_base::out) ||
113 (mode == (std::ios_base::out | std::ios_base::binary)) ||
114 (mode == (std::ios_base::out | std::ios_base::trunc)) ||
115 (mode == (std::ios_base::out | std::ios_base::trunc |
116 std::ios_base::binary))) &&
117 "Illegal file mode for json writer");
118
119 const header_payload& header = payload.header;
120 tagged_header_payload tagged_header(
121 static_cast<const common_header_payload&>(header));
122
123 // Write every detector component that contains data
124 std::vector<std::string> file_names{};
125
126 const auto has_component_data =
127 []<typename payload_t, typename header_t>(
128 const std::optional<payload_t>& sub_payload,
129 const std::optional<header_t>& sub_header) -> bool {
130 return sub_payload.has_value() && sub_payload->is_valid() &&
131 sub_header.has_value() && sub_header->write_component();
132 };
133
134 if (payload.geometry.is_valid() && header.geometry.has_value() &&
135 header.geometry->write_component()) {
136 tagged_header.tag = "geometry";
137 file_names.push_back(write_component(tagged_header, header.geometry,
138 payload.geometry, file_path, mode));
139 }
140 if (has_component_data(payload.homogeneous_material,
141 header.homogeneous_material)) {
142 tagged_header.tag = "homogeneous_material";
143 file_names.push_back(
144 write_component(tagged_header, header.homogeneous_material,
145 *payload.homogeneous_material, file_path, mode));
146 }
147 if (has_component_data(payload.material_maps, header.material_maps)) {
148 tagged_header.tag = "material_maps";
149 file_names.push_back(write_component(tagged_header, header.material_maps,
150 *payload.material_maps, file_path,
151 mode));
152 }
153 if (has_component_data(payload.surface_grids, header.surface_grids)) {
154 tagged_header.tag = "surface_grids";
155 file_names.push_back(write_component(tagged_header, header.surface_grids,
156 *payload.surface_grids, file_path,
157 mode));
158 }
159
160 return file_names;
161 }
162
163 private:
175 template <typename sub_header_t, typename data_t>
176 std::string write_component(const tagged_header_payload& header,
177 const std::optional<sub_header_t>& sub_header,
178 const data_t& data,
179 const std::filesystem::path& file_path,
180 const std::ios_base::openmode mode) const {
181 // Write the header and the detector data into the json stream by using
182 // the conversion functions defined in "detray/io/json/json_io.hpp"
183 nlohmann::ordered_json out_json;
184 // Write header information in this order, so that nothing get overwritten
185 out_json["header"] = *sub_header;
186 out_json["header"]["common"] = header;
187 out_json["data"] = data;
188
189 // Create a new file
190 const std::string file_stem{header.detector + "_" +
191 std::string(header.tag)};
192 const std::string file_ext{file_extension()};
193 io::file_handle file{file_path / file_stem, file_ext, mode};
194
195 // Write json stream to file
196 *file << std::setw(4) << out_json << std::endl;
197
198 return file_stem + file_ext;
199 }
200};
