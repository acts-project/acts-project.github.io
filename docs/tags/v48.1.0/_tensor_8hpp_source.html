106template <Acts::Concepts::arithmetic T>
107class Tensor {
108 public:
110 using Shape = std::array<std::size_t, 2>;
111
116 static Tensor Create(Shape shape, const ExecutionContext &execContext) {
117 auto ptr = detail::createTensorMemory(shape[0] * shape[1] * sizeof(T),
118 execContext);
119 return Tensor(shape, std::move(ptr), execContext);
120 }
121
127 Tensor clone(const ExecutionContext &to) const {
128 auto clonedPtr = detail::cloneTensorMemory(m_ptr, nbytes(), m_device, to);
129 return Tensor(m_shape, std::move(clonedPtr), to);
130 }
131
134 T *data() { return static_cast<T *>(m_ptr.get()); }
135
138 const T *data() const { return static_cast<const T *>(m_ptr.get()); }
139
142 Shape shape() const { return m_shape; }
143
146 std::size_t size() const { return m_shape[0] * m_shape[1]; }
147
150 std::size_t nbytes() const { return size() * sizeof(T); }
151
154 Device device() const { return m_device; }
155
156 private:
157 Tensor(Shape shape, detail::TensorPtr ptr, const ExecutionContext &ctx)
158 : m_shape(shape), m_ptr(std::move(ptr)), m_device(ctx.device) {}
159
160 Shape m_shape{};
161 detail::TensorPtr m_ptr;
162 Device m_device{};
163};

176template <Acts::Concepts::arithmetic T>
177void checkMaskCompatibility(const Tensor<T> &tensor, const Tensor<bool> &mask,
178 std::size_t matchDim) {
179 if (tensor.shape()[matchDim] != mask.shape()[0]) {
180 throw std::invalid_argument(
181 "tensor shape[" + std::to_string(matchDim) +
182 "]=" + std::to_string(tensor.shape()[matchDim]) +
183 " does not match mask length " + std::to_string(mask.shape()[0]));
184 }
185 if (mask.shape()[1] != 1) {
186 throw std::invalid_argument("mask must have shape [N, 1], got [N, " +
187 std::to_string(mask.shape()[1]) + "]");
188 }
189 if (tensor.device() != mask.device()) {
190 throw std::invalid_argument(
191 "tensor and mask must reside on the same device");
192 }
193}

195} // namespace detail
