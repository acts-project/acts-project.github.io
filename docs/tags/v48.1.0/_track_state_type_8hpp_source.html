289 friend std::ostream& operator<<(std::ostream& os,
290 const TrackStateTypeBase& t) {
291 bool first = true;
292 const auto append = [&](const char* name, bool condition) {
293 if (condition) {
294 if (!first) {
295 os << ",";
296 }
297 os << name;
298 first = false;
299 }
300 };
301 append("HasParams", t.hasParameters());
302 if (t.isMaterial()) {
303 append("IsMat", true);
304 } else {
305 append("HasMat", t.hasMaterial());
306 }
307 if (t.isMeasurement()) {
308 append("IsMeas", true);
309 } else if (t.isOutlier()) {
310 append("IsOutlier", true);
311 } else if (t.isHole()) {
312 append("IsHole", true);
313 } else {
314 append("HasMeas", t.hasMeasurement());
315 }
316 append("IsSharedHit", t.isSharedHit());
317 append("IsSplitHit", t.isSplitHit());
318 append("NoExpectedHit", t.hasNoExpectedHit());
319 return os;
320 }

363 void assertConsistency() const {
364 assert(!(test(HasNoExpectedHit) &&
365 (test(HasMeasurement) || test(IsOutlier) || test(IsHole) ||
366 test(IsSharedHit) || test(IsSplitHit))) &&
367 "TrackStateType - NoExpectedHit cannot be set with other "
368 "measurement flags");
369 assert(!(test(IsOutlier) && test(IsHole)) &&
370 "TrackStateType - Outlier and Hole cannot be set simultaneously");
371 assert(
372 !(test(IsHole) && test(HasMeasurement)) &&
373 "TrackStateType - Hole and Measurement cannot be set simultaneously");
374 assert(!(test(IsOutlier) && !test(HasMeasurement)) &&
375 "TrackStateType - Outlier flag requires Measurement flag to be set");
376 assert(!(test(IsSharedHit) && test(IsSplitHit)) &&
377 "TrackStateType - SharedHit and SplitHit cannot be set "
378 "simultaneously");
379 }
