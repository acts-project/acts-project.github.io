63template <typename grid_type>
64nlohmann::json toJson(const grid_type& grid) {
65 nlohmann::json jGrid;
66
67 auto axes = grid.axes();
68 nlohmann::json jAxes;
69 for (unsigned int ia = 0u; ia < grid_type::DIM; ++ia) {
70 auto jAxis = AxisJsonConverter::toJson(*axes[ia]);
71 jAxes.push_back(jAxis);
72 }
73 jGrid["axes"] = jAxes;
74
75 nlohmann::json jData;
76 // 1D connections
77 if constexpr (grid_type::DIM == 1u) {
78 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
79 typename grid_type::index_t lbin;
80 lbin[0u] = ib0;
81 jData.push_back(std::tie(lbin, grid.atLocalBins(lbin)));
82 }
83 }
84 // 2D connections
85 if constexpr (grid_type::DIM == 2u) {
86 for (unsigned int ib0 = 1u; ib0 <= axes[0u]->getNBins(); ++ib0) {
87 for (unsigned int ib1 = 1u; ib1 <= axes[1u]->getNBins(); ++ib1) {
88 typename grid_type::index_t lbin;
89 lbin[0u] = ib0;
90 lbin[1u] = ib1;
91 jData.push_back(std::tie(lbin, grid.atLocalBins(lbin)));
92 }
93 }
94 }
95 jGrid["data"] = jData;
96
97 return jGrid;
98}

111template <typename axis_generator_type,
112 typename value_type = std::vector<std::size_t>>
113auto fromJson(const nlohmann::json& jGrid,
114 const axis_generator_type& aGenerator) {
115 // Generate the grid
116 using GridType = typename axis_generator_type::template grid_type<value_type>;
117 GridType grid(aGenerator());
118 nlohmann::json jData = jGrid["data"];
119 // Index filling
120 if constexpr (GridType::DIM == 1u) {
121 for (const auto& jd : jData) {
122 std::array<std::size_t, 1u> lbin = jd[0u];
123 if (!jd[1u].is_null()) {
124 grid.atLocalBins(lbin) = jd[1u].get<value_type>();
125 }
126 }
127 }
128 if constexpr (GridType::DIM == 2u) {
129 for (const auto& jd : jData) {
130 std::array<std::size_t, 2u> lbin = jd[0u];
131 if (!jd[1u].is_null()) {
132 grid.atLocalBins(lbin) = jd[1u].get<value_type>();
133 }
134 }
135 }
136 return grid;
137}

150template <typename value_type>
151nlohmann::json toJsonAny(const IGrid& grid, AnyGridConstView<value_type> view) {
152 std::size_t dim = grid.dimensions();
153 if (dim != 1u && dim != 2u) {
154 throw std::invalid_argument(
155 "GridJsonConverter::toJsonAny: only 1D and 2D grids are supported.");
156 }
157
158 nlohmann::json jGrid;
159
160 nlohmann::json jAxes;
161 for (std::size_t ia = 0u; ia < dim; ++ia) {
162 jAxes.push_back(AxisJsonConverter::toJson(grid.axis(ia)));
163 }
164 jGrid["axes"] = jAxes;
165
166 nlohmann::json jData;
167 if (dim == 1u) {
168 for (std::size_t ib0 = 1u; ib0 <= grid.axis(0u).getNBins(); ++ib0) {
169 IGrid::AnyIndexType lbin = {ib0};
170 jData.push_back(std::tie(lbin, view.atLocalBins(lbin)));
171 }
172 } else {
173 for (std::size_t ib0 = 1u; ib0 <= grid.axis(0u).getNBins(); ++ib0) {
174 for (std::size_t ib1 = 1u; ib1 <= grid.axis(1u).getNBins(); ++ib1) {
175 IGrid::AnyIndexType lbin = {ib0, ib1};
176 jData.push_back(std::tie(lbin, view.atLocalBins(lbin)));
177 }
178 }
179 }
180 jGrid["data"] = jData;
181
182 return jGrid;
183}
