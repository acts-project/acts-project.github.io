59template <typename Scalar>
60Scalar etaFromTheta(Scalar theta) {
61 if (theta <= EtaThetaConversionTraits<Scalar>::minTheta) {
62 return std::numeric_limits<Scalar>::infinity();
63 }
64 if (theta >= EtaThetaConversionTraits<Scalar>::maxTheta) {
65 return -std::numeric_limits<Scalar>::infinity();
66 }
67
68 return -std::log(std::tan(theta / 2));
69}

79template <typename Scalar>
80Scalar thetaFromEta(Scalar eta) {
81 if (eta <= EtaThetaConversionTraits<Scalar>::minEta) {
82 return std::numbers::pi_v<Scalar>;
83 }
84 if (eta >= EtaThetaConversionTraits<Scalar>::maxEta) {
85 return 0;
86 }
87
88 return 2 * std::atan(std::exp(-eta));
89}
