363 bool prepareCurvilinearState(State& state) const {
364 // test whether the accumulated path has still its initial value.
365 if (state.pathAccumulated != 0) {
366 return true;
367 }
368
369 // dr/ds :
370 state.derivative.template head<3>() = direction(state);
371 // dt / ds
372 state.derivative(eFreeTime) = fastHypot(
373 1., state.particleHypothesis.mass() / absoluteMomentum(state));
374 // d (dr/ds) / ds : == 0
375 state.derivative.template segment<3>(4) = Acts::Vector3::Zero().transpose();
376 // d qop / ds == 0
377 state.derivative(eFreeQOverP) = 0.;
378
379 return true;
380 }

444 // This is simply ignored if `nullptr`.
449 Result<double> step(State& state, Direction propDir,
450 const IVolumeMaterial* material) const {
451 static_cast<void>(material);
452
453 // use the adjusted step size
454 const auto h = state.stepSize.value() * propDir;
455 const auto m = state.particleHypothesis.mass();
456 const auto p = absoluteMomentum(state);
457 // time propagates along distance as 1/b = sqrt(1 + m²/p²)
458 const auto dtds = fastHypot(1, m / p);
459 // Update the track parameters according to the equations of motion
460 Vector3 dir = direction(state);
461 state.pars.template segment<3>(eFreePos0) += h * dir;
462 state.pars[eFreeTime] += h * dtds;
463
464 // Propagate the jacobian
465 if (state.cov.has_value()) {
466 // The step transport matrix in global coordinates
467 FreeMatrix D = FreeMatrix::Identity();
468 D.block<3, 3>(0, 4) = SquareMatrix<3>::Identity() * h;
469 // Extend the calculation by the time propagation
470 // d(t)/d(q/p) = h m^2 (q/p) / (q^2 dt/ds), with the q^2 folded into p
471 // via p = |q| / |q/p|.
472 D(3, 7) = h * m * m / (p * p * state.pars[eFreeQOverP] * dtds);
473 // Set the derivative factor the time
474 state.derivative(3) = dtds;
475 // Update jacobian and derivative
476 state.jacTransport = D * state.jacTransport;
477 state.derivative.template head<3>() = dir;
478 }
479
480 // state the path length
481 state.pathAccumulated += h;
482 ++state.nSteps;
483
484 ++state.statistics.nAttemptedSteps;
485 ++state.statistics.nSuccessfulSteps;
486 if (propDir != Direction::fromScalarZeroAsPositive(h)) {
487 ++state.statistics.nReverseSteps;
488 }
489 state.statistics.pathLength += h;
490 state.statistics.absolutePathLength += std::abs(h);
491
492 return h;
493 }
