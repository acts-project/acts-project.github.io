32template <typename grid_type, typename cast_container>
33typename grid_type::point_t castPosition(const Vector3& position,
34 const cast_container& globalCasts) {
35 // Fill the grid point from global, unrolling the cast loop
36 return [&]<std::size_t... idx>(std::index_sequence<idx...> /*indices*/) {
37 return typename grid_type::point_t{
38 VectorHelpers::cast(position, globalCasts[idx])...};
39 }(std::make_integer_sequence<std::size_t, grid_type::DIM>{});
40}

64template <typename grid_type, typename local_indices>
65typename grid_type::point_t accessLocal(const Vector2& lposition,
66 const local_indices& laccess) {
67 if constexpr (grid_type::DIM > 2u) {
68 throw std::invalid_argument(
69 "GridAccessHelper: only 1-D and 2-D grids are possible for local "
70 "access.");
71 }
72 // Fill the grid point from local according to the accessors
73 typename grid_type::point_t accessed{};
74 fillLocal(lposition, laccess, accessed,
75 std::make_integer_sequence<std::size_t, grid_type::DIM>{});
76 return accessed;
77}
