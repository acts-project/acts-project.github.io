339inline std::ostream& operator<<(std::ostream& os,
340 const TrackSelector::Config& cuts) {
341 // for printing cuts set up with `within`
342 auto printMinMax = [&](const char* name, const auto& min, const auto& max) {
343 os << " - " << min << " <= " << name << " < " << max << "\n";
344 };
345 // for printing cuts set up with `checkMin`
346 auto printMin = [&](const char* name, const auto& min) {
347 os << " - " << min << " <= " << name << "\n";
348 };
349 // for printing cuts set up with `checkMax`
350 auto printMax = [&](const char* name, const auto& max) {
351 os << " - " << name << " <= " << max << "\n";
352 };
353
354 printMinMax("loc0", cuts.loc0Min, cuts.loc0Max);
355 printMinMax("loc1", cuts.loc1Min, cuts.loc1Max);
356 printMinMax("time", cuts.timeMin, cuts.timeMax);
357 printMinMax("phi", cuts.phiMin, cuts.phiMax);
358 printMinMax("eta", cuts.etaMin, cuts.etaMax);
359 printMinMax("absEta", cuts.absEtaMin, cuts.absEtaMax);
360 printMinMax("pt", cuts.ptMin, cuts.ptMax);
361 printMax("nHoles", cuts.maxHoles);
362 printMax("nOutliers", cuts.maxOutliers);
363 printMax("nHoles + nOutliers", cuts.maxHolesAndOutliers);
364 printMax("nSharedHits", cuts.maxSharedHits);
365 printMax("chi2", cuts.maxChi2);
366 printMin("nMeasurements", cuts.minMeasurements);
367 return os;

370inline TrackSelector::EtaBinnedConfig& TrackSelector::EtaBinnedConfig::addCuts(
371 double etaMax, const std::function<void(Config&)>& callback) {
372 if (etaMax <= absEtaEdges.back()) {
373 throw std::invalid_argument{
374 "Abs Eta bin edges must be in increasing order"};
375 }
376
377 if (etaMax < 0.0) {
378 throw std::invalid_argument{"Abs Eta bin edges must be positive"};
379 }
380
381 absEtaEdges.push_back(etaMax);
382 cutSets.emplace_back();
383 if (callback) {
384 callback(cutSets.back());
385 }
386 return *this;
387}

406inline std::size_t TrackSelector::EtaBinnedConfig::binIndexNoCheck(
407 double eta) const {
408 auto binIt = std::ranges::upper_bound(absEtaEdges, std::abs(eta));
409 std::size_t index = std::ranges::distance(absEtaEdges.begin(), binIt);
410 if (index == 0) {
411 index = absEtaEdges.size() + 1; // positive value to check for underflow
412 }
413 return index - 1;

445template <TrackProxyConcept track_proxy_t>
446bool TrackSelector::isValidTrack(const track_proxy_t& track) const {
447 auto checkMin = [](auto x, auto min) { return min <= x; };
448 auto checkMax = [](auto x, auto max) { return x <= max; };
449 auto within = [](double x, double min, double max) {
450 return (min <= x) && (x < max);
451 };
452
453 const auto theta = track.theta();
454
455 constexpr double kUnset = -std::numeric_limits<double>::infinity();
456
457 double _eta = kUnset;
458 double _absEta = kUnset;
459
460 auto absEta = [&]() {
461 if (_absEta == kUnset) {
462 _eta = AngleHelpers::etaFromTheta(theta);
463 _absEta = std::abs(_eta);
464 }
465 return _absEta;
466 };
467
468 const Config* cutsPtr{nullptr};
469 if (!m_isUnbinned) {
470 // return false if |eta| is outside its range, or nan.
471 if (!(absEta() >= m_cfg.absEtaEdges.front() &&
472 _absEta < m_cfg.absEtaEdges.back())) {
473 return false;
474 }
475 cutsPtr = &m_cfg.getCuts(_eta);
476 } else {
477 cutsPtr = &m_cfg.cutSets.front();
478 }
479
480 const Config& cuts = *cutsPtr;
481
482 auto parameterCuts = [&]() {
483 return within(track.transverseMomentum(), cuts.ptMin, cuts.ptMax) &&
484 (!m_isUnbinned ||
485 (within(absEta(), cuts.absEtaMin, cuts.absEtaMax) &&
486 within(_eta, cuts.etaMin, cuts.etaMax))) &&
487 within(track.phi(), cuts.phiMin, cuts.phiMax) &&
488 within(track.loc0(), cuts.loc0Min, cuts.loc0Max) &&
489 within(track.loc1(), cuts.loc1Min, cuts.loc1Max) &&
490 within(track.time(), cuts.timeMin, cuts.timeMax);
491 };
492
493 auto trackCuts = [&]() {
494 return checkMin(track.nMeasurements(), cuts.minMeasurements) &&
495 checkMax(track.nHoles(), cuts.maxHoles) &&
496 checkMax(track.nOutliers(), cuts.maxOutliers) &&
497 checkMax(track.nHoles() + track.nOutliers(),
498 cuts.maxHolesAndOutliers) &&
499 checkMax(track.nSharedHits(), cuts.maxSharedHits) &&
500 checkMax(track.chi2(), cuts.maxChi2) &&
501 cuts.measurementCounter.isValidTrack(track);
502 };
503
504 if (cuts.requireReferenceSurface) {
505 return track.hasReferenceSurface() && parameterCuts() && trackCuts();
506 } else {
507 return trackCuts();
508 }
509}

511inline TrackSelector::TrackSelector(
512 const TrackSelector::EtaBinnedConfig& config)
513 : m_cfg(config) {
514 if (m_cfg.cutSets.size() != m_cfg.nEtaBins()) {
515 throw std::invalid_argument{
516 "TrackSelector cut / eta bin configuration is inconsistent"};
517 }
518
519 if (m_cfg.nEtaBins() == 1) {
520 static const std::vector<double> infVec = {0, inf};
521 m_isUnbinned =
522 m_cfg.absEtaEdges == infVec; // single bin, no eta edges given
523 }
524
525 if (!m_isUnbinned) {
526 for (const auto& cuts : m_cfg.cutSets) {
527 if (cuts.etaMin != -inf || cuts.etaMax != inf || cuts.absEtaMin != 0.0 ||
528 cuts.absEtaMax != inf) {
529 throw std::invalid_argument{
530 "Explicit eta cuts are only valid for single eta bin"};
531 }
532 }
533 }
534}

539template <TrackProxyConcept track_proxy_t>
540bool TrackSelector::MeasurementCounter::isValidTrack(
541 const track_proxy_t& track) const {
542 // No hit cuts, accept everything
543 if (counters.empty()) {
544 return true;
545 }
546
547 boost::container::small_vector<unsigned int, 4> counterValues;
548 counterValues.resize(counters.size(), 0);
549
550 for (const auto& ts : track.trackStatesReversed()) {
551 if (!ts.typeFlags().isMeasurement()) {
552 continue;
553 }
554
555 const auto geoId = ts.referenceSurface().geometryId();
556
557 for (std::size_t i = 0; i < counters.size(); i++) {
558 const auto& [counterMap, threshold] = counters[i];
559 if (const auto it = counterMap.find(geoId); it != counterMap.end()) {
560 counterValues[i]++;
561 }
562 }
563 }
564
565 for (std::size_t i = 0; i < counters.size(); i++) {
566 const auto& [counterMap, threshold] = counters[i];
567 const unsigned int value = counterValues[i];
568 if (value < threshold) {
569 return false;
570 }
571 }
572
573 return true;
574}
