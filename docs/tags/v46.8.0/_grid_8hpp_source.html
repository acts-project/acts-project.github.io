434 template <class Point>
435 T interpolate(const Point& point) const {
436 // there are 2^DIM corner points used during the interpolation
437 constexpr std::size_t nCorners = 1 << DIM;
438
439 // construct vector of pairs of adjacent bin centers and values
440 std::array<value_type, nCorners> neighbors{};
441
442 // get local indices for current bin
443 // value of bin is interpreted as being the field value at its lower left
444 // corner
445 const auto& llIndices = localBinsFromPosition(point);
446
447 // get global indices for all surrounding corner points
448 const auto& closestIndices = rawClosestPointsIndices(llIndices);
449
450 // get values on grid points
451 std::size_t i = 0;
452 for (std::size_t index : closestIndices) {
453 neighbors.at(i++) = at(index);
454 }
455
456 return Acts::interpolate(point, lowerLeftBinEdge(llIndices),
457 upperRightBinEdge(llIndices), neighbors);
458 }

527 std::size_t size(bool fullCounter = true) const {
528 index_t nBinsArray = numLocalBins();
529 std::size_t current_size = 1;
530 // add under-and overflow bins for each axis and multiply all bins
531 if (fullCounter) {
532 for (const auto& value : nBinsArray) {
533 current_size *= value + 2;
534 }
535 }
536 // ignore under-and overflow bins for each axis and multiply all bins
537 else {
538 for (const auto& value : nBinsArray) {
539 current_size *= value;
540 }
541 }
542 return current_size;
543 }
