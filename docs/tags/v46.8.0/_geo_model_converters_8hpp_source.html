90inline std::shared_ptr<const IGeoShapeConverter> geoShapesConverters(
91 int geoShapeId) {
92 static const std::unordered_map<int,
93 std::shared_ptr<const IGeoShapeConverter>>
94 converters{
95 {GeoBox::getClassTypeID(), std::make_shared<GeoBoxConverter>()},
96 {GeoShapeIntersection::getClassTypeID(),
97 std::make_shared<GeoIntersectionAnnulusConverter>()},
98 {GeoShapeShift::getClassTypeID(),
99 std::make_shared<GeoShiftConverter>()},
100 {GeoTrd::getClassTypeID(), std::make_shared<GeoTrdConverter>()},
101 {GeoTube::getClassTypeID(), std::make_shared<GeoTubeConverter>()},
102 {GeoSimplePolygonBrep::getClassTypeID(),
103 std::make_shared<GeoPolygonConverter>()},
104 {GeoShapeSubtraction::getClassTypeID(),
105 std::make_shared<GeoSubtractionConverter>()},
106 {GeoShapeUnion::getClassTypeID(),
107 std::make_shared<GeoUnionDoubleTrdConverter>()}};
108 auto itr = converters.find(geoShapeId);
109
110 return itr != converters.end() ? itr->second : nullptr;
111};
