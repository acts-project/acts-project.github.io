127template <std::size_t Dim>
128class ProfileHistogram {
129 public:
137 ProfileHistogram(std::string name, std::string title,
138 const std::array<AxisVariant, Dim>& axes,
139 std::string sampleAxisTitle,
140 Range1D<double> sampleRange = {})
141 : m_name(std::move(name)),
142 m_title(std::move(title)),
143 m_sampleAxisTitle(std::move(sampleAxisTitle)),
144 m_sampleRange(sampleRange),
145 m_hist(boost::histogram::make_profile(axes.begin(), axes.end())) {}
146
151 void fill(const std::array<double, Dim>& values, double sample) {
152 if (!m_sampleRange.contains(sample)) {
153 return;
154 }
155
156 std::apply(
157 [&](auto... v) { m_hist(v..., boost::histogram::sample(sample)); },
158 std::tuple_cat(values));
159 }
160
163 const std::string& name() const { return m_name; }
164
167 const std::string& title() const { return m_title; }
168
171 static constexpr std::size_t rank() { return Dim; }
172
175 const std::string& sampleAxisTitle() const { return m_sampleAxisTitle; }
176
179 const BoostProfileHist& histogram() const { return m_hist; }
180
181 private:
182 std::string m_name;
183 std::string m_title;
184 std::string m_sampleAxisTitle;
185 Range1D<double> m_sampleRange;
186
187 BoostProfileHist m_hist;
188};

208 Efficiency(std::string name, std::string title,
209 const std::array<AxisVariant, Dim>& axes)
210 : m_name(std::move(name)),
211 m_title(std::move(title)),

219 void fill(const std::array<double, Dim>& values, bool accepted) {
220 std::apply(
221 [&](auto... v) {
222 m_total(v...);
