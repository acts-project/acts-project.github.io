361class DefaultFilterPolicy final : public OutputFilterPolicy {
362 public:
366 explicit DefaultFilterPolicy(Level lvl) : m_level(lvl) {
367 if (lvl > getFailureThreshold()) {
368 throw ThresholdFailure(
369 "Requested debug level is incompatible with "
370 "the ACTS_LOG_FAILURE_THRESHOLD=" +

493 public:
498 explicit TimedOutputDecorator(std::unique_ptr<OutputPrintPolicy> wrappee,
499 const std::string& format = "%X")
500 : OutputDecorator(std::move(wrappee)), m_format(format) {}
501
509 void flush(const Level& lvl, const std::string& input) override {

511 os << std::left << std::setw(12) << now() << input;
512 OutputDecorator::flush(lvl, os.str());
513 }
514
518 std::unique_ptr<OutputPrintPolicy> clone(
519 const std::string& name) const override {
520 return std::make_unique<TimedOutputDecorator>(m_wrappee->clone(name),
521 m_format);

524 private:
528 std::string now() const {
529 char buffer[20];
530 time_t t{};
531 std::time(&t);
532 struct tm tbuf {};
533 std::strftime(buffer, sizeof(buffer), m_format.c_str(),
534 localtime_r(&t, &tbuf));
535 return buffer;
536 }

539 std::string m_format;

624class DefaultPrintPolicy final : public OutputPrintPolicy {
625 public:
631 explicit DefaultPrintPolicy(std::ostream* out = &std::cout) : m_out(out) {}
632
637 void flush(const Level& lvl, const std::string& input) final {
638 // Mutex to serialize access to std::cout
639 static std::mutex s_stdoutMutex;
640 std::unique_lock lock{s_stdoutMutex,
641 std::defer_lock}; // prep empty, we might not need it
642
643 if (m_out == &std::cout) {
644 lock.lock(); // lock only if we are printing to std::cout
645 }
646
647 (*m_out) << input << std::endl;
648 if (lvl >= getFailureThreshold()) {
649 throw ThresholdFailure(
650 "Previous debug message exceeds the "
651 "ACTS_LOG_FAILURE_THRESHOLD=" +
652 std::string{levelName(getFailureThreshold())} +
653 " configuration, bailing out. See "
654 "https://acts.readthedocs.io/en/latest/core/misc/"
655 "logging.html#logging-thresholds");
656 }
657 }
658
665 const std::string& name() const override {
666 throw std::runtime_error{
667 "Default print policy doesn't have a name. Is there no named output in "
668 "the decorator chain?"};
669 };
670
673 std::unique_ptr<OutputPrintPolicy> clone(
674 const std::string& /*name*/) const override {
675 return std::make_unique<DefaultPrintPolicy>(m_out);
676 };
677
678 private:
680 std::ostream* m_out;
681};

693class Logger {
694 public:
699 Logger(std::unique_ptr<Logging::OutputPrintPolicy> pPrint,
700 std::unique_ptr<Logging::OutputFilterPolicy> pFilter)
701 : m_printPolicy(std::move(pPrint)), m_filterPolicy(std::move(pFilter)) {}
702
708 bool doPrint(const Logging::Level& lvl) const {
709 return m_filterPolicy->doPrint(lvl);
710 }
711
716 void log(const Logging::Level& lvl, const std::string& input) const {
717 if (doPrint(lvl)) {

720 }

724 const Logging::OutputPrintPolicy& printPolicy() const {
725 return *m_printPolicy;
726 }
727
730 const Logging::OutputFilterPolicy& filterPolicy() const {
731 return *m_filterPolicy;
732 }
733
736 Logging::Level level() const { return m_filterPolicy->level(); }
737
740 const std::string& name() const { return m_printPolicy->name(); }

746 std::unique_ptr<Logger> clone(
747 const std::optional<std::string>& _name = std::nullopt,
748 const std::optional<Logging::Level>& _level = std::nullopt) const {
749 return std::make_unique<Logger>(
750 m_printPolicy->clone(_name.value_or(name())),
751 m_filterPolicy->clone(_level.value_or(level())));
752 }
753
758 std::unique_ptr<Logger> clone(Logging::Level _level) const {

761
767 std::unique_ptr<Logger> cloneWithSuffix(

770 return clone(name() + suffix, _level.value_or(level()));
771 }
772

778 private:
