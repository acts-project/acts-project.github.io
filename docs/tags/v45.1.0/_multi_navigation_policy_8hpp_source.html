26 : MultiNavigationPolicy{[](auto... args) {

71 bool isValid(const GeometryContext& gctx, const NavigationArguments args,
72 NavigationPolicyState& state,
73 const Logger& logger) const override {
74 ACTS_VERBOSE("MultiNavigationPolicy isValid check, forward to "
75 << m_policyPtrs.size() << " policies.");
76
77 auto& thisState = state.as<State>();
78 if (thisState.policyStates.size() != m_policyPtrs.size()) {
79 ACTS_ERROR("MultiNavigationPolicy isValid: number of states ("
80 << thisState.policyStates.size()
81 << ") does not match number of policies ("
82 << m_policyPtrs.size() << ").");
83 throw std::runtime_error(
84 "MultiNavigationPolicy isValid: inconsistent state size.");
85 }
86
87 for (auto [policy, policyState] :
88 zip(m_policyPtrs, thisState.policyStates)) {
89 if (!policy->isValid(gctx, args, policyState, logger)) {
90 return false;
91 }
92 }
93 // If no policy rejected, return true
94 return true;
95 }
96
102 void createState(const GeometryContext& gctx, const NavigationArguments args,
103 NavigationPolicyStateManager& stateManager,
104 const Logger& logger) const override {
105 ACTS_VERBOSE("MultiNavigationPolicy createState, create states for "
106 << m_policyPtrs.size() << " policies.");
107
108 // Push child states first, then at the end push this policy's state,
109 // containing the references
110
111 std::vector<NavigationPolicyState> states;
112 states.reserve(m_policyPtrs.size());
113
114 for (const auto& policy : m_policyPtrs) {
115 ACTS_VERBOSE("Creating child state for policy ");
116 policy->createState(gctx, args, stateManager, logger);
117 states.emplace_back(stateManager.currentState());
118 }
119
120 ACTS_VERBOSE("Created "
121 << states.size()
122 << " child states for MultiNavigationPolicy (of which "
123 << std::ranges::count_if(
124 states, [](const auto& s) { return !s.empty(); })
125 << " are non-empty)");
126
127 // Important, push at the end
128 stateManager.pushState<State>(std::move(states));
129 }
130
135 void popState(NavigationPolicyStateManager& stateManager,
136 const Logger& logger) const override {
137 // By default, we didn't push anything, so we don't need to pop anything
138 ACTS_VERBOSE("MultiNavigationPolicy popState called, popping for "
139 << m_policyPtrs.size() << " child policies");
140
141 // Pops this policy's state first
142 stateManager.popState();
143
144 // Then pops all child states
145 for (const auto& policy : m_policyPtrs) {
146 policy->popState(stateManager, logger);
147 }
148 }
