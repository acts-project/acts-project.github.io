101template <std::size_t Dim>
102class ProfileHistogram {
103 public:
111 ProfileHistogram(std::string name, std::string title,
112 const std::array<AxisVariant, Dim>& axes,
113 std::string sampleAxisTitle,
114 Range1D<double> sampleRange = {})
115 : m_name(std::move(name)),
116 m_title(std::move(title)),
117 m_sampleAxisTitle(std::move(sampleAxisTitle)),
118 m_sampleRange(sampleRange),
119 m_hist(boost::histogram::make_profile(axes.begin(), axes.end())) {}
120
125 void fill(const std::array<double, Dim>& values, double sample) {
126 if (!m_sampleRange.contains(sample)) {
127 return;
128 }
129
130 std::apply(
131 [&](auto... v) { m_hist(v..., boost::histogram::sample(sample)); },
132 std::tuple_cat(values));
133 }
134
136 const std::string& name() const { return m_name; }
137
139 const std::string& title() const { return m_title; }
140
142 static constexpr std::size_t rank() { return Dim; }
143
145 const std::string& sampleAxisTitle() const { return m_sampleAxisTitle; }
146
148 const BoostProfileHist& histogram() const { return m_hist; }
149
150 private:
151 std::string m_name;
152 std::string m_title;
153 std::string m_sampleAxisTitle;
154 Range1D<double> m_sampleRange;
155
156 BoostProfileHist m_hist;
157};

169template <std::size_t Dim>
170class Efficiency {
171 public:
177 Efficiency(std::string name, std::string title,
178 const std::array<AxisVariant, Dim>& axes)
179 : m_name(std::move(name)),
180 m_title(std::move(title)),
181 m_accepted(boost::histogram::make_histogram(axes.begin(), axes.end())),
182 m_total(boost::histogram::make_histogram(axes.begin(), axes.end())) {}
183
188 void fill(const std::array<double, Dim>& values, bool accepted) {
189 std::apply(
190 [&](auto... v) {
191 m_total(v...);
192 if (accepted) {
193 m_accepted(v...);
194 }
195 },
196 std::tuple_cat(values));
197 }
198
200 const std::string& name() const { return m_name; }
201
203 const std::string& title() const { return m_title; }
204
206 static constexpr std::size_t rank() { return Dim; }
207
209 const BoostHist& acceptedHistogram() const { return m_accepted; }
210
212 const BoostHist& totalHistogram() const { return m_total; }
213
214 private:
215 std::string m_name;
216 std::string m_title;
217
218 BoostHist m_accepted;
219 BoostHist m_total;
220};
