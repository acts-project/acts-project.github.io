76 DETRAY_HOST_DEVICE
77 transform3(const vector3 &t, const vector3 &x, const vector3 &y,
78 const vector3 &z, bool get_inverse = true) {
79 _data(0, 0) = x[0];
80 _data(1, 0) = x[1];
81 _data(2, 0) = x[2];
82 _data(0, 1) = y[0];
83 _data(1, 1) = y[1];
84 _data(2, 1) = y[2];
85 _data(0, 2) = z[0];
86 _data(1, 2) = z[1];
87 _data(2, 2) = z[2];
88 _data(0, 3) = t[0];
89 _data(1, 3) = t[1];
90 _data(2, 3) = t[2];
91
92 if (get_inverse) {
93 int ifail = 0;
94 _data_inv = _data.Inverse(ifail);
95 SMATRIX_CHECK(ifail);
96 }
97 }

140 DETRAY_HOST
141 transform3(const matrix44 &m, const matrix44 &m_inv)
142 : _data{m}, _data_inv{m_inv} {
143 // The assertion will not hold for (casts to) int
144 if constexpr (std::floating_point<scalar_type>) {
145 [[maybe_unused]] constexpr auto epsilon{
146 std::numeric_limits<scalar_type>::epsilon()};
147 assert(algebra::approx_equal(matrix44(m * m_inv),
148 matrix44(ROOT::Math::SMatrixIdentity()),
149 16.f * epsilon, 1e-6f));
150 }
151 }

171 _data(0, 3) = ma[3];

173 _data(2, 3) = ma[11];
174 _data(3, 3) = ma[15];

177 _data_inv = _data.Inverse(ifail);
