73template <concepts::scalar scalar_t>
74struct constrained_step {
76 template <step::constraint type>
77 requires(type != step::constraint::e_all)
78 DETRAY_HOST_DEVICE void set(const scalar_t step_size) {
79 _constraints[type] = math::min(_constraints[type], math::fabs(step_size));
80 }
81
85 template <step::constraint type = step::constraint::e_all>
86 DETRAY_HOST_DEVICE scalar_t
87 size(const step::direction dir = step::direction::e_forward) const {
88 if constexpr (type == step::constraint::e_all) {
89 return static_cast<scalar_t>(dir) * min();
90 } else {
91 return static_cast<scalar_t>(dir) * _constraints[type];
92 }
93 }
94
96 template <step::constraint type = step::constraint::e_actor>
97 DETRAY_HOST_DEVICE void release() {
98 if constexpr (type == step::constraint::e_all) {
99 _constraints = {std::numeric_limits<scalar_t>::max(),
100 std::numeric_limits<scalar_t>::max(),
101 std::numeric_limits<scalar_t>::max(),
102 std::numeric_limits<scalar_t>::max()};
103 } else {
104 _constraints[type] = std::numeric_limits<scalar_t>::max();
105 }
106 }
107
109 DETRAY_HOST_DEVICE scalar_t min() const {
110 scalar_t min_constr = std::numeric_limits<scalar_t>::max();
111 min_constr =
112 math::min(min_constr, _constraints[step::constraint::e_accuracy]);
113 min_constr = math::min(min_constr, _constraints[step::constraint::e_actor]);
114 min_constr =
115 math::min(min_constr, _constraints[step::constraint::e_aborter]);
116 return math::min(min_constr, _constraints[step::constraint::e_user]);
117 }
118
120 darray<scalar_t, 4> _constraints = {std::numeric_limits<scalar_t>::max(),
121 std::numeric_limits<scalar_t>::max(),
122 std::numeric_limits<scalar_t>::max(),
123 std::numeric_limits<scalar_t>::max()};
124};
