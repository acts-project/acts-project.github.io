107 DETRAY_HOST
108 void add_material(material_id id, data_type &&mat_data,
109 std::vector<std::size_t> &&n_bins,
110 std::vector<std::vector<scalar_type>> &&axis_spans,
111 std::vector<index_type> &&indices) {
112 DETRAY_VERBOSE_HOST("Received material map data");
113
114 auto [sf_index, mat, thickness] = std::move(mat_data).get_data();
115
116 assert(!detail::is_invalid_value(sf_index));
117
118 m_links.emplace(sf_index, std::make_pair(id, std::move(indices)));
119 m_n_bins.emplace(sf_index, std::move(n_bins));
120 m_axis_spans.emplace(sf_index, std::move(axis_spans));
121 m_materials.emplace(sf_index, std::move(mat));
122 m_thickness.emplace(sf_index, std::move(thickness));
123 }

151 template <typename bin_data_t, std::size_t N>
152 DETRAY_HOST auto operator()(
153 typename detector_t::surface_lookup_container &surfaces,
154 std::map<dindex, std::vector<bin_data_t>> &material_map,
155 std::map<dindex, darray<std::size_t, N>> &n_bins,
156 std::map<dindex, darray<std::vector<scalar_type>, N>> &axis_spans) {
157 DETRAY_VERBOSE_HOST("Add material maps...");
158
159 using link_t = typename detector_t::surface_type::material_link;
160
161 // Check data consistency
162 const std::size_t n_materials{this->n_materials()};
163 if (n_materials == 0) {
164 return;
165 }
166
167 assert(surfaces.size() >= n_materials);
168
169 DETRAY_VERBOSE_HOST("-> Have " << m_materials.size()
170 << " configured material map(s) to assign");
171
172 // Add the material only to those surfaces that the data links against
173 for (auto &[i, materials] : m_materials) {
174 const auto sf_idx{static_cast<dindex>(i)};
175
176 DETRAY_DEBUG_HOST("--> #" << sf_idx << " sf = " << surfaces.at(sf_idx));
177
178 // Copy the number of bins to the builder
179 assert(m_n_bins.at(sf_idx).size() == N);
180 n_bins.emplace(sf_idx, darray<std::size_t, N>{});
181 std::ranges::copy_n(m_n_bins.at(sf_idx).begin(), N,
182 n_bins.at(sf_idx).begin());
183
184 // Copy the axis spans to the builder (if present)
185 axis_spans.emplace(sf_idx, darray<std::vector<scalar_type>, N>{});
186 for (std::size_t in = 0; in < N; ++in) {
187 if (m_axis_spans.at(sf_idx).size() > in) {
188 axis_spans.at(sf_idx).at(in) = m_axis_spans.at(sf_idx).at(in);
189 }
190 }
191
192 // Combine the material slab with its local bin index
193 for (const auto [j, mat] : detray::views::enumerate(materials)) {
194 scalar_type t{m_thickness.at(sf_idx)[j]};
195 auto &loc_bin_idx{m_links.at(sf_idx).second.at(j)};
196
197 material_slab<scalar_type> mat_slab{mat, t};
198 bin_data_t data{loc_bin_idx, mat_slab};
199
200 auto search = material_map.find(sf_idx);
201 if (search == material_map.end()) {
202 material_map.emplace(sf_idx, std::vector<bin_data_t>{data});
203 } else {
204 search->second.push_back(data);
205 }
206 }
207
208 auto map_id{m_links.at(sf_idx).first};
209 DETRAY_DEBUG_HOST("--> Set surface material id: " << map_id);
210 surfaces.at(sf_idx).material() = link_t{map_id, dindex_invalid};
211 }
212
213 DETRAY_DEBUG_HOST("Finished");
214 }
215
216 private:
218 std::map<std::size_t, std::pair<material_id, std::vector<index_type>>>
219 m_links{};
221 std::map<std::size_t, std::vector<std::size_t>> m_n_bins{};
223 std::map<std::size_t, std::vector<std::vector<scalar_type>>> m_axis_spans{};
225 std::map<std::size_t, std::vector<scalar_type>> m_thickness{};
228 std::map<std::size_t, std::vector<material<scalar_type>>> m_materials{};
229};
