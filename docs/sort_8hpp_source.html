22template <std::random_access_iterator RandomIt>
23DETRAY_HOST_DEVICE inline void insertion_sort(RandomIt first, RandomIt last) {
24 for (auto it = first; it != last; it++) {
25 // Searching the upper bound, i.e., first
26 // element greater than *it from beginning
27 auto const insertion_point = detray::detail::upper_bound(first, it, *it);
28
29 // Shifting the unsorted part
30 std::ranges::rotate(insertion_point, it, it + 1);
31 }
32}

36DETRAY_HOST_DEVICE inline void insertion_sort(vector_t<TYPE> &vec) {
37 detray::detail::insertion_sort(vec.begin(), vec.end());
38}

40template <std::random_access_iterator RandomIt, class Comp = std::less<void>>
41DETRAY_HOST_DEVICE inline void selection_sort(RandomIt first, RandomIt last,
42 Comp &&comp = Comp()) {
43 for (RandomIt i = first; i < (last - 1); ++i) {
44 RandomIt k = i;
45
46 for (RandomIt j = i + 1; j < last; ++j) {
47 if (comp(*j, *k)) {
48 k = j;
49 }
50 }
51
52 if (k != i) {
53 auto t = *i;
54 *i = *k;
55 *k = t;
56 }
57 }
58}
