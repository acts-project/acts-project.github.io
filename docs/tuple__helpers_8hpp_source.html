272template <typename T1, typename T2>
273struct is_permutation : public std::false_type {};
274
275template <>
276struct is_permutation<detray::dtuple<>, detray::dtuple<>>
277 : public std::true_type {};
278
280template <typename... Args1, typename... Args2>
281struct is_permutation<detray::dtuple<Args1...>, detray::dtuple<Args2...>> {
283 using T1 = detray::dtuple<Args1...>;
285 using T2 = detray::dtuple<Args2...>;
286
289 template <typename o_tuple_t, typename T, typename... U>
290 static consteval bool compare() {
291 if constexpr (detray::detail::has_type_v<T, o_tuple_t>) {
292 if constexpr (sizeof...(U) == 0u) {
293 return true;
294 } else {
295 return compare<o_tuple_t, U...>();
296 }
297 } else {
298 return false;
299 }
300 }
301
303 static constexpr bool value =
304 ((detray::detail::tuple_size_v<T1> == detray::detail::tuple_size_v<T2>) &&
305 (compare<T1, Args2...>() && compare<T2, Args1...>()));
306};
307
308template <typename T1, typename T2>
309constexpr bool is_permutation_v = is_permutation<T1, T2>::value;
