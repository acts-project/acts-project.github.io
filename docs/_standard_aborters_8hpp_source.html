23struct PathLimitReached {
25 double internalLimit = std::numeric_limits<double>::max();
26
38 template <typename propagator_state_t, typename stepper_t,
39 typename navigator_t>
40 bool checkAbort(propagator_state_t& state, const stepper_t& stepper,
41 const navigator_t& navigator, const Logger& logger) const {
42 static_cast<void>(navigator);
43
44 // Check if the maximum allowed step size has to be updated
45 double distance =
46 std::abs(internalLimit) - std::abs(stepper.pathLength(state.stepping));
47 double tolerance = state.options.surfaceTolerance;
48 bool limitReached = (std::abs(distance) < std::abs(tolerance));
49 if (limitReached) {
50 ACTS_VERBOSE("PathLimit aborter | " << "Path limit reached at distance "
51 << distance);
52 return true;
53 }
54 stepper.updateStepSize(state.stepping, distance,
55 ConstrainedStep::Type::Actor);
56 ACTS_VERBOSE("PathLimit aborter | "
57 << "Target stepSize (path limit) updated to "
58 << stepper.outputStepSize(state.stepping));
59 return false;
60 }
61};

122struct VolumeConstraintAborter {
132 template <typename propagator_state_t, typename stepper_t,
133 typename navigator_t>
134 bool checkAbort(propagator_state_t& state, const stepper_t& /*stepper*/,
135 const navigator_t& navigator, const Logger& logger) const {
136 const auto& constrainToVolumeIds = state.options.constrainToVolumeIds;
137 const auto& endOfWorldVolumeIds = state.options.endOfWorldVolumeIds;
138
139 if (constrainToVolumeIds.empty() && endOfWorldVolumeIds.empty()) {
140 return false;
141 }
142 const auto* currentVolume = navigator.currentVolume(state.navigation);
143
144 // We need a volume to check its ID
145 if (currentVolume == nullptr) {
146 return false;
147 }
148
149 const auto currentVolumeId =
150 static_cast<std::uint32_t>(currentVolume->geometryId().volume());
151
152 if (!constrainToVolumeIds.empty() &&
153 !rangeContainsValue(constrainToVolumeIds, currentVolumeId)) {
154 ACTS_VERBOSE(
155 "VolumeConstraintAborter aborter | Abort with volume constrain "
156 << currentVolumeId);
157 return true;
158 }
159
160 if (!endOfWorldVolumeIds.empty() &&
161 rangeContainsValue(endOfWorldVolumeIds, currentVolumeId)) {
162 ACTS_VERBOSE(
163 "VolumeConstraintAborter aborter | Abort with additional end of "
164 "world volume "
165 << currentVolumeId);
166 return true;
167 }
168
169 return false;
170 }
171};

174struct AnySurfaceReached {
184 template <typename propagator_state_t, typename stepper_t,
185 typename navigator_t>
186 bool checkAbort(propagator_state_t& state, const stepper_t& stepper,
187 const navigator_t& navigator, const Logger& logger) const {
188 static_cast<void>(stepper);
189 static_cast<void>(logger);
190
191 const Surface* startSurface = navigator.startSurface(state.navigation);
192 const Surface* targetSurface = navigator.targetSurface(state.navigation);
193 const Surface* currentSurface = navigator.currentSurface(state.navigation);
194
195 // `startSurface` is excluded because we want to reach a new surface
196 // `targetSurface` is excluded because another aborter should handle it
197 if (currentSurface != nullptr && currentSurface != startSurface &&
198 currentSurface != targetSurface) {
199 return true;
200 }
201
202 return false;
203 }
204};
