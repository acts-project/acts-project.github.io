301 template <typename E = entry_type>
302 DETRAY_HOST_DEVICE constexpr void push_back(E&& entry) {
303 assert(m_capacity > 0);
304 assert(m_data->size <= m_capacity);
305 assert(m_global_storage);
306
307 if (m_data->size < m_capacity) {
308 *(const_cast<entry_type*>(m_global_storage) + m_data->size) =
309 std::forward<E>(entry);
310 ++(const_cast<data*>(m_data)->size);
311 }
312 }
313
318 DETRAY_HOST_DEVICE
319 constexpr auto init(entry_type entry = detail::invalid_value<entry_type>())
320 -> dynamic_array& {
321 if (capacity() == 0u) {
322 return *this;
323 }
324
325 // Initialize the storage element
326 std::ranges::fill(this, detail::invalid_value<entry_type>());
327
328 if (entry == detail::invalid_value<entry_type>()) {
329 const_cast<data*>(m_data)->size = 0u;
330 return *this;
331 }
332
333 // The bin has at least a capacity of 1
334 this->front() = entry;
335 const_cast<data*>(m_data)->size = 1u;
336
337 return *this;
338 }

345 template <typename storage_t>
346 DETRAY_HOST_DEVICE constexpr auto init(const storage_t& content)
347 -> dynamic_array& {
348 const_cast<data*>(m_data)->size = 0u;
349 for (dindex i{0u}; i < m_capacity; ++i) {
350 if (content[i] != detail::invalid_value<entry_type>()) {
351 const_cast<entry_type*>(m_global_storage)[i] = content[i];
352 ++(const_cast<data*>(m_data)->size);
353 }
354 }
355 return *this;
356 }

363 DETRAY_HOST_DEVICE
364 constexpr bool operator==(const dynamic_array& rhs) const {
365 // Check if the bin points to the same data
366 if (m_data == rhs.m_data || *m_data == *rhs.m_data) {
367 return true;
368 }
369 if (m_data->size != rhs.m_data->size) {
370 return false;
371 }
372 // It could still point to different data, but the
373 // content is the same
374 // Loop over the size of the bin and compare
375 for (dindex i{0u}; i < m_data->size; ++i) {
376 if (m_global_storage[i] != rhs.m_global_storage[i]) {
377 return false;
378 }
379 }
380 return true;
381 }
382
383 private:
386 entry_ptr_t m_global_storage{nullptr};
388 data_ptr_t m_data{nullptr};
390 dindex m_capacity{0u};
391};
