77 template <concepts::algebra algebra_t>
78 DETRAY_HOST_DEVICE constexpr result_type<dbool<algebra_t>> check_boundaries(
79 const bounds_type<dscalar<algebra_t>>& /*bounds*/,
80 const dtransform3D<algebra_t>& /*trf*/,
81 const dpoint3D<algebra_t>& /*glob_p*/,
82 const dscalar<algebra_t> /*tol*/ = 0.f,
83 const dscalar<algebra_t> /*edge_tol*/ = 0.f) const {

96 template <typename bounds_t, concepts::point point_t,
97 concepts::scalar scalar_t>
98 DETRAY_HOST_DEVICE constexpr auto check_boundaries(
99 const bounds_t& /*bounds*/, const point_t& /*loc_p*/,
100 const scalar_t tol = 0.f, const scalar_t edge_tol = 0.f) const {
101 using bool_t = decltype(tol < edge_tol);
102
103 if constexpr (std::same_as<result_type<bool_t>, bool_t>) {
104 return bool_t{true};
105 } else {
106 return result_type<bool_t>{true, true};
107 }
108 }
