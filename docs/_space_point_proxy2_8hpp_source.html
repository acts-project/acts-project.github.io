121 void assignSourceLinks(std::span<const SourceLink> sourceLinks) const
122 requires(!ReadOnly)
123 {
124 if (m_index >= m_container->size()) {
125 throw std::out_of_range("Index out of range in SpacePointContainer2");
126 }
127 if (!m_container->m_sourceLinkOffsetColumn.has_value() ||
128 !m_container->m_sourceLinkCountColumn.has_value()) {
129 throw std::logic_error("No source links column available");
130 }
131 if (accessImpl(m_container->m_sourceLinkCountColumn) != 0) {
132 throw std::logic_error(
133 "Source links already assigned to the space point");
134 }
135
136 accessImpl(m_container->m_sourceLinkOffsetColumn) =
137 static_cast<std::uint32_t>(m_container->m_sourceLinks.size());
138 accessImpl(m_container->m_sourceLinkCountColumn) =
139 static_cast<std::uint8_t>(sourceLinks.size());
140 m_container->m_sourceLinks.insert(m_container->m_sourceLinks.end(),
141 sourceLinks.begin(), sourceLinks.end());
142 }
143
146 std::span<SourceLink> sourceLinks() const noexcept
147 requires(!ReadOnly)
148 {
149 const std::size_t offset =
150 accessImpl(m_container->m_sourceLinkOffsetColumn);
151 const std::size_t count = accessImpl(m_container->m_sourceLinkCountColumn);
152 return std::span<SourceLink>(m_container->m_sourceLinks.data() + offset,
153 count);
154 }
