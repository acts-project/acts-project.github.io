120 void assignSourceLinks(std::span<const SourceLink> sourceLinks) const
121 requires(!ReadOnly)
122 {
123 if (m_index >= m_container->size()) {
124 throw std::out_of_range("Index out of range in SpacePointContainer2");
125 }
126 if (!m_container->m_sourceLinkOffsetColumn.has_value() ||
127 !m_container->m_sourceLinkCountColumn.has_value()) {
128 throw std::logic_error("No source links column available");
129 }
130 if (accessImpl(m_container->m_sourceLinkCountColumn) != 0) {
131 throw std::logic_error(
132 "Source links already assigned to the space point");
133 }
134
135 accessImpl(m_container->m_sourceLinkOffsetColumn) =
136 static_cast<std::uint32_t>(m_container->m_sourceLinks.size());
137 accessImpl(m_container->m_sourceLinkCountColumn) =
138 static_cast<std::uint8_t>(sourceLinks.size());
139 m_container->m_sourceLinks.insert(m_container->m_sourceLinks.end(),
140 sourceLinks.begin(), sourceLinks.end());
141 }
142
145 std::span<SourceLink> sourceLinks() const noexcept
146 requires(!ReadOnly)
147 {
148 const std::size_t offset =
149 accessImpl(m_container->m_sourceLinkOffsetColumn);
150 const std::size_t count = accessImpl(m_container->m_sourceLinkCountColumn);
151 return std::span<SourceLink>(m_container->m_sourceLinks.data() + offset,
152 count);
153 }
