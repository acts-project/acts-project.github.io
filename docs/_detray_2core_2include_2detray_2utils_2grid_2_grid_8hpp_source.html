380 DETRAY_HOST_DEVICE constexpr auto operator==(const grid_impl &rhs) const
381 -> bool {
382 // Check axes: they need to be identical
383 if (m_axes != rhs.m_axes) {
384 return false;
385 }
386 // Loop over global bin index and compare the two
387 for (glob_bin_index i = 0; i < nbins(); ++i) {
388 if (bin(i) != rhs.bin(i)) {
389 return false;
390 }
391 }
392 return true;
393 }
394
396 DETRAY_HOST
397 friend std::ostream &operator<<(std::ostream &os, const grid_impl &gr) {
398 os << gr.axes() << std::endl;
399
400 for (glob_bin_index gbin = 0; gbin < gr.nbins(); ++gbin) {
401 os << "bin " << gbin << ": ";
402 for (const auto &bin_content : gr.bin(gbin)) {
403 os << bin_content << ", ";
404 }
405 os << std::endl;
406 }
407
408 return os;
409 }
