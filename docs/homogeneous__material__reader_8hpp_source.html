72 static void from_payload_impl(detector_builder<typename detector_t::metadata,
73 volume_builder>& det_builder,
74 const detector_payload& det_data)
75 requires(detray::concepts::has_material_slabs<detector_t> ||
76 detray::concepts::has_material_rods<detector_t>)
77 {
78 DETRAY_VERBOSE_HOST("Reading payload object...");
79
80 using scalar_t = dscalar<typename detector_t::algebra_type>;
81 using mat_id = typename detector_t::material::id;
82
83 if (!det_data.homogeneous_material.has_value()) {
84 std::string err_str{"No data in homogeneous material payload"};
85 DETRAY_FATAL_HOST(err_str);
86 throw std::invalid_argument(err_str);
87 }
88
89 const payload_type& det_mat_data = *det_data.homogeneous_material;
90
91 DETRAY_DEBUG_HOST("Converting material for " << det_mat_data.volumes.size()
92 << " volumes");
93
94 // Convert the material volume by volume
95 for (const auto& mv_data : det_mat_data.volumes) {
96 const auto vol_idx{
97 detail::basic_converter::from_payload(mv_data.volume_link)};
98
99 DETRAY_DEBUG_HOST(" - volume index from payload is " << vol_idx);
100
101 if (!det_builder.has_volume(vol_idx)) {
102 std::stringstream err_stream;
103 err_stream << "Volume " << vol_idx << ": "
104 << "Cannot build homogeneous material for volume "
105 << "(volume not registered in detector builder)";
106 DETRAY_FATAL_HOST(err_stream.str());
107 throw std::invalid_argument(err_stream.str());
108 }
109
110 // Decorate the current volume builder with material
111 auto vm_builder =
112 det_builder
113 .template decorate<homogeneous_material_builder<detector_t>>(
114 vol_idx);
115 DETRAY_DEBUG_HOST(
116 "Retrieved volume builder from detector build using vol_idx="
117 << vol_idx);
118
119 // Add the material data to the factory
120 auto mat_factory =
121 std::make_shared<homogeneous_material_factory<detector_t>>();
122
123 DETRAY_DEBUG_HOST("Adding " << mv_data.surface_mat.size()
124 << " material slabs to material factory");
125 for (const auto& mat_data : mv_data.surface_mat) {
126 // Determine the material type id
127 mat_id mat_type{mat_id::e_none};
128 if constexpr (detray::concepts::has_material_rods<detector_t>) {
129 if (mat_data.type == io::material_id::rod) {
130 mat_type = mat_id::e_material_rod;
131 }
132 }
133 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
134 if (mat_data.type == io::material_id::slab) {
135 mat_type = mat_id::e_material_slab;
136 }
137 }
138 if (mat_type == mat_id::e_none) {
139 std::stringstream err_str{};
140 err_str << "Homogeneous material ID \"" << mat_data.type
141 << "\" in payload does not match any material type in "
142 "detector ("
143 << detray::types::get_name<typename detector_t::metadata>()
144 << "). Check the detector metadata!\nMaterial store:\n"
145 << detray::types::get_name<typename detector_t::material>();
146 DETRAY_FATAL_HOST(err_str.str());
147 throw std::invalid_argument(err_str.str());
148 }
149
150 const auto mat_idx{mat_data.index_in_coll.has_value()
151 ? mat_data.index_in_coll.value()
152 : detray::detail::invalid_value<std::size_t>()};
153
154 DETRAY_DEBUG_HOST("-> Surface link is: " << mat_idx);
155
156 mat_factory->add_material(
157 mat_type, from_payload_impl<scalar_t>(mat_data), mat_idx);
158 }
159
160 // Add the material to the volume
161 DETRAY_DEBUG_HOST("Adding material factory to the builder");
162 vm_builder->add_surfaces(mat_factory);
163 }
164 }

168 template <detray::concepts::scalar scalar_t>
169 static material_data<scalar_t> from_payload_impl(
170 const surface_material_payload& slab_data) {
171 return {static_cast<scalar_t>(slab_data.thickness),
172 from_payload_impl<scalar_t>(slab_data.mat),
173 detail::basic_converter::from_payload(slab_data.surface)};
174 }
175
177 template <detray::concepts::scalar scalar_t>
178 static auto from_payload_impl(const material_param_payload& mat_data) {
179 return material<scalar_t>{
180 static_cast<scalar_t>(mat_data.params[0]),
181 static_cast<scalar_t>(mat_data.params[1]),
182 static_cast<scalar_t>(mat_data.params[2]),
183 static_cast<scalar_t>(mat_data.params[3]),
184 static_cast<scalar_t>(mat_data.params[4]),
185 // The molar density (params[5]) is calculated on the fly
186 static_cast<material_state>(mat_data.params[6])};
187 }
188};
