51 template <class detector_t>
52 static void from_payload(detector_builder<typename detector_t::metadata,
53 volume_builder>& det_builder,
54 const payload_type& det_mat_data) {
55 DETRAY_VERBOSE_HOST("Reading payload object...");
56
57 using scalar_t = dscalar<typename detector_t::algebra_type>;
58 using mat_id = typename detector_t::material::id;
59
60 DETRAY_DEBUG_HOST("Converting material for " << det_mat_data.volumes.size()
61 << " volumes");
62
63 // Convert the material volume by volume
64 for (const auto& mv_data : det_mat_data.volumes) {
65 const auto vol_idx{
66 detail::basic_converter::from_payload(mv_data.volume_link)};
67
68 DETRAY_DEBUG_HOST(" - volume index from payload is " << vol_idx);
69
70 if (!det_builder.has_volume(vol_idx)) {
71 std::stringstream err_stream;
72 err_stream << "Volume " << vol_idx << ": "
73 << "Cannot build homogeneous material for volume "
74 << "(volume not registered in detector builder)";
75 DETRAY_FATAL_HOST(err_stream.str());
76 throw std::invalid_argument(err_stream.str());
77 }
78
79 // Decorate the current volume builder with material
80 auto vm_builder =
81 det_builder
82 .template decorate<homogeneous_material_builder<detector_t>>(
83 vol_idx);
84 DETRAY_DEBUG_HOST(
85 "Retrieved volume builder from detector build using vol_idx="
86 << vol_idx);
87
88 // Add the material data to the factory
89 auto mat_factory =
90 std::make_shared<homogeneous_material_factory<detector_t>>();
91
92 DETRAY_DEBUG_HOST("Adding " << mv_data.surface_mat.size()
93 << " material slabs to material factory");
94 for (const auto& mat_data : mv_data.surface_mat) {
95 // Determine the material type id
96 mat_id mat_type{mat_id::e_none};
97 if constexpr (detray::concepts::has_material_rods<detector_t>) {
98 if (mat_data.type == io::material_id::rod) {
99 mat_type = mat_id::e_material_rod;
100 }
101 }
102 if constexpr (detray::concepts::has_material_slabs<detector_t>) {
103 if (mat_data.type == io::material_id::slab) {
104 mat_type = mat_id::e_material_slab;
105 }
106 }
107 assert(mat_type != mat_id::e_none);
108
109 const auto mat_idx{mat_data.index_in_coll.has_value()
110 ? mat_data.index_in_coll.value()
111 : detray::detail::invalid_value<std::size_t>()};
112
113 DETRAY_DEBUG_HOST("-> Surface link is: " << mat_idx);
114
115 mat_factory->add_material(mat_type, from_payload<scalar_t>(mat_data),
116 mat_idx);
117 }
118
119 // Add the material to the volume
120 DETRAY_DEBUG_HOST("Adding material factory to the builder");
121 vm_builder->add_surfaces(mat_factory);
122 }
123 }

127 template <detray::concepts::scalar scalar_t>
128 static material_data<scalar_t> from_payload(
129 const surface_material_payload& slab_data) {
130 return {static_cast<scalar_t>(slab_data.thickness),
131 from_payload<scalar_t>(slab_data.mat),
132 detail::basic_converter::from_payload(slab_data.surface)};
133 }
134
136 template <detray::concepts::scalar scalar_t>
137 static auto from_payload(const material_param_payload& mat_data) {
138 return material<scalar_t>{
139 static_cast<scalar_t>(mat_data.params[0]),
140 static_cast<scalar_t>(mat_data.params[1]),
141 static_cast<scalar_t>(mat_data.params[2]),
142 static_cast<scalar_t>(mat_data.params[3]),
143 static_cast<scalar_t>(mat_data.params[4]),
144 // The molar density (params[5]) is calculated on the fly
145 static_cast<material_state>(mat_data.params[6])};
146 }
147};
