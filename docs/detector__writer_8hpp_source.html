31template <class detector_t>
32void write_detector(detector_t& det, const typename detector_t::name_map& names,
33 detector_writer_config& cfg) {
34 // How to open the file
35 const std::ios_base::openmode out_mode{std::ios_base::out |
36 std::ios_base::binary};
37 auto mode =
38 cfg.replace_files() ? (out_mode | std::ios_base::trunc) : out_mode;
39
40 const auto file_path = detray::io::create_path(cfg.path());
41
42 if (det.material_store().all_empty()) {
43 cfg.write_material(false);
44 }
45
46 // Get the writer
47 io::detail::detector_components_writer<detector_t> writer{};
48 if (cfg.format() == io::format::json) {
49 detail::add_json_writers(writer, cfg);
50 }
51
52 if (cfg.compactify_json()) {
53 DETRAY_ERROR_HOST("Compactifying json files is not yet implemented");
54 }
55
56 writer.write(det, names, cfg.source(), mode, file_path);
57}
