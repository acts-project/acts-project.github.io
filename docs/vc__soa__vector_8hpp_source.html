104template <std::size_t N, concepts::value value_t,
105 template <typename, std::size_t> class array_t>
106DETRAY_HOST_DEVICE constexpr Vc::Vector<value_t> dot(
107 const algebra::storage::vector<N, Vc::Vector<value_t>, array_t> &a,
108 const algebra::storage::vector<N, Vc::Vector<value_t>, array_t> &b) {
109 auto ret = a[0] * b[0];
110
111 for (unsigned int i{1u}; i < N; i++) {
112 ret = Vc::fma(a[i], b[i], ret);
113 }
114
115 return ret;
116}

161template <std::size_t N, concepts::value value_t,
162 template <typename, std::size_t> class array_t>
163 requires(N >= 3)
164DETRAY_HOST_DEVICE constexpr auto eta(
165 const algebra::storage::vector<N, Vc::Vector<value_t>, array_t> &v) {
166 // atanh does not exist in Vc
167 auto atanh_func = [](value_t e) { return std::atanh(e); };
168
169 return (v[2] / norm(v)).apply(atanh_func);
170
171 // Faster, but less accurate
172 // return (Vc::reciprocal(norm(v)) * v[2]).apply(atanh_func);
173
174 // Even faster, but even less accurate
175 // return (Vc::rsqrt(dot(v, v)) * v[2]).apply(atanh_func);
176}

187template <std::size_t N, concepts::value value_t,
188 template <typename, std::size_t> class array_t>
189DETRAY_HOST_DEVICE constexpr Vc::Vector<value_t> sum(
190 const algebra::storage::vector<N, Vc::Vector<value_t>, array_t> &v) {
191 Vc::Vector<value_t> res{v[0]};
192
193 for (std::size_t i = 1u; i < Vc::Vector<value_t>::size(); ++i) {
194 res = res + v[i];
195 }
196
197 return res;
198}
