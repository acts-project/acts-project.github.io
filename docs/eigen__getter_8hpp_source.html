125template <std::size_t I, std::size_t J, concepts::matrix M>
126DETRAY_HOST_DEVICE decltype(auto) element(M &matrix) {
127 if constexpr (concepts::has_compile_time_2d_access<M>) {
128 return matrix.template element<I, J>();
129 } else {
130 using index_t = detray::traits::index_t<std::decay_t<M>>;
131 return element(matrix, static_cast<index_t>(I), static_cast<index_t>(J));
132 }
133}
134
135template <std::size_t I, concepts::vector V>
136DETRAY_HOST_DEVICE decltype(auto) element(V &vector) {
137 if constexpr (concepts::has_compile_time_1d_access<V>) {
138 return vector.template element<I>();
139 } else {
140 using index_t = detray::traits::index_t<std::decay_t<V>>;
141 return element(vector, static_cast<index_t>(I));
142 }

204 template <typename derived_type1, typename derived_type2,
205 concepts::index index_t_1, concepts::index index_t_2>
206 DETRAY_HOST_DEVICE void set(Eigen::MatrixBase<derived_type1> &m,
207 const Eigen::MatrixBase<derived_type2> &b,
208 const index_t_1 row, const index_t_2 col) const {
209 using block_t = Eigen::MatrixBase<derived_type2>;
210 constexpr auto R{block_t::RowsAtCompileTime};
211 constexpr auto C{block_t::ColsAtCompileTime};
212 m.template block<R, C>(static_cast<Eigen::Index>(row),
213 static_cast<Eigen::Index>(col)) = b;
214 }

216}; // struct block_getter
