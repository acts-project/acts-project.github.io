110template <std::size_t I, std::size_t J, concepts::matrix M>
111DETRAY_HOST_DEVICE decltype(auto) element(M &matrix) {
112 if constexpr (concepts::has_compile_time_2d_access<M>) {
113 return matrix.template element<I, J>();
114 } else {
115 using index_t = detray::traits::index_t<std::decay_t<M>>;
116 return element(matrix, static_cast<index_t>(I), static_cast<index_t>(J));
117 }
118}

120template <std::size_t I, concepts::vector V>
121DETRAY_HOST_DEVICE decltype(auto) element(V &vector) {
122 if constexpr (concepts::has_compile_time_1d_access<V>) {
123 return vector.template element<I>();
124 } else {
125 using index_t = detray::traits::index_t<std::decay_t<V>>;
126 return element(vector, static_cast<index_t>(I));

189 template <typename derived_type1, typename derived_type2,
190 concepts::index index_t_1, concepts::index index_t_2>
191 DETRAY_HOST_DEVICE void set(Eigen::MatrixBase<derived_type1> &m,
192 const Eigen::MatrixBase<derived_type2> &b,
193 const index_t_1 row, const index_t_2 col) const {
194 using block_t = Eigen::MatrixBase<derived_type2>;
195 constexpr auto R{block_t::RowsAtCompileTime};
196 constexpr auto C{block_t::ColsAtCompileTime};
197 m.template block<R, C>(static_cast<Eigen::Index>(row),
198 static_cast<Eigen::Index>(col)) = b;
199 }
