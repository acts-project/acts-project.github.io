96 template <typename... Args>
97 requires(std::same_as<std::string, std::remove_cvref_t<Args>> || ...)
98 detector_reader_config& add_files(Args&&... file_names) {
99 std::array<std::string, sizeof...(Args)> tmp_files{file_names...};
100
101 m_files.reserve(m_files.size() + tmp_files.size());
102 m_files.insert(std::end(m_files), std::begin(tmp_files),
103 std::end(tmp_files));
104 return *this;
105 }

110 m_builder_opts.deduplicate(toggle);
111 return *this;

116 detector_reader_config& do_check(const bool check) {
117 m_do_check = check;
118 return *this;
119 }
120
124 detector_reader_config& verbose_check(const bool verbose) {
125 if (verbose && !m_do_check) {
126 m_do_check = true;
127 }
128 m_verbose = verbose;
129 return *this;
130 }
