44template <typename track_t,
45 typename generator_t =
46 detail::random_numbers<dscalar<typename track_t::algebra_type>>>
47class random_track_generator
48 : public detray::ranges::view_interface<
49 random_track_generator<track_t, generator_t>> {
50 using algebra_t = typename track_t::algebra_type;
51 using scalar_t = dscalar<algebra_t>;
52 using point3_t = dpoint3D<algebra_t>;
53 using vector3_t = dvector3D<algebra_t>;

108 DETRAY_HOST_DEVICE
109 track_t operator*() const {
110 if (!m_rnd_numbers) {
111 std::string err_str{"Invalid random number generator"};
112 DETRAY_FATAL_HOST(err_str);
113 throw std::invalid_argument(err_str);
114 }
115
116 const auto& ori = m_cfg.origin();
117 const auto& ori_stddev = m_cfg.origin_stddev();
118
119 const point3_t vtx =
120 m_cfg.do_vertex_smearing()
121 ? point3_t{m_rnd_numbers->normal(ori[0], ori_stddev[0]),
122 m_rnd_numbers->normal(ori[1], ori_stddev[1]),
123 m_rnd_numbers->normal(ori[2], ori_stddev[2])}
124 : point3_t{ori[0], ori[1], ori[2]};
125
126 scalar_t p_mag{(*m_rnd_numbers)(m_cfg.mom_range())};
127 scalar_t phi{(*m_rnd_numbers)(m_cfg.phi_range())};
128 scalar_t theta{(*m_rnd_numbers)(m_cfg.theta_range())};
129 scalar_t sin_theta{math::sin(theta)};
130
131 // Momentum direction from angles
132 vector3_t mom{math::cos(phi) * sin_theta, math::sin(phi) * sin_theta,
133 math::cos(theta)};
134
135 if constexpr (std::is_same_v<track_t, detail::ray<algebra_t>>) {
136 mom = vector::normalize(mom);
137 } else {
138 sin_theta = (sin_theta == scalar_t{0.f})
139 ? std::numeric_limits<scalar_t>::epsilon()
140 : sin_theta;
141 mom = (m_cfg.is_pT() ? 1.f / sin_theta : 1.f) * p_mag *
142 vector::normalize(mom);
143 }
144
145 // Randomly flip the charge sign
146 darray<double, 2> signs{1., -1.};
147 const auto sign{static_cast<scalar_t>(
148 signs[m_cfg.randomize_charge() ? m_rnd_numbers->coin_toss() : 0u])};
149
150 return track_t{vtx, m_cfg.time(), mom, sign * m_cfg.charge()};
151 }
152
154 std::shared_ptr<generator_t> m_rnd_numbers;
155
157 std::size_t m_tracks{0u};
158
160 configuration m_cfg{};
161 };
