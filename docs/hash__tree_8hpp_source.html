23template <typename value_t>
24struct default_hash {
26 using hash_type = std::size_t;
28 using data_hasher = std::hash<value_t>;
30 using node_hasher = std::hash<hash_type>;
31
35 auto operator()(const value_t &v) { return data_hasher{}(v); }
36
41 auto operator()(const hash_type &left, const hash_type &right) {
42 return node_hasher{}(left) + node_hasher{}(right);
43 }
44};

55template <typename input_collection_t,
56 typename data_t = typename input_collection_t::value_type,
57 typename hash_function_t = default_hash<data_t>,
58 typename hash_t =
59 decltype(std::declval<hash_function_t>()(data_t{0})),

138 private:
140 void build(const input_collection_t &input_data) noexcept(false) {
141 // Build leaves from input data type
142 for (const auto &data : input_data) {
143 _tree.emplace_back(_hash(data));
144 }
145 // Need an even number of leaves to build the tree correctly
146 if (input_data.size() % 2u != 0u) {
147 _tree.emplace_back(_hash(0));
148 }
149 // Size of the tree is already known (all iterators stay valid in
150 // recursion)
151 // we might need to add one dummy node per level
152 auto n_levels = static_cast<dindex>(math::log(input_data.size()));
153 _tree.reserve(2u * _tree.size() + n_levels);
154 // Build next level
155 build(_tree.begin(), static_cast<dindex>(_tree.size()));
156 }
157
163 template <typename iterator_t>
164 void build(iterator_t &&first_child, dindex n_prev_level) {
165 // base case
166 if (n_prev_level <= 1u) {
167 return;
168 }
169
170 auto last_child = first_child + static_cast<std::ptrdiff_t>(n_prev_level);
171
172 // Run over previous tree level to build the next level
173 for (auto current_child = first_child; current_child != last_child;
174 current_child += 2u) {
175 auto parent_digest =
176 _hash(current_child->key(), (current_child + 1u)->key());
177 hashed_node parent = _tree.emplace_back(parent_digest);
178
179 // Parent node index is at the back of the tree
180 current_child->set_parent(static_cast<dindex>(_tree.size()) - 1u);
181 (current_child + 1)->set_parent(static_cast<dindex>(_tree.size()) - 1u);
182
183 // Set the indices as distances in the contiguous container
184 auto left_child_idx = std::distance(current_child, _tree.begin());
185 parent.set_children(static_cast<dindex>(left_child_idx),
186 static_cast<dindex>(left_child_idx) + 1u);
187 }
188 dindex n_level = n_prev_level / 2u;
189 // Need dummy leaf node for next level?
190 if (n_level % 2u != 0u && n_level > 1u) {
191 _tree.emplace_back(0);
192 n_level++;
193 }
194 // begin next time where we ended this time
195 build(last_child, n_level);
196 }
197
199 hash_function_t _hash{hash_function_t{}};
200
202 vector_t<hashed_node> _tree = {};
203};
