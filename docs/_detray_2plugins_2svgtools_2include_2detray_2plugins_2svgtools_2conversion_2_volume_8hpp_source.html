50template <concepts::detector detector_t, typename view_t>
51auto volume(const typename detector_t::geometry_context& context,
52 const detector_t& detector,
53 const detray::tracking_volume<detector_t>& d_volume,
54 const view_t& view,
55 const styling::volume_style& style =
56 styling::tableau_colorblind::volume_style,
57 bool hide_portals = false, bool hide_passives = false,
58 bool hide_grids = false, bool hide_material = true,
59 const std::array<dindex, 2>& search_window = {2u, 2u}) {
60 using point3_container_t = std::vector<typename detector_t::point3_type>;
61
62 actsvg::proto::volume<point3_container_t> p_volume;
63 p_volume._index = d_volume.index();
64
65 // Prepare surfaces to be displayed in grids
66 std::vector<actsvg::proto::surface<point3_container_t>> p_sensitves;
67
68 // Convert grid, if present
69 auto [p_grid, grid_type] = svgtools::conversion::surface_grid(
70 detector, p_volume._index, view, style._grid_style);
71
72 for (const auto& desc : d_volume.surfaces()) {
73 const auto sf = detray::geometry::surface{detector, desc};
74
75 if (sf.is_portal()) {
76 if (!hide_portals) {
77 auto p_portals = svgtools::conversion::portal(context, detector, sf,
78 view, style._portal_style,
79 false, hide_material);
80
81 std::ranges::copy(p_portals, std::back_inserter(p_volume._portals));
82 }
83 } else if (!(sf.is_passive() && hide_passives)) {
84 const auto& sf_style = sf.is_sensitive() ? style._sensitive_surface_style
85 : style._passive_surface_style;
86
87 auto p_surfaces = svgtools::conversion::surface(
88 context, detector, sf, view, sf_style, hide_material);
89 assert(p_surfaces.size() == sf.n_masks());
90
91 for (std::size_t i = 0u; i < p_surfaces.size(); ++i) {
92 auto& p_surface = p_volume._v_surfaces.emplace_back(p_surfaces.at(i));
93
94 std::string sf_info{"* index " + std::to_string(sf.index()) +
95 "* mask " + std::to_string(i)};
96
97 p_surface._aux_info["module_info"] = {sf_info};
98 p_surface._aux_info["grid_info"] = {sf_info};
99
100 // Put the sensitive surfaces in the grids
101 if (sf.is_sensitive()) {
102 p_sensitves.push_back(p_surface);
103 }
104 }
105 }
106 }
107
108 // Add the proto grid to the proto volume and find bin associations
109 if (!hide_grids && p_grid.has_value()) {
110 p_volume._surface_grid = *p_grid;
111 p_volume._grid_associations = {
112 get_bin_association(detector, d_volume, search_window)};
113 }
114
115 p_volume._surfaces = {std::move(p_sensitves)};
116
117 return std::tuple(p_volume, grid_type);
118}
