30template <concepts::algebra algebra_t>
31struct perigee_stopper : public base_actor {
33 using scalar_t = dscalar<algebra_t>;
34
36 struct state {
40 scalar_t m_stopping_radius{10.f * unit<scalar_t>::mm};
42 // @TODO Make smaller once overstepping is solved
43 scalar_t m_on_perigee_tol{100.f * unit<scalar_t>::um};
45 unsigned int m_inner_vol_idx{0u};
46 };
47
53 template <typename propagator_state_t>
54 DETRAY_HOST_DEVICE void operator()(state &actor_state,
55 propagator_state_t &prop_state) const {
56 using detector_t = typename propagator_state_t::detector_type;
57 using perigee_intersector_t = ray_intersector<line_circular, algebra_t>;
58
59 // Nothing left to do. Propagation will exit successfully on its own
60 auto &navigation = prop_state.navigation();
61 if (navigation.finished()) {
62 return;
63 }
64
65 // Only check this in the innermost volume and during backward nav.
66 if (navigation.volume() != actor_state.m_inner_vol_idx ||
67 navigation.direction() != navigation::direction::e_backward) {
68 return;
69 }
70
71 // Volume that contains the IP
72 const tracking_volume inner_vol{navigation.detector(),
73 actor_state.m_inner_vol_idx};
74
75 // Stop at the perigee (cylindrical detectors only)
76 if (inner_vol.id() != volume_id::e_cylinder) {
77 return;
78 }
79
80 // At least the exit portal should be reachable
81 if (navigation.cache_exhausted()) {
82 navigation.abort("Perigee stopper has no next candidate");
83 prop_state.heartbeat(false);
84 return;
85 }
86
87 auto &stepping = prop_state.stepping();
88 auto &track = stepping();
89
90 // Linear track approximation
91 const detail::ray<algebra_t> trk_approx{track.pos(), -1.f * track.dir()};
92
93 // Check the stopping radius (outside the track will not be stopped)
94 assert(actor_state.m_stopping_radius > 0.f);
95 constexpr scalar_t max_hz{detail::invalid_value<scalar_t>()};
96
97 using perigee_mask_t = mask<line_circular, algebra_t>;
98 const perigee_mask_t perigee_mask{
99 static_cast<typename perigee_mask_t::links_type>(
100 actor_state.m_inner_vol_idx),
101 actor_state.m_stopping_radius, max_hz};
102
103 // The perigee is not linked to any detector surface
104 constexpr typename detector_t::surface_type inv_sf{};
105 const dtransform3D<algebra_t> identity{};
106 constexpr scalar_t mask_tolerance{0.f};
107 constexpr scalar_t overstep_tolerance{-detail::invalid_value<scalar_t>()};
108
109 const auto perigee_intr =
110 perigee_intersector_t{}(trk_approx, inv_sf, perigee_mask, identity,
111 mask_tolerance, overstep_tolerance);
112
113 scalar_t dist_to_cand{std::as_const(navigation).target().path()};
114 if (perigee_intr.is_probably_inside() &&
115 math::fabs(perigee_intr.path()) < math::fabs(dist_to_cand)) {
116 assert(actor_state.m_on_perigee_tol > 0.f);
117
118 // The track has reached the perigee: "exit success"
119 if (math::fabs(perigee_intr.path()) <= actor_state.m_on_perigee_tol) {
120 navigation.exit();
121 prop_state.heartbeat(false);
122 } else {
123 // @TODO: Use a guided navigator for this in order to catch
124 // overstepping correctly
125 stepping.template set_constraint<step::constraint::e_actor>(
126 perigee_intr.path());
127 }
128 }
129 }
130};
