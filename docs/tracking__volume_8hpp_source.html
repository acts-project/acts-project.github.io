172 template <object_id I, typename functor_t, typename... Args>
173 DETRAY_HOST_DEVICE constexpr void visit_accelerator(Args &&...args) const {
174 static_assert(I < object_id::e_all);
175
176 if (const auto &link{m_desc.template accel_link<I>()}; !link.is_invalid()) {
177 // Run over the surfaces in a single acceleration data structure
178 // and apply the functor to the resulting neighborhood
179 m_detector.accelerator_store().template visit<functor_t>(
180 link, std::forward<Args>(args)...);
181 }
182 }

190 template <typename functor_t, int I = static_cast<int>(object_id::e_all) - 1,
191 typename... Args>
192 DETRAY_HOST_DEVICE constexpr void visit_accelerators(Args &&...args) const {
193 // Get the acceleration data structures for this volume and only visit,
194 // if object type is contained in volume
195 for (std::size_t id = 0u; id < static_cast<std::size_t>(object_id::e_size);
196 ++id) {
197 if (const auto &link{m_desc.accel_link()[id]}; !link.is_invalid()) {
198 // Run over the surfaces in a single acceleration data structure
199 // and apply the functor to the resulting neighborhood
200 m_detector.accelerator_store().template visit<functor_t>(
201 link, std::forward<Args>(args)...);
202 }
203 }
204 }

214 template <surface_id I, typename functor_t, typename... Args>
215 DETRAY_HOST_DEVICE constexpr void visit_surfaces(Args &&...args) const {
216 using surface_getter_t = detail::apply_to_surfaces<functor_t>;
217
218 // Dispatch to the correct acceleration structure
219 if constexpr (I == surface_id::e_portal) {
220 visit_accelerator<object_id::e_portal, surface_getter_t>(
221 std::forward<Args>(args)...);
222 } else if constexpr (I == surface_id::e_sensitive) {
223 visit_accelerator<object_id::e_sensitive, surface_getter_t>(
224 std::forward<Args>(args)...);
225 } else if constexpr (I == surface_id::e_passive) {
226 visit_accelerator<object_id::e_passive, surface_getter_t>(
227 std::forward<Args>(args)...);
228 } else {
229 // Visit all surface types, but not other geometric objects
230 // (e.g. daughter volumes)
231 visit_accelerator<object_id::e_portal, surface_getter_t>(
232 std::forward<Args>(args)...);
233 if constexpr (object_id::e_portal != object_id::e_passive) {
234 visit_accelerator<object_id::e_passive, surface_getter_t>(
235 std::forward<Args>(args)...);
236 }
237 visit_accelerator<object_id::e_sensitive, surface_getter_t>(
238 std::forward<Args>(args)...);
239 }
240 }

271 template <object_id I, typename functor_t, typename track_t,
272 concepts::arithmetic window_size_t, typename... Args>
273 DETRAY_HOST_DEVICE constexpr void visit_neighborhood(
274 const track_t &track, const search_window<window_size_t, 2> &win_size,
275 const context_t &ctx, Args &&...args) const {
276 if constexpr (I == object_id::e_all) {
277 visit_accelerators<detail::apply_to_neighbourhood<functor_t>>(
278 m_detector, m_desc, track, win_size, ctx,
279 std::forward<Args>(args)...);
280 } else {
281 visit_accelerator<I, detail::apply_to_neighbourhood<functor_t>>(
282 m_detector, m_desc, track, win_size, ctx,
283 std::forward<Args>(args)...);
284 }
285 }

306 DETRAY_HOST bool self_check(std::ostream &os) const {
307 if (id() == volume_id::e_unknown) {
308 os << "DETRAY ERROR (HOST): Unknown volume shape type in volume:\n"
309 << *this << std::endl;
310 return false;
311 }
312 if (detail::is_invalid_value(index())) {
313 os << "DETRAY ERROR (HOST): Volume index undefined in volume:\n"
314 << *this << std::endl;
315 return false;
316 }
317 if (index() >= m_detector.volumes().size()) {
318 os << "DETRAY ERROR (HOST): Volume index out of bounds in volume:\n"
319 << *this << std::endl;
320 return false;
321 }
322 if (detail::is_invalid_value(m_desc.transform())) {
323 os << "DETRAY ERROR (HOST): Volume transform undefined in volume:\n"
324 << *this << std::endl;
325 return false;
326 }
327 if (m_desc.transform() >= m_detector.transform_store().size()) {
328 os << "DETRAY ERROR (HOST): Volume transform index out of bounds "
329 "in volume:\n"
330 << *this << std::endl;
331 return false;
332 }
333 // Only check, if there is material in the detector
334 if (!m_detector.material_store().all_empty() && has_material() &&
335 m_desc.material().is_invalid_index()) {
336 os << "DETRAY ERROR (HOST): Volume does not have valid material "
337 "link:\n"
338 << *this << std::endl;
339 return false;
340 }
341 const auto &acc_link = m_desc.accel_link();
342 if (detail::is_invalid_value(acc_link[0])) {
343 os << "DETRAY ERROR (HOST): Link to portal lookup broken: " << acc_link[0]
344 << "\n in volume: " << *this << std::endl;
345 return false;
346 }
347 if (const auto &pt_link = m_desc.template sf_link<surface_id::e_portal>();
348 detail::is_invalid_value(pt_link)) {
349 os << "DETRAY ERROR (HOST): Link to portal surfaces broken: " << pt_link
350 << "\n in volume: " << *this << std::endl;
351 return false;
352 }
353 // Check consistency of surface ranges
354 std::vector<dindex_range> sf_ranges = {
355 m_desc.template sf_link<surface_id::e_portal>()};
356
357 // Only add the other ranges in case they are not empty
358 if (const auto &sens_range =
359 m_desc.template sf_link<surface_id::e_sensitive>();
360 sens_range[0] != sens_range[1]) {
361 sf_ranges.push_back(sens_range);
362 }
363 if (const auto &psv_range =
364 m_desc.template sf_link<surface_id::e_passive>();
365 psv_range[0] != psv_range[1]) {
366 sf_ranges.push_back(psv_range);
367 }
368
369 // Sort and check that the ranges are contiguous
370 auto compare_ranges = [](const dindex_range &rg1, const dindex_range &rg2) {
371 return rg1[0] < rg2[0];
372 };
373
374 std::ranges::sort(sf_ranges, compare_ranges);
375
376 if ((sf_ranges.size() > 1 && sf_ranges[0][1] != sf_ranges[1][0]) ||
377 (sf_ranges.size() > 2 && sf_ranges[1][1] != sf_ranges[2][0])) {
378 os << "DETRAY ERROR (HOST): Surface index ranges not contiguous: "
379 << m_desc.sf_link() << "\n in volume: " << *this << std::endl;
380 return false;
381 }
382
383 // Warnings
384 bool suspicious_links = false;
385 std::stringstream warnings{};
386 for (std::size_t i = 1u; i < acc_link.size(); ++i) {
387 // An acceleration data structure link was set, but is invalid
388 if (!acc_link[i].is_invalid_id() && acc_link[i].is_invalid_index()) {
389 suspicious_links = true;
390 warnings << "Link to acceleration data structure "
391 << static_cast<int>(acc_link[i].id()) << " is invalid"
392 << std::endl;
393 }
394 }
395 if (suspicious_links) {
396 DETRAY_WARN_HOST(warnings.str() << " in volume: " << *this);
397 }
398
399 return true;
400 }
