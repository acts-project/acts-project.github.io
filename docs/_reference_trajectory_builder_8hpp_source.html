189 template <typename propagator_state_t, typename stepper_t,
190 typename navigator_t>
191 Result<void> act(propagator_state_t& state, const stepper_t& stepper,
192 const navigator_t& navigator, result_type& result,
193 const Logger& /*logger*/) const {
194 assert(result.trajectory && "No MultiTrajectory set");
195
196 if (result.finished) {
197 return Result<void>::success();
198 }
199
200 ACTS_VERBOSE("ReferenceTrajectory step at pos: "
201 << stepper.position(state.stepping).transpose()
202 << " dir: " << stepper.direction(state.stepping).transpose()
203 << " momentum: "
204 << stepper.absoluteMomentum(state.stepping));
205
206 if (result.pathLimitReached.internalLimit ==
207 std::numeric_limits<double>::max()) {
208 detail::setupLoopProtection(state, stepper, result.pathLimitReached,
209 true, logger());
210 }
211
212 if (const Surface* surface = navigator.currentSurface(state.navigation);
213 surface != nullptr) {
214 ACTS_VERBOSE("Handle Surface " << surface->geometryId() << " "
215 << state.options.direction);
216 auto res = handleSurface(*surface, state, stepper, navigator, result);
217 if (!res.ok()) {
218 ACTS_DEBUG("Error in " << state.options.direction
219 << " filter: " << res.error());
220 return res.error();
221 }
222 }
223
224 const bool isEndOfWorldReached =
225 endOfWorldReached.checkAbort(state, stepper, navigator, logger());
226 const bool isVolumeConstraintReached = volumeConstraintAborter.checkAbort(
227 state, stepper, navigator, logger());
228 const bool isPathLimitReached = result.pathLimitReached.checkAbort(
229 state, stepper, navigator, logger());
230 const bool isTargetReached =
231 targetReached.checkAbort(state, stepper, navigator, logger());
232 if (isEndOfWorldReached || isVolumeConstraintReached ||
233 isPathLimitReached || isTargetReached) {
234 ACTS_VERBOSE(
235 "Finalizing reference trajectory: "
236 << (isEndOfWorldReached ? "end of world reached; " : "")
237 << (isVolumeConstraintReached ? "volume constraint reached; " : "")
238 << (isPathLimitReached ? "path limit reached; " : "")
239 << (isTargetReached ? "target surface reached; " : ""));
240
241 if (isTargetReached) {
242 ACTS_VERBOSE("Setting parameters at target surface");
243
244 auto res = stepper.boundState(state.stepping, *targetReached.surface);
245 if (!res.ok()) {
246 ACTS_DEBUG("Error while acquiring bound state for target surface: "
247 << res.error() << " " << res.error().message());
248 return res.error();
249 } else {
250 const auto& [boundParams, jacobian, pathLength] = *res;
251 result.referenceParameters = boundParams;
252 }
253 }
254
255 result.finished = true;
256 }
257
258 return Result<void>::success();
259 }

264 const navigator_t& /*navigator*/, const result_type& result,

269 template <typename propagator_state_t, typename stepper_t,
270 typename navigator_t>
271 Result<void> handleSurface(const Surface& surface,
272 propagator_state_t& state,
273 const stepper_t& stepper,
274 const navigator_t& navigator,
275 result_type& result) const {
276 stepper.transportCovarianceToBound(state.stepping, surface,
277 freeToBoundCorrection);
278
279 const Result<detail::PointwiseMaterialEffects> materialInteractionPreRes =
280 detail::performMaterialInteraction(
281 state, stepper, surface,
282 detail::determineMaterialUpdateMode(
283 state, navigator, MaterialUpdateMode::PreUpdate),
284 NoiseUpdateMode::addNoise, multipleScattering, energyLoss,
285 logger());
286 if (!materialInteractionPreRes.ok()) {
287 ACTS_DEBUG("Error during pre-update material interaction: "
288 << materialInteractionPreRes.error());
289 return materialInteractionPreRes.error();
290 }
291
292 TrackStatePropMask mask =
293 TrackStatePropMask::Predicted | TrackStatePropMask::Jacobian;
294 TrackStateProxy trackStateProxy =
295 result.trajectory->makeTrackState(mask, result.lastTrackStateIndex);
296
297 ConstTrackStateProxy trackStateProxyConst{trackStateProxy};
298
299 trackStateProxy.setReferenceSurface(surface.getSharedPtr());
300 auto res = stepper.boundState(state.stepping, surface, false,
301 freeToBoundCorrection);
302 if (!res.ok()) {
303 ACTS_DEBUG("Propagate to surface " << surface.geometryId()
304 << " failed: " << res.error());
305 return res.error();
306 }
307 const auto& [boundParams, jacobian, pathLength] = *res;
308
309 trackStateProxy.predicted() = boundParams.parameters();
310 trackStateProxy.predictedCovariance() = state.stepping.cov;
311 trackStateProxy.jacobian() = jacobian;
312 trackStateProxy.pathLength() = pathLength;
313
314 auto typeFlags = trackStateProxy.typeFlags();
315 typeFlags.setHasParameters();
316 if (surface.hasMaterial()) {
317 typeFlags.setHasMaterial();
318 }
319
320 result.lastTrackStateIndex = trackStateProxy.index();
321
322 const Result<detail::PointwiseMaterialEffects>
323 materialInteractionPostRes = detail::performMaterialInteraction(
324 state, stepper, surface,
325 detail::determineMaterialUpdateMode(
326 state, navigator, MaterialUpdateMode::PostUpdate),
327 NoiseUpdateMode::addNoise, multipleScattering, energyLoss,
328 logger());
329 if (!materialInteractionPostRes.ok()) {
330 ACTS_DEBUG("Error during post-update material interaction: "
331 << materialInteractionPostRes.error());
332 return materialInteractionPostRes.error();
333 }
334
335 return Result<void>::success();
336 }

385 template <typename track_proxy_t, typename source_link_range_t>
386 void attachSourceLinks(
387 track_proxy_t trackProxy, const source_link_range_t& sourceLinkRange,
388 const SourceLinkSurfaceAccessor& surfaceAccessor) const {
389 const std::size_t nMeasurements = std::ranges::distance(sourceLinkRange);
390
391 ACTS_VERBOSE("Preparing " << nMeasurements << " input measurements");
392 std::unordered_map<const Surface*, SourceLink> inputMeasurements;
393 for (const SourceLink& sourceLink : sourceLinkRange) {
394 const Surface* surface = surfaceAccessor(sourceLink);
395 inputMeasurements.try_emplace(surface, sourceLink);
396 }
397
398 for (auto trackState : trackProxy.trackStates()) {
399 if (!trackState.hasReferenceSurface()) {
400 continue;
401 }
402 const Surface& surface = trackState.referenceSurface();
403
404 if (!surface.isSensitive()) {
405 continue;
406 }
407
408 auto typeFlagsMap = trackState.typeFlags();
409
410 if (const auto it = inputMeasurements.find(&surface);
411 it == inputMeasurements.end()) {
412 typeFlagsMap.setIsHole();
413 } else {
414 SourceLink sourceLink = it->second;
415
416 trackState.setUncalibratedSourceLink(std::move(sourceLink));
417 typeFlagsMap.setHasMeasurement();
418 }
419 }
420 }

463 template <typename track_proxy_t>
464 Result<void> filter(const GeometryContext& geoContext,
465 track_proxy_t trackProxy, const Updater& updater) const {
466 std::optional<TrackStateProxy> lastTrackState;
467 BoundVector accumulatedBoundDeltas = BoundVector::Zero();
468 BoundMatrix predictedCovariance = BoundMatrix::Zero();
469
470 for (auto trackState : trackProxy.trackStates()) {
471 if (lastTrackState.has_value()) {
472 // Transport the last delta to the current surface using the Jacobian of
473 // the track state
474
475 accumulatedBoundDeltas = trackState.jacobian() * accumulatedBoundDeltas;
476 trackState.predicted() += accumulatedBoundDeltas;
477
478 predictedCovariance = trackState.jacobian() * predictedCovariance *
479 trackState.jacobian().transpose();
480
481 {
482 const FreeVector freeParams = transformBoundToFreeParameters(
483 trackState.referenceSurface(), geoContext,
484 trackState.predicted());
485
486 const Result<MaterialSlab> materialSlabRes =
487 detail::evaluateMaterialSlab(
488 geoContext, trackState.referenceSurface(),
489 Direction::Forward(), freeParams.segment<3>(eFreePos0),
490 freeParams.segment<3>(eFreeDir0),
491 MaterialUpdateMode::PreUpdate);
492 if (!materialSlabRes.ok()) {
493 ACTS_DEBUG(
494 "Error evaluating material slab: " << materialSlabRes.error());
495 return materialSlabRes.error();
496 }
497 const MaterialSlab materialSlab = materialSlabRes.value();
498
499 const detail::PointwiseMaterialEffects materialEffects =
500 detail::computeMaterialEffects(
501 materialSlab, trackProxy.particleHypothesis(),
502 freeParams.segment<3>(eFreeDir0), freeParams[eFreeQOverP],
503 true, true, true);
504
505 predictedCovariance(eBoundPhi, eBoundPhi) +=
506 materialEffects.variancePhi;
507 predictedCovariance(eBoundTheta, eBoundTheta) +=
508 materialEffects.varianceTheta;
509 predictedCovariance(eBoundQOverP, eBoundQOverP) +=
510 materialEffects.varianceQoverP;
511 }
512
513 trackState.predictedCovariance() = predictedCovariance;
514 }
515
516 if (!trackState.typeFlags().hasMeasurement()) {
517 trackState.shareFrom(trackState, TrackStatePropMask::Predicted,
518 TrackStatePropMask::Filtered);
519 } else {
520 trackState.addComponents(TrackStatePropMask::Filtered);
521
522 const Result<void> updateResult =
523 updater(geoContext, trackState, logger());
524 if (!updateResult.ok()) {
525 ACTS_DEBUG("Error in filter: " << updateResult.error());
526 return updateResult.error();
527 }
528 }
529
530 lastTrackState = trackState;
531 accumulatedBoundDeltas += trackState.filtered() - trackState.predicted();
532 predictedCovariance = trackState.filteredCovariance();
533
534 {
535 const FreeVector freeParams = transformBoundToFreeParameters(
536 trackState.referenceSurface(), geoContext, trackState.filtered());
537
538 const Result<MaterialSlab> materialSlabRes =
539 detail::evaluateMaterialSlab(
540 geoContext, trackState.referenceSurface(), Direction::Forward(),
541 freeParams.segment<3>(eFreePos0),
542 freeParams.segment<3>(eFreeDir0),
543 MaterialUpdateMode::PostUpdate);
544 if (!materialSlabRes.ok()) {
545 ACTS_DEBUG(
546 "Error evaluating material slab: " << materialSlabRes.error());
547 return materialSlabRes.error();
548 }
549 const MaterialSlab materialSlab = materialSlabRes.value();
550
551 const detail::PointwiseMaterialEffects materialEffects =
552 detail::computeMaterialEffects(
553 materialSlab, trackProxy.particleHypothesis(),
554 freeParams.segment<3>(eFreeDir0), freeParams[eFreeQOverP], true,
555 true, true);
556
557 predictedCovariance(eBoundPhi, eBoundPhi) +=
558 materialEffects.variancePhi;
559 predictedCovariance(eBoundTheta, eBoundTheta) +=
560 materialEffects.varianceTheta;
561 predictedCovariance(eBoundQOverP, eBoundQOverP) +=
562 materialEffects.varianceQoverP;
563 }
564 }
565
566 return Result<void>::success();
567 }

570 auto makePropagatorOptions(const Options& actorOptions,
571 const std::vector<const Surface*>* sSequence,
572 const Surface* targetSurface) const {
573 using Actors = ActorList<Actor>;
574 using PropagatorOptions = typename propagator_t::template Options<Actors>;
575
576 PropagatorOptions propagatorOptions(actorOptions.geoContext,
577 actorOptions.magFieldContext);
578 propagatorOptions.setPlainOptions(actorOptions.propagatorPlainOptions);
579
580 if constexpr (!isDirectNavigator) {
581 if (sSequence != nullptr) {
582 for (const Surface* surface : *sSequence) {
583 propagatorOptions.navigation.appendExternalSurface(*surface);
584 }
585 }
586 } else {
587 assert(sSequence != nullptr &&
588 "DirectNavigator requires a surface sequence for "
589 "ReferenceTrajectory");
590 propagatorOptions.navigation.externalSurfaces = *sSequence;
591 }
592
593 auto& actor = propagatorOptions.actorList.template get<Actor>();
594 actor.targetReached.surface = targetSurface;
595 actor.multipleScattering = actorOptions.multipleScattering;
596 actor.energyLoss = actorOptions.energyLoss;
597 actor.freeToBoundCorrection = actorOptions.freeToBoundCorrection;
598 actor.actorLogger = m_actorLogger.get();
599
600 return propagatorOptions;
601 }

603 template <typename propagator_options_t,
604 TrackContainerFrontend track_container_t>
605 auto buildImpl(const BoundTrackParameters& sParameters,
606 const propagator_options_t& propagatorOptions,
607 track_container_t& trackContainer) const
608 -> Result<typename track_container_t::TrackProxy> {
609 auto propagatorState = m_propagator.makeState(propagatorOptions);
610
611 auto propagatorInitResult =
612 m_propagator.initialize(propagatorState, sParameters);
613 if (!propagatorInitResult.ok()) {
614 ACTS_DEBUG("Propagation initialization failed: "
615 << propagatorInitResult.error());
616 return propagatorInitResult.error();
617 }
618
619 auto& actorResult = propagatorState.template get<ResultType>();
620 actorResult.trajectory = &trackContainer.trackStateContainer();
621
622 auto result = m_propagator.propagate(propagatorState);
623
624 if (!result.ok()) {
625 ACTS_DEBUG("Propagation failed: " << result.error());
626 return result.error();
627 }
628
629 auto track = trackContainer.makeTrack();
630 track.tipIndex() = actorResult.lastTrackStateIndex;
631 if (actorResult.referenceParameters.has_value()) {
632 const auto& params = *actorResult.referenceParameters;
633 track.parameters() = params.parameters();
634 track.covariance() = params.covariance().value();
635 track.setReferenceSurface(params.referenceSurface().getSharedPtr());
636 }
637
638 track.linkForward();
639
640 return track;
641 }
