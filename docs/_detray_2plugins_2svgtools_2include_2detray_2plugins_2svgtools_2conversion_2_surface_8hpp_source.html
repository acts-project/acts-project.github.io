33template <typename mask_t>
34inline void set_measures(
35 actsvg::proto::surface<std::vector<typename mask_t::point3_type>>&
36 p_surface,
37 const mask_t& m) {
38 auto cast_scalar = [](const typename mask_t::scalar_type& v) {
39 return static_cast<actsvg::scalar>(v);
40 };
41 std::ranges::transform(m.values(), std::back_inserter(p_surface._measures),
42 cast_scalar);
43}

46template <concepts::transform3D transform3_t, typename mask_t>
47inline void set_vertices(
48 actsvg::proto::surface<std::vector<typename transform3_t::point3>>&
49 p_surface,
50 const transform3_t& trf, const mask_t& m) {
51 // Approximate any arcs in the mask shape with ten line segments
52 auto vertices = m.vertices(10u);
53 for (std::size_t i = 0; i < vertices.size(); i++) {
54 p_surface._vertices.push_back(trf.point_to_global(vertices.at(i)));
55 }
56}

61template <concepts::transform3D transform3_t, typename mask_t>
62auto inline surface(const transform3_t& transform, const mask_t& m) {
63 using point3_t = typename mask_t::point3_type;
64 using p_surface_t = actsvg::proto::surface<std::vector<point3_t>>;
65
66 p_surface_t p_surface;
67 p_surface._type = p_surface_t::type::e_polygon;
68 set_measures(p_surface, m);
69 set_vertices(p_surface, transform, m);
70
71 const point3_t center = transform.translation();
72 const auto r{static_cast<actsvg::scalar>(vector::perp(center))};
73 p_surface._radii = {r, r};
74
75 return p_surface;
76}

79template <concepts::transform3D transform3_t, typename shape_t,
80 concepts::algebra algebra_t>
81 requires std::is_same_v<shape_t, cylinder2D> ||
82 std::is_same_v<shape_t, concentric_cylinder2D>
83auto inline surface(const transform3_t& transform,
84 const mask<shape_t, algebra_t>& m) {
85 using point3_t = typename mask<shape_t, algebra_t>::point3_type;
86 using p_surface_t = actsvg::proto::surface<std::vector<point3_t>>;
87
88 p_surface_t p_surface;
89
90 const auto r = static_cast<actsvg::scalar>(m[shape_t::e_r]);
91 const auto nhz = static_cast<actsvg::scalar>(m[shape_t::e_lower_z]);
92 const auto phz = static_cast<actsvg::scalar>(m[shape_t::e_upper_z]);
93 const auto center = transform.translation();
94 const auto z0 = static_cast<actsvg::scalar>(center[2]);
95 auto hz = static_cast<actsvg::scalar>(0.5f * (phz - nhz));
96
97 // ACTS-like cylinder definition: symmetric around translation
98 if (math::fabs(nhz - phz) <= std::numeric_limits<actsvg::scalar>::epsilon()) {
99 p_surface._zparameters = {z0, hz};
100
101 } else {
102 // detray-like cylinder definition: asymmetric around translation
103 p_surface._zparameters = {nhz + hz + z0, hz};
104 }
105
106 p_surface._type = p_surface_t::type::e_cylinder;
107 p_surface._radii = {0.f, r};
108 p_surface._transform._tr = {static_cast<actsvg::scalar>(center[0]),
109 static_cast<actsvg::scalar>(center[1])};
110 set_measures(p_surface, m);
111
112 return p_surface;
113}

118 using shape_t = ring2D;

156template <concepts::transform3D transform3_t, bool kSquareCrossSect,
157 concepts::algebra algebra_t>
158auto surface(const transform3_t& transform,
159 const mask<line<kSquareCrossSect>, algebra_t>& m) {
160 using shape_t = line<kSquareCrossSect>;
161 using point3_t = typename mask<shape_t, algebra_t>::point3_type;
162 using p_surface_t = actsvg::proto::surface<std::vector<point3_t>>;
163
164 p_surface_t p_surface;
165
166 // All line surfaces are drawn as a circles(straws) in xy-view
167 const auto r{static_cast<actsvg::scalar>(m[shape_t::e_cross_section])};
168 const auto hz{static_cast<actsvg::scalar>(m[shape_t::e_half_z])};
169 const auto center = transform.translation();
170
171 p_surface._type = p_surface_t::type::e_straw;
172 p_surface._radii = {1.f, r};
173 p_surface._zparameters = {-hz, hz};
174 p_surface._transform._tr = {static_cast<actsvg::scalar>(center[0]),
175 static_cast<actsvg::scalar>(center[1])};
176 set_measures(p_surface, m);
177
178 return p_surface;
179}

244template <concepts::detector detector_t, typename view_t>
245auto surface(const typename detector_t::geometry_context& context,
246 const detector_t& detector,
247 const detray::geometry::surface<detector_t>& d_surface,
248 const view_t& view,
249 const styling::surface_style& style =
250 styling::tableau_colorblind::surface_style_sensitive,
251 bool hide_material = false) {
252 using point3_t = typename detector_t::point3_type;
253 using proto_surface_t = actsvg::proto::surface<std::vector<point3_t>>;
254
255 std::vector<proto_surface_t> p_surfaces{};
256 p_surfaces.reserve(d_surface.n_masks());
257
258 for (std::size_t i = 0u; i < d_surface.n_masks(); ++i) {
259 auto p_surface = d_surface.template visit_mask<surface_converter>(
260 d_surface.transform(context), i);
261
262 p_surface._name = "surface_" + std::to_string(d_surface.index()) +
263 "_mask_" + std::to_string(i);
264 p_surface._sf_type = d_surface.is_sensitive()
265 ? proto_surface_t::sf_type::e_sensitive
266 : proto_surface_t::sf_type::e_passive;
267
268 svgtools::styling::apply_style(p_surface, style);
269
270 p_surfaces.push_back(p_surface);
271 }
272
273 // Add the surface material only once
274 if (!hide_material && d_surface.has_material()) {
275 p_surfaces.at(0)._material = svgtools::conversion::surface_material(
276 detector, d_surface, view, style._material_style);
277 }
278
279 return p_surfaces;
280}
