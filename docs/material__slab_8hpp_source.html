24template <concepts::scalar scalar_t>
25struct material_slab {
27 using scalar_type = scalar_t;
29 using material_type = material<scalar_t>;
30
31 constexpr material_slab() = default;
32
36 constexpr material_slab(const material_type& material, scalar_type thickness)
37 : m_material(material),
38 m_thickness(thickness),
39 m_thickness_in_X0(thickness / material.X0()),
40 m_thickness_in_L0(thickness / material.L0()) {}
41
46 DETRAY_HOST_DEVICE
47 constexpr bool operator==(const material_slab& rhs) const {
48 return (m_material == rhs.get_material() && m_thickness == rhs.thickness());
49 }
50
53 DETRAY_HOST_DEVICE
54 constexpr explicit operator bool() const {
55 if (m_thickness <= std::numeric_limits<scalar_type>::epsilon() ||
56 m_thickness == std::numeric_limits<scalar_type>::max() ||
57 m_material == vacuum<scalar_type>() ||
58 m_material.mass_density() == 0.f || m_material.molar_density() == 0.f) {
59 return false;
60 }
61 return true;
62 }
63
66 DETRAY_HOST_DEVICE
67 constexpr const material_type& get_material() const { return m_material; }
70 DETRAY_HOST_DEVICE
71 constexpr scalar_type thickness() const { return m_thickness; }
74 DETRAY_HOST_DEVICE
75 constexpr scalar_type thickness_in_X0() const { return m_thickness_in_X0; }
78 DETRAY_HOST_DEVICE
79 constexpr scalar_type thickness_in_L0() const { return m_thickness_in_L0; }
80
85 DETRAY_HOST_DEVICE constexpr scalar_type path_segment(
86 const scalar_type cos_inc_angle,
87 const scalar_type /*unused*/ = 0.f) const {
88 return m_thickness / cos_inc_angle;
89 }
90
95 DETRAY_HOST_DEVICE constexpr scalar_type path_segment_in_X0(
96 const scalar_type cos_inc_angle,
97 const scalar_type /*unused*/ = 0.f) const {
98 return m_thickness_in_X0 / cos_inc_angle;
99 }
100
105 DETRAY_HOST_DEVICE constexpr scalar_type path_segment_in_L0(
106 const scalar_type cos_inc_angle,
107 const scalar_type /*unused*/ = 0.f) const {
108 return m_thickness_in_L0 / cos_inc_angle;
109 }
110
112 DETRAY_HOST
113 friend std::ostream& operator<<(std::ostream& os, const material_slab& mat) {
114 os << "slab: ";
115 os << mat.get_material();
116 os << " | thickness = " << mat.thickness() << " mm";
117
118 return os;
119 }
120
121 private:
122 material_type m_material = {};
123 scalar_type m_thickness = std::numeric_limits<scalar_type>::epsilon();
124 scalar_type m_thickness_in_X0 = std::numeric_limits<scalar_type>::epsilon();
125 scalar_type m_thickness_in_L0 = std::numeric_limits<scalar_type>::epsilon();
126};
