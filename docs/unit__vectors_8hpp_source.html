21template <concepts::vector3D vector3_t>
22struct unit_vectors {
31 DETRAY_HOST_DEVICE inline vector3_t make_curvilinear_unit_u(
32 const vector3_t& dir) {
33 vector3_t unit_u{0.f, 0.f, 0.f};
34 // explicit version of U = Z x T
35 unit_u[0] = -dir[1];
36 unit_u[1] = dir[0];
37
38 // if the absolute scale is tiny, the initial direction vector is
39 // aligned with the z-axis. the ZxT product is ill-defined since any
40 // vector in the x-y plane would be orthogonal to the direction. fix the
41 // U unit vector along the x-axis to avoid this numerical instability.
42 if (const auto scale = vector::norm(unit_u); scale < 1e-6f) {
43 unit_u[0] = 1;
44 unit_u[1] = 0;
45 } else {
46 unit_u = unit_u * (1.f / scale);
47 }
48
49 return unit_u;
50 }
51
68 DETRAY_HOST_DEVICE inline darray<vector3_t, 2> make_curvilinear_unit_vectors(
69 const vector3_t& dir) {
70 darray<vector3_t, 2> uv;
71 uv[0] = make_curvilinear_unit_u(dir);
72 uv[1] = vector::normalize(vector::cross(dir, uv[0]));
73
74 return uv;
75 }
76};
