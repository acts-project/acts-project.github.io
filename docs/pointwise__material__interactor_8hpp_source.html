90 template <typename mat_group_t, typename index_t>
91 DETRAY_HOST_DEVICE inline bool operator()(
92 [[maybe_unused]] const mat_group_t &material_group,
93 [[maybe_unused]] const index_t &mat_index, [[maybe_unused]] state &s,
94 [[maybe_unused]] const pdg_particle<scalar_type> &ptc,
95 [[maybe_unused]] const bound_track_parameters<algebra_t> &bound_params,
96 [[maybe_unused]] const scalar_type cos_inc_angle,
97 [[maybe_unused]] const scalar_type approach) const {
98 using material_t = typename mat_group_t::value_type;
99
100 // Filter material types for which to do "pointwise" interactions
101 if constexpr (concepts::surface_material<material_t>) {
102 const auto mat = detail::material_accessor::get(
103 material_group, mat_index, bound_params.bound_local());
104
105 // Return early in case of zero thickness
106 if (mat.thickness() <= std::numeric_limits<scalar_type>::epsilon()) {
107 DETRAY_DEBUG_HOST_DEVICE("Zero thickness material: skipping");
108 return false;
109 }
110
111 DETRAY_VERBOSE_HOST_DEVICE(
112 "-> Material path per X0: %f",
113 mat.path_segment_in_X0(cos_inc_angle, approach));
114
115 const scalar_type qop = bound_params.qop();
116
117 // Set to exactly zero if the particle is not charged
118 if (qop == 0.f) {
119 // TODO: Implement interactions for neutral particles here
120
121 DETRAY_DEBUG_HOST_DEVICE(
122 "Currently no interactions for neutral particles "
123 "implemented: skipping");
124
125 return false;
126 }
127
128 const scalar_type path_segment{
129 mat.path_segment(cos_inc_angle, approach)};
130
131 detail::relativistic_quantities rq(ptc, qop);
132
133 // Energy Loss
134 if (s.do_energy_loss) {
135 s.e_loss = interaction_type().compute_energy_loss_bethe_bloch(
136 path_segment, mat.get_material(), ptc, rq);
137 }
138
139 // @todo: include the radiative loss (Bremsstrahlung)
140 if (s.do_energy_loss && s.do_covariance_transport) {
141 s.sigma_qop =
142 interaction_type().compute_energy_loss_landau_sigma_QOverP(
143 path_segment, mat.get_material(), ptc, rq);
144 }
145
146 // Covariance update
147 if (s.do_multiple_scattering) {
148 // @todo: use momentum before or after energy loss in
149 // backward mode?
150 s.projected_scattering_angle =
151 interaction_type().compute_multiple_scattering_theta0(
152 mat.path_segment_in_X0(cos_inc_angle, approach), ptc, rq);
153 }
154
155 return true;
156
157 } else {
158 // For non-pointwise material interactions, do nothing
159 return false;
160 }
161 }
162 };

169 template <typename propagator_state_t>
170 DETRAY_HOST_DEVICE inline void operator()(
171 state &interactor_state, propagator_state_t &prop_state,
172 parameter_transporter_result<algebra_t> &res) const {
173 const auto &navigation = prop_state.navigation();
174
175 // Do material interaction when the track is on material surface
176 if (!navigation.encountered_sf_material()) {
177 return;
178 }
179
180 DETRAY_VERBOSE_HOST_DEVICE("Actor: Resolve material effects:");
181
182 interactor_state.reset();
183
184 const auto &stepping = prop_state.stepping();
185
186 const bool success = this->update(
187 prop_state.context(), stepping.particle_hypothesis(),
188 res.destination_params(), interactor_state,
189 static_cast<int>(navigation.direction()), navigation.current_surface());
190
191 res.status = success ? actor::status::e_success : res.status;
192 }

203 template <typename context_t, typename surface_t>
204 DETRAY_HOST_DEVICE inline bool update(
205 const context_t gctx, const pdg_particle<scalar_type> &ptc,
206 bound_track_parameters<algebra_t> &bound_params, state &interactor_state,
207 const int nav_dir, const surface_t &sf) const {
208 // Closest approach of the track to a line surface. Otherwise this is
209 // ignored.
210 const auto approach{bound_params[e_bound_loc0]};
211 const scalar_type cos_inc_angle{
212 cos_angle(gctx, sf, bound_params.dir(), bound_params.bound_local())};
213
214 const bool success = sf.template visit_material<kernel>(
215 interactor_state, ptc, bound_params, cos_inc_angle, approach);
216
217 if (success) {
218 auto &covariance = bound_params.covariance();
219
220 if (interactor_state.do_energy_loss) {
221 update_qop(bound_params, ptc, interactor_state.e_loss, nav_dir);
222
223 if (interactor_state.do_covariance_transport) {
224 update_qop_variance(covariance, interactor_state.sigma_qop);
225 }
226 }
227
228 if (interactor_state.do_covariance_transport) {
229 update_angle_variance(covariance, bound_params.dir(),
230 interactor_state.projected_scattering_angle);
231 }
232 }
233
234 assert(!bound_params.is_invalid());
235
236 return success;
237 }

245 DETRAY_HOST_DEVICE
246 inline void update_qop(bound_param_vector_type &vector,
247 const pdg_particle<scalar_type> &ptc,
248 const scalar_type e_loss, const int sign) const {
249 const scalar_type m = ptc.mass();
250 const scalar_type q = ptc.charge();
251 const scalar_type p = vector.p(q);
252
253 // Get new Energy
254 const scalar_type next_e{
255 math::sqrt(m * m + p * p) -
256 math::copysign(e_loss, static_cast<scalar_type>(sign))};
257
258 DETRAY_DEBUG_HOST_DEVICE("-> new energy: %f GeV", next_e);
259
260 // Put particle at rest if energy loss is too large
261 const scalar_type next_p{(m < next_e) ? math::sqrt(next_e * next_e - m * m)
262 : 0.f};
263
264 // For neutral particles, qoverp = 1/p
265 constexpr auto inv{detail::invalid_value<scalar_type>()};
266 const scalar_type next_qop{(q != 0.f) ? q / next_p : 1.f / next_p};
267 vector.set_qop((next_p == 0.f) ? inv : next_qop);
268
269 DETRAY_DEBUG_HOST_DEVICE("-> new abs. momentum: %f GeV", next_p);
270
271 DETRAY_DEBUG_HOST_DEVICE("-> Update qop: before: %f e/GeV, after: %f e/GeV",
272 q / p, vector.qop());
273 }

279 DETRAY_HOST_DEVICE inline void update_qop_variance(
280 bound_matrix_type &covariance, const scalar_type sigma_qop) const {
281 const scalar_type variance_qop{sigma_qop * sigma_qop};
282
283 DETRAY_DEBUG_HOST_DEVICE("-> qop variance: %f (e/GeV)^2", variance_qop);
284
285 getter::element(covariance, e_bound_qoverp, e_bound_qoverp) += variance_qop;
286
287 DETRAY_DEBUG_HOST_DEVICE(
288 "-> Update qop variance: before: %f (e/GeV)^2, after: %f "
289 "(e/GeV)^2",
290 getter::element(covariance, e_bound_qoverp, e_bound_qoverp) -
291 variance_qop,
292 getter::element(covariance, e_bound_qoverp, e_bound_qoverp));
293 }

300 DETRAY_HOST_DEVICE inline void update_angle_variance(
301 bound_matrix_type &covariance, const vector3_type &dir,
302 const scalar_type projected_scattering_angle) const {
303 const scalar_type var_scattering_angle{projected_scattering_angle *
304 projected_scattering_angle};
305
306 DETRAY_DEBUG_HOST_DEVICE("-> Scattering angle variance: %f rad^2",
307 var_scattering_angle);
308
309 constexpr auto inv{detail::invalid_value<scalar_type>()};
310
311 DETRAY_DEBUG_HOST_DEVICE("-> Update phi variance:");
312 DETRAY_DEBUG_HOST_DEVICE(
313 "--> before: %f rad^2",
314 getter::element(covariance, e_bound_phi, e_bound_phi));
315
316 getter::element(covariance, e_bound_phi, e_bound_phi) +=
317 (dir[2] == 1.f) ? inv : var_scattering_angle / (1.f - dir[2] * dir[2]);
318
319 DETRAY_DEBUG_HOST_DEVICE(
320 "--> after: %f rad^2",
321 getter::element(covariance, e_bound_phi, e_bound_phi));
322
323 DETRAY_DEBUG_HOST_DEVICE("-> Update theta variance:");
324 DETRAY_DEBUG_HOST_DEVICE(
325 "--> before: %f rad^2",
326 getter::element(covariance, e_bound_theta, e_bound_theta));
327
328 getter::element(covariance, e_bound_theta, e_bound_theta) +=
329 var_scattering_angle;
330
331 DETRAY_DEBUG_HOST_DEVICE(
332 "--> after: %f rad^2",
333 getter::element(covariance, e_bound_theta, e_bound_theta));
334 }
