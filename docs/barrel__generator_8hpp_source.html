144template <concepts::detector detector_t, typename mask_shape_t = rectangle2D>
145class barrel_generator final : public surface_factory_interface<detector_t> {
146 using algebra_t = typename detector_t::algebra_type;
147 using scalar_t = dscalar<algebra_t>;
148 using transform3_t = dtransform3D<algebra_t>;
149 using point3_t = dpoint3D<algebra_t>;
150 using vector3_t = dvector3D<algebra_t>;
151
152 public:
154 DETRAY_HOST
155 explicit barrel_generator(const barrel_generator_config<scalar_t> &cfg)
156 : m_cfg{cfg} {}
157
159 DETRAY_HOST
160 auto size() const -> dindex override {
161 return static_cast<dindex>(m_cfg.binning().first * m_cfg.binning().second);
162 }
163
166 DETRAY_HOST
167 void clear() override { /*Do nothing*/ };
168 DETRAY_HOST
169 void push_back(
170 surface_data<detector_t> && /*unused*/) override { /*Do nothing*/ }
171 DETRAY_HOST
173 auto push_back(std::vector<surface_data<detector_t>> && /*unused*/)
174 -> void override { /*Do nothing*/ }
175
176
184 DETRAY_HOST
186 auto operator()(typename detector_t::volume_type &volume,
187 typename detector_t::surface_lookup_container &surfaces,
188 typename detector_t::transform_container &transforms,
189 typename detector_t::mask_container &masks,
190 typename detector_t::geometry_context ctx = {})
191 -> dindex_range override {
192 DETRAY_VERBOSE_HOST("Generate silicon tracker barrel modules...");
193 DETRAY_VERBOSE_HOST("-> Generate " << size() << " surfaces");
194
195 using surface_t = typename detector_t::surface_type;
196 using nav_link_t = typename surface_t::navigation_link;
197 using mask_link_t = typename surface_t::mask_link;
198 using material_link_t = typename surface_t::material_link;
199
200 const dindex surfaces_offset{static_cast<dindex>(surfaces.size())};
201 constexpr auto invalid_src_link{detail::invalid_value<std::uint64_t>()};
202
203 // The type id of the surface mask shape
204 constexpr auto mask_id{
205 types::id<typename detector_t::masks, mask<mask_shape_t, algebra_t>>};
206
207 // The material will be added in a later step
208 constexpr auto no_material{surface_t::material_id::e_none};
209
210 auto volume_idx{volume.index()};
211 // Modules link back to mother volume in navigation
212 const auto mask_volume_link{static_cast<nav_link_t>(volume_idx)};
213
214 // surface grid bins
215 const unsigned int n_phi_bins{m_cfg.binning().first};
216 const unsigned int n_z_bins{m_cfg.binning().second};
217
218 // module positions
219 std::vector<point3_t> mod_centers;
220 mod_centers.reserve(n_phi_bins * n_z_bins);
221
222 // prep work
223 const scalar_t phi_step{2.f * constant<scalar_t>::pi /
224 static_cast<scalar_t>(n_phi_bins)};
225 const scalar_t min_phi{-constant<scalar_t>::pi + 0.5f * phi_step};
226
227 // @TODO: Only work for rectangles
228 const scalar_t z_start{
229 -0.5f * static_cast<scalar_t>(n_z_bins - 1u) *
230 (2.f * m_cfg.module_bounds().at(1) - m_cfg.z_overlap())};
231 const scalar_t z_step{(math::fabs(z_start) - z_start) /
232 static_cast<scalar_t>(n_z_bins - 1)};
233
234 // loop over the z bins
235 for (unsigned int z_bin = 0u; z_bin < n_z_bins; ++z_bin) {
236 // prepare z and r
237 const scalar_t mod_z{z_start + static_cast<scalar_t>(z_bin) * z_step};
238 const scalar_t mod_r{
239 (z_bin % 2u) != 0u ? m_cfg.radius() - 0.5f * m_cfg.radial_stagger()
240 : m_cfg.radius() + 0.5f * m_cfg.radial_stagger()};
241
242 for (unsigned int phiBin = 0u; phiBin < n_phi_bins; ++phiBin) {
243 // calculate the current phi value
244 const scalar_t mod_phi{min_phi +
245 static_cast<scalar_t>(phiBin) * phi_step};
246 mod_centers.push_back(point3_t{mod_r * math::cos(mod_phi),
247 mod_r * math::sin(mod_phi), mod_z});
248 }
249 }
250
251 // Create geometry data
252 for (auto &mod_center : mod_centers) {
253 // Surfaces with the linking into the local containers
254 mask_link_t mask_link = {mask_id, {masks.template size<mask_id>(), 1u}};
255 material_link_t material_link{no_material, dindex_invalid};
256 const auto trf_index = transforms.size(ctx);
257
258 surfaces.push_back({trf_index, mask_link, material_link, volume_idx,
259 surface_id::e_sensitive},
260 invalid_src_link);
261
262 // Build the transform
263 // The local phi
264 const scalar_t mod_phi{vector::phi(mod_center)};
265 // Local z axis is the normal vector
266 const scalar_t tilt_phi{m_cfg.tilt_phi()};
267 const vector3_t mod_local_z{math::cos(mod_phi + tilt_phi),
268 math::sin(mod_phi + tilt_phi),
269 static_cast<scalar_t>(0)};
270 // Local x axis the normal to local y,z
271 const vector3_t mod_local_x{-math::sin(mod_phi + tilt_phi),
272 math::cos(mod_phi + tilt_phi),
273 static_cast<scalar_t>(0)};
274
275 // Create the module transform
276 transforms.emplace_back(ctx, mod_center, mod_local_z, mod_local_x);
277 }
278
279 // Add the mask
280 masks.template emplace_back<mask_id>(empty_context{}, m_cfg.module_bounds(),
281 mask_volume_link);
282
283 return {surfaces_offset, static_cast<dindex>(surfaces.size())};
284 }
285
286 private:
288 barrel_generator_config<scalar_t> m_cfg{};
289};
