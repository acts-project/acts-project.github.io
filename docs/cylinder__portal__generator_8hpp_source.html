154template <concepts::detector detector_t>
155class cylinder_portal_generator final
156 : public surface_factory_interface<detector_t> {
157 using algebra_t = typename detector_t::algebra_type;
158 using scalar_t = dscalar<algebra_t>;
159 using point3_t = dpoint3D<algebra_t>;
160 using vector3_t = dvector3D<algebra_t>;
161 using transform3_t = dtransform3D<algebra_t>;
162
163 using surface_type = typename detector_t::surface_type;
164 using mask_id = typename detector_t::masks::id;
165 using mask_link_t = typename surface_type::mask_link;
166 using material_id = typename detector_t::material::id;
167 using material_link_t = typename surface_type::material_link;

170 struct bounding_box_creator {
171 using aabb_t = axis_aligned_bounding_volume<cuboid3D, algebra_t>;
172
173 template <typename mask_group_t, typename idx_range_t>
174 DETRAY_HOST_DEVICE inline void operator()(
175 const mask_group_t &mask_group, const idx_range_t &idx_range,
176 const scalar_t envelope, const transform3_t &trf,
177 std::vector<aabb_t> &boxes) const {
178 for (const auto &mask : detray::ranges::subrange(mask_group, idx_range)) {
179 // Local minimum bounding box
180 aabb_t box{mask, boxes.size(), envelope};
181 // Bounding box in global coordinates (might no longer be
182 // minimum)
183 boxes.push_back(box.transform(trf));
184 }
185 }
186 };

238 DETRAY_HOST
239 auto operator()(typename detector_t::volume_type &volume,
240 typename detector_t::surface_lookup_container &surfaces,
241 typename detector_t::transform_container &transforms,
242 typename detector_t::mask_container &masks,
243 typename detector_t::geometry_context ctx = {})
244 -> dindex_range override {
245 DETRAY_VERBOSE_HOST("Generate cylinder portals...");
246
247 using aabb_t = axis_aligned_bounding_volume<cuboid3D, algebra_t>;
248
249 // Only build portals for cylinder volumes
250 assert(volume.id() == volume_id::e_cylinder);
251 const dindex vol_idx{volume.index()};
252
253 const std::size_t n_surfaces{surfaces.size()};
254
255 if (!m_cfg.do_autofit()) {
256 // Without autofit, the portal bounds have to be given explicitly
257 assert(!(m_cfg.fixed_inner_radius() == 0.f &&
258 m_cfg.fixed_outer_radius() == 0.f) ||
259 m_cfg.fixed_half_length() != 0.f);
260 } else {
261 // Need surfaces in volume to do autofit
262 assert(n_surfaces != 0u);
263 }
264
265 scalar_t inner_r{m_cfg.fixed_inner_radius()};
266 scalar_t outer_r{m_cfg.fixed_outer_radius()};
267 scalar_t lower_z{-m_cfg.fixed_half_length()};
268 scalar_t upper_z{m_cfg.fixed_half_length()};
269
270 if (m_cfg.do_autofit()) {
271 // The bounding boxes around the module surfaces
272 std::vector<aabb_t> boxes;
273 boxes.reserve(n_surfaces);
274
275 for (const auto &sf : surfaces) {
276 masks.template visit<bounding_box_creator>(
277 sf.mask(), m_cfg.envelope(), transforms.at(sf.transform(), ctx),
278 boxes);
279 }
280
281 // Build an aabb in the global space around the surface aabbs
282 aabb_t world_box{boxes, boxes.size(), m_cfg.envelope()};
283 // translation
284 const point3_t center = world_box.template center<point3_t>();
285
286 // The world box local frame is the global coordinate frame
287 const point3_t box_min = world_box.template loc_min<point3_t>();
288 const point3_t box_max = world_box.template loc_max<point3_t>();
289
290 // Get the half lengths for the cylinder height and disc translation
291 const point3_t h_lengths = 0.5f * (box_max - box_min);
292 const scalar_t h_x{math::fabs(h_lengths[0])};
293 const scalar_t h_y{math::fabs(h_lengths[1])};
294 const scalar_t h_z{math::fabs(h_lengths[2])};
295
296 const scalar_t outer_r_min{math::max(h_x, h_y)};
297 const scalar_t mean_radius{get_mean_radius(surfaces, transforms)};
298
299 outer_r = outer_r_min;
300 inner_r = mean_radius - (outer_r - mean_radius);
301 lower_z = math::min(center[2] - h_z, center[2] + h_z);
302 upper_z = math::max(center[2] - h_z, center[2] + h_z);
303 }
304
305 // Observe boundary conditions
306 if (m_cfg.fixed_inner_radius() > 0.f) {
307 inner_r = m_cfg.fixed_inner_radius();
308 }
309 if (m_cfg.fixed_outer_radius() > 0.f) {
310 outer_r = m_cfg.fixed_outer_radius();
311 }
312 if (m_cfg.fixed_half_length() > 0.f) {
313 lower_z = -m_cfg.fixed_half_length();
314 upper_z = m_cfg.fixed_half_length();
315 }
316
317 m_boundaries = {inner_r, outer_r, lower_z, upper_z};
318
319 // If inner radius is 0, skip adding the inner cylinder
320 if (m_cfg.build_inner()) {
321 add_cylinder_portal(surfaces, transforms, masks, ctx, vol_idx,
322 m_cfg.link_south(), inner_r, lower_z, upper_z);
323 }
324 add_cylinder_portal(surfaces, transforms, masks, ctx, vol_idx,
325 m_cfg.link_north(), outer_r, lower_z, upper_z);
326
327 add_disc_portal(surfaces, transforms, masks, ctx, vol_idx,
328 m_cfg.link_west(), inner_r, outer_r, lower_z);
329 add_disc_portal(surfaces, transforms, masks, ctx, vol_idx,
330 m_cfg.link_east(), inner_r, outer_r, upper_z);
331
332 return {static_cast<dindex>(n_surfaces),
333 static_cast<dindex>(surfaces.size())};
334 }

336 private:
338 void add_cylinder_portal(
339 typename detector_t::surface_lookup_container &surfaces,
340 typename detector_t::transform_container &transforms,
341 typename detector_t::mask_container &masks,
342 typename detector_t::geometry_context ctx, dindex vol_idx,
343 dindex vol_link, const scalar_t r, const scalar_t lower_z,
344 const scalar_t upper_z) const {
345 constexpr auto invalid_src_link{detail::invalid_value<std::uint64_t>()};
346
347 const scalar_t min_z{math::min(lower_z, upper_z)};
348 const scalar_t max_z{math::max(lower_z, upper_z)};
349
350 // translation
351 const point3_t tsl{0.f, 0.f, 0.f};
352
353 // Add transform and mask data
354 transforms.emplace_back(ctx, tsl);
355 masks.template emplace_back<mask_id::e_concentric_cylinder2D>(
356 empty_context{}, vol_link, r, min_z, max_z);
357
358 // Add surface links
359 mask_link_t mask_link{};
360 const auto mask_idx{static_cast<dindex>(
361 masks.template size<mask_id::e_concentric_cylinder2D>() - 1u)};
362 set_mask_link(mask_link, mask_id::e_concentric_cylinder2D, mask_idx);
363
364 material_link_t material_link{material_id::e_none, dindex_invalid};
365
366 surfaces.push_back(
367 {static_cast<dindex>(transforms.size(ctx) - 1u), mask_link,
368 material_link, vol_idx, surface_id::e_portal},
369 invalid_src_link);
370 }
371
373 void add_disc_portal(typename detector_t::surface_lookup_container &surfaces,
374 typename detector_t::transform_container &transforms,
375 typename detector_t::mask_container &masks,
376 typename detector_t::geometry_context ctx,
377 dindex vol_idx, dindex vol_link, const scalar_t inner_r,
378 const scalar_t outer_r, const scalar_t z) const {
379 constexpr auto invalid_src_link{detail::invalid_value<std::uint64_t>()};
380
381 const scalar_t min_r{math::min(inner_r, outer_r)};
382 const scalar_t max_r{math::max(inner_r, outer_r)};
383
384 // translation
385 point3_t tsl{static_cast<scalar_t>(0), static_cast<scalar_t>(0), z};
386
387 // Add transform and mask data
388 transforms.emplace_back(ctx, tsl);
389 masks.template emplace_back<mask_id::e_ring2D>(empty_context{}, vol_link,
390 min_r, max_r);
391
392 // Add surface links
393 mask_link_t mask_link{};
394 const auto mask_idx{
395 static_cast<dindex>(masks.template size<mask_id::e_ring2D>() - 1u)};
396 set_mask_link(mask_link, mask_id::e_ring2D, mask_idx);
397
398 material_link_t material_link{material_id::e_none, dindex_invalid};
399
400 surfaces.push_back(
401 {static_cast<dindex>(transforms.size(ctx) - 1u), mask_link,
402 material_link, vol_idx, surface_id::e_portal},
403 invalid_src_link);
404 }

407 template <typename surface_container_t, typename transform_container_t>
408 scalar_t get_mean_radius(const surface_container_t &surfaces,
409 const transform_container_t &transforms) const {
410 double mean{0.};
411
412 for (const auto &sf_desc : surfaces) {
413 const auto &trf = transforms.at(sf_desc.transform());
414 mean += vector::perp(trf.translation());
415 }
416
417 return static_cast<scalar_t>(mean / static_cast<double>(surfaces.size()));
418 }
419
421 void set_mask_link(mask_link_t &mask_link, const mask_id id,
422 const dindex lower_idx) const {
423 mask_link.set_id(id);
424 if constexpr (concepts::interval<typename mask_link_t::index_type>) {
425 mask_link.set_index({lower_idx, 1u});
426 } else {
427 mask_link.set_index(lower_idx);
428 }
429 }

432 cylinder_portal_config<scalar_t> m_cfg;
434 boundaries m_boundaries{};
