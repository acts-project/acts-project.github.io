49template <typename bounds_t>
50std::shared_ptr<const bounds_t> fromJson(const nlohmann::json& j) {
51 const std::size_t kValues = bounds_t::BoundValues::eSize;
52 std::array<double, kValues> bValues{};
53 std::vector<double> bVector = j["values"];
54 std::copy_n(bVector.begin(), kValues, bValues.begin());
55 return std::make_shared<const bounds_t>(bValues);
56}
