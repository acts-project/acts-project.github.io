167 DETRAY_HOST_DEVICE
168 transform3(const matrix44 &m, const matrix44 &m_inv)
169 : _data{m}, _data_inv{m_inv} {
170 // The assertion will not hold for (casts to) int
171 if constexpr (std::floating_point<scalar_type>) {
172 // The concrete type of matrix mult is not available at this point
173 auto prod = generic::math::zero<matrix44>();
174
175 constexpr element_getter elem{};
176
177 for (index_t i = 0; i < 4; ++i) {
178 for (index_t j = 0; j < 4; ++j) {
179 for (index_t k = 0; k < 4; ++k) {
180 elem(prod, k, j) += elem(m, k, i) * elem(m_inv, i, j);
181 }
182 }
183 }
184
185 [[maybe_unused]] constexpr auto epsilon{
186 std::numeric_limits<scalar_type>::epsilon()};
187 assert(algebra::approx_equal(prod, generic::math::identity<matrix44>(),
188 16.f * epsilon, 1e-6f));
189 }
190 }

220 DETRAY_HOST_DEVICE
221 constexpr bool operator==(const transform3 &rhs) const {
222 for (index_t j = 0; j < 4; j++) {
223 // Check only the rows that can differ
224 for (index_t i = 0; i < 3; i++) {
225 if (element_getter{}(_data, i, j) !=
226 element_getter{}(rhs._data, i, j)) {
227 return false;
228 }
229 }
230 }
231
232 return true;
233 }

241 static constexpr vector3 rotate(const matrix44 &m, const vector2 &v) {
242 vector3 ret{0.f, 0.f, 0.f};
243
244 element_getter{}(ret, 0) +=
245 element_getter{}(m, 0, 0) * element_getter{}(v, 0);
246 element_getter{}(ret, 1) +=
247 element_getter{}(m, 1, 0) * element_getter{}(v, 0);
248 element_getter{}(ret, 2) +=
249 element_getter{}(m, 2, 0) * element_getter{}(v, 0);
250
251 element_getter{}(ret, 0) +=
252 element_getter{}(m, 0, 1) * element_getter{}(v, 1);
253 element_getter{}(ret, 1) +=
254 element_getter{}(m, 1, 1) * element_getter{}(v, 1);
255 element_getter{}(ret, 2) +=
256 element_getter{}(m, 2, 1) * element_getter{}(v, 1);
257
258 return ret;
259 }
