322 template <typename actor_states_t, bool is_owning>
323 requires(concepts::is_state_of<actor_states_t, actor_chain_type>)
324 DETRAY_HOST_DEVICE bool propagate(
325 state_base<is_owning> &propagation,
326 actor_states_t actor_state_refs = dtuple<>{}) const {
327 auto &navigation = propagation.navigation();
328 auto &stepping = propagation.stepping();
329 auto &context = propagation.context();
330 const auto &track = stepping();
331 assert(!track.is_invalid());
332
333 DETRAY_VERBOSE_HOST("Starting propagation for track:\n" << track);
334
335 bool is_init = false;
336 if (this->is_paused(propagation)) {
337 DETRAY_VERBOSE_HOST("Resuming propagation...");
338 } else {
339 // Initialize the navigation
340 DETRAY_VERBOSE_HOST("Initialize navigation...");
341 m_navigator.init(track, navigation, m_cfg.navigation, context);
342 propagation.heartbeat(navigation.is_alive());
343
344 is_init = true;
345 }
346
347 // Run while there is a heartbeat. In order to help the compiler
348 // optimize this, and in order to make the code more GPU-friendly,
349 // this code is run as a flat loop, but this loop has a defined
350 // structure. Indeed, the structure is always to run either the
351 // actors or the stepper (in alternating order) followed by the
352 // navigation update.
353 //
354 // A = actors
355 // N = navigation update
356 // S = propagation step
357 //
358 // ANSNANSNANSNANSNANSNANS...
359 scalar_type path_length{0.f};
360 unsigned int stall_counter{0u};
361 for (unsigned int i = 0; i % 2 == 0 || propagation.is_alive(); ++i) {
362 if (i % 2 == 0) {
363 DETRAY_VERBOSE_HOST_DEVICE("Propagation step: %d", i / 2);
364 DETRAY_VERBOSE_HOST_DEVICE("-> Path length: %f mm",
365 stepping.path_length());
366
367 // Run all registered actors/aborters
368 run_actors(actor_state_refs, propagation);
369
370 // Don't run another navigation update, if already exited
371 if (!propagation.is_alive()) {
372 continue;
373 }
374
375 path_length = stepping.path_length();
376
377 assert(!track.is_invalid());
378 } else {
379 assert(!track.is_invalid());
380
381 // Set access to the volume material for the stepper
382 auto vol = navigation.current_volume();
383 const material<scalar_type> *vol_mat_ptr =
384 vol.has_material() ? vol.material_parameters(track.pos()) : nullptr;
385
386 // Break automatic step size scaling by the stepper when a
387 // surface was reached and whenever the navigation is
388 // (re-)initialized
389 const bool reset_stepsize{navigation.is_on_surface() || is_init};
390
391 // Take the step
392 DETRAY_VERBOSE_HOST("Calling stepper...");
393 // GCC reports 'propagator' as maybe-uninitialized here once this call
394 // is inlined at -O3. It is a false positive: m_cfg is bound in the
395 // constructor's init list and m_stepper/m_navigator have default member
396 // initializers, so the object is always fully initialized. The pragma
397 // has to sit here, at the diagnostic's own location, rather than in the
398 // including header -- that way it applies no matter which header first
399 // pulls this file into a translation unit.
400#if defined(__GNUC__) && !defined(__clang__)
401#pragma GCC diagnostic push
402#pragma GCC diagnostic ignored "-Wmaybe-uninitialized"
403#endif
404 propagation.heartbeat(propagation.heartbeat() &&
405 m_stepper.step(navigation(), stepping,
406 m_cfg.stepping, reset_stepsize,
407 vol_mat_ptr));
408#if defined(__GNUC__) && !defined(__clang__)
409#pragma GCC diagnostic pop
410#endif
411
412 // Reduce navigation trust level according to stepper update
413 DETRAY_VERBOSE_HOST("-> Evaluate stepper navigation policy:");
414 typename stepper_t::policy_type{}(stepping.policy_state(), propagation);
415
416 if (i > 0) {
417 is_init = false;
418 }
419
420 // Check if the propagation makes progress
421 if (math::fabs(stepping.path_length()) <=
422 math::fabs(path_length) +
423 m_cfg.navigation.intersection.path_tolerance) {
424 if (stall_counter >= 10u) {
425 propagation.heartbeat(false);
426 navigation.abort("Propagation stalled");
427 DETRAY_ERROR_HOST("Propagation stalled");
428 } else if (stall_counter > 2u) {
429 // Print a warning if the propagation starts stalling
430 // (no overlap)
431 DETRAY_WARN_HOST("Propagation is stalling (counter "
432 << stall_counter << ")");
433 DETRAY_VERBOSE_DEVICE("Propagation is stalling (counter %d)",
434 stall_counter);
435 DETRAY_WARN_HOST(print(propagation));
436 DETRAY_WARN_HOST("-> Track: " << stepping());
437 }
438 DETRAY_DEBUG_HOST_DEVICE("-> Step stalled. Counter %d",
439 stall_counter);
440 stall_counter++;
441 } else {
442 stall_counter = 0u;
443 }
444 }
445
446 // Find next candidate
447 DETRAY_VERBOSE_HOST("Calling navigator...");
448 const bool nav_is_init =
449 m_navigator.update(track, navigation, m_cfg.navigation, context);
450 is_init = is_init || nav_is_init;
451
452 propagation.heartbeat(propagation.heartbeat() && navigation.is_alive());
453
454 if (i % 2 == 0 && i > 0 && propagation.debug()) {
455 DETRAY_VERBOSE_HOST(print(propagation));
456 }
457 }
458
459 // Pass on the whether the propagation was successful
460 DETRAY_VERBOSE_HOST("Finished propagation for track:\n" << track);
461 if (finished(propagation)) {
462 DETRAY_VERBOSE_HOST_DEVICE("Status: SUCCESS");
463 } else if (is_paused(propagation)) {
464 DETRAY_VERBOSE_HOST_DEVICE("Status: PAUSED");
465 } else {
466 DETRAY_VERBOSE_HOST_DEVICE("Status: ABORT");
467 }
468
469 return finished(propagation) || is_paused(propagation);
470 }

486 template <bool is_owning>
487 DETRAY_HOST std::string print(state_base<is_owning> &propagation) const {
488 const auto &navigation = propagation.navigation();
489 const auto &stepping = propagation.stepping();
490
491 std::stringstream debug_stream{};
492 debug_stream << std::left << std::setw(10);
493 debug_stream << "\nstatus: " << navigation.status() << std::endl;
494
495 debug_stream << "volume: " << std::setw(10);
496 if (detail::is_invalid_value(navigation.volume())) {
497 debug_stream << "invalid";
498 } else {
499 debug_stream << navigation.volume();
500 }
501 debug_stream << std::endl;
502
503 debug_stream << "navigation:" << std::endl;
504 if (navigation.is_on_surface()) {
505 debug_stream << std::setw(10)
506 << " -> on surface: " << navigation.geometry_identifier();
507 } else {
508 debug_stream << std::setw(10) << " -> target: "
509 << navigation.target().surface().identifier();
510 }
511 debug_stream << std::endl;
512 debug_stream << std::setw(10) << " -> path: " << navigation() << " mm"
513 << std::endl;
514
515 debug_stream << "stepping:" << std::endl;
516 debug_stream << std::setw(10) << " -> step size: " << stepping.step_size()
517 << " mm" << std::endl;
518 debug_stream << " -> " << detail::ray<algebra_type>(stepping());
519
520 return debug_stream.str();
521 }
