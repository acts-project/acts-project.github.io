88 DETRAY_HOST_DEVICE
89 transform3(const vector3 &t, const vector3 &x, const vector3 &y,
90 const vector3 &z, bool get_inverse = true) {
91 // The matrix needs to be initialized to the identity matrix first. We
92 // only modify the top 4x3 portion of the matrix, so it doesn't matter
93 // what values it initially had. However, the bottom row is required to
94 // have the values of [0, 0, 0, 1], so that's why we set `_data` to the
95 // identity matrix first.
96 _data.eye2();
97
98 _data(Fastor::fseq<0, 3>(), 0) = x;
99 _data(Fastor::fseq<0, 3>(), 1) = y;
100 _data(Fastor::fseq<0, 3>(), 2) = z;
101 _data(Fastor::fseq<0, 3>(), 3) = t;
102
103 if (get_inverse) {
104 _data_inv = Fastor::inverse(_data);
105 }
106 }

115 transform3(const vector3 &t, const vector3 &z, const vector3 &x,

122 DETRAY_HOST
123 explicit transform3(const vector3 &t) {
124 // The matrix needs to be initialized to the identity matrix first. In
125 // this case, the `transform3` requires `_data` to look just like an
126 // identity matrix except for the third column, which is the one we are
127 // modifying here.
128 _data.eye2();
129
130 _data(Fastor::fseq<0, 3>(), 3) = t;
131
132 _data_inv = Fastor::inverse(_data);
133 }

147 DETRAY_HOST
148 transform3(const matrix44 &m, const matrix44 &m_inv)
149 : _data{m}, _data_inv{m_inv} {
150 // The assertion will not hold for (casts to) int
151 if constexpr (std::floating_point<scalar_type>) {
152 [[maybe_unused]] constexpr auto epsilon{
153 std::numeric_limits<scalar_type>::epsilon()};
154
155 [[maybe_unused]] matrix44 identity_matrix;
156 identity_matrix.eye2();
157 assert(algebra::approx_equal(m * m_inv, identity_matrix, 16.f * epsilon,
158 1e-6f));
159 }
160 }

228 DETRAY_HOST
229 constexpr point3 point_to_global(const point2 &v) const {
230 Fastor::Tensor<scalar_type, 4> vector_4;
231 vector_4(Fastor::fseq<0, 2>()) = v;
232 vector_4[2] = static_cast<scalar_type>(0);
233 vector_4[3] = static_cast<scalar_type>(1);
234 return Fastor::Tensor<scalar_type, 3>(
235 Fastor::matmul(_data, vector_4)(Fastor::fseq<0, 3>()));
236 }
