198 BinningData(AxisDirection axisDir, const IAxis& axis)
199 : binvalue(axisDir), subBinningData(nullptr) {
200 type = axis.getType() == AxisType::Equidistant ? equidistant : arbitrary;
201 option = axis.getBoundaryType() == AxisBoundaryType::Closed ? closed : open;
202 min = static_cast<float>(axis.getMin());
203 max = static_cast<float>(axis.getMax());
204 m_bins = axis.getNBins();
205 step = (max - min) / static_cast<float>(m_bins);
206 zdim = (m_bins == 1);
207 m_boundaries.reserve(axis.getBinEdges().size());
208 for (const auto& edge : axis.getBinEdges()) {
209 m_boundaries.push_back(static_cast<float>(edge));

213 // Set the search function pointer based on axis type
214 m_functionPtr = (type == equidistant) ? &searchEquidistantWithBoundary
215 : &searchInVectorWithBoundary;

303 float value(const Vector3& position) const {
304 using VectorHelpers::eta;
305 using VectorHelpers::perp;
306 using VectorHelpers::phi;
307 // ordered after occurrence
308 if (binvalue == AxisDirection::AxisR ||
309 binvalue == AxisDirection::AxisTheta) {
310 return static_cast<float>(perp(position));
311 }
312 if (binvalue == AxisDirection::AxisRPhi) {
313 return static_cast<float>(perp(position) * phi(position));
314 }
315 if (binvalue == AxisDirection::AxisEta) {
316 return static_cast<float>(eta(position));
317 }
318 if (toUnderlying(binvalue) < 3) {
319 return static_cast<float>(position[toUnderlying(binvalue)]);
320 }
321 // phi gauging
322 return static_cast<float>(phi(position));
323 }

426 std::size_t searchWithSubStructure(float value) const {
427 // find the masterbin with the correct function pointer
428 std::size_t masterbin = (*m_functionPtr)(value, *this);
429 // additive sub binning -
430 if (subBinningAdditive) {
431 // no gauging done, for additive sub structure
432 return masterbin + subBinningData->search(value);
433 }
434 // gauge the value to the subBinData
435 float gvalue = value - static_cast<float>(masterbin) *
436 (subBinningData->max - subBinningData->min);
437 // now go / additive or multiplicative
438 std::size_t subbin = subBinningData->search(gvalue);
439 // now return
440 return masterbin * subBinningData->bins() + subbin;
441 }

479 BinningData scale(float factor) const {
480 BinningData scaled = *this;
481 scaled.min *= factor;
482 scaled.max *= factor;
483 scaled.step *= factor;
484 for (auto& boundary : scaled.m_boundaries) {
485 boundary *= factor;
486 }
487 for (auto& boundary : scaled.m_totalBoundaries) {
488 boundary *= factor;
489 }
490 if (scaled.subBinningData) {
491 scaled.subBinningData = std::make_unique<const BinningData>(
492 scaled.subBinningData->scale(factor));
493 }
494 return scaled;

522 void checkSubStructure() {
523 // sub structure is only checked when sBinData is defined
524 if (subBinningData) {
525 m_totalBoundaries.clear();
526 // (A) additive sub structure
527 if (subBinningAdditive) {
528 // one bin is replaced by the sub bins
529 m_totalBins = m_bins + subBinningData->bins() - 1;
530 // the tricky one - exchange one bin by many others
531 m_totalBoundaries.reserve(m_totalBins + 1);
532 // get the sub bin boundaries
533 const std::vector<float>& subBinBoundaries =
534 subBinningData->boundaries();
535 float sBinMin = subBinBoundaries[0];
536 // get the min value of the sub bin boundaries
537 std::vector<float>::const_iterator mbvalue = m_boundaries.begin();
538 for (; mbvalue != m_boundaries.end(); ++mbvalue) {
539 // should define numerically stable
540 if (std::abs((*mbvalue) - sBinMin) < 10e-10) {
541 // copy the sub bin boundaries into the vector
542 m_totalBoundaries.insert(m_totalBoundaries.begin(),
543 subBinBoundaries.begin(),
544 subBinBoundaries.end());
545 ++mbvalue;
546 } else {
547 m_totalBoundaries.push_back(*mbvalue);
548 }
549 }
550 } else { // (B) multiplicative sub structure
551 // every bin is just replaced by the sub binning structure
552 m_totalBins = m_bins * subBinningData->bins();
553 m_totalBoundaries.reserve(m_totalBins + 1);
554 // get the sub bin boundaries if there are any
555 const std::vector<float>& subBinBoundaries =
556 subBinningData->boundaries();
557 // create the boundary vector
558 m_totalBoundaries.push_back(min);
559 for (std::size_t ib = 0; ib < m_bins; ++ib) {
560 float offset = static_cast<float>(ib) * step;
561 for (std::size_t isb = 1; isb < subBinBoundaries.size(); ++isb) {
562 m_totalBoundaries.push_back(offset + subBinBoundaries[isb]);
563 }
564 }
565 }
566 // sort the total boundary vector
567 std::ranges::sort(m_totalBoundaries);
568 }
569 }

571 // Equidistant search
572 // - fastest method
573 static std::size_t searchEquidistantWithBoundary(float value,
574 const BinningData& bData) {
575 // vanilla
576
577 int bin = static_cast<int>((value - bData.min) / bData.step);
578 // special treatment of the 0 bin for closed
579 if (bData.option == closed) {
580 if (value < bData.min) {
581 return (bData.m_bins - 1);
582 }
583 if (value > bData.max) {
584 return 0;
585 }
586 }
587 // if outside boundary : return boundary for open, opposite bin for closed
588 bin =
589 bin < 0
590 ? ((bData.option == open) ? 0 : static_cast<int>(bData.m_bins - 1))
591 : bin;
592 return static_cast<std::size_t>(
593 (bin <= static_cast<int>(bData.m_bins - 1))
594 ? static_cast<std::size_t>(bin)
595 : ((bData.option == open) ? (bData.m_bins - 1) : 0));
596 }
597
598 // Search in arbitrary boundary
599 static std::size_t searchInVectorWithBoundary(float value,
600 const BinningData& bData) {
601 // lower boundary
602 if (value <= bData.m_boundaries[0]) {
603 return (bData.option == closed) ? (bData.m_bins - 1) : 0;
604 }
605 // higher boundary
606 if (value >= bData.max) {
607 return (bData.option == closed) ? 0 : (bData.m_bins - 1);
608 }
609
610 auto lb = std::ranges::lower_bound(bData.m_boundaries, value);
611 return static_cast<std::size_t>(
612 std::ranges::distance(bData.m_boundaries.begin(), lb) - 1);
613 }
