65 DETRAY_HOST_DEVICE
66 helix(const point3_type &pos, const scalar_type time, const vector3_type &dir,
67 const scalar_type qop, const vector3_type &mag_field)
68 : _pos(pos),
69 _time(time),
70 _qop(qop),
71 _B{vector::norm(mag_field)},
72 _h0{vector::normalize(mag_field)},
73 _t0{dir} {
74 assert((math::fabs(vector::norm(_t0) - 1.f) < 1e-5f) &&
75 "The helix direction must be normalized");
76
77 // Momentum
78 const vector3_type mom =
79 (1.f / static_cast<scalar_type>(math::fabs(qop))) * _t0;
80
81 // Normalized _h0 X _t0
82 _n0 = vector::normalize(vector::cross(_h0, _t0));
83
84 // Magnitude of _h0 X _t0
85 _alpha = vector::norm(vector::cross(_h0, _t0));
86
87 // Dot product of _h0 X _t0
88 _delta = vector::dot(_h0, _t0);
89
90 // Path length scaler
91 _K = -_qop * _B;
92
93 // Get longitudinal momentum parallel to B field
94 scalar_type pz = vector::dot(mom, _h0);
95
96 // Get transverse momentum perpendicular to B field
97 vector3_type pT = mom - pz * _h0;
98
99 // R [mm] = pT [GeV] / B [T] in natural unit
100 _R = vector::norm(pT) / _B;
101
102 // Handle the case of pT ~ 0
103 if (vector::norm(pT) < 1e-6f) {
104 _vz_over_vt = detail::invalid_value<scalar_type>();
105 } else {
106 // Get vz over vt in new coordinate
107 _vz_over_vt = pz / vector::norm(pT);
108 }
109 }

144 DETRAY_HOST_DEVICE
145 point3_type pos(const scalar_type s) const {
146 // Handle the case of pT ~ 0
147 if (_vz_over_vt == detail::invalid_value<scalar_type>()) {
148 return _pos + s * _h0;
149 }
150
151 point3_type ret = _pos;
152 ret = ret + _delta / _K * (_K * s - math::sin(_K * s)) * _h0;
153 ret = ret + math::sin(_K * s) / _K * _t0;
154 ret = ret + _alpha / _K * (1.f - math::cos(_K * s)) * _n0;
155
156 return ret;
157 }

167 DETRAY_HOST_DEVICE
168 vector3_type dir(const scalar_type s) const {
169 // Handle the case of pT ~ 0
170 if (_vz_over_vt == detail::invalid_value<scalar_type>()) {
171 return _t0;
172 }
173
174 vector3_type ret{0.f, 0.f, 0.f};
175
176 ret = ret + _delta * (1 - math::cos(_K * s)) * _h0;
177 ret = ret + math::cos(_K * s) * _t0;
178 ret = ret + _alpha * math::sin(_K * s) * _n0;
179
180 return vector::normalize(ret);
181 }

211 DETRAY_HOST_DEVICE
212 free_matrix_t jacobian(const scalar_type s) const {
213 free_matrix_t ret = matrix::zero<free_matrix_t>();
214
215 const matrix_type<3, 3> I33 = matrix::identity<matrix_type<3, 3>>();
216 const matrix_type<3, 3> Z33 = matrix::zero<matrix_type<3, 3>>();
217
218 // Notations
219 // r = position
220 // t = direction
221 // l = qoverp
222
223 // Get drdr
224 auto drdr = I33;
225 getter::set_block(ret, drdr, e_free_pos0, e_free_pos0);
226
227 // Get dtdr
228 auto dtdr = Z33;
229 getter::set_block(ret, dtdr, e_free_dir0, e_free_pos0);
230
231 // Get drdt
232 auto drdt = Z33;
233
234 const scalar_type sin_ks = math::sin(_K * s);
235 const scalar_type cos_ks = math::cos(_K * s);
236 drdt = drdt + sin_ks / _K * I33;
237
238 matrix_type<3, 1> H0 = matrix::zero<matrix_type<3, 1>>();
239 getter::element(H0, 0u, 0u) = _h0[0u];
240 getter::element(H0, 1u, 0u) = _h0[1u];
241 getter::element(H0, 2u, 0u) = _h0[2u];
242 const matrix_type<1, 3> H0_T = matrix::transpose(H0);
243 const matrix_type<3, 3> H0H0_T = H0 * H0_T;
244
245 drdt = drdt + (_K * s - sin_ks) / _K * H0H0_T;
246
247 drdt = drdt + (cos_ks - 1.f) / _K * matrix::column_wise_cross(I33, _h0);
248
249 getter::set_block(ret, drdt, e_free_pos0, e_free_dir0);
250
251 // Get dtdt
252 auto dtdt = Z33;
253 dtdt = dtdt + cos_ks * I33;
254 dtdt = dtdt + (1 - cos_ks) * H0H0_T;
255 dtdt = dtdt - sin_ks * matrix::column_wise_cross(I33, _h0);
256
257 getter::set_block(ret, dtdt, e_free_dir0, e_free_dir0);
258
259 // Get drdl
260 vector3_type drdl = 1.f / _qop * (s * this->dir(s) + _pos - this->pos(s));
261
262 getter::set_block(ret, drdl, e_free_pos0, e_free_qoverp);
263
264 // Get dtdl
265 vector3_type dtdl =
266 -_B * s *
267 (sin_ks * ((H0H0_T - I33) * _t0) + cos_ks * vector::cross(_h0, _t0));
268
269 getter::set_block(ret, dtdl, e_free_dir0, e_free_qoverp);
270
271 // 3x3 and 7x7 element is 1 (Maybe?)
272 getter::element(ret, e_free_time, e_free_time) = 1.f;
273 getter::element(ret, e_free_qoverp, e_free_qoverp) = 1.f;
274
275 return ret;
276 }
