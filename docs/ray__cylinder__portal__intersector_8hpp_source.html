73 template <typename mask_t>
74 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
75 const trajectory_type<algebra_t> &ray,
76 const dtransform3D<algebra_t> & /*trf*/, const mask_t &mask,
77 const dscalar<algebra_t> overstep_tol = 0.f) const {
78 using scalar_t = dscalar<algebra_t>;
79 using point3_t = dpoint3D<algebra_t>;
80 using vector3_t = dvector3D<algebra_t>;
81
82 const scalar_t r{mask[mask_t::shape::e_r]};
83 constexpr scalar_t inv{detail::invalid_value<dvalue<algebra_t>>()};
84
85 const point3_t &ro = ray.pos();
86 const vector3_t &rd = ray.dir();
87 scalar_t path{inv};
88
89 const scalar_t rd_perp_2{rd[0] * rd[0] + rd[1] * rd[1]};
90
91 // The ray is parallel to the cylinder axis (z-axis)...
92 if (rd_perp_2 < std::numeric_limits<scalar_t>::epsilon()) [[unlikely]] {
93 return {};
94 }
95
96 // ...otherwise, two solutions should exist, if the descriminator is
97 // greater than zero
98 const scalar_t ro_perp_2{ro[0] * ro[0] + ro[1] * ro[1]};
99
100 // Only calculate the path, when not already on surface
101 // 'rad_diff' leads to numerical instabilites in single precision on GPU
102 if (const double rad_diff{r * r - ro_perp_2};
103 math::fabs(rad_diff) > 1.f * unit<scalar_t>::um) {
104 const scalar_t rd_perp_inv_2{1.f / rd_perp_2};
105 const scalar_t k{-rd_perp_inv_2 * (ro[0] * rd[0] + ro[1] * rd[1])};
106 const auto discr{static_cast<scalar_t>(rd_perp_inv_2 * rad_diff + k * k)};
107
108 // No intersection found
109 if (discr < 0.f) [[unlikely]] {
110 return {};
111 }
112
113 const scalar_t sqrt_discr{math::sqrt(discr)};
114 const scalar_t s1{k + sqrt_discr};
115 const scalar_t s2{k - sqrt_discr};
116
117 // Take the nearest solution
118 path = (math::fabs(s1) < math::fabs(s2)) ? s1 : s2;
119
120 // If the near solution is outside the overstepping tolerance, take
121 // the far solution (if it exists)
122 if (path < overstep_tol && sqrt_discr > 0.f) {
123 path = (math::fabs(s1) >= math::fabs(s2)) ? s1 : s2;
124 }
125 } else {
126 path = static_cast<scalar_t>(rad_diff);
127 }
128
129 if (path < overstep_tol) {
130 path = inv;
131 }
132
133 // Only need the global z-component for the mask check
134 return {path, point_type{inv, ro[2] + path * rd[2]}};
135 }
136};
