56template <unsigned int ROWS, unsigned int COLS, concepts::scalar scalar_t>
57 requires(ROWS == 3)
58DETRAY_HOST_DEVICE constexpr matrix_type<scalar_t, ROWS, COLS>
59column_wise_cross(const ROOT::Math::SMatrix<scalar_t, ROWS, COLS> &m,
60 const ROOT::Math::SVector<scalar_t, ROWS> &v) {
61 matrix_type<scalar_t, ROWS, COLS> ret;
62 for (unsigned int j = 0u; j < COLS; j++) {
63 ROOT::Math::SVector<scalar_t, ROWS> col = ROOT::Math::Cross(
64 m.template SubCol<ROOT::Math::SVector<scalar_t, ROWS>>(j, 0u), v);
65 ret.Place_in_col(col, 0, j);
66 }
67
68 return ret;
69}

85template <unsigned int ROWS, unsigned int COLS, concepts::scalar scalar_t>
86DETRAY_HOST_DEVICE constexpr matrix_type<scalar_t, ROWS, COLS>
87column_wise_multiply(const ROOT::Math::SMatrix<scalar_t, ROWS, COLS> &m,
88 const ROOT::Math::SVector<scalar_t, ROWS> &v) {
89 matrix_type<scalar_t, ROWS, COLS> ret;
90 for (unsigned int j = 0u; j < COLS; j++) {
91 ROOT::Math::SVector<scalar_t, ROWS> col =
92 m.template SubCol<ROOT::Math::SVector<scalar_t, ROWS>>(j, 0u) * v;
93 ret.Place_in_col(col, 0, j);
94 }
95
96 return ret;
97}

140 matrix_type<scalar_t, ROWS, COLS> L;
