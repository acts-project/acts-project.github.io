171template <concepts::detector detector_t>
172class homogeneous_material_factory final
173 : public factory_decorator<detector_t> {
174 using mask_id = typename detector_t::masks::id;
175 using material_id = typename detector_t::material::id;
176
177 using base_factory = factory_decorator<detector_t>;
178 using placeholder_factory_t = surface_factory<detector_t, unmasked<>>;
179
180 public:
182 using scalar_type = dscalar<typename detector_t::algebra_type>;
183
184 using base_factory::operator();
185
189 DETRAY_HOST
190 explicit homogeneous_material_factory(
191 std::unique_ptr<surface_factory_interface<detector_t>> sf_factory =
192 std::make_unique<placeholder_factory_t>())
193 : base_factory(std::move(sf_factory)) {
194 static_assert(concepts::has_material_slabs<detector_t> ||
195 concepts::has_material_rods<detector_t>,
196 "No homogeneous surface material in detector type");
197 }
198
201 DETRAY_HOST
202 auto n_materials() const -> dindex {
203 const dindex n_surfaces{static_cast<dindex>(m_links.size())};
204
205 // Need exactly one material per surface
206 assert(m_indices.empty() || (m_indices.size() == n_surfaces));
207 assert(m_links.size() == n_surfaces);
208 assert(m_materials.size() == n_surfaces);
209 assert(m_thickness.size() == n_surfaces);
210
211 return n_surfaces;
212 }
213
215 DETRAY_HOST
216 auto links() const -> const std::vector<material_id> & { return m_links; }
217
219 DETRAY_HOST
220 auto materials() const -> const std::vector<material<scalar_type>> & {
221 return m_materials;
222 }
223
225 DETRAY_HOST
226 auto thickness() const -> const std::vector<scalar_type> & {
227 return m_thickness;
228 }

235 DETRAY_HOST
236 void add_material(material_id id, material_data<scalar_type> &&mat_data,
237 std::size_t index = detail::invalid_value<std::size_t>()) {
238 DETRAY_DEBUG_HOST("homogeneous_material_factory::add_material single");
239
240 auto [sf_index, mat, thickness] = std::move(mat_data).get_data();
241
242 DETRAY_DEBUG_HOST("- sf_index=" << sf_index);
243 DETRAY_DEBUG_HOST("- mat=" << mat.at(0));
244 DETRAY_DEBUG_HOST("- thickness=" << thickness.at(0));
245
246 // Only one homogeneous material slab/rod per surface
247 assert(mat.size() == 1u);
248 assert(thickness.size() == 1u);
249
250 m_links.push_back(std::make_pair(id, static_cast<dindex>(index)));
251 m_indices.push_back(sf_index);
252 m_materials.push_back(mat.at(0));
253 m_thickness.push_back(thickness.at(0));
254 }
255
260 DETRAY_HOST
261 void add_material(material_id id,
262 std::vector<material_data<scalar_type>> &&mat_data_vec) {
263 DETRAY_DEBUG_HOST("homogeneous_material_factory::add_material multiple");
264 // Read the material containers
265 m_links.reserve(m_links.size() + mat_data_vec.size());
266 m_materials.reserve(m_materials.size() + mat_data_vec.size());
267 m_thickness.reserve(m_thickness.size() + mat_data_vec.size());
268
269 // Add the material components
270 for (auto &mat_data : mat_data_vec) {
271 this->add_material(id, std::move(mat_data));
272 }
273 }
274
276 DETRAY_HOST
277 auto clear() -> void override {
278 m_links.clear();
279 m_materials.clear();
280 m_thickness.clear();
281 }

299 DETRAY_HOST
300 auto operator()(typename detector_t::surface_lookup_container &surfaces,
301 typename detector_t::material_container &materials) {
302 DETRAY_VERBOSE_HOST("Add homogeneous material...");
303
304 using link_t = typename detector_t::surface_type::material_link;
305 DETRAY_DEBUG_HOST("-> Accumulated " << m_materials.size() << " materials");
306
307 if (m_materials.empty()) {
308 return;
309 }
310
311 DETRAY_DEBUG_HOST("-> n_materials=" << n_materials());
312 DETRAY_DEBUG_HOST("-> links:");
313 for ([[maybe_unused]] const auto &[i, link] :
314 detray::views::enumerate(m_links)) {
315 DETRAY_DEBUG_HOST("-> #" << i << ": " << link.first << " "
316 << link.second);
317 }
318
319 // Check that the surfaces were set up correctly
320 const std::size_t n_materials{this->n_materials()};
321 assert(surfaces.size() >= n_materials);
322
323 // If no concrete surface ordering was passed, use index sequence
324 // and add the materials to the trailing elements in the surfaces cont.
325 if (m_indices.empty() ||
326 std::ranges::find(m_indices, detail::invalid_value<std::size_t>()) !=
327 m_indices.end()) {
328 DETRAY_DEBUG_HOST("Indices were empty or invalid, recalculating");
329 m_indices.resize(n_materials);
330 std::iota(std::begin(m_indices), std::end(m_indices),
331 surfaces.size() - n_materials);
332 }
333 DETRAY_DEBUG_HOST("-> Indices: " << DETRAY_LOG_VECTOR(m_indices));
334
335 // Add the material to the surfaces that the data links against.
336 //
337 // The i-th entries of @c m_indices, @c m_materials, @c m_thickness and
338 // @c m_links all describe the same material instance, so the factory data
339 // is indexed by the position in that sequence, while @c m_indices holds
340 // the index of the surface the material belongs to. The surfaces that
341 // carry material therefore do not have to be contiguous in the volume.
342 for (const auto [i, sf_idx] : detray::views::enumerate(m_indices)) {
343 auto &sf_desc = surfaces.at(static_cast<dindex>(sf_idx));
344
345 if (sf_desc.material().id() < detector_t::material::id::e_none) {
346 DETRAY_WARN_HOST("Surface descriptor already has a material link:"
347 << sf_desc.material().id()
348 << " skipping configured homogeneous assignment");
349 continue;
350 }
351
352 const material<scalar_type> &mat = m_materials.at(i);
353 scalar_type t = m_thickness.at(i);
354
355 DETRAY_DEBUG_HOST("-> Adding: i=" << i << ", sf_index=" << sf_idx
356 << ", sf=" << sf_desc);
357 DETRAY_DEBUG_HOST(" mat=" << mat << " thickness=" << t);
358
359 dindex mat_idx{0u};
360
361 if constexpr (concepts::has_material_slabs<detector_t>) {
362 if (m_links.at(i).first == material_id::e_material_slab) {
363 auto &mat_coll =
364 materials.template get<material_id::e_material_slab>();
365
366 material_slab<scalar_type> mat_slab{mat, t};
367 mat_idx = this->insert_in_container(mat_coll, mat_slab,
368 m_links.at(i).second);
369 }
370 }
371 if constexpr (concepts::has_material_rods<detector_t>) {
372 if (m_links.at(i).first == material_id::e_material_rod) {
373 auto &mat_coll =
374 materials.template get<material_id::e_material_rod>();
375
376 material_rod<scalar_type> mat_rod{mat, t};
377 mat_idx = this->insert_in_container(mat_coll, mat_rod,
378 m_links.at(i).second);
379 }
380 }
381 DETRAY_DEBUG_HOST("-> After insert: mat_idx=" << mat_idx);
382 link_t new_link{m_links.at(i).first, mat_idx};
383
384 DETRAY_DEBUG_HOST("-> Existing link: " << sf_desc.material());
385 DETRAY_DEBUG_HOST("-> Setting new link: " << new_link);
386
387 // Set the initial surface material link (will be updated when
388 // added to the detector)
389 sf_desc.material() = new_link;
390
391 DETRAY_DEBUG_HOST("-> Added material to surface: " << sf_desc);
392 }
393 }
394
395 private:
397 std::vector<std::pair<material_id, dindex>> m_links{};
399 std::vector<std::size_t> m_indices{};
401 std::vector<scalar_type> m_thickness{};
403 std::vector<material<scalar_type>> m_materials{};
404};
