298 friend std::ostream& operator<<(std::ostream& os,
299 const TrackStateTypeBase& t) {
300 bool first = true;
301 const auto append = [&](const char* name, bool condition) {
302 if (condition) {
303 if (!first) {
304 os << ",";
305 }
306 os << name;
307 first = false;
308 }
309 };
310 append("HasParams", t.hasParameters());
311 if (t.isMaterial()) {
312 append("IsMat", true);
313 } else {
314 append("HasMat", t.hasMaterial());
315 }
316 if (t.isMeasurement()) {
317 append("IsMeas", true);
318 } else if (t.isOutlier()) {
319 append("IsOutlier", true);
320 } else if (t.isHole()) {
321 append("IsHole", true);
322 } else {
323 append("HasMeas", t.hasMeasurement());
324 }
325 append("IsSharedHit", t.isSharedHit());
326 append("IsSplitHit", t.isSplitHit());
327 append("NoExpectedHit", t.hasNoExpectedHit());
328 return os;
329 }

372 void assertConsistency() const {
373 assert(!(test(HasNoExpectedHit) &&
374 (test(HasMeasurement) || test(IsOutlier) || test(IsHole) ||
375 test(IsSharedHit) || test(IsSplitHit))) &&
376 "TrackStateType - NoExpectedHit cannot be set with other "
377 "measurement flags");
378 assert(!(test(IsOutlier) && test(IsHole)) &&
379 "TrackStateType - Outlier and Hole cannot be set simultaneously");
380 assert(
381 !(test(IsHole) && test(HasMeasurement)) &&
382 "TrackStateType - Hole and Measurement cannot be set simultaneously");
383 assert(!(test(IsOutlier) && !test(HasMeasurement)) &&
384 "TrackStateType - Outlier flag requires Measurement flag to be set");
385 }
