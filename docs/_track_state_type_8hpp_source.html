281 friend std::ostream& operator<<(std::ostream& os,
282 const TrackStateTypeBase& t) {
283 bool first = true;
284 const auto append = [&](const char* name, bool condition) {
285 if (condition) {
286 if (!first) {
287 os << ",";
288 }
289 os << name;
290 first = false;
291 }
292 };
293 append("HasParams", t.hasParameters());
294 if (t.isMaterial()) {
295 append("IsMat", true);
296 } else {
297 append("HasMat", t.hasMaterial());
298 }
299 if (t.isMeasurement()) {
300 append("IsMeas", true);
301 } else if (t.isOutlier()) {
302 append("IsOutlier", true);
303 } else if (t.isHole()) {
304 append("IsHole", true);
305 } else {
306 append("HasMeas", t.hasMeasurement());
307 }
308 append("IsSharedHit", t.isSharedHit());
309 append("IsSplitHit", t.isSplitHit());
310 append("NoExpectedHit", t.hasNoExpectedHit());
311 return os;
312 }

355 void assertConsistency() const {
356 assert(!(test(HasNoExpectedHit) &&
357 (test(HasMeasurement) || test(IsOutlier) || test(IsHole) ||
358 test(IsSharedHit) || test(IsSplitHit))) &&
359 "TrackStateType - NoExpectedHit cannot be set with other "
360 "measurement flags");
361 assert(!(test(IsOutlier) && test(IsHole)) &&
362 "TrackStateType - Outlier and Hole cannot be set simultaneously");
363 assert(
364 !(test(IsHole) && test(HasMeasurement)) &&
365 "TrackStateType - Hole and Measurement cannot be set simultaneously");
366 assert(!(test(IsOutlier) && !test(HasMeasurement)) &&
367 "TrackStateType - Outlier flag requires Measurement flag to be set");
368 }
