102template <typename stepper_options_t, typename navigator_options_t,
103 typename actor_t = DefaultActor>
104struct PropagatorOptions : public detail::PurePropagatorPlainOptions {
106 using stepper_options_type = stepper_options_t;
108 using navigator_options_type = navigator_options_t;
110 using actor_type = actor_t;
111
115 PropagatorOptions(const GeometryContext& gctx,
116 const MagneticFieldContext& mctx)
117 : geoContext(gctx),
118 magFieldContext(mctx),
119 stepping(gctx, mctx),
120 navigation(gctx) {}
121
124 explicit PropagatorOptions(const PropagatorPlainOptions& pOptions)
125 : geoContext(pOptions.geoContext),
126 magFieldContext(pOptions.magFieldContext),
127 stepping(pOptions.geoContext, pOptions.magFieldContext),
128 navigation(pOptions.geoContext) {
129 setPlainOptions(pOptions);
130 }
131
133 explicit operator PropagatorPlainOptions() const {
134 PropagatorPlainOptions pOptions(geoContext, magFieldContext);
135 static_cast<PurePropagatorPlainOptions&>(pOptions) =
136 static_cast<const PurePropagatorPlainOptions&>(*this);
137 pOptions.stepping = static_cast<const StepperPlainOptions&>(stepping);
138 pOptions.navigation = static_cast<const NavigatorPlainOptions&>(navigation);
139 return pOptions;
140 }
141
145 void setPlainOptions(const PropagatorPlainOptions& pOptions) {
146 static_cast<PurePropagatorPlainOptions&>(*this) =
147 static_cast<const PurePropagatorPlainOptions&>(pOptions);
148
149 stepping.setPlainOptions(pOptions.stepping);
150 navigation.setPlainOptions(pOptions.navigation);
151 }
152
154 std::reference_wrapper<const GeometryContext> geoContext;
155
157 std::reference_wrapper<const MagneticFieldContext> magFieldContext;
158
160 stepper_options_t stepping;
161
163 navigator_options_t navigation;
164
166 actor_t actor;
167};
