303 DETRAY_HOST_DEVICE
304 auto local_min_bounds(
305 const scalar_type env = std::numeric_limits<scalar_type>::epsilon()) const
306 -> mask<cuboid3D, algebra_t, unsigned int> {
307 const auto bounds =
308 get_shape().template local_min_bounds<algebra_t>(_values, env);
309 static_assert(bounds.size() == cuboid3D::e_size,
310 "Shape returns incompatible bounds for bound box");
311 return {bounds, std::numeric_limits<unsigned int>::max()};
312 }

335 DETRAY_HOST
336 friend mask operator+(const mask& left, const mask& right) {
337 links_type vol_link{left.volume_link() == right.volume_link()
338 ? left.volume_link()
339 : detail::invalid_value<links_type>()};
340 mask_values merged_vals =
341 mask::get_shape().merge(left.values(), right.values());
342
343 return {std::move(merged_vals), vol_link};
344 }

348 DETRAY_HOST
349 constexpr bool self_check(std::ostream& os) const {
350 const bool result = get_shape().check_consistency(_values, os);
351
352 if (!result) {
353 os << to_string();
354 }
355
356 return result;
357 }
