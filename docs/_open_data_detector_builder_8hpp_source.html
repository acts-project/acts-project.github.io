65inline int layerIndexFromName(std::string_view elemName,
66 const std::regex& layerFilter) {
67 std::cmatch match;
68 if (std::regex_search(elemName.begin(), elemName.end(), match, layerFilter) &&
69 match.size() > 1) {
70 return std::stoi(match[1].str());
71 }
72
73 if (std::regex groupedLayerNameFilter{"layer(\\d+)"};
74 std::regex_search(elemName.begin(), elemName.end(), match,
75 groupedLayerNameFilter) &&
76 match.size() > 1) {
77 return std::stoi(match[1].str());
78 }
79
80 return 0;
81}
