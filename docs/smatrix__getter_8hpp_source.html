148template <std::size_t I, std::size_t J, concepts::matrix M>
149DETRAY_HOST_DEVICE decltype(auto) element(M &matrix) {
150 if constexpr (concepts::has_compile_time_2d_access<M>) {
151 return matrix.template element<I, J>();
152 } else {
153 using index_t = detray::traits::index_t<std::decay_t<M>>;
154 return element(matrix, static_cast<index_t>(I), static_cast<index_t>(J));
155 }
156}
157
158template <std::size_t I, concepts::vector V>
159DETRAY_HOST_DEVICE decltype(auto) element(V &vector) {
160 if constexpr (concepts::has_compile_time_1d_access<V>) {
161 return vector.template element<I>();
162 } else {
163 using index_t = detray::traits::index_t<std::decay_t<V>>;
164 return element(vector, static_cast<index_t>(I));
165 }

169struct block_getter {
175 template <unsigned int ROWS, unsigned int COLS, unsigned int oROWS,
176 unsigned int oCOLS, concepts::scalar scalar_t>
177 DETRAY_HOST_DEVICE ROOT::Math::SMatrix<scalar_t, ROWS, COLS> operator()(
178 const ROOT::Math::SMatrix<scalar_t, oROWS, oCOLS> &m, unsigned int row,
179 unsigned int col) const {
180 return m.template Sub<ROOT::Math::SMatrix<scalar_t, ROWS, COLS>>(row, col);
181 }
182
188 template <unsigned int SIZE, unsigned int ROWS, unsigned int COLS,
189 concepts::scalar scalar_t>
190 DETRAY_HOST_DEVICE ROOT::Math::SVector<scalar_t, SIZE> vector(
191 const ROOT::Math::SMatrix<scalar_t, ROWS, COLS> &m, unsigned int row,
192 unsigned int col) const {
193 // TODO: SMatrix bug?
194 // return m.template SubCol<ROOT::Math::SVector<scalar_t, SIZE>>(col,
195 // row);
196
197 assert(col < COLS);
198 assert(row + SIZE <= ROWS);
199
200 ROOT::Math::SVector<scalar_t, SIZE> ret;
201
202 for (unsigned int irow = row; irow < row + SIZE; ++irow) {
203 ret[irow - row] = m(irow, col);
204 }
205
206 return ret;
207 }
208
214 template <unsigned int ROWS, unsigned int COLS, concepts::scalar scalar_t,
215 concepts::matrix input_matrix_type>
216 DETRAY_HOST_DEVICE void set(
217 input_matrix_type &m, const ROOT::Math::SMatrix<scalar_t, ROWS, COLS> &b,
218 std::size_t row, std::size_t col) const {
219 for (unsigned int i = 0; i < ROWS; ++i) {
220 for (unsigned int j = 0; j < COLS; ++j) {
221 m(i + static_cast<unsigned int>(row),
222 j + static_cast<unsigned int>(col)) = b(i, j);
223 }
224 }
225 }
226
232 template <unsigned int ROWS, concepts::scalar scalar_t,
233 concepts::matrix input_matrix_type>
234 DETRAY_HOST_DEVICE void set(input_matrix_type &m,
235 const ROOT::Math::SVector<scalar_t, ROWS> &b,
236 unsigned int row, unsigned int col) const {
237 for (unsigned int i = 0; i < ROWS; ++i) {
238 m(i + row, col) = b[i];
239 }
240 }
241
242}; // struct block_getter
