44 DETRAY_HOST_DEVICE
45 static inline point3_type global_to_local_3D(const transform3_type &trf,
46 const point3_type &p,
47 const vector3_type &dir) {
48 const auto local3 = trf.point_to_local(p);
49
50 // Line direction
51 const vector3_type z = trf.z();
52
53 // Line center
54 const point3_type t = trf.translation();
55
56 // Radial vector
57 const vector3_type r = vector::cross(z, dir);
58
59 // Assign the sign depending on the position w.r.t line
60 // Right: -1
61 // Left: 1
62 const scalar_type sign = vector::dot(r, t - p) > 0.f ? -1.f : 1.f;
63
64 return {sign * vector::perp(local3), local3[2], vector::phi(local3)};
65 }

74 DETRAY_HOST_DEVICE
75 static inline loc_point global_to_local(const transform3_type &trf,
76 const point3_type &p,
77 const vector3_type &dir) {
78 const point3_type local3 =
79 line2D<algebra_t>::global_to_local_3D(trf, p, dir);
80
81 return {local3[0], local3[1]};
82 }

89 DETRAY_HOST_DEVICE static inline point3_type local_to_global(
90 const transform3_type &trf, const point3_type &p) {
91 const scalar_type R = math::fabs(p[0]);
92 const point3_type local = {R * math::cos(p[2]), R * math::sin(p[2]), p[1]};
93
94 return trf.point_to_global(local);
95 }

103 template <typename mask_t>
104 DETRAY_HOST_DEVICE static inline point3_type local_to_global(
105 const transform3_type &trf, const mask_t & /*mask*/, const loc_point &p,
106 const vector3_type &dir) {
107 // Line direction
108 const vector3_type z = trf.z();
109
110 // Radial vector
111 const vector3_type r = vector::cross(z, dir);
112
113 // Local Z position in global cartesian coordinate
114 const point3_type locZ_in_global = trf.point_to_global(point3_type{
115 static_cast<scalar_type>(0), static_cast<scalar_type>(0), p[1]});
116
117 return locZ_in_global + p[0] * vector::normalize(r);
118 }
