259 DETRAY_HOST_DEVICE
260 inline const auto &volume(const point3_type &p) const {
261 // Allow to call the volume search data structure
262 // TODO: Add volume accelerator builder
263 volume_type v_desc{};
264 v_desc.template set_accel_link<geo_obj_ids::e_volume>(
265 accel::id::e_volume_default, 0u);
266 tracking_volume world{*this, v_desc};
267
268 dindex volume_index{0u};
269 world.template visit_accelerator<geo_obj_ids::e_volume, volume_search>(
270 p, &volume_index);
271 return _volumes[volume_index];
272 }

350 struct volume_search {
352 template <concepts::accelerator_collection accel_coll_t,
353 typename accel_index_t>
354 DETRAY_HOST_DEVICE inline void operator()(const accel_coll_t &coll,
355 const accel_index_t index,
356 const point3_type &p,
357 dindex *const result) const {
358 using accel_type = typename accel_coll_t::value_type;
359
360 if constexpr (concepts::volume_accelerator<accel_type>) {
361 const auto volume_accelerator = coll[index];
362
363 // Only one entry per bin
364 *result = volume_accelerator.search(p).value();
365 }
366 }
367 };
