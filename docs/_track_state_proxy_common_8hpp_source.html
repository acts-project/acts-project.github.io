360 template <std::ranges::sized_range index_range_t>
361 void setProjectorSubspaceIndices(const index_range_t& subspaceIndices)
362 requires(!read_only &&
363 std::convertible_to<std::ranges::range_value_t<index_range_t>,
364 std::uint8_t>)
365 {
366 assert(derived().template has<detail_tsp::kProjectorKey>());
367 assert(subspaceIndices.size() <= eBoundSize);
368 BoundSubspaceIndices boundSubspace{};
369 std::ranges::transform(subspaceIndices, boundSubspace.begin(),
370 [](auto i) { return static_cast<std::uint8_t>(i); });
371 derived()
372 .template component<SerializedSubspaceIndices,
373 detail_tsp::kProjectorKey>() =
374 serializeSubspaceIndices(boundSubspace);
375 }

510 template <typename val_t, typename cov_t>
511 void allocateCalibrated(const Eigen::DenseBase<val_t>& val,
512 const Eigen::DenseBase<cov_t>& cov)
513 requires(!read_only && Concepts::eigen_base_is_fixed_size<val_t> &&
514 Concepts::eigen_bases_have_same_num_rows<val_t, cov_t> &&
515 Concepts::eigen_base_is_square<cov_t> &&
516 Eigen::PlainObjectBase<val_t>::RowsAtCompileTime <=
517 static_cast<std::underlying_type_t<BoundIndices>>(eBoundSize))
518 {
519 constexpr std::size_t measdim =
520 static_cast<std::size_t>(val_t::RowsAtCompileTime);
521 derived().allocateCalibrated(measdim);
522 calibrated<measdim>() = val;
523 calibratedCovariance<measdim>() = cov;
524 }
