354 template <std::ranges::sized_range index_range_t>
355 void setProjectorSubspaceIndices(const index_range_t& subspaceIndices)
356 requires(!read_only &&
357 std::convertible_to<std::ranges::range_value_t<index_range_t>,
358 std::uint8_t>)
359 {
360 assert(derived().template has<detail_tsp::kProjectorKey>());
361 assert(subspaceIndices.size() <= eBoundSize);
362 BoundSubspaceIndices boundSubspace{};
363 std::transform(subspaceIndices.begin(), subspaceIndices.end(),
364 boundSubspace.begin(),
365 [](auto i) { return static_cast<std::uint8_t>(i); });
366 derived()
367 .template component<SerializedSubspaceIndices,
368 detail_tsp::kProjectorKey>() =
369 serializeSubspaceIndices(boundSubspace);
370 }

505 template <typename val_t, typename cov_t>
506 void allocateCalibrated(const Eigen::DenseBase<val_t>& val,
507 const Eigen::DenseBase<cov_t>& cov)
508 requires(!read_only && Concepts::eigen_base_is_fixed_size<val_t> &&
509 Concepts::eigen_bases_have_same_num_rows<val_t, cov_t> &&
510 Concepts::eigen_base_is_square<cov_t> &&
511 Eigen::PlainObjectBase<val_t>::RowsAtCompileTime <=
512 static_cast<std::underlying_type_t<BoundIndices>>(eBoundSize))
513 {
514 constexpr std::size_t measdim =
515 static_cast<std::size_t>(val_t::RowsAtCompileTime);
516 derived().allocateCalibrated(measdim);
517 calibrated<measdim>() = val;
518 calibratedCovariance<measdim>() = cov;
519 }
