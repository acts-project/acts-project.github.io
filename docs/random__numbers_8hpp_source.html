23template <concepts::scalar scalar_t,

64 DETRAY_HOST auto operator()(const darray<scalar_t, 2> range = {
65 -std::numeric_limits<scalar_t>::max(),
66 std::numeric_limits<scalar_t>::max()}) {
67 const scalar_t min{range[0]};
68 const scalar_t max{range[1]};
69 assert(min <= max);
70
71 // Uniform
72 if constexpr (std::is_same_v<distribution_t,
73 std::uniform_real_distribution<scalar_t>>) {
74 return distribution_t(min, max)(m_engine);
75
76 // Normal
77 } else if constexpr (std::is_same_v<distribution_t,
78 std::normal_distribution<scalar_t>>) {
79 scalar_t mu{min + 0.5f * (max - min)};
80 return distribution_t(mu, 0.5f / 3.0f * (max - min))(m_engine);
81 }
82 }
83
88 DETRAY_HOST auto normal(const scalar_t mean, const scalar_t stddev) {
89 return (stddev == static_cast<scalar_t>(0))
90 ? mean
91 : std::normal_distribution<scalar_t>(mean, stddev)(m_engine);
92 }
93
96 DETRAY_HOST std::uint8_t coin_toss() {
97 return std::uniform_int_distribution<std::uint8_t>(0u, 1u)(m_engine);
98 }
99
102 static constexpr seed_type default_seed() { return engine_t::default_seed; }
103};
