43template <typename input_converter_t>
44void add_input_file_converters(
45 detray::io::detail::detector_components_converter& payload_converter,
46 const std::vector<std::string>& file_names) {
47 assert(!file_names.empty());
48 // Register each file according to the file extension
49 for (const std::filesystem::path file_name : file_names) {
50 // File names have already been filtered
51 assert(!file_name.empty());
52
53 // Only add readers for json files
54 std::string file_ext{file_name.extension()};
55 if (file_ext == input_converter_t{}.file_extension()) {
56 payload_converter.template add_converter<input_converter_t>(file_name);
57 }
58 }
59}

71template <typename... input_converter_ts>
72void convert_to_payload(detray::io::detector_payload& payload,
73 const std::vector<std::string>& file_names) {
74 static_assert(sizeof...(input_converter_ts) > 0);
75
76 // Hold all required readers (one for every component) and optional converters
77 detray::io::detail::detector_components_converter payload_converter{};
78
79 // Filter out malformed paths from filenames
80 std::vector<std::string> filtered_files{};
81 filtered_files.reserve(file_names.size());
82 const auto file_filter = [](const std::string& f) {
83 const bool file_pred{detray::io::file_exists(f)};
84 // Empty file names are treated as defaults, do not spam warnings for them
85 if (!file_pred && !f.empty()) {
86 DETRAY_WARN_HOST(
87 "Skipping detector component: Input file does not exist\n(\""
88 << f << "\")");
89 }
90 return file_pred;
91 };
92 std::ranges::copy_if(file_names, std::back_inserter(filtered_files),
93 file_filter);
94
95 // Remove duplicates
96 std::ranges::sort(filtered_files);
97 auto [first, last] = std::ranges::unique(filtered_files);
98 filtered_files.erase(first, last);
99
100 // Check if data from any input files needs to be converted
101 if (!filtered_files.empty()) {
102 // Register the readers for the files in json format
103 (detray::io::detail::add_input_file_converters<input_converter_ts>(
104 payload_converter, filtered_files),
105 ...);
106
107 // Make sure that all files will be read
108 if (payload_converter.converter_map().size() != filtered_files.size()) {
109 std::stringstream err_str{};
110 for (const auto& conv_pair : payload_converter.converter_map()) {
111 err_str << "-> " << conv_pair.first << std::endl;
112 }
113 DETRAY_ERROR_HOST(
114 "Not all files were registered to a converter: Please check e.g. "
115 "that the file extensions match the converter types!\n"
116 << "Successfully registered files:\n"
117 << err_str.str());
118 }
119 }
120
121 payload_converter.convert(payload);
122}

153template <class detector_t, std::size_t GCAP = 0u, std::size_t GDIM = 2u,
154 std::size_t MDIM = 2u,
155 template <typename> class volume_builder_t = volume_builder,
156 typename... input_converter_ts>
157auto read_detector(vecmem::memory_resource& resc,
158 const detray::io::detector_reader_config& cfg,
159 detray::io::detector_payload& payload) noexcept(false) {
160 // Convert the input file data and add it to the detector payload object
161 if constexpr (sizeof...(input_converter_ts) > 0) {
162 // File list may be empty, in this case this call will do nothing
163 detray::io::convert_to_payload<input_converter_ts...>(payload, cfg.files());
164 } else {
165 // No converters present: Input files cannot be handled
166 if (!cfg.files().empty()) {
167 const std::string err_msg{
168 "No converters compiled into the 'read_detector' function: Cannot "
169 "handle file inputs!\nUse e.g. the 'read_detector_json' function for "
170 "json files or add custom input converter types to 'read_detector'."};
171 DETRAY_FATAL_HOST(err_msg);
172 throw std::invalid_argument(err_msg);
173 }
174 }
175
176 // Collect all input data about the detector in the detector builder
177 // (from converters and external payload)
178 detray::io::detail::detector_components_reader<detector_t, GCAP, GDIM, MDIM>
179 detector_components;
180 detray::detector_builder<typename detector_t::metadata, volume_builder_t>
181 det_builder;
182
183 // Transcribe the data into the detector builder
184 detector_components.read(det_builder, payload);
185
186 // Build and return the detector
187 typename detector_t::name_map names{};
188 auto det = det_builder.build(resc, cfg.builder_options(), names);
189
190 if (cfg.do_check()) {
191 // This will throw an exception in case of inconsistencies
192 detray::detail::check_consistency(det, cfg.verbose_check(), names);
193 }
194
195 DETRAY_DEBUG_HOST("\n" << detray::utils::print_detector(det, names));
196
197 return std::make_pair(std::move(det), std::move(names));
198}
