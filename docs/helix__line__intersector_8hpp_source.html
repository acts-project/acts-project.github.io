32template <detray::concepts::aos algebra_t>
33struct helix_intersector_impl<line2D<algebra_t>, algebra_t> {
34 private:
35 using scalar_t = dscalar<algebra_t>;
36
37 public:
39 using algebra_type = algebra_t;

42 template <typename surface_descr_t>
43 using intersection_type =
44 intersection2D<surface_descr_t, algebra_t, intersection::contains_pos>;
45
47 template <typename other_algebra_t>
48 using trajectory_type = detail::helix<other_algebra_t>;
49
50 // Maximum number of solutions this intersector can produce
52 static constexpr std::uint8_t n_solutions{1u};
53
55 using result_type = intersection_point_err<algebra_t>;
56
64 DETRAY_HOST_DEVICE constexpr result_type point_of_intersection(
65 const trajectory_type<algebra_t> &h, const dtransform3D<algebra_t> &trf,
66 const scalar_t /*unused*/ = 0.f) const {
67 using point3_t = dpoint3D<algebra_t>;
68 using vector3_t = dvector3D<algebra_t>;
69
70 // line axis direction
71 const vector3_t l = getter::vector<3>(trf.matrix(), 0u, 2u);
72
73 // line center
74 const point3_t c = trf.translation();
75
76 // initial track direction
77 const vector3_t t0 = h.dir(0.f);
78
79 // initial track position
80 const point3_t r0 = h.pos(0.f);
81
82 // Projection of line to track direction
83 const scalar_t lt0{vector::dot(l, t0)};
84
85 const scalar_t denom{1.f - (lt0 * lt0)};
86
87 // Case for wire is parallel to track
88 // @NOTE We might not have to call this which is meant to be for ray
89 // intersection...
90 if (denom < 1e-5f) {
91 DETRAY_ERROR_HOST("Helix line intersector encountered invalid value!");
92 return {};
93 }
94
95 // vector from track position to line center
96 const vector3_t D = c - r0;
97
98 // D projection on line direction
99 const scalar_t P{vector::dot(D, l)};
100
101 // D projection on track direction
102 const scalar_t Q{vector::dot(D, t0)};
103
104 // Path length to the point of closest approach on the track
105 // @NOTE Ray intersection algorithm is used for the initial guess on
106 // the path length
107 scalar_t s_ini{1.f / denom * (Q - P * lt0)};
108
110 auto line_inters_func = [&h, &c, &l](const scalar_t x) {
111 // track direction
112 const vector3_t t = h.dir(x);
113
114 // track position
115 const point3_t r = h.pos(x);
116
117 // Projection of (track position - center) to the line
118 const scalar_t A = vector::dot(r - c, l);
119
120 // Vector orthogonal to the line and passing the track position
121 // w = r - (c + ((r - c) * l)l)
122 const vector3_t w = r - (c + A * l);
123
124 // f(s) = t * w = 0
125 const scalar_t f = vector::dot(t, w);
126
127 // dtds = d^2r/ds^2 = qop * (t X b_field)
128 const vector3_t dtds = h.qop() * vector::cross(t, h.b_field());
129 // dwds = t - (t * l)l
130 const vector3_t dwds = t - vector::dot(t, l) * l;
131
132 // f'(s) = dtds * w + t * dwds
133 const scalar_t dfds = vector::dot(dtds, w) + vector::dot(t, dwds);
134
135 return std::make_tuple(f, dfds);
136 };
137
138 // Run the root finding algorithm
139 scalar_t s;
140 scalar_t ds;
141 if (run_rtsafe) {
142 std::tie(s, ds) =
143 newton_raphson_safe(line_inters_func, s_ini, convergence_tolerance,
144 max_n_tries, max_path);
145 } else {
146 std::tie(s, ds) =
147 newton_raphson(line_inters_func, s_ini, convergence_tolerance,
148 max_n_tries, max_path);
149 }
150
151 return {s, h.pos(s), ds};
152 }
153
155 scalar_t convergence_tolerance{1.f * unit<scalar_t>::um};
157 std::size_t max_n_tries{1000u};
159 scalar_t max_path{5.f * unit<scalar_t>::m};
161 bool run_rtsafe{true};
162};
