98template <concepts::detector detector_t>
99class homogeneous_material_generator final
100 : public factory_decorator<detector_t> {
101 using scalar_t = dscalar<typename detector_t::algebra_type>;
102
103 public:
107 DETRAY_HOST
108 homogeneous_material_generator(
109 std::unique_ptr<surface_factory_interface<detector_t>> factory,
110 const hom_material_config<scalar_t> cfg)
111 : factory_decorator<detector_t>(std::move(factory)), m_cfg{cfg} {
112 static_assert(concepts::has_material_slabs<detector_t> ||
113 concepts::has_material_rods<detector_t>,
114 "No homogeneous surface material in detector type");
115 }
116
126 DETRAY_HOST
127 auto operator()(typename detector_t::volume_type &volume,
128 typename detector_t::surface_lookup_container &surfaces,
129 typename detector_t::transform_container &transforms,
130 typename detector_t::mask_container &masks,
131 typename detector_t::geometry_context ctx = {})
132 -> dindex_range override {
133 const dindex_range idx_range =
134 (*this->get_factory())(volume, surfaces, transforms, masks, ctx);
135
136 m_surface_range = idx_range;
137
138 return idx_range;
139 }
140
149 DETRAY_HOST
150 auto operator()(typename detector_t::surface_lookup_container &surfaces,
151 typename detector_t::material_container &materials) {
152 DETRAY_VERBOSE_HOST("Generate homogeneous material...");
153
154 using material_id = typename detector_t::material::id;
155 using link_t = typename detector_t::surface_type::material_link;
156
157 assert(surfaces.size() >= (m_surface_range[1] - m_surface_range[0]));
158
159 DETRAY_VERBOSE_HOST("-> Material surface range: "
160 << m_surface_range[0] << " - " << m_surface_range[1]);
161
162 // Add the material to the surfaces that the data links against
163 for (auto &sf : detray::ranges::subrange(surfaces, m_surface_range)) {
164 const material<scalar_t> *mat_ptr{nullptr};
165
166 // Get the correct material for this surface type
167 constexpr vacuum<scalar_t> vac{};
168 switch (sf.id()) {
169 case surface_id::e_passive: {
170 const auto &mat = m_cfg.passive_material();
171 mat_ptr = (mat != vac) ? &mat : nullptr;
172 break;
173 }
174 case surface_id::e_sensitive: {
175 const auto &mat = m_cfg.sensitive_material();
176 mat_ptr = (mat != vac) ? &mat : nullptr;
177 break;
178 }
179 case surface_id::e_portal: {
180 const auto &mat = m_cfg.portal_material();
181 mat_ptr = (mat != vac) ? &mat : nullptr;
182 break;
183 }
184 case surface_id::e_unknown: {
185 std::stringstream err_stream{};
186 err_stream << "-> Encountered surface of unknown type during "
187 "material generation: "
188 << sf << std::endl;
189
190 DETRAY_FATAL_HOST(err_stream.str());
191 throw std::runtime_error(err_stream.str());
192 break;
193 }
194 default: {
195 break;
196 }
197 }
198
199 // Found suitable material for this surface?
200 if (mat_ptr == nullptr) {
201 continue;
202 }
203
204 // Handle line shaped surfaces differently
205 bool is_line{false};
206 link_t mat_link;
207
208 if constexpr (concepts::has_material_rods<detector_t>) {
209 using mask_id = typename detector_t::masks::id;
210
211 // If the current surface is a line, generate a material rod
212 const mask_id sf_mask_id = sf.mask().id();
213 if (sf_mask_id == mask_id::e_straw_tube ||
214 sf_mask_id == mask_id::e_drift_cell) {
215 is_line = true;
216
217 auto &mat_coll =
218 materials.template get<material_id::e_material_rod>();
219 mat_coll.emplace_back(*mat_ptr, m_cfg.thickness());
220
221 mat_link = {material_id::e_material_rod,
222 static_cast<dindex>(mat_coll.size() - 1u)};
223 }
224 }
225
226 // For all surfaces that are not lines, generate a material slab
227 if constexpr (concepts::has_material_slabs<detector_t>) {
228 if (!is_line) {
229 auto &mat_coll =
230 materials.template get<material_id::e_material_slab>();
231 mat_coll.emplace_back(*mat_ptr, m_cfg.thickness());
232
233 mat_link = {material_id::e_material_slab,
234 static_cast<dindex>(mat_coll.size() - 1u)};
235 }
236 }
237
238 // Set the initial surface material link (will be updated when
239 // added to the detector)
240 sf.material() = mat_link;
241
242 DETRAY_DEBUG_HOST("-> Added material to surface: " << sf);
243 }
244 }
245
246 private:
248 hom_material_config<scalar_t> m_cfg;
250 dindex_range m_surface_range{};
251};
