338 template <typename propagator_state_t, typename stepper_t,
339 typename navigator_t>
340 Result<void> act(propagator_state_t& state, const stepper_t& stepper,
341 const navigator_t& navigator, result_type& result,
342 const Logger& /*logger*/) const {
343 assert(result.fittedStates && "No MultiTrajectory set");
344
345 if (result.finished) {
346 return Result<void>::success();
347 }
348
349 ACTS_VERBOSE("KalmanFitter step at pos: "
350 << stepper.position(state.stepping).transpose()
351 << " dir: " << stepper.direction(state.stepping).transpose()
352 << " momentum: "
353 << stepper.absoluteMomentum(state.stepping));
354
355 // Initialize path limit reached aborter
356 if (result.pathLimitReached.internalLimit ==
357 std::numeric_limits<double>::max()) {
358 detail::setupLoopProtection(state, stepper, result.pathLimitReached,
359 true, logger());
360 }
361
362 // Update:
363 // - Waiting for a current surface
364 const Surface* surface = navigator.currentSurface(state.navigation);
365 if (surface != nullptr) {
366 // Check if the surface is in the measurement map
367 // -> Get the measurement / calibrate
368 // -> Create the predicted state
369 // -> Check outlier behavior, if non-outlier:
370 // -> Perform the kalman update
371 // -> Fill track state information & update stepper information
372
373 ACTS_VERBOSE("Perform " << state.options.direction << " filter step");
374 auto res = filter(*surface, state, stepper, navigator, result);
375 if (!res.ok()) {
376 ACTS_DEBUG("Error in " << state.options.direction
377 << " filter: " << res.error());
378 return res.error();
379 }
380 }
381
382 // Finalization:
383 // when all track states have been handled or an aborter is triggered
384 const bool isTrackComplete =
385 result.measurementStates == inputMeasurements.size();
386 const bool isEndOfWorldReached =
387 endOfWorldReached.checkAbort(state, stepper, navigator, logger());
388 const bool isVolumeConstraintReached = volumeConstraintAborter.checkAbort(
389 state, stepper, navigator, logger());
390 const bool isPathLimitReached = result.pathLimitReached.checkAbort(
391 state, stepper, navigator, logger());
392 const bool isTargetReached =
393 targetReached.checkAbort(state, stepper, navigator, logger());
394 if (isTrackComplete || isEndOfWorldReached || isVolumeConstraintReached ||
395 isPathLimitReached || isTargetReached) {
396 ACTS_VERBOSE(
397 "Finalizing Kalman fit: "
398 << (isTrackComplete ? "track complete; " : "")
399 << (isEndOfWorldReached ? "end of world reached; " : "")
400 << (isVolumeConstraintReached ? "volume constraint reached; " : "")
401 << (isPathLimitReached ? "path limit reached; " : "")
402 << (isTargetReached ? "target surface reached; " : ""));
403
404 if (isTargetReached) {
405 ACTS_VERBOSE("Setting fitted parameters at target surface");
406
407 // Bind the parameter to the target surface
408 auto res = stepper.boundState(state.stepping, *targetReached.surface);
409 if (!res.ok()) {
410 ACTS_DEBUG("Error while acquiring bound state for target surface: "
411 << res.error() << " " << res.error().message());
412 return res.error();
413 } else {
414 const auto& [boundParams, jacobian, pathLength] = *res;
415 result.fittedParameters = boundParams;
416 }
417 }
418
419 result.finished = true;
420 }
421
422 return Result<void>::success();
423 }

428 const navigator_t& /*navigator*/, const result_type& result,
429 const Logger& /*logger*/) const {
430 return result.finished;

444 template <typename propagator_state_t, typename stepper_t,
445 typename navigator_t>
446 Result<void> filter(const Surface& surface, propagator_state_t& state,
447 const stepper_t& stepper, const navigator_t& navigator,
448 result_type& result) const {
449 const bool precedingMeasurementExists = result.measurementStates > 0;
450 const bool surfaceIsSensitive = surface.isSensitive();
451 const bool surfaceHasMaterial = surface.surfaceMaterial() != nullptr;
452
453 // Try to find the surface in the measurement surfaces
454 const auto sourceLinkIt = inputMeasurements.find(&surface);
455 if (sourceLinkIt != inputMeasurements.end()) {
456 // Screen output message
457 ACTS_VERBOSE("Measurement surface " << surface.geometryId()
458 << " detected.");
459 // Transport the covariance to the surface
460 stepper.transportCovarianceToBound(state.stepping, surface,
461 freeToBoundCorrection);
462
463 // Update state and stepper with pre material effects
464 detail::performMaterialInteraction(
465 state, stepper, navigator,
466 detail::determineMaterialUpdateMode(state, navigator,
467 MaterialUpdateMode::PreUpdate),
468 NoiseUpdateMode::addNoise, multipleScattering, energyLoss,
469 logger());
470
471 // Create a track state with the desired components
472 TrackStatePropMask mask =
473 TrackStatePropMask::Predicted | TrackStatePropMask::Filtered |
474 TrackStatePropMask::Jacobian | TrackStatePropMask::Calibrated;
475 typename traj_t::TrackStateProxy trackStateProxy =
476 result.fittedStates->makeTrackState(mask, result.lastTrackIndex);
477
478 typename traj_t::ConstTrackStateProxy trackStateProxyConst{
479 trackStateProxy};
480
481 // Set the trackStateProxy components with the state from the ongoing
482 // propagation
483 trackStateProxy.setReferenceSurface(surface.getSharedPtr());
484 // Bind the transported state to the current surface
485 auto res = stepper.boundState(state.stepping, surface, false,
486 freeToBoundCorrection);
487 if (!res.ok()) {
488 ACTS_DEBUG("Propagate to surface " << surface.geometryId()
489 << " failed: " << res.error());
490 return res.error();
491 }
492 const auto& [boundParams, jacobian, pathLength] = *res;
493
494 // Fill the track state
495 trackStateProxy.predicted() = boundParams.parameters();
496 trackStateProxy.predictedCovariance() = state.stepping.cov;
497
498 trackStateProxy.jacobian() = jacobian;
499 trackStateProxy.pathLength() = pathLength;
500
501 // We have predicted parameters, so calibrate the uncalibrated input
502 // measurement
503 extensions.calibrator(state.geoContext, *calibrationContext,
504 sourceLinkIt->second, trackStateProxy);
505
506 // Get and set the type flags
507 auto typeFlags = trackStateProxy.typeFlags();
508 typeFlags.setHasParameters();
509 if (surface.surfaceMaterial() != nullptr) {
510 typeFlags.setHasMaterial();
511 }
512
513 // Check if the state is an outlier.
514 // If not:
515 // - run Kalman update
516 // - tag it as a measurement
517 // - update the stepping state.
518 // Else, just tag it as an outlier
519 if (!extensions.outlierFinder(trackStateProxyConst)) {
520 // Run Kalman update
521 auto updateRes =
522 extensions.updater(state.geoContext, trackStateProxy, logger());
523 if (!updateRes.ok()) {
524 ACTS_DEBUG("Update step failed: " << updateRes.error());
525 return updateRes.error();
526 }
527 // Set the measurement type flag
528 typeFlags.setIsMeasurement();
529 } else {
530 ACTS_VERBOSE(
531 "Filtering step successful. But measurement is determined "
532 "to be an outlier. Stepping state is not updated.");
533 // Set the outlier type flag
534 typeFlags.setIsOutlier();
535 trackStateProxy.shareFrom(trackStateProxy,
536 TrackStatePropMask::Predicted,
537 TrackStatePropMask::Filtered);
538 }
539
540 result.lastTrackIndex = trackStateProxy.index();
541
542 // Update the stepper if it is not an outlier
543 if (trackStateProxy.typeFlags().isMeasurement()) {
544 // Update the stepping state with filtered parameters
545 ACTS_VERBOSE("Filtering step successful, updated parameters are:\n"
546 << trackStateProxy.filtered().transpose());
547 // update stepping state using filtered parameters after kalman
548 stepper.update(state.stepping,
549 MultiTrajectoryHelpers::freeFiltered(
550 state.options.geoContext, trackStateProxy),
551 trackStateProxy.filtered(),
552 trackStateProxy.filteredCovariance(), surface);
553 // We count the state with measurement
554 ++result.measurementStates;
555 }
556
557 // Update state and stepper with post material effects
558 detail::performMaterialInteraction(
559 state, stepper, navigator,
560 detail::determineMaterialUpdateMode(state, navigator,
561 MaterialUpdateMode::PostUpdate),
562 NoiseUpdateMode::addNoise, multipleScattering, energyLoss,
563 logger());
564 // We count the processed state
565 ++result.processedStates;
566 // Update the number of holes count only when encountering a
567 // measurement
568 result.measurementHoles = result.missedActiveSurfaces.size();
569 // Since we encountered a measurement update the lastMeasurementIndex to
570 // the lastTrackIndex.
571 result.lastMeasurementIndex = result.lastTrackIndex;
572
573 } else if ((precedingMeasurementExists && surfaceIsSensitive) ||
574 surfaceHasMaterial) {
575 // We only create track states here if there is already measurement
576 // detected or if the surface has material (no holes before the first
577 // measurement)
578
579 // Create a track state with the desired components
580 TrackStatePropMask mask =
581 TrackStatePropMask::Predicted | TrackStatePropMask::Jacobian;
582 typename traj_t::TrackStateProxy trackStateProxy =
583 result.fittedStates->makeTrackState(mask, result.lastTrackIndex);
584
585 // Set the trackStateProxy components with the state from the ongoing
586 // propagation
587 trackStateProxy.setReferenceSurface(surface.getSharedPtr());
588 // Bind the transported state to the current surface
589 auto res = stepper.boundState(state.stepping, surface, true,
590 freeToBoundCorrection);
591 if (!res.ok()) {
592 return res.error();
593 }
594 const auto& [boundParams, jacobian, pathLength] = *res;
595
596 // Fill the track state
597 trackStateProxy.predicted() = boundParams.parameters();
598 trackStateProxy.predictedCovariance() = state.stepping.cov;
599
600 trackStateProxy.jacobian() = jacobian;
601 trackStateProxy.pathLength() = pathLength;
602
603 // Set the filtered parameter index to be the same with predicted
604 // parameter
605 trackStateProxy.shareFrom(trackStateProxy,
606 TrackStatePropMask::Predicted,
607 TrackStatePropMask::Filtered);
608
609 // Set the track state flags
610 auto typeFlags = trackStateProxy.typeFlags();
611 typeFlags.setHasParameters();
612
613 if (surfaceHasMaterial) {
614 typeFlags.setHasMaterial();
615 }
616
617 if (surfaceIsSensitive && precedingMeasurementExists) {
618 ACTS_VERBOSE("Detected hole on " << surface.geometryId());
619 // If the surface is sensitive, set the hole type flag
620 typeFlags.setIsHole();
621 } else if (surfaceIsSensitive) {
622 ACTS_VERBOSE("Skip hole (no preceding measurements) on surface "
623 << surface.geometryId());
624 } else if (surfaceHasMaterial) {
625 ACTS_VERBOSE("Detected in-sensitive surface "
626 << surface.geometryId());
627 }
628
629 result.lastTrackIndex = trackStateProxy.index();
630
631 if (trackStateProxy.typeFlags().isHole()) {
632 // Count the missed surface
633 result.missedActiveSurfaces.push_back(&surface);
634 }
635
636 ++result.processedStates;
637
638 // Update state and stepper with (possible) material effects
639 detail::performMaterialInteraction(
640 state, stepper, navigator,
641 detail::determineMaterialUpdateMode(state, navigator,
642 MaterialUpdateMode::FullUpdate),
643 NoiseUpdateMode::addNoise, multipleScattering, energyLoss,
644 logger());
645 }
646
647 return Result<void>::success();
648 }

711 template <typename source_link_iterator_t>
712 auto make_propagator_options(source_link_iterator_t it,
713 source_link_iterator_t end,
714 const KalmanFitterOptions<traj_t>& kfOptions,
715 const std::vector<const Surface*>* sSequence,
716 const Surface* targetSurface,
717 bool reverseDirection) const {
718 using KalmanActor = Actor;
719 using Actors = ActorList<KalmanActor>;
720 using PropagatorOptions = typename propagator_t::template Options<Actors>;
721
722 const std::size_t nMeasurements = std::distance(it, end);
723
724 // To be able to find measurements later, we put them into a map
725 // We need to copy input SourceLinks anyway, so the map can own them.
726 ACTS_VERBOSE("Preparing " << nMeasurements << " input measurements");
727 std::unordered_map<const Surface*, SourceLink> inputMeasurements;
728 for (; it != end; ++it) {
729 SourceLink sl = *it;
730 const Surface* surface = kfOptions.extensions.surfaceAccessor(sl);
731 inputMeasurements.try_emplace(surface, std::move(sl));
732 }
733
734 // Create relevant options for the propagation options
735 PropagatorOptions propagatorOptions(kfOptions.geoContext,
736 kfOptions.magFieldContext);
737
738 // Set the trivial propagator options
739 propagatorOptions.setPlainOptions(kfOptions.propagatorPlainOptions);
740
741 if (reverseDirection) {
742 propagatorOptions.direction = propagatorOptions.direction.invert();
743 }
744
745 if constexpr (!isDirectNavigator) {
746 // Add the measurement surface as external surface to navigator.
747 // We will try to hit those surface by ignoring boundary checks.
748 for (const auto& [surface, _] : inputMeasurements) {
749 propagatorOptions.navigation.insertExternalSurface(*surface);
750 }
751 } else {
752 assert(sSequence != nullptr &&
753 "DirectNavigator requires a surface sequence for KalmanFitter");
754 // Set the surface sequence
755 propagatorOptions.navigation.surfaces = *sSequence;
756 }
757
758 // Catch the actor and set the measurements
759 auto& kalmanActor = propagatorOptions.actorList.template get<KalmanActor>();
760 kalmanActor.inputMeasurements = std::move(inputMeasurements);
761 kalmanActor.targetReached.surface = targetSurface;
762 kalmanActor.multipleScattering = kfOptions.multipleScattering;
763 kalmanActor.energyLoss = kfOptions.energyLoss;
764 kalmanActor.freeToBoundCorrection = kfOptions.freeToBoundCorrection;
765 kalmanActor.calibrationContext = &kfOptions.calibrationContext.get();
766 kalmanActor.extensions = kfOptions.extensions;
767 kalmanActor.actorLogger = m_actorLogger.get();
768
769 return propagatorOptions;
770 }

772 template <typename propagator_options_t,
773 TrackContainerFrontend track_container_t>
774 auto filter_impl(const BoundTrackParameters& sParameters,
775 const propagator_options_t& propagatorOptions,
776 track_container_t& trackContainer) const
777 -> Result<typename track_container_t::TrackProxy> {
778 auto propagatorState = m_propagator.makeState(propagatorOptions);
779
780 auto propagatorInitResult =
781 m_propagator.initialize(propagatorState, sParameters);
782 if (!propagatorInitResult.ok()) {
783 ACTS_DEBUG("Propagation initialization failed: "
784 << propagatorInitResult.error());
785 return propagatorInitResult.error();
786 }
787
788 auto& kalmanResult =
789 propagatorState.template get<KalmanFitterResult<traj_t>>();
790 kalmanResult.fittedStates = &trackContainer.trackStateContainer();
791
792 // Run the fitter
793 auto result = m_propagator.propagate(propagatorState);
794
795 if (!result.ok()) {
796 ACTS_DEBUG("Propagation failed: " << result.error());
797 return result.error();
798 }
799
802 if (!kalmanResult.measurementStates) {
803 ACTS_DEBUG("KalmanFilter failed: No measurement states found");
804 return KalmanFitterError::NoMeasurementFound;
805 }
806
807 auto track = trackContainer.makeTrack();
808 track.tipIndex() = kalmanResult.lastMeasurementIndex;
809 if (kalmanResult.fittedParameters) {
810 const auto& params = kalmanResult.fittedParameters.value();
811 track.parameters() = params.parameters();
812 track.covariance() = params.covariance().value();
813 track.setReferenceSurface(params.referenceSurface().getSharedPtr());
814 }
815
816 calculateTrackQuantities(track);
817
818 return track;
819 }

834 template <typename source_link_iterator_t,
835 TrackContainerFrontend track_container_t>
836 auto fit_impl(source_link_iterator_t it, source_link_iterator_t end,
837 const BoundTrackParameters& sParameters,
838 const KalmanFitterOptions<traj_t>& kfOptions,
839 const std::vector<const Surface*>* sSequence,
840 track_container_t& trackContainer) const
841 -> Result<typename track_container_t::TrackProxy> {
842 using TrackProxy = typename track_container_t::TrackProxy;
843 using TrackStateProxy = typename track_container_t::TrackStateProxy;
844
845 auto forwardPropagatorOptions =
846 make_propagator_options(it, end, kfOptions, sSequence, nullptr, false);
847
848 auto forwardFilterResult =
849 filter_impl(sParameters, forwardPropagatorOptions, trackContainer);
850
851 if (!forwardFilterResult.ok()) {
852 ACTS_DEBUG("KalmanFilter failed: "
853 << forwardFilterResult.error() << ", "
854 << forwardFilterResult.error().message());
855 return forwardFilterResult.error();
856 }
857
858 TrackProxy forwardTrack = forwardFilterResult.value();
859
860 TrackStateProxy firstMeasurementState =
861 trackContainer.trackStateContainer().getTrackState(
862 findFirstMeasurementState(forwardTrack).value().index());
863 TrackStateProxy lastMeasurementState =
864 trackContainer.trackStateContainer().getTrackState(
865 findLastMeasurementState(forwardTrack).value().index());
866 lastMeasurementState.shareFrom(lastMeasurementState,
867 TrackStatePropMask::Filtered,
868 TrackStatePropMask::Smoothed);
869
870 TrackProxy track = forwardTrack;
871
872 const bool doReverseFilter =
873 kfOptions.reverseFiltering ||
874 kfOptions.extensions.reverseFilteringLogic(
875 typename traj_t::ConstTrackStateProxy(lastMeasurementState));
876 if (doReverseFilter) {
877 ACTS_VERBOSE("Smooth track by reversed filtering");
878
879 auto reverseStartParameters = forwardTrack.createParametersFromState(
880 typename traj_t::ConstTrackStateProxy(lastMeasurementState));
881 reverseStartParameters.covariance().value() *=
882 kfOptions.reverseFilteringCovarianceScaling;
883 auto reversePropagatorOptions = make_propagator_options(
884 it, end, kfOptions, sSequence, kfOptions.referenceSurface, true);
885 auto reverseFilterResult = filter_impl(
886 reverseStartParameters, reversePropagatorOptions, trackContainer);
887
888 if (!reverseFilterResult.ok()) {
889 ACTS_DEBUG("Reversed KalmanFilter failed: "
890 << reverseFilterResult.error() << ", "
891 << reverseFilterResult.error().message());
892 return reverseFilterResult.error();
893 }
894
895 TrackProxy reverseTrack = reverseFilterResult.value();
896
897 TrackStateProxy reverseLastMeasurementState =
898 trackContainer.trackStateContainer().getTrackState(
899 findLastMeasurementState(reverseTrack).value().index());
900
901 if (&firstMeasurementState.referenceSurface() !=
902 &reverseLastMeasurementState.referenceSurface()) {
903 ACTS_DEBUG(
904 "Inconsistent reference surfaces between forward and "
905 "reversed filtered tracks");
906 return Result<TrackProxy>::failure(
907 KalmanFitterError::InconsistentTrackStates);
908 }
909 firstMeasurementState.shareFrom(reverseLastMeasurementState,
910 TrackStatePropMask::Filtered,
911 TrackStatePropMask::Smoothed);
912
913 if (reverseTrack.hasReferenceSurface()) {
914 track.parameters() = reverseTrack.parameters();
915 track.covariance() = reverseTrack.covariance();
916 track.setReferenceSurface(
917 reverseTrack.referenceSurface().getSharedPtr());
918 }
919
920 trackContainer.removeTrack(reverseTrack.index());
921 } else {
922 ACTS_VERBOSE("Smooth track directly without reversed filtering");
923
924 auto smoothRes = kfOptions.extensions.smoother(
925 kfOptions.geoContext, trackContainer.trackStateContainer(),
926 forwardTrack.tipIndex(), logger());
927 if (!smoothRes.ok()) {
928 ACTS_DEBUG("Smoothing step failed: " << smoothRes.error() << ", "
929 << smoothRes.error().message());
930 return smoothRes.error();
931 }
932 }
933
934 if (!track.hasReferenceSurface() && kfOptions.referenceSurface != nullptr) {
935 typename propagator_t::template Options<> extrapolationOptions(
936 kfOptions.geoContext, kfOptions.magFieldContext);
937 auto extrapolationResult = extrapolateTrackToReferenceSurface(
938 track, *kfOptions.referenceSurface, m_propagator,
939 extrapolationOptions, kfOptions.referenceSurfaceStrategy, logger());
940
941 if (!extrapolationResult.ok()) {
942 ACTS_DEBUG("Extrapolation to reference surface failed: "
943 << extrapolationResult.error() << ", "
944 << extrapolationResult.error().message());
945 return extrapolationResult.error();
946 }
947 }
948
949 if (trackContainer.hasColumn(hashString("smoothed"))) {
950 track.template component<bool, hashString("smoothed")>() =
951 !doReverseFilter;
952 }
953 if (trackContainer.hasColumn(hashString("reversed"))) {
954 track.template component<bool, hashString("reversed")>() =
955 doReverseFilter;
956 }
957
958 return track;
959 }
960};
