47 static inline std::array<double, 3> decomposeCovariance(
48 const SquareMatrix<2>& covariance) {
49 double c00 = covariance(eBoundLoc0, eBoundLoc0);
50 double c01 = covariance(eBoundLoc0, eBoundLoc1);
51 double c11 = covariance(eBoundLoc1, eBoundLoc1);
52
53 double cdsq = std::pow((c00 - c11), 2) / 4.;
54 double cosq = c01 * c01;
55
56 // Calculate the eigen values w.r.t reference frame
57 double lambda0 = (c00 + c11) / 2. + std::sqrt(cdsq + cosq);
58 double lambda1 = (c00 + c11) / 2. - std::sqrt(cdsq + cosq);
59 double theta = std::atan2(lambda0 - c00, c01);
60
61 return {lambda0, lambda1, theta};
62 }

74 static inline std::vector<Vector3> createEllipse(
75 double lambda0, double lambda1, double theta, std::size_t lseg,
76 double offset, const Vector2& lposition = Vector2(0., 0.),
77 const Transform3& transform = Transform3::Identity()) {
78 double ctheta = std::cos(theta);
79 double stheta = std::sin(theta);
80
81 double l1sq = std::sqrt(lambda0);
82 double l2sq = std::sqrt(lambda1);
83
84 // Now generate the ellipse points
85 std::vector<Vector3> ellipse;
86 ellipse.reserve(lseg);
87 double thetaStep = 2 * std::numbers::pi / lseg;
88 for (std::size_t it = 0; it < lseg; ++it) {
89 double phi = -std::numbers::pi + it * thetaStep;
90 double cphi = std::cos(phi);
91 double sphi = std::sin(phi);
92 double x = lposition.x() + (l1sq * ctheta * cphi - l2sq * stheta * sphi);
93 double y = lposition.y() + (l1sq * stheta * cphi + l2sq * ctheta * sphi);
94 ellipse.push_back(transform * Vector3(x, y, offset));
95 }
96 return ellipse;
97 }

138 template <typename parameters_t>
139 static inline void drawBoundTrackParameters(
140 IVisualization3D& helper, const parameters_t& parameters,
141 const GeometryContext& gctx =
142 GeometryContext::dangerouslyDefaultConstruct(),
143 double momentumScale = 1., double locErrorScale = 1.,
144 double angularErrorScale = 1.,
145 const ViewConfig& parConfig = s_viewParameter,
146 const ViewConfig& covConfig = s_viewParameter,
147 const ViewConfig& surfConfig = s_viewSensitive) {
148 if (surfConfig.visible) {
149 GeometryView3D::drawSurface(helper, parameters.referenceSurface(), gctx,
150 Transform3::Identity(), surfConfig);
151 }
152
153 // Draw the parameter shaft and cone
154 auto position = parameters.position(gctx);
155 auto direction = parameters.direction();
156 double p = parameters.absoluteMomentum();
157
158 ViewConfig lparConfig = parConfig;
159 lparConfig.lineThickness = 0.05;
160 Vector3 parLength = p * momentumScale * direction;
161
162 GeometryView3D::drawArrowBackward(
163 helper, position, position + 0.5 * parLength, 100., 1.0, lparConfig);
164
165 GeometryView3D::drawArrowForward(helper, position + 0.5 * parLength,
166 position + parLength, 4., 2.5, lparConfig);
167
168 if (parameters.covariance().has_value()) {
169 auto paramVec = parameters.parameters();
170 auto lposition = paramVec.template block<2, 1>(0, 0);
171
172 // Draw the local covariance
173 const auto& covariance = *parameters.covariance();
174 drawCovarianceCartesian(
175 helper, lposition, covariance.template block<2, 2>(0, 0),
176 parameters.referenceSurface().localToGlobalTransform(gctx),
177 locErrorScale, covConfig);
178
179 drawCovarianceAngular(
180 helper, position, direction, covariance.template block<2, 2>(2, 2),
181 0.9 * p * momentumScale, angularErrorScale, covConfig);
182 }
183 }

196 IVisualization3D& helper, const Vector2& lposition,
197 const SquareMatrix2& covariance, const Transform3& transform,

226 template <typename traj_t>
227 static void drawMultiTrajectory(
228 IVisualization3D& helper, const traj_t& multiTraj,
229 const std::size_t& entryIndex,
230 const GeometryContext& gctx =
231 GeometryContext::dangerouslyDefaultConstruct(),
232 double momentumScale = 1., double locErrorScale = 1.,
233 double angularErrorScale = 1.,
234 const ViewConfig& surfaceConfig = s_viewSensitive,
235 const ViewConfig& measurementConfig = s_viewMeasurement,
236 const ViewConfig& predictedConfig = s_viewPredicted,
237 const ViewConfig& filteredConfig = s_viewFiltered,
238 const ViewConfig& smoothedConfig = s_viewSmoothed) {
239 // @TODO: Refactor based on Track class
240
241 // TODO get particle hypothesis from track
242 ParticleHypothesis particleHypothesis = ParticleHypothesis::pion();
243
244 // Visit the track states on the trajectory
245 multiTraj.visitBackwards(entryIndex, [&](const auto& state) {
246 // Only draw the measurement states
247 if (!state.typeFlags().hasMeasurement()) {
248 return true;
249 }
250
251 // Use smaller scaling factors for the first state
252 // @Todo: add parameter for the first state error scaling
253 if (state.index() == 0) {
254 locErrorScale = locErrorScale * 0.1;
255 angularErrorScale = angularErrorScale * 0.1;
256 }
257
258 // First, if necessary, draw the surface
259 if (surfaceConfig.visible) {
260 GeometryView3D::drawSurface(helper, state.referenceSurface(), gctx,
261 Transform3::Identity(), surfaceConfig);
262 }
263
264 // Second, if necessary and present, draw the calibrated measurement (only
265 // draw 2D measurement here)
266 // @Todo: how to draw 1D measurement?
267 if (state.hasCalibrated() && state.calibratedSize() == 2) {
268 const Vector2& lposition = state.template calibrated<2>();
269 const SquareMatrix2 covariance =
270 state.template calibratedCovariance<2>();
271 drawMeasurement(helper, lposition, covariance,
272 state.referenceSurface().localToGlobalTransform(gctx),
273 locErrorScale, measurementConfig);
274 }
275
276 // Last, if necessary and present, draw the track parameters
277 // (a) predicted track parameters
278 if (predictedConfig.visible && state.hasPredicted()) {
279 drawBoundTrackParameters(
280 helper,
281 BoundTrackParameters(state.referenceSurface().getSharedPtr(),
282 state.predicted(), state.predictedCovariance(),
283 particleHypothesis),
284 gctx, momentumScale, locErrorScale, angularErrorScale,
285 predictedConfig, predictedConfig, {.visible = false});
286 }
287 // (b) filtered track parameters
288 if (filteredConfig.visible && state.hasFiltered()) {
289 drawBoundTrackParameters(
290 helper,
291 BoundTrackParameters(state.referenceSurface().getSharedPtr(),
292 state.filtered(), state.filteredCovariance(),
293 particleHypothesis),
294 gctx, momentumScale, locErrorScale, angularErrorScale,
295 filteredConfig, filteredConfig, {.visible = false});
296 }
297 // (c) smoothed track parameters
298 if (smoothedConfig.visible && state.hasSmoothed()) {
299 drawBoundTrackParameters(
300 helper,
301 BoundTrackParameters(state.referenceSurface().getSharedPtr(),
302 state.smoothed(), state.smoothedCovariance(),
303 particleHypothesis),
304 gctx, momentumScale, locErrorScale, angularErrorScale,
305 smoothedConfig, smoothedConfig, {.visible = false});
306 }
307 return true;
308 });
309 }
