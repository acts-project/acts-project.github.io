79 struct ProtoAxis {
81 BinningType bType = BinningType::equidistant;
83 AxisDirection axisDir = AxisDirection::AxisX;
85 std::size_t nBins = 0;
87 AxisScalar min = 0;
89 AxisScalar max = 0;
91 std::vector<AxisScalar> binEdges;
92
96 std::size_t getBin(AxisScalar x) const {
97 if (binEdges.empty()) {
98 // equidistant
99 AxisScalar w = (max - min) / nBins;
100 return static_cast<std::size_t>(std::floor((x - min) / w));
101 } else {
102 // variable
103 const auto it = std::ranges::upper_bound(binEdges, x);
104 return static_cast<std::size_t>(
105 std::ranges::distance(binEdges.begin(), it)) -
106 1;
107 }
108 }
109 };

283 static bool isSurfaceEquivalent(const GeometryContext& gctx,
284 AxisDirection aDir, const Surface* a,
285 const Surface* b) {
286 using namespace UnitLiterals;
287 using VectorHelpers::perp;
288
289 if (aDir == AxisDirection::AxisPhi) {
290 // Take the two binning positions
291 Vector3 pos1 = a->referencePosition(gctx, AxisDirection::AxisR);
292 Vector3 pos2 = b->referencePosition(gctx, AxisDirection::AxisR);
293
294 // Project them on the (x, y) plane, where Phi angles are calculated
295 auto proj1 = pos1.head<2>(), proj2 = pos2.head<2>();
296
297 // Basic dot and cross products identities give us the cosine and sine
298 // of these angles, time the squared vector norm
299 auto cos_dPhi_n2 = proj1.dot(proj2);
300 auto sin_dPhi_n2 = proj1.x() * proj2.y() - proj2.x() * proj1.y();
301
302 // ...so by injecting them into atan2, we get the angle between them
303 auto dPhi = std::atan2(sin_dPhi_n2, cos_dPhi_n2);
304 return std::abs(dPhi) < std::numbers::pi / 180.;
305 }
306
307 if (aDir == AxisDirection::AxisZ) {
308 return (std::abs(a->referencePosition(gctx, AxisDirection::AxisR).z() -
309 b->referencePosition(gctx, AxisDirection::AxisR).z()) <
310 1_um);
311 }
312
313 if (aDir == AxisDirection::AxisR) {
314 return (std::abs(perp(a->referencePosition(gctx, AxisDirection::AxisR)) -
315 perp(b->referencePosition(gctx, AxisDirection::AxisR))) <
316 1_um);
317 }
318
319 return false;
320 }

415 using ISGL = SurfaceArray::ISurfaceGridLookup;

426 maxNeighborDistance);
