258 static bool isSurfaceEquivalent(const GeometryContext& gctx,
259 AxisDirection aDir, const Surface* a,
260 const Surface* b) {
261 using namespace UnitLiterals;
262 using VectorHelpers::perp;
263
264 if (aDir == AxisDirection::AxisPhi) {
265 // Take the two binning positions
266 Vector3 pos1 = a->referencePosition(gctx, AxisDirection::AxisR);
267 Vector3 pos2 = b->referencePosition(gctx, AxisDirection::AxisR);
268
269 // Project them on the (x, y) plane, where Phi angles are calculated
270 auto proj1 = pos1.head<2>(), proj2 = pos2.head<2>();
271
272 // Basic dot and cross products identities give us the cosine and sine
273 // of these angles, time the squared vector norm
274 auto cos_dPhi_n2 = proj1.dot(proj2);
275 auto sin_dPhi_n2 = proj1.x() * proj2.y() - proj2.x() * proj1.y();
276
277 // ...so by injecting them into atan2, we get the angle between them
278 auto dPhi = std::atan2(sin_dPhi_n2, cos_dPhi_n2);
279 return std::abs(dPhi) < std::numbers::pi / 180.;
280 }
281
282 if (aDir == AxisDirection::AxisZ) {
283 return (std::abs(a->referencePosition(gctx, AxisDirection::AxisR).z() -
284 b->referencePosition(gctx, AxisDirection::AxisR).z()) <
285 1_um);
286 }
287
288 if (aDir == AxisDirection::AxisR) {
289 return (std::abs(perp(a->referencePosition(gctx, AxisDirection::AxisR)) -
290 perp(b->referencePosition(gctx, AxisDirection::AxisR))) <
291 1_um);
292 }
293
294 return false;
295 }
