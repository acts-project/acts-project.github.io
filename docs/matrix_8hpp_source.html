143template <concepts::matrix matrix_t, std::size_t COLS = matrix_t::columns()>
144DETRAY_HOST_DEVICE constexpr matrix_t zero() noexcept {
145 matrix_t m;
146
147 DETRAY_UNROLL_N(COLS)
148 for (std::size_t j = 0u; j < COLS; ++j) {
149 // Fill zero initialized vector
150 m[j] = typename matrix_t::vector_type{};
151 }

184template <std::size_t ROW, std::size_t COL, concepts::scalar scalar_t,
185 template <typename, std::size_t> class array_t, std::size_t... I>
186DETRAY_HOST_DEVICE constexpr auto transpose(
187 const matrix<array_t, scalar_t, ROW, COL> &m,
188 std::index_sequence<I...> /*unused*/) noexcept {
189 using matrix_T_t = matrix<array_t, scalar_t, COL, ROW>;
190 using column_t = typename matrix_T_t::vector_type;
191
192 matrix_T_t res_m;
193
194 DETRAY_UNROLL_N(ROW)
195 for (std::size_t j = 0u; j < ROW; ++j) {
196 res_m[j] = column_t{m[I][j]...};
197 }
198
199 return res_m;
200}

283template <std::size_t ROW, std::size_t COL, concepts::scalar scalar_t,
284 template <typename, std::size_t> class array_t>
285DETRAY_HOST_DEVICE constexpr decltype(auto) operator*(
286 const matrix<array_t, scalar_t, ROW, COL> &lhs,
287 const vector<COL, scalar_t, array_t> &v) noexcept {
288 // Init vector
289 vector<ROW, scalar_t, array_t> res_v{v[0] * lhs[0]};
290
291 // Add the rest per column
292 DETRAY_UNROLL_N(COL)
293 for (std::size_t j = 1u; j < COL; ++j) {
294 // fma
295 res_v = res_v + v[j] * lhs[j];
296 }
297
298 return res_v;
299}

302template <std::size_t LROW, std::size_t COL, std::size_t RCOL,
303 concepts::scalar scalar_t,
304 template <typename, std::size_t> class array_t>
305DETRAY_HOST_DEVICE constexpr decltype(auto) operator*(
306 const matrix<array_t, scalar_t, LROW, COL> &lhs,
307 const matrix<array_t, scalar_t, COL, RCOL> &rhs) noexcept {
308 matrix<array_t, scalar_t, LROW, RCOL> res_m;
309
310 DETRAY_UNROLL_N(RCOL)
311 for (std::size_t j = 0u; j < RCOL; ++j) {
312 // Init column j
313 res_m[j] = rhs[j][0] * lhs[0];
314
315 // Add the rest per column
316 DETRAY_UNROLL_N(COL)
317 for (std::size_t i = 1u; i < COL; ++i) {
318 // fma
319 res_m[j] = res_m[j] + rhs[j][i] * lhs[i];
320 }
321 }
322
323 return res_m;
324}
