614 DETRAY_HOST_DEVICE constexpr void run_inspector(
615 [[maybe_unused]] const navigation::config &cfg,
616 [[maybe_unused]] const point3_t &track_pos,
617 [[maybe_unused]] const vector3_t &track_dir,
618 [[maybe_unused]] const char *message) {
619 [[maybe_unused]] auto derived = static_cast<const derived_t &>(*this);
620
621 if constexpr (!std::is_same_v<inspector_t, navigation::void_inspector>) {
622 cast_impl().inspector()(derived, cfg, track_pos, track_dir, message);
623 }
624
625 DETRAY_DEBUG_HOST("" << message << "\n"
626 << detray::navigation::print_state(derived)
627 << detray::navigation::print_candidates(
628 derived, cfg, track_pos, track_dir));
629 }

639 DETRAY_HOST_DEVICE constexpr void run_inspector(

642 [[maybe_unused]] const vector3_t &track_dir,
