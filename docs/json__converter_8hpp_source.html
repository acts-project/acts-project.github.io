38class json_input_converter final : public input_file_converter_interface {
39 public:
41 json_input_converter() : input_file_converter_interface(".json") {}
42
46 void add_from_file(io::file_handle& file,
47 detector_payload& payload) override {
48 // Reads the data from file and returns the corresponding io payloads
49 nlohmann::json in_json;
50 *file >> in_json;
51
52 // Peek at the header to determine the kind of payload is in the file
53 const tagged_header_payload header = in_json["header"]["common"];
54 // If not already done, add common header information to detector payload
55 if (payload.header.version.empty()) {
56 payload.detector_name = header.detector;
57 payload.header = header;
58 }
59
60 if (header.version < io::detail::minimal_io_version) {
61 DETRAY_WARN_HOST("File was generated with a different detray version");
62 }
63
64 nlohmann::json& json_payload = in_json["data"];
65 if (header.tag == "geometry") {
66 payload.geometry = json_payload;
67 } else if (header.tag == "homogeneous_material") {
68 payload.homogeneous_material = json_payload;
69 } else if (header.tag == "material_maps") {
70 payload.material_maps = json_payload;
71 } else if (header.tag == "surface_grids") {
72 payload.surface_grids = json_payload;
73 } else {
74 std::stringstream err_stream{};
75 err_stream << "Unsupported file tag '" << header.tag << "' in input file";
76 DETRAY_FATAL_HOST(err_stream.str());
77 throw std::invalid_argument(err_stream.str());
78 }
79 }
80};

94template <class detector_t, class backend_t>
95 requires concepts::writer_backend<detector_t, backend_t>
96class json_converter<detector_t, backend_t> final
97 : public writer_interface<detector_t> {
98 using io_backend = backend_t;
99
100 public:
102 json_converter() : writer_interface<detector_t>(".json") {}
103
111 std::string write(const detector_t& det,
112 const typename detector_t::name_map& names,
113 const std::string& source,
114 const std::ios_base::openmode mode = std::ios::out |
115 std::ios::binary,
116 const std::filesystem::path& file_path = {"./"}) override {
117 // Assert output stream
118 assert(((mode == std::ios_base::out) ||
119 (mode == (std::ios_base::out | std::ios_base::binary)) ||
120 (mode == (std::ios_base::out | std::ios_base::trunc)) ||
121 (mode == (std::ios_base::out | std::ios_base::trunc |
122 std::ios_base::binary))) &&
123 "Illegal file mode for json writer");
124
125 std::string det_name = det.name(names);
126
127 // Create a new file
128 std::string file_stem{det_name + "_" + std::string(io_backend::tag)};
129 io::file_handle file{file_path / file_stem, this->file_extension(), mode};
130
131 // Write some general information
132 nlohmann::ordered_json out_json;
133 out_json["header"] = io_backend::to_header_payload(det);
134 out_json["header"]["common"] =
135 io::detail::basic_converter::to_tagged_header_payload<detector_t>(
136 det_name, std::string(io_backend::tag), source);
137
138 // Write the detector data into the json stream by using the
139 // conversion functions defined in "detray/io/json/json_io.hpp"
140 out_json["data"] = io_backend::to_payload(det, names);
141
142 // Write to file
143 *file << std::setw(4) << out_json << std::endl;
144
145 return file_stem + this->file_extension();
146 }
147};
