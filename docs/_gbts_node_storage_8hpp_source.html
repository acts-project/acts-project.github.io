193 const OuterStripSpacePointCalibrationDetailsDerived& strip(
194 SpacePointIndex node) const {
195 // A strip layer fed from a container without the pairs leaves the node
196 // without one while its bin still says it is a strip bin.
197 assert(node < m_stripIndex.size() &&
198 m_stripIndex[node] != detail::kNoStrip &&
199 "node carries no stereo pair");
200 return m_strips[m_stripIndex[node]];
