220 const OuterStripSpacePointCalibrationDetailsDerived& strip(
221 SpacePointIndex node) const {
222 // A strip layer fed from a container without the pairs leaves the node
223 // without one while its bin still says it is a strip bin.
224 assert(node < m_stripIndex.size() &&
225 m_stripIndex[node] != detail::kNoStrip &&
226 "node carries no stereo pair");
227 return m_strips[m_stripIndex[node]];
228 }
