209 const OuterStripSpacePointCalibrationDetailsDerived& strip(
210 SpacePointIndex node) const {
211 // A strip layer fed from a container without the pairs leaves the node
212 // without one while its bin still says it is a strip bin.
213 assert(node < m_stripIndex.size() &&
214 m_stripIndex[node] != detail::kNoStrip &&
215 "node carries no stereo pair");
216 return m_strips[m_stripIndex[node]];
