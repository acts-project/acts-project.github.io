152 DETRAY_HOST_DEVICE
153 friend constexpr auto operator<=>(const identifier lhs,
154 const identifier rhs) noexcept {
155 const auto l{lhs.index()};
156 const auto r{rhs.index()};
157 if (l < r || (l == r && l < r)) {
158 return std::strong_ordering::less;
159 }
160 if (l > r || (l == r && l > r)) {
161 return std::strong_ordering::greater;
162 }
163 return std::strong_ordering::equivalent;
164 }

166 private:
167 // clang-format off
168 static constexpr value_t k_volume_mask = 0xfff0000000000000; // (2^12)-1 = 4095 volumes
169 static constexpr value_t k_id_mask = 0x000f000000000000; // (2^4)-1 = 15 surface categories
170 static constexpr value_t k_index_mask = 0x0000fffff8000000; // (2^21)-1 = 2097151 surfaces
171 static constexpr value_t k_transform_mask = 0x0000000007ffffc0; // (2^21)-1 = 2097151 transforms
172 static constexpr value_t k_extra_mask = 0x000000000000003f; // (2^6)-1 = 63 extra tags
173 // clang-format on
174
175 // The masks together must cover all bits
176 static_assert(k_volume_mask + k_id_mask + k_index_mask + k_transform_mask +
177 k_extra_mask ==
178 ~static_cast<value_t>(0));
179
181 value_t m_value{~static_cast<value_t>(0)};
182
184 DETRAY_HOST
185 friend std::ostream& operator<<(std::ostream& os, const identifier c) {
186 if (c.is_invalid()) {
187 os << "INVALID: ";
188 }
189
190 os << "vol = " << c.volume();
191 os << " | id = " << c.id() << "(" << static_cast<int>(c.id()) << ")";
192 os << " | index = " << c.index();
193 os << " | trf = " << c.transform();
194 os << " | extra = " << c.extra();
195
196 return os;
197 }
