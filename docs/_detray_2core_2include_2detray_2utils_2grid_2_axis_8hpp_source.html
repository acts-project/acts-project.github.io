117 DETRAY_HOST_DEVICE
118 constexpr dindex nbins() const {
119 // The open axis boundary has extra over- and underflow bins that are
120 // automatically added beyond the axis span
121 if constexpr (bounds_type::type == axis::bounds::e_open) {
122 return m_binning.nbins() + 2u;
123 } else {
124 return m_binning.nbins();

134 return m_binning.bin_width(std::forward<Args>(args)...);

144 DETRAY_HOST_DEVICE
145 dindex bin(const scalar_type v) const {
146 int b{m_bounds.map(m_binning.bin(v), m_binning.nbins())};
147
148 if constexpr (bounds_type::type == axis::bounds::e_circular) {
149 b = m_bounds.wrap(b, m_binning.nbins());
150 }
151
152 return static_cast<dindex>(b);
153 }

164 template <typename neighbor_t>
165 DETRAY_HOST_DEVICE bin_range range(const scalar_type v,
166 const darray<neighbor_t, 2> &nhood) const {
167 return m_bounds.map(m_binning.range(v, nhood), m_binning.nbins());
168 }

173 return m_binning.bin_edges(ibin);

220 static constexpr dindex dim = sizeof...(axis_ts);
222 static constexpr bool is_owning = ownership;

344 template <std::size_t I>
345 DETRAY_HOST_DEVICE types::get<axis_reg, types::id_cast<axis_reg, I>>
346 get_axis() const {
347 if constexpr (std::same_as<edge_offset_range_t,
348 vecmem::vector<dsized_index_range>>) {
349#if defined(__CUDACC__)

379 DETRAY_HOST_DEVICE constexpr auto nbins_per_axis() const -> loc_bin_index {
380 // Empty bin indices to be filled
381 loc_bin_index n_bins{};
382 // Get the number of bins for every axis
383 (get_axis_nbins(get_axis<axis_ts>(), n_bins), ...);
384
385 return n_bins;

389 DETRAY_HOST_DEVICE constexpr auto nbins() const -> dindex {
390 const auto n_bins_per_axis = nbins_per_axis();
391 dindex n_bins{1u};
392 for (dindex i = 0u; i < dim; ++i) {
393 n_bins *= n_bins_per_axis[i];
394 }
395 return n_bins;

499 template <typename axis_t>
500 DETRAY_HOST_DEVICE void get_axis_nbins(const axis_t &ax,
501 loc_bin_index &n_bins) const {
502 // Get the index corresponding to the axis label (e.g. bin_x <=> 0)
503 constexpr auto loc_idx{
504 types::index_cast<axis_reg, axis_t::bounds_type::label>};
505 n_bins[loc_idx] = ax.nbins();
506 }
507
517 template <typename axis_t>
518 DETRAY_HOST_DEVICE void get_axis_bin(const axis_t &ax, const point_type &p,
519 loc_bin_index &bin_indices) const {
520 // Get the index corresponding to the axis label (e.g. bin_x <=> 0)
521 constexpr auto loc_idx{
522 types::index_cast<axis_reg, axis_t::bounds_type::label>};
523 bin_indices[loc_idx] = ax.bin(p[loc_idx]);
524 }
525
538 template <typename axis_t, typename neighbor_t>
539 DETRAY_HOST_DEVICE void get_axis_bin_ranges(
540 const axis_t &ax, const point_type &p, const darray<neighbor_t, 2> &nhood,
541 multi_bin_range<dim> &bin_ranges) const {
542 // Get the index corresponding to the axis label (e.g. bin_range_x = 0)
543 constexpr auto loc_idx{
544 types::index_cast<axis_reg, axis_t::bounds_type::label>};
545 bin_ranges[loc_idx] = ax.range(p[loc_idx], nhood);
546
547 assert(bin_ranges[loc_idx][0] <= bin_ranges[loc_idx][1]);
548 }
549
551 edge_offset_range_t m_edge_offsets{};
553 edge_range_t m_edges{};
554};
