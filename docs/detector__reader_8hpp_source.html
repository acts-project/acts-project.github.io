35template <class detector_t, std::size_t CAP = 0u, std::size_t DIM = 2u>
36void read_components_from_file(const std::vector<std::string>& file_names,
37 detector_builder<typename detector_t::metadata,
38 volume_builder>& det_builder) {
39 // Hold all required readers (one for every component)
40 detail::detector_components_reader<detector_t> readers;
41
42 // Register the readers for the files in json format
43 detail::add_json_readers<CAP, DIM>(readers, file_names);
44
45 // Make sure that all files will be read
46 if (readers.size() != file_names.size()) {
47 std::stringstream err_str{};
48 for (const auto& [file_name, reader] : readers.readers_map()) {
49 err_str << "-> " << file_name << std::endl;
50 }
51 DETRAY_ERROR_HOST("Not all files were registered to a reader. "
52 << "Successfully registered files:\n"
53 << err_str.str());
54 }
55
56 // Read the data into the detector builder
57 readers.read(det_builder);
58}

72template <class detector_t, std::size_t CAP = 0u, std::size_t DIM = 2u,
73 template <typename> class volume_builder_t = volume_builder>
74auto read_detector(vecmem::memory_resource& resc,
75 const detector_reader_config& cfg) noexcept(false) {
76 // Map the volume names to their indices
77 typename detector_t::name_map names{};
78
79 detector_builder<typename detector_t::metadata, volume_builder_t> det_builder;
80
81 DETRAY_INFO_HOST("Reading detector files... ");
82
83 // Register readers for the respective detector component and file format
84 // and read the data into the detector_builder
85 read_components_from_file<detector_t, CAP, DIM>(cfg.files(), det_builder);
86
87 DETRAY_INFO_HOST("Done reading files");
88
89 // Build and return the detector
90 auto det = det_builder.build(resc, cfg.builder_options(), names);
91
92 if (cfg.do_check()) {
93 // This will throw an exception in case of inconsistencies
94 detray::detail::check_consistency(det, cfg.verbose_check(), names);
95 }
96
97 DETRAY_DEBUG_HOST("\n" << detray::utils::print_detector(det, names));
98
99 return std::make_pair(std::move(det), std::move(names));
100}
